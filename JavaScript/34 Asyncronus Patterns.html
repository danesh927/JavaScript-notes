ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಅಸಿಂಕ್ರೊನಸ್ (Asynchronous) ಕೆಲಸಗಳನ್ನು ನಿರ್ವಹಿಸಲು ಪ್ರಮುಖವಾಗಿ 3 ವಿಧಾನಗಳಿವೆ (Patterns). ನಾನ್-ಐಟಿ ಹಿನ್ನೆಲೆಯವರಿಗಾಗಿ ದೈನಂದಿನ ಉದಾಹರಣೆಗಳಿಂದ ಶುರುಮಾಡಿ, ಫ್ರಂಟ್-ಎಂಡ್ (Front-end) ಡೆವಲಪರ್ ಹಂತದವರೆಗೆ ಈ ಮೂರೂ ವಿಧಾನಗಳನ್ನು ಆಳವಾಗಿ ಅರ್ಥಮಾಡಿಕೊಳ್ಳೋಣ.

1. ಕಾಲ್‌ಬ್ಯಾಕ್ (Callbacks) - [ಹಳೆಯ ವಿಧಾನ]

ನಾನ್-ಐಟಿ (Non-IT) ಉದಾಹರಣೆ:
ನೀವು ನಿಮ್ಮ ಬೈಕ್ ಅನ್ನು ಸರ್ವಿಸ್‌ಗೆ ಬಿಡುತ್ತೀರಿ. ಮೆಕ್ಯಾನಿಕ್‌ಗೆ, "ನಾನು ಹೋಗುತ್ತೇನೆ, ಬೈಕ್ ರೆಡಿ ಆದ ಮೇಲೆ ನನಗೆ ಕಾಲ್ ಬ್ಯಾಕ್ (Call back) ಮಾಡು" ಎಂದು ಹೇಳುತ್ತೀರಿ. ಮೆಕ್ಯಾನಿಕ್ ರಿಪೇರಿ ಮುಗಿಸಿದ ಮೇಲೆ ನಿಮಗೆ ಕರೆ ಮಾಡಿ ತಿಳಿಸುತ್ತಾನೆ. ಅಲ್ಲಿಯವರೆಗೆ ನೀವು ನಿಮ್ಮ ಬೇರೆ ಕೆಲಸಗಳನ್ನು ಮಾಡಿಕೊಂಡಿರಬಹುದು.

ಪ್ರೊಫೆಷನಲ್ (Professional) ವಿವರಣೆ:
ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಫಂಕ್ಷನ್‌ಗಳು 'ಫಸ್ಟ್-ಕ್ಲಾಸ್ ಸಿಟಿಜನ್ಸ್'. ಅಂದರೆ, ಒಂದು ಫಂಕ್ಷನ್ ಅನ್ನು ಮತ್ತೊಂದು ಫಂಕ್ಷನ್‌ಗೆ 'ಆರ್ಗ್ಯುಮೆಂಟ್' ಆಗಿ ಕಳುಹಿಸಬಹುದು. ಮುಖ್ಯ ಕೆಲಸ ಮುಗಿದ ನಂತರ ಆ ಆರ್ಗ್ಯುಮೆಂಟ್ ಆಗಿ ಬಂದ ಫಂಕ್ಷನ್ ಅನ್ನು ರನ್ ಮಾಡುವುದೇ 'ಕಾಲ್‌ಬ್ಯಾಕ್'.

  • ಸಮಸ್ಯೆ (Callback Hell / Pyramid of Doom):
    ಫ್ರಂಟ್-ಎಂಡ್ ಡೆವಲಪ್‌ಮೆಂಟ್‌ನಲ್ಲಿ, ನೀವು ಮೊದಲು ಯೂಸರ್ ಡೇಟಾ ತಂದು, ಆಮೇಲೆ ಅವರ ಪೋಸ್ಟ್‌ಗಳನ್ನು ತಂದು, ಆಮೇಲೆ ಕಮೆಂಟ್‌ಗಳನ್ನು ತರಬೇಕಾದರೆ, ಒಂದರೊಳಗೆ ಮತ್ತೊಂದು ಕಾಲ್‌ಬ್ಯಾಕ್ ಬರೆಯುತ್ತಾ ಹೋದರೆ ಕೋಡ್ ತ್ರಿಕೋನಾಕಾರದಲ್ಲಿ (Pyramid) ಬೆಳೆಯುತ್ತದೆ. ಇದನ್ನು ಓದುವುದು ಮತ್ತು ಡಿಬಗ್ ಮಾಡುವುದು ಬಹಳ ಕಷ್ಟ.
// ಫ್ರಂಟ್-ಎಂಡ್ ಡೆವಲಪರ್ ಎದುರಿಸುವ 'Callback Hell' ಉದಾಹರಣೆ
getUserData(function(user) {
    getPosts(user.id, function(posts) {
        getComments(posts[0].id, function(comments) {
            updateDOM(comments); // DOM Manipulation
        });
    });
});

2. ಪ್ರಾಮಿಸ್ (Promises) - [ಸುಧಾರಿತ ವಿಧಾನ]

ನಾನ್-ಐಟಿ (Non-IT) ಉದಾಹರಣೆ:
ನೀವು ಒಂದು ಆನ್‌ಲೈನ್ ಆ್ಯಪ್‌ನಲ್ಲಿ (ಉದಾ: Swiggy/Zomato) ಊಟ ಆರ್ಡರ್ ಮಾಡುತ್ತೀರಿ. ಅವರು ನಿಮಗೆ ಊಟ ತಲುಪಿಸುವ ಒಂದು "ಭರವಸೆ" (Promise) ಕೊಡುತ್ತಾರೆ ಮತ್ತು ಆ್ಯಪ್‌ನಲ್ಲಿ ಒಂದು ಸ್ಟೇಟಸ್ ಕಾಣಿಸುತ್ತದೆ.
ಇದು 3 ಹಂತಗಳಲ್ಲಿ ಇರುತ್ತದೆ:

  1. Pending (ಕಾಯುವಿಕೆ): ಊಟ ತಯಾರಾಗುತ್ತಿದೆ, ಇನ್ನು ಬಂದಿಲ್ಲ.
  2. Fulfilled (ಯಶಸ್ಸು): ಊಟ ಮನೆಗೆ ಬಂತು (ಭರವಸೆ ಈಡೇರಿತು).
  3. Rejected (ವಿಫಲ): ಮಳೆಯಿಂದಾಗಿ ಆರ್ಡರ್ ಕ್ಯಾನ್ಸಲ್ ಆಯಿತು (ಭರವಸೆ ಮುರಿಯಿತು).

ಪ್ರೊಫೆಷನಲ್ (Professional) ವಿವರಣೆ:
ಕಾಲ್‌ಬ್ಯಾಕ್ ಸಮಸ್ಯೆಯನ್ನು ಬಗೆಹರಿಸಲು ES6 (2015) ನಲ್ಲಿ Promises ಪರಿಚಯಿಸಲಾಯಿತು. Promise ಎಂಬುದು ಭವಿಷ್ಯದಲ್ಲಿ ಸಿಗುವ ಒಂದು ಮೌಲ್ಯವನ್ನು (Value) ಪ್ರತಿನಿಧಿಸುವ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಆಬ್ಜೆಕ್ಟ್.

  • ಡೇಟಾ ಯಶಸ್ವಿಯಾಗಿ ಬಂದರೆ .then() ಬಳಸಿ ಅದನ್ನು ಪಡೆಯುತ್ತೇವೆ.
  • ಏನಾದರೂ ಎರರ್ ಬಂದರೆ .catch() ಬಳಸಿ ಅದನ್ನು ನಿಭಾಯಿಸುತ್ತೇವೆ (Error Handling).
  • ಇದರಿಂದ ಕೋಡ್ ಸಮತಟ್ಟಾಗಿ (Flat) ಮತ್ತು ಓದಲು ಸುಲಭವಾಗಿ ಇರುತ್ತದೆ.
// Promise ಅನ್ನು ಬಳಸಿ ಡೇಟಾ ತರುವುದು (Fetch API)
fetch('https://api.example.com/user')
    .then(response => response.json()) // ಇದು ಯಶಸ್ವಿಯಾದರೆ (Fulfilled)
    .then(user => fetch(`https://api.example.com/posts/${user.id}`)) // Promise Chaining
    .then(response => response.json())
    .then(posts => {
        document.getElementById('post-container').innerHTML = posts[0].title; // DOM Update
    })
    .catch(error => {
        console.error("ಏನೋ ತಪ್ಪಾಗಿದೆ!", error); // ವಿಫಲವಾದರೆ (Rejected)
    });

3. ಅಸಿಂಕ್ / ಅವೇಟ್ (Async / Await) - [ಆಧುನಿಕ ಮತ್ತು ಅತ್ಯುತ್ತಮ ವಿಧಾನ]

ನಾನ್-ಐಟಿ (Non-IT) ಉದಾಹರಣೆ:
ನೀವು ವಾಷಿಂಗ್ ಮೆಷಿನ್‌ಗೆ ಬಟ್ಟೆ ಹಾಕುತ್ತೀರಿ. ಬಟ್ಟೆ ಒಗೆಯುವುದು ಮುಗಿಯುವವರೆಗೆ (Await) ನೀವು ಬೇರೆ ಬಟ್ಟೆಗಳನ್ನು ಮಡಚುವ ಕೆಲಸ ಮಾಡುತ್ತೀರಿ. ನೋಡಲು ನೀವು ಒಂದಾದ ಮೇಲೊಂದು ಕೆಲಸ (Synchronous) ಮಾಡುತ್ತಿರುವಂತೆ ಕಾಣುತ್ತದೆ, ಆದರೆ ವಾಷಿಂಗ್ ಮೆಷಿನ್ ತನ್ನ ಕೆಲಸವನ್ನು ಹಿನ್ನೆಲೆಯಲ್ಲಿ (Asynchronous) ಮಾಡುತ್ತಿರುತ್ತದೆ. ನಿಮ್ಮ ದಿನಚರಿ ಎಲ್ಲಿಯೂ ನಿಲ್ಲುವುದಿಲ್ಲ (Not blocked).

ಪ್ರೊಫೆಷನಲ್ (Professional) ವಿವರಣೆ:
ಇದು ES8 (2017) ನಲ್ಲಿ ಬಂದ ಹೊಸ ಫೀಚರ್. ಇದು ವಾಸ್ತವವಾಗಿ ಪ್ರಾಮಿಸ್‌ಗಳ (Promises) ಮೇಲೆಯೇ ಕೆಲಸ ಮಾಡುತ್ತದೆ, ಆದರೆ ಕೋಡ್ ಅನ್ನು ಬರೆಯುವ ಶೈಲಿ ಸಿಂಕ್ರೊನಸ್ ಕೋಡ್‌ನಂತೆಯೇ (Synchronous-looking) ಕಾಣುತ್ತದೆ.

  • ಯಾವುದೇ ಫಂಕ್ಷನ್ ಹಿಂದೆ async ಎಂದು ಬರೆದರೆ, ಆ ಫಂಕ್ಷನ್ ಯಾವಾಗಲೂ ಒಂದು Promise ಅನ್ನು ರಿಟರ್ನ್ ಮಾಡುತ್ತದೆ.
  • await ಕೀ-ವರ್ಡ್ ಅನ್ನು ಕೇವಲ async ಫಂಕ್ಷನ್ ಒಳಗೆ ಮಾತ್ರ ಬಳಸಬಹುದು. ಇದು ಪ್ರಾಮಿಸ್‌ನ ಫಲಿತಾಂಶ ಬರುವವರೆಗೂ ಆ ಸಾಲಿನಲ್ಲಿ ಎಕ್ಸಿಕ್ಯೂಷನ್ ಅನ್ನು (ಹಿನ್ನೆಲೆಯಲ್ಲಿ) ತಡೆಹಿಡಿಯುತ್ತದೆ.
  • ಎರರ್‌ಗಳನ್ನು ನಿಭಾಯಿಸಲು ಸಾಂಪ್ರದಾಯಿಕ try...catch ಬ್ಲಾಕ್ ಅನ್ನು ಬಳಸಲಾಗುತ್ತದೆ.
// ಅತ್ಯಂತ ಸುಲಭವಾದ ಮತ್ತು ಕ್ಲೀನ್ ಆದ Async/Await ವಿಧಾನ
async function displayUserContent() {
    try {
        // ಡೇಟಾ ಬರುವವರೆಗೆ ಹಿನ್ನೆಲೆಯಲ್ಲಿ ಕಾಯುತ್ತದೆ (await)
        const userResponse = await fetch('https://api.example.com/user');
        const user = await userResponse.json();

        const postResponse = await fetch(`https://api.example.com/posts/${user.id}`);
        const posts = await postResponse.json();

        // Browser DOM Architecture ಪ್ರಕಾರ ಎಲಿಮೆಂಟ್ ಸೆಲೆಕ್ಟ್ ಮಾಡಿ ಅಪ್‌ಡೇಟ್ ಮಾಡುವುದು
        const container = document.querySelector('#content');
        container.textContent = `User: ${user.name}, Latest Post: ${posts[0].title}`;
        
    } catch (error) {
        // ನೆಟ್‌ವರ್ಕ್ ಹೋದರೆ ಅಥವಾ API ಫೇಲ್ ಆದರೆ ಇಲ್ಲಿಗೆ ಬರುತ್ತದೆ
        console.error("ಡೇಟಾ ತರುವಲ್ಲಿ ವಿಫಲವಾಗಿದೆ:", error);
    }
}

displayUserContent();

ಡೆವಲಪರ್‌ಗಳಿಗಾಗಿ ಸಾರಾಂಶ:

  • ❌ Callbacks: ಹಳೆಯ ಕೋಡ್‌ಬೇಸ್‌ಗಳಲ್ಲಿ ಅಥವಾ ಈವೆಂಟ್ ಲಿಸ್ನರ್‌ಗಳಲ್ಲಿ (addEventListener) ಮಾತ್ರ ಬಳಸಲಾಗುತ್ತದೆ. Nested ಆದಾಗ ನಿರ್ವಹಣೆ ಕಷ್ಟ.
  • ⚠️ Promises: .then() ಚೈನಿಂಗ್ ಮೂಲಕ ಕಾಲ್‌ಬ್ಯಾಕ್ ಹೆಲ್ ಅನ್ನು ತಪ್ಪಿಸುತ್ತದೆ.
  • ✅ Async/Await: ಇಂದಿನ ಮಾಡರ್ನ್ ಫ್ರಂಟ್-ಎಂಡ್ ಡೆವಲಪ್‌ಮೆಂಟ್‌ನಲ್ಲಿ API ಕಾಲ್‌ಗಳನ್ನು ಮಾಡಲು (ಉದಾ: ಯೂಸರ್ ಲಾಗಿನ್, ಡೇಟಾ ಫೆಚ್) ಶೇ. 99% ಇದನ್ನೇ ಬಳಸಲಾಗುತ್ತದೆ. ಇದು ಕೋಡ್ ಅನ್ನು ಸ್ವಚ್ಛವಾಗಿ ಮತ್ತು ನಿರ್ವಹಿಸಲು ಸುಲಭವಾಗಿ ಇಡುತ್ತದೆ.

ವಿಸ್ತೃತ ತಾಂತ್ರಿಕ ಮಾಹಿತಿ (Extended Core Concepts)

ಉತ್ತಮ ಫ್ರಂಟ್-ಎಂಡ್ ಡೆವಲಪರ್ ಆಗಲು ಅತ್ಯಗತ್ಯವಾದ ಮೂಲಭೂತ ಕಾನ್ಸೆಪ್ಟ್‌ಗಳು

4. ಕೋರ್ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಮತ್ತು DOM ಮ್ಯಾನಿಪುಲೇಷನ್

DOM (Document Object Model) ಎಂದರೇನು?
ಬ್ರೌಸರ್ ನೀವು ಬರೆದ HTML ಅನ್ನು ಓದಿ, ಅದನ್ನು ಒಂದು 'ಟ್ರೀ ಸ್ಟ್ರಕ್ಚರ್' (Tree Structure) ಆಗಿ ಪರಿವರ್ತಿಸುತ್ತದೆ. ಈ ಟ್ರೀ ಯನ್ನು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಬಳಸಿ ಬದಲಾಯಿಸುವುದನ್ನೇ DOM Manipulation ಎನ್ನುತ್ತೇವೆ. ಇದರಿಂದ ವೆಬ್‌ಪೇಜ್ ರಿಫ್ರೆಶ್ ಆಗದೆಯೇ ಹೊಸ ವಿಷಯಗಳನ್ನು ಪರದೆಯ ಮೇಲೆ ತೋರಿಸಬಹುದು.

ಪ್ರಮುಖ Element Selection ವಿಧಾನಗಳು:

  • getElementById('id'): ಐಡಿ ಮೂಲಕ ನಿರ್ದಿಷ್ಟ ಎಲಿಮೆಂಟ್ ಹುಡುಕಲು (ಅತಿ ವೇಗವಾದ ವಿಧಾನ).
  • querySelector('.class'): CSS ಸೆಲೆಕ್ಟರ್ ಮೂಲಕ ಮೊದಲ ಎಲಿಮೆಂಟ್ ಅನ್ನು ಹುಡುಕಲು (ಅತ್ಯಂತ ಹೆಚ್ಚು ಬಳಕೆಯಾಗುವ ಆಧುನಿಕ ವಿಧಾನ).
  • querySelectorAll('div'): ಆ ಹೆಸರಿನ ಎಲ್ಲಾ ಎಲಿಮೆಂಟ್‌ಗಳನ್ನು ಒಂದು ಲಿಸ್ಟ್ (NodeList) ಆಗಿ ತರಲು.
// DOM ಮ್ಯಾನಿಪುಲೇಷನ್ ಉದಾಹರಣೆ
// 1. ಎಲಿಮೆಂಟ್ ಆಯ್ಕೆ ಮಾಡುವುದು
const titleElement = document.querySelector('.main-title');

// 2. ಅದರ ಒಳಗಿನ ಟೆಕ್ಸ್ಟ್ ಬದಲಾಯಿಸುವುದು
titleElement.textContent = "ಹೊಸ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಟೈಟಲ್";

// 3. ಸ್ಟೈಲ್ (CSS) ಬದಲಾಯಿಸುವುದು
titleElement.style.color = "blue";
titleElement.style.fontSize = "24px";

// 4. ಹೊಸ ಎಲಿಮೆಂಟ್ ಸೃಷ್ಟಿಸಿ ಸೇರಿಸುವುದು
const newDiv = document.createElement('div');
newDiv.innerHTML = "<strong>ಹೊಸ ಪ್ಯಾರಾಗ್ರಾಫ್!</strong>";
document.body.appendChild(newDiv);

5. ಈವೆಂಟ್ ಹ್ಯಾಂಡ್ಲಿಂಗ್ (Event Handling)

ಯೂಸರ್ ವೆಬ್‌ಪೇಜ್ ಜೊತೆ ಏನೇ ಇಂಟರಾಕ್ಟ್ ಮಾಡಿದರೂ (ಕ್ಲಿಕ್ ಮಾಡುವುದು, ಟೈಪ್ ಮಾಡುವುದು, ಸ್ಕ್ರಾಲ್ ಮಾಡುವುದು) ಅದು ಒಂದು "Event". ಅದನ್ನು ಹಿಡಿದು (Listen) ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಮೂಲಕ ಪ್ರತಿಕ್ರಿಯಿಸುವುದೇ 'Event Handling'. ಇದಕ್ಕಾಗಿ ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಯಾವಾಗಲೂ addEventListener ಅನ್ನು ಬಳಸಲಾಗುತ್ತದೆ.

// ಬಟನ್ ಕ್ಲಿಕ್ ಈವೆಂಟ್ ನಿರ್ವಹಣೆ
const submitBtn = document.querySelector('#submit-button');

// ಈವೆಂಟ್ ಲಿಸ್ನರ್ ಸೇರಿಸುವುದು (Event = 'click')
submitBtn.addEventListener('click', function(event) {
    // ಫಾರ್ಮ್ ರಿಫ್ರೆಶ್ ಆಗುವುದನ್ನು ತಡೆಯಲು
    event.preventDefault(); 
    
    console.log("ಬಟನ್ ಕ್ಲಿಕ್ ಮಾಡಲಾಗಿದೆ!");
    
    // ಕ್ಲಿಕ್ ಮಾಡಿದಾಗ ಬಟನ್ ಬಣ್ಣ ಬದಲಾಯಿಸಿ
    event.target.style.backgroundColor = 'green';
});

// ಕೀಬೋರ್ಡ್ ಈವೆಂಟ್ (ಉದಾಹರಣೆಗೆ Enter ಕೀ ಒತ್ತಿದಾಗ)
document.addEventListener('keydown', (e) => {
    if(e.key === 'Enter') {
        console.log("Enter ಕೀ ಒತ್ತಿದ್ದೀರಿ!");
    }
});

6. ಬ್ರೌಸರ್ ಸ್ಟೋರೇಜ್ (Client-Side Storage API)

ಬಳಕೆದಾರನ ಬ್ರೌಸರ್‌ನಲ್ಲೇ ಕೆಲವು ಡೇಟಾವನ್ನು ಸೇವ್ ಮಾಡಿಡಲು HTML5 ಸ್ಟೋರೇಜ್ API ಗಳನ್ನು ಬಳಸುತ್ತೇವೆ. ಇದು ಲಾಗಿನ್ ಟೋಕನ್, ಡಾರ್ಕ್ ಥೀಮ್ ಸೆಟ್ಟಿಂಗ್ ಅಥವಾ ಕಾರ್ಟ್ ಡೇಟಾವನ್ನು ಉಳಿಸಲು ಬಹಳ ಮುಖ್ಯ.

Local Storage

ಬ್ರೌಸರ್ ಕ್ಲೋಸ್ ಮಾಡಿದರೂ ಡೇಟಾ ಅಳಿಸಿಹೋಗುವುದಿಲ್ಲ. ಶಾಶ್ವತವಾಗಿ (ನೀವಾಗಿ ಡಿಲೀಟ್ ಮಾಡುವವರೆಗೆ) ಇರುತ್ತದೆ.

Session Storage

ಟ್ಯಾಬ್ ಅಥವಾ ಬ್ರೌಸರ್ ಕ್ಲೋಸ್ ಮಾಡಿದ ತಕ್ಷಣ ಡೇಟಾ ಅಳಿಸಿಹೋಗುತ್ತದೆ. ತಾತ್ಕಾಲಿಕ ಬಳಕೆಗೆ ಮಾತ್ರ.

// LocalStorage ಬಳಕೆಯ ಉದಾಹರಣೆ: ಡೇಟಾ ಸೇವ್ ಮಾಡುವುದು
// ಗಮನಿಸಿ: ಸ್ಟೋರೇಜ್ ಯಾವಾಗಲೂ ಸ್ಟ್ರಿಂಗ್ (String) ರೂಪದಲ್ಲಿ ಮಾತ್ರ ಡೇಟಾ ಸ್ವೀಕರಿಸುತ್ತದೆ.

const userProfile = { name: "ರಮೇಶ್", role: "Developer" };

// Object ಅನ್ನು String ಆಗಿ ಕನ್ವರ್ಟ್ ಮಾಡಿ ಸೇವ್ ಮಾಡುವುದು
localStorage.setItem('user', JSON.stringify(userProfile));

// ಡೇಟಾ ವಾಪಸ್ ಪಡೆಯುವುದು
const savedData = localStorage.getItem('user');

if (savedData) {
    // String ಅನ್ನು ಮತ್ತೆ Object ಆಗಿ ಕನ್ವರ್ಟ್ ಮಾಡುವುದು
    const parsedUser = JSON.parse(savedData);
    console.log("Welcome back, " + parsedUser.name);
}

// ನಿರ್ದಿಷ್ಟ ಡೇಟಾ ಅಳಿಸಲು
localStorage.removeItem('user');

// ಸಂಪೂರ್ಣ ಸ್ಟೋರೇಜ್ ಖಾಲಿ ಮಾಡಲು
localStorage.clear();
