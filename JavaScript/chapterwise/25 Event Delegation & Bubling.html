Event Delegation ಎನ್ನುವುದು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ಅತ್ಯಂತ ಶಕ್ತಿಶಾಲಿ ಮತ್ತು ವೃತ್ತಿಪರ ತಂತ್ರಗಳಲ್ಲಿ ಒಂದಾಗಿದೆ. ಸರಳವಾಗಿ ಹೇಳಬೇಕೆಂದರೆ: ಪ್ರತಿಯೊಂದು ಮಗುವಿಗೂ (Child) ಪ್ರತ್ಯೇಕವಾಗಿ ಕಾವಲುಗಾರನನ್ನು (Event Listener) ನೇಮಿಸುವ ಬದಲು, ತಂದೆಗೆ (Parent) ಕೇವಲ ಒಬ್ಬ ಕಾವಲುಗಾರನನ್ನು ನೇಮಿಸುವುದು.

ಇದನ್ನು Beginner ಹಂತದಿಂದ Pro ಹಂತದವರೆಗೆ ಯಾವುದೇ ಮಾಹಿತಿಯನ್ನು ಬಿಡದೆ ಇಲ್ಲಿ ಆಳವಾಗಿ ವಿವರಿಸಲಾಗಿದೆ.

Event Delegation

The Master Technique for Performance

Level 1 Delegation ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ?

ಇದು Event Bubbling ಎಂಬ ನಿಯಮದ ಮೇಲೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ. ನೀವು ಒಂದು ಮಗುವಿನ ಮೇಲೆ ಕ್ಲಿಕ್ ಮಾಡಿದರೆ, ಆ ಕ್ಲಿಕ್ ಅದರ ಪೇರೆಂಟ್‌ಗೂ ತಲುಪುತ್ತದೆ. ಆದ್ದರಿಂದ, ನಾವು ಮಗುವಿಗೆ ಈವೆಂಟ್ ಹಾಕುವ ಬದಲು ಪೇರೆಂಟ್‌ಗೆ ಈವೆಂಟ್ ಹಾಕುತ್ತೇವೆ. ಆಮೇಲೆ e.target ಬಳಸಿ ಯಾವ ಮಗುವಿನ ಮೇಲೆ ಕ್ಲಿಕ್ ಆಗಿದೆ ಎಂದು ಪತ್ತೆ ಹಚ್ಚುತ್ತೇವೆ.

ರಿಯಲ್-ಲೈಫ್ ಉದಾಹರಣೆ:

ನಿಮ್ಮ ಕ್ಲಾಸ್‌ನಲ್ಲಿ 50 ವಿದ್ಯಾರ್ಥಿಗಳಿದ್ದಾರೆ (Children). ಟೀಚರ್ (ಬ್ರೌಸರ್) 50 ವಿದ್ಯಾರ್ಥಿಗಳ ಬಳಿ ಹೋಗಿ "ಹೋಂವರ್ಕ್ ಮಾಡಿದ್ದೀಯಾ?" ಎಂದು ಕೇಳುವುದು ಕಷ್ಟ (ಇದೇ 50 Event Listeners ಹಾಕುವುದು). ಬದಲಾಗಿ, ಕ್ಲಾಸ್ ಲೀಡರ್‌ಗೆ (Parent) "ಯಾರು ಹೋಂವರ್ಕ್ ಮಾಡಿಲ್ಲ ಹೇಳು" ಎಂದು ಹೇಳುವುದು ಜಾಣತನ. ಇದೇ Event Delegation!

Level 2 ಲಿಸ್ಟ್‌ಗಳಲ್ಲಿ (Lists) Event Delegation

ನಿಮ್ಮ ಬಳಿ ಒಂದು <ul> ಇದೆ ಮತ್ತು ಅದರೊಳಗೆ 1000 <li> ಐಟಂಗಳಿವೆ ಎಂದು ಕಲ್ಪಿಸಿಕೊಳ್ಳಿ.

A. ತಪ್ಪು ವಿಧಾನ (The Bad Way)

ಹೊಸ ಡೆವಲಪರ್‌ಗಳು ಲೂಪ್ (Loop) ಬಳಸಿ ಪ್ರತಿ <li> ಗೂ ಈವೆಂಟ್ ಹಾಕುತ್ತಾರೆ.

const allItems = document.querySelectorAll("li");

// 1000 ಬಾರಿ ಲೂಪ್ ಆಗಿ, ಬ್ರೌಸರ್‌ನ ಮೆಮೊರಿಯಲ್ಲಿ 1000 Listeners!
allItems.forEach(item => {
  item.addEventListener("click", function() {
    console.log("ಕ್ಲಿಕ್ ಮಾಡಿದ್ದು: " + this.textContent);
  });
});
  • ಸಮಸ್ಯೆ 1: RAM (ಮೆಮೊರಿ) ವಿಪರೀತ ತಿನ್ನುತ್ತದೆ. ವೆಬ್‌ಸೈಟ್ ಸ್ಲೋ.
  • ಸಮಸ್ಯೆ 2: ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಮೂಲಕ ಹೊಸದಾಗಿ 1001ನೇ <li> ಸೇರಿಸಿದರೆ, ಅದಕ್ಕೆ ಈ ಈವೆಂಟ್ ಕೆಲಸ ಮಾಡುವುದಿಲ್ಲ!

B. ಸರಿಯಾದ ವಿಧಾನ (The Event Delegation Way)

ನಾವು ಕೇವಲ ಒಂದೇ ಒಂದು ಈವೆಂಟ್ ಅನ್ನು <ul> (Parent) ಗೆ ಹಾಕುತ್ತೇವೆ.

const list = document.getElementById("myList");

// ಕೇವಲ 1 ಕಾವಲುಗಾರ (Event Listener)
list.addEventListener("click", function(e) {
  // ನಮಗೆ ಬೇಕಾದ <li> ಮೇಲೆ ಮಾತ್ರ ಕ್ಲಿಕ್ ಆಗಿದೆಯೇ ಎಂದು ಚೆಕ್ ಮಾಡುವುದು
  if (e.target.tagName === "LI") {
    console.log("ಕ್ಲಿಕ್: " + e.target.textContent);
    e.target.style.color = "red";
  }
});
  • ಲಾಭ: 1000 ಲಿಸನರ್‌ಗಳ ಬದಲು ಕೇವಲ 1 ಲಿಸನರ್. ಮೆಮೊರಿ ಉಳಿತಾಯ.
  • ಮ್ಯಾಜಿಕ್: ನಾಳೆ ಹೊಸದಾಗಿ ಎಷ್ಟೇ <li> ಸೇರಿಸಿದರೂ, ಅವೆಲ್ಲದಕ್ಕೂ ಇದು ತಾನಾಗಿಯೇ ಕೆಲಸ ಮಾಡುತ್ತದೆ!

Level 3 ಫಾರ್ಮ್‌ಗಳಲ್ಲಿ (Forms) Event Delegation

ಒಂದು ಫಾರ್ಮ್‌ನಲ್ಲಿ 15 ಇನ್‌ಪುಟ್‌ಗಳಿದ್ದರೆ, ಬಳಕೆದಾರರು ಟೈಪ್ ಮಾಡುವಾಗ ಲೈವ್ ಆಗಿ ವ್ಯಾಲಿಡೇಶನ್ ಮಾಡಲು 15 ಬೇರೆ ಬೇರೆ input ಲಿಸನರ್ ಹಾಕುವ ಅಗತ್ಯವಿಲ್ಲ. ಇಡೀ ಫಾರ್ಮ್‌ಗೆ ಒಂದೇ ಲಿಸನರ್ ಸಾಕು. 

<form id="registrationForm">
  <input type="text" name="username">
  <input type="email" name="email">
  <input type="password" name="password">
</form>
const form = document.getElementById("registrationForm");

// ಇಡೀ ಫಾರ್ಮ್‌ಗೆ ಒಂದೇ 'input' ಈವೆಂಟ್
form.addEventListener("input", function(e) {
  const fieldName = e.target.name;
  const fieldValue = e.target.value;
  
  // ಪಾಸ್‌ವರ್ಡ್ ಬಾಕ್ಸ್ ಆಗಿದ್ದರೆ ಮಾತ್ರ ಒಂದು ಷರತ್ತು (Condition)
  if (fieldName === "password") {
    if (fieldValue.length < 8) {
      e.target.style.border = "2px solid red";
    } else {
      e.target.style.border = "2px solid green";
    }
  }
});

Pro Level closest() ಜೊತೆ Delegation (The Master Trick)

ರಿಯಲ್-ವರ್ಲ್ಡ್ ಪ್ರಾಜೆಕ್ಟ್‌ಗಳಲ್ಲಿ, ಬಟನ್‌ಗಳ ಒಳಗೆ ಐಕಾನ್‌ಗಳು (<i> ಅಥವಾ <svg>) ಮತ್ತು ಟೆಕ್ಸ್ಟ್ (<span>) ಇರುತ್ತವೆ.

<div id="buttonContainer">
  <button class="delete-btn">
    <i class="icon-trash"></i> ಡಿಲೀಟ್ ಮಾಡಿ
  </button>
</div>
ನೀವು ಐಕಾನ್ ಮೇಲೆ ಕ್ಲಿಕ್ ಮಾಡಿದರೆ, e.target ಐಕಾನ್ (<i>) ಆಗಿರುತ್ತದೆಯೇ ಹೊರತು ಬಟನ್ ಆಗಿರುವುದಿಲ್ಲ! ಆಗ e.target.tagName === "BUTTON" ಎಂಬ ಷರತ್ತು ಫೇಲ್ ಆಗುತ್ತದೆ. ಇದನ್ನು ಸರಿಪಡಿಸಲು Pro ಡೆವಲಪರ್‌ಗಳು e.target.closest() ಬಳಸುತ್ತಾರೆ. 
document.getElementById("buttonContainer")
  .addEventListener("click", function(e) {
  
  // ಕ್ಲಿಕ್ ಆದ ಜಾಗದಿಂದ ಮೇಲ್ಮುಖವಾಗಿ ಹುಡುಕಿ ಬಟನ್ ಹಿಡಿಯುವುದು
  const clickedBtn = e.target.closest(".delete-btn");

  // ಕ್ಲಿಕ್ ಆಗಿದ್ದು ಬಟನ್ ಅಥವಾ ಅದರೊಳಗಿನ ಐಕಾನ್ ಮೇಲೆ ಆಗಿದ್ದರೆ:
  if (clickedBtn) {
    console.log("ಡಿಲೀಟ್ ಬಟನ್ ಕ್ಲಿಕ್ ಆಯ್ತು!");
    // clickedBtn.parentElement.remove()
  }
});

Event Delegation ಪರ್ಫಾರ್ಮೆನ್ಸ್ ಸಾರಾಂಶ

  • 1. ಕಡಿಮೆ ಮೆಮೊರಿ (RAM): 100 ಲಿಸನರ್‌ಗಳು 100 ಆಬ್ಜೆಕ್ಟ್‌ಗಳನ್ನು ಸೃಷ್ಟಿಸುತ್ತವೆ. ಡೆಲಿಗೇಶನ್ ಕೇವಲ 1 ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸುತ್ತದೆ.
  • 2. ಡೈನಾಮಿಕ್ ಎಲಿಮೆಂಟ್ಸ್: AJAX/Fetch ಮೂಲಕ ಸರ್ವರ್‌ನಿಂದ ತಂದು ಹೊಸದಾಗಿ ಸೇರಿಸಿದ ಎಲಿಮೆಂಟ್‌ಗಳಿಗೆ ಮತ್ತೆ ಲಿಸನರ್ ಅಟ್ಯಾಚ್ ಮಾಡುವ ತಲೆನೋವಿಲ್ಲ.
  • 3. Memory Leaks ಇರಲ್ಲ: 100 ಎಲಿಮೆಂಟ್‌ ಡಿಲೀಟ್ ಮಾಡುವಾಗ 100 ಲಿಸನರ್ ರಿಮೂವ್ ಮಾಡುವ ಭಯ ಇರುವುದಿಲ್ಲ.

Event Bubbling

The DOM Event Flow

ನೀರಿನಲ್ಲಿ ಗಾಳಿಯ ಗುಳ್ಳೆ (Bubble) ಕೆಳಗಿನಿಂದ ಹೇಗೆ ಮೇಲಕ್ಕೆ ಬರುತ್ತದೆಯೋ, ಅದೇ ರೀತಿ DOM ನಲ್ಲಿ ಒಂದು ಎಲಿಮೆಂಟ್ ಮೇಲೆ ನಡೆದ ಈವೆಂಟ್, ಅದರ ಮೇಲಿರುವ ತಂದೆ (Parent), ಅಜ್ಜ (Grandparent) ಮತ್ತು ಕೊನೆಗೆ document ವರೆಗೂ ಮೇಲ್ಮುಖವಾಗಿ ಹರಡುತ್ತದೆ. ಇದನ್ನೇ Event Bubbling ಎನ್ನುತ್ತಾರೆ.

Level 1 Bubbling ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ?

ನೀವು ಬಟನ್ ಮೇಲೆ ಕ್ಲಿಕ್ ಮಾಡಿದರೆ, ನೀವು ಕೇವಲ ಬಟನ್ ಮಾತ್ರ ಕ್ಲಿಕ್ ಮಾಡಿದಂತಲ್ಲ, ಪರೋಕ್ಷವಾಗಿ ಆ div ಅನ್ನು ಮತ್ತು ಇಡೀ body ಅನ್ನು ಕೂಡ ಕ್ಲಿಕ್ ಮಾಡಿದಂತೆಯೇ ಲೆಕ್ಕ! 

document.getElementById("grandparent")
  .addEventListener("click", () => console.log("Grandparent ಕ್ಲಿಕ್!"));
  
document.getElementById("parent")
  .addEventListener("click", () => console.log("Parent ಕ್ಲಿಕ್!"));
  
document.getElementById("child")
  .addEventListener("click", () => console.log("Child (ಬಟನ್) ಕ್ಲಿಕ್!"));

Child Button ಕ್ಲಿಕ್ ಮಾಡಿದರೆ ಔಟ್‌ಪುಟ್:

  1. Child (ಬಟನ್) ಕ್ಲಿಕ್ ಆಯ್ತು!
  2. Parent ಕ್ಲಿಕ್ ಆಯ್ತು!
  3. Grandparent ಕ್ಲಿಕ್ ಆಯ್ತು!

Level 2 ಬಬ್ಲಿಂಗ್ ಅನ್ನು ನಿಲ್ಲಿಸುವುದು (stopPropagation)

ಕೆಲವೊಮ್ಮೆ ನಮಗೆ ಈ ಬಬ್ಲಿಂಗ್ ಬೇಡವಾಗಿರುತ್ತದೆ. "ನಾನು ಕೇವಲ ಬಟನ್ ಕ್ಲಿಕ್ ಮಾಡಿದ್ದೇನೆ, ಆ ಕ್ಲಿಕ್ ಅದರ ಪೇರೆಂಟ್‌ಗೆ ಹೋಗಬಾರದು" ಎಂದು ನೀವು ಅಂದುಕೊಂಡರೆ, ಆ ಗುಳ್ಳೆಯನ್ನು ಒಡೆಯಬಹುದು! ಅದಕ್ಕಾಗಿ e.stopPropagation() ಬಳಸಲಾಗುತ್ತದೆ. 

document.getElementById("child").addEventListener("click", (e) => {
  e.stopPropagation(); // ಇಲ್ಲಿಗೆ ಬಬ್ಲಿಂಗ್ (ಗುಳ್ಳೆ) ಒಡೆದುಹೋಗುತ್ತದೆ!
  console.log("Child ಮಾತ್ರ ಕ್ಲಿಕ್ ಆಯ್ತು. ಈವೆಂಟ್ ಮೇಲಕ್ಕೆ ಹೋಗಲಿಲ್ಲ.");
});

document.getElementById("parent").addEventListener("click", () => {
  console.log("Parent ಕ್ಲಿಕ್ ಆಯ್ತು!"); // ಇದು ರನ್ ಆಗುವುದಿಲ್ಲ!
});

Level 3 e.target vs e.currentTarget

e.target

ಈವೆಂಟ್ ಎಲ್ಲಿಂದ ಹುಟ್ಟಿತು? (ನಿಖರವಾಗಿ ನೀವು ಎಲ್ಲಿ ಕ್ಲಿಕ್ ಮಾಡಿದಿರಿ). ಇದು ಮೇಲಕ್ಕೆ ಹೋದರೂ ಬದಲಾಗುವುದಿಲ್ಲ.

e.currentTarget

ಪ್ರಸ್ತುತ ಈವೆಂಟ್ ಲಿಸನರ್ ಎಲ್ಲಿ ರನ್ ಆಗುತ್ತಿದೆ? (ಯಾವ ಪೇರೆಂಟ್ ಮೂಲಕ ಗುಳ್ಳೆ ಹಾದುಹೋಗುತ್ತಿದೆ).

document.getElementById("grandparent").addEventListener("click", function(e) {
  console.log("ಗುಳ್ಳೆ ಹುಟ್ಟಿದ ಜಾಗ (target):", e.target.id); // 'child' 
  console.log("ಈಗ ಗುಳ್ಳೆ ಇರುವ ಜಾಗ (currentTarget):", e.currentTarget.id); // 'grandparent' 
});

Expert Level Event Capturing (Trickling)

ಮ್ಯಾಜಿಕ್ ಎಂದರೆ, ನೀವು ಕ್ಲಿಕ್ ಮಾಡಿದಾಗ ಕೇವಲ Event Bubbling (ಕೆಳಗಿನಿಂದ ಮೇಲೆ) ಮಾತ್ರ ನಡೆಯುವುದಿಲ್ಲ. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಈವೆಂಟ್ ಫ್ಲೋ (Event Flow) 3 ಹಂತಗಳಲ್ಲಿ ನಡೆಯುತ್ತದೆ:

  • 1
    Capturing Phase (ಮೇಲಿನಿಂದ ಕೆಳಗೆ): ವಿಂಡೋದಿಂದ ಶುರುವಾಗಿ ನೀವು ಕ್ಲಿಕ್ ಮಾಡಿದ ಎಲಿಮೆಂಟ್ ಅನ್ನು ಹುಡುಕಿಕೊಂಡು ಕೆಳಗೆ ಬರುತ್ತದೆ.
  • 2
    Target Phase: ನೀವು ಕ್ಲಿಕ್ ಮಾಡಿದ ಎಲಿಮೆಂಟ್ ಅನ್ನು ತಲುಪುತ್ತದೆ.
  • 3
    Bubbling Phase (ಕೆಳಗಿನಿಂದ ಮೇಲೆ): ಮತ್ತೆ ವಾಪಸ್ ಮೇಲಕ್ಕೆ ಹೋಗುತ್ತದೆ (Default).

ಡೀಫಾಲ್ಟ್ ಆಗಿ addEventListener 3ನೇ ಹಂತದಲ್ಲಿ ಮಾತ್ರ ಕೆಲಸ ಮಾಡುತ್ತದೆ. ಆದರೆ ವೃತ್ತಿಪರರಾಗಿ, 1ನೇ ಹಂತದಲ್ಲೇ ಈವೆಂಟ್ ಹಿಡಿಯಲು 3ನೇ ಪ್ಯಾರಾಮೀಟರ್ ಆಗಿ { capture: true } ಬಳಸಬೇಕು. 

// ಇದು ಮೇಲಿನಿಂದ ಕೆಳಗೆ ಬರುವಾಗಲೇ ರನ್ ಆಗುತ್ತದೆ!
document.getElementById("grandparent")
  .addEventListener("click", () => {
  console.log("Grandparent - Capturing Phase");
}, { capture: true });

document.getElementById("child")
  .addEventListener("click", () => {
  console.log("Child - Target Phase");
});
Golden Pro Rule:

ನೀವು Event Delegation ಬಳಸುತ್ತಿದ್ದರೆ, ದಯವಿಟ್ಟು e.stopPropagation() ಬಳಸಬೇಡಿ. ಏಕೆಂದರೆ ನೀವು ಮಗುವಿನ ಬಳಿ ಗುಳ್ಳೆ ಒಡೆದರೆ, ಅದು ಪೇರೆಂಟ್ ಬಳಿ ಇರುವ Delegation ಲಿಸನರ್‌ಗೆ ತಲುಪುವುದೇ ಇಲ್ಲ, ಮತ್ತು ನಿಮ್ಮ ಕೋಡ್ ಕೆಲಸ ಮಾಡುವುದಿಲ್ಲ!
