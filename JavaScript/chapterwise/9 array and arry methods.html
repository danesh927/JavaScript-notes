ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ (JavaScript) ನಲ್ಲಿ Array ಮತ್ತು ಅದರ Methods

ಸಂಪೂರ್ಣ ಮಾರ್ಗದರ್ಶಿ (Beginner to Pro)

Array ಎಂದರೆ ಒಂದೇ ವೇರಿಯೇಬಲ್ (variable) ನಲ್ಲಿ ಒಂದಕ್ಕಿಂತ ಹೆಚ್ಚು ಮೌಲ್ಯಗಳನ್ನು (values) ಸಂಗ್ರಹಿಸಲು ಬಳಸುವ ಒಂದು ವ್ಯವಸ್ಥೆ. ಉದಾಹರಣೆಗೆ, ನಿಮ್ಮ ಬಳಿ 5 ಹಣ್ಣುಗಳ ಹೆಸರುಗಳಿದ್ದರೆ, 5 ಬೇರೆ ಬೇರೆ ವೇರಿಯೇಬಲ್ ಮಾಡುವ ಬದಲು ಒಂದೇ Array ನಲ್ಲಿ ಹಾಕಬಹುದು.

Array ತಯಾರಿಸುವ 5 ವಿಭಿನ್ನ ವಿಧಾನಗಳು

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ (JavaScript) Array ಯನ್ನು ತಯಾರಿಸಲು (Create/Initialize ಮಾಡಲು) ಪ್ರಮುಖವಾಗಿ 5 ವಿಭಿನ್ನ ವಿಧಾನಗಳಿವೆ. ಪ್ರತಿಯೊಂದನ್ನು ಉದಾಹರಣೆಯೊಂದಿಗೆ ಕೆಳಗೆ ವಿವರಿಸಲಾಗಿದೆ:

1. Array Literal ವಿಧಾನ (ಅತ್ಯಂತ ಸಾಮಾನ್ಯ ಮತ್ತು ಸುಲಭ)

ಬ್ರಾಕೆಟ್ [] ಅನ್ನು ಬಳಸಿ Array ರಚಿಸುವುದು ಅತ್ಯಂತ ವೇಗವಾದ ಮತ್ತು ಪ್ರೋಗ್ರಾಮರ್‌ಗಳು ಅತಿ ಹೆಚ್ಚು ಬಳಸುವ ವಿಧಾನವಾಗಿದೆ.

JS
// ಡೇಟಾ ಜೊತೆಗೆ Array ರಚನೆ
let fruits = ["Apple", "Banana", "Mango"];

// ಒಂದು ಖಾಲಿ (Empty) Array ರಚನೆ
let emptyArray = []; 
emptyArray[0] = "Orange"; // ನಂತರ ಡೇಟಾ ಸೇರಿಸಬಹುದು

2. new Array() Constructor ಬಳಸಿ

new ಎಂಬ ಕೀವರ್ಡ್ (keyword) ಬಳಸಿ ಕೂಡ Array ತಯಾರಿಸಬಹುದು. ಆದರೆ ಇದನ್ನು ಬಳಸುವುದು ಸ್ವಲ್ಪ ಕಡಿಮೆ.

JS
let cars = new Array("Volvo", "BMW", "Audi");
console.log(cars); // ಫಲಿತಾಂಶ: ["Volvo", "BMW", "Audi"]
ಎಚ್ಚರಿಕೆ (Catch): ಈ ವಿಧಾನದಲ್ಲಿ ಕೇವಲ ಒಂದೇ ಒಂದು ಸಂಖ್ಯೆಯನ್ನು (Number) ಹಾಕಿದರೆ ಅದು ಆ ಸಂಖ್ಯೆಯ Array ಆಗುವುದಿಲ್ಲ, ಬದಲಿಗೆ ಅಷ್ಟು ಉದ್ದದ (Length) ಖಾಲಿ Array ಸೃಷ್ಟಿಯಾಗುತ್ತದೆ!
JS
let numbers = new Array(5); 
// ಇದು [5] ಎಂಬ Array ಅಲ್ಲ! ಬದಲಿಗೆ 5 ಖಾಲಿ ಜಾಗಗಳಿರುವ (empty slots) Array.
console.log(numbers.length); // 5
console.log(numbers); // [empty × 5]

3. Array.of() (ES6 ಹೊಸ ವಿಧಾನ)

ಮೇಲೆ new Array() ನಲ್ಲಿ ಆಗುವ ತೊಂದರೆಯನ್ನು (ಒಂದೇ ಸಂಖ್ಯೆ ಕೊಟ್ಟಾಗ ಆಗುವ ಗೊಂದಲ) ನಿವಾರಿಸಲು Array.of() ಅನ್ನು ಪರಿಚಯಿಸಲಾಯಿತು. ನೀವು ಏನೇ ಕೊಟ್ಟರೂ ಅದು ಅದನ್ನು ನಿಖರವಾಗಿ Array ಒಳಗೆ ಹಾಕುತ್ತದೆ.

JS
let num = Array.of(5);
console.log(num); // ಫಲಿತಾಂಶ: [5] (ಇದು 5 ಖಾಲಿ ಜಾಗಗಳನ್ನಲ್ಲ, ಬದಲಿಗೆ 5 ಎಂಬ ಸಂಖ್ಯೆಯನ್ನೇ ಸೇರಿಸಿದೆ)

let mix = Array.of(1, "Two", 3);
console.log(mix); // ಫಲಿತಾಂಶ: [1, "Two", 3]

4. Array.from() (ಇತರ ಡೇಟಾದಿಂದ Array ಆಗಿ ಪರಿವರ್ತಿಸಲು)

ಒಂದು String (ಪದ) ಅಥವಾ Array ತರಹ ಕಾಣುವ ಬೇರೆ ಡೇಟಾವನ್ನು ನಿಜವಾದ Array ಆಗಿ ಪರಿವರ್ತಿಸಲು (Convert) ಇದನ್ನು ಬಳಸಲಾಗುತ್ತದೆ.

JS
// ಒಂದು ಪದವನ್ನು ಅಕ್ಷರಶಃ ಒಡೆದು Array ಮಾಡುವುದು
let name = "RAVI";
let nameArray = Array.from(name);

console.log(nameArray); 
// ಫಲಿತಾಂಶ: ["R", "A", "V", "I"]

5. Spread Operator [...] ಬಳಸಿ (ನಕಲು ಮಾಡಲು ಅಥವಾ ಸೇರಿಸಲು)

ಈಗಾಗಲೇ ಇರುವ ಒಂದು Array ಯ ಡೇಟಾವನ್ನು ತೆಗೆದುಕೊಂಡು ಹೊಸ Array ಯನ್ನು ತಯಾರಿಸಲು ಈ ... (Spread operator) ಅನ್ನು ಬಳಸಲಾಗುತ್ತದೆ.

JS
let arr1 = [1, 2, 3];

// arr1 ನಲ್ಲಿರುವ ಡೇಟಾವನ್ನು ಹೊಸ Array ಗೆ ಹಾಕಿ, ಜೊತೆಗೆ ಹೊಸ ಸಂಖ್ಯೆ ಸೇರಿಸುವುದು
let newArr = [...arr1, 4, 5]; 

console.log(newArr); 
// ಫಲಿತಾಂಶ: [1, 2, 3, 4, 5]
ಯಾವುದು ಬೆಸ್ಟ್?
ಹೆಚ್ಚಿನ ಎಲ್ಲಾ ಪ್ರಾಜೆಕ್ಟ್‌ಗಳಲ್ಲಿ, ಸರಳತೆ ಮತ್ತು ವೇಗದ ಕಾರಣಕ್ಕಾಗಿ 1ನೇ ವಿಧಾನವಾದ [] (Array Literal) ಅನ್ನೇ 99% ರಷ್ಟು ಬಳಸಲಾಗುತ್ತದೆ. ಉಳಿದವುಗಳನ್ನು ನಿರ್ದಿಷ್ಟ ಸಂದರ್ಭಗಳಲ್ಲಿ (ಉದಾಹರಣೆಗೆ String ಇಂದ Array ಮಾಡಬೇಕಾದಾಗ Array.from()) ಬಳಸಲಾಗುತ್ತದೆ.

1. Beginner Level (ಮೂಲ ಹಂತ)

ಈ ಹಂತದಲ್ಲಿ Array ಒಳಗೆ ಹೊಸ ಡೇಟಾ ಸೇರಿಸುವುದು ಮತ್ತು ತೆಗೆಯುವುದನ್ನು ಕಲಿಯೋಣ.

push()
ಕೊನೆಯಲ್ಲಿ ಸೇರಿಸಲು: Array ನ ಕೊನೆಯ ಭಾಗಕ್ಕೆ ಹೊಸ item ಸೇರಿಸುತ್ತದೆ.
JS
let arr = [1, 2, 3];
arr.push(4); 
// ಫಲಿತಾಂಶ: [1, 2, 3, 4]
pop()
ಕೊನೆಯಿಂದ ತೆಗೆಯಲು: Array ನ ಕೊನೆಯ item ಅನ್ನು ಅಳಿಸಿಹಾಕುತ್ತದೆ.
JS
let arr = [1, 2, 3];
arr.pop(); 
// ಫಲಿತಾಂಶ: [1, 2]
unshift()
ಆರಂಭದಲ್ಲಿ ಸೇರಿಸಲು: Array ನ ಮೊದಲಿಗೆ ಹೊಸ item ಸೇರಿಸುತ್ತದೆ.
JS
let arr = [1, 2, 3];
arr.unshift(0); 
// ಫಲಿತಾಂಶ: [0, 1, 2, 3]
shift()
ಆರಂಭದಿಂದ ತೆಗೆಯಲು: Array ನ ಮೊದಲ item ಅನ್ನು ಅಳಿಸಿಹಾಕುತ್ತದೆ.
JS
let arr = [1, 2, 3];
arr.shift(); 
// ಫಲಿತಾಂಶ: [2, 3]

2. Intermediate Level (ಮಧ್ಯಮ ಹಂತ)

Array ಅನ್ನು ಹುಡುಕುವುದು, ಕತ್ತರಿಸುವುದು ಮತ್ತು ಜೋಡಿಸುವುದು.

slice()
ಒಂದು ಭಾಗವನ್ನು ಮಾತ್ರ ಪಡೆಯಲು: ಇದು ಮೂಲ Array ಅನ್ನು ಬದಲಾಯಿಸದೆ, ನಮಗೆ ಬೇಕಾದ ಭಾಗವನ್ನು ಮಾತ್ರ ಕತ್ತರಿಸಿ ಹೊಸ Array ಕೊಡುತ್ತದೆ.
JS
let numbers = [10, 20, 30, 40, 50];
let sliced = numbers.slice(1, 4); // index 1 ರಿಂದ 3 ರವರೆಗೆ
// ಫಲಿತಾಂಶ: [20, 30, 40]
splice()
ಮಧ್ಯದಲ್ಲಿ ಸೇರಿಸಲು/ತೆಗೆಯಲು: ಮೂಲ Array ಅನ್ನು ಬದಲಾಯಿಸುತ್ತದೆ. ನಿರ್ದಿಷ್ಟ ಜಾಗದಿಂದ ಡೇಟಾ ತೆಗೆಯಲು ಅಥವಾ ಸೇರಿಸಲು ಬಳಸಬಹುದು.
JS
let colors = ["Red", "Green", "Blue"];
colors.splice(1, 1, "Yellow"); 
// ಫಲಿತಾಂಶ: ["Red", "Yellow", "Blue"]
concat()
ಸೇರಿಸುವುದು (Merge): ಎರಡು ಅಥವಾ ಹೆಚ್ಚಿನ Array ಗಳನ್ನು ಸೇರಿಸಿ ಒಂದೇ Array ಮಾಡುತ್ತದೆ.
JS
let a = [1, 2];
let b = [3, 4];
let c = a.concat(b); 
// ಫಲಿತಾಂಶ: [1, 2, 3, 4]
includes()
ಹುಡುಕಾಟ (True/False): ಒಂದು ನಿರ್ದಿಷ್ಟ value Array ನಲ್ಲಿ ಇದೆಯೋ ಇಲ್ಲವೋ ಎಂದು ತಿಳಿಸುತ್ತದೆ.
JS
let pets = ["Dog", "Cat", "Fish"];
console.log(pets.includes("Cat")); // true
indexOf()
ಎಷ್ಟನೇ ಸ್ಥಾನ (Index position): ಒಂದು item Array ನಲ್ಲಿ ಎಷ್ಟನೇ ಸ್ಥಾನದಲ್ಲಿದೆ ಎಂದು ಹುಡುಕುತ್ತದೆ.
JS
let pets = ["Dog", "Cat", "Fish"];
console.log(pets.indexOf("Fish")); // 2
join()
Array ಇಂದ String ಗೆ ಬದಲಾಯಿಸಲು: Array ನಲ್ಲಿರುವ ಎಲ್ಲಾ ಪದಗಳನ್ನು/ಬೆಲೆಗಳನ್ನು ಸೇರಿಸಿ ಒಂದೇ ವಾಕ್ಯ (String) ಮಾಡುತ್ತದೆ.
JS
let words = ["I", "love", "JavaScript"];
let sentence = words.join(" "); 
// ಫಲಿತಾಂಶ: "I love JavaScript"
reverse()
ಉಲ್ಟಾ ಮಾಡಲು (Reverse): Array ನಲ್ಲಿರುವ items ಗಳನ್ನು ಹಿಂದಿನಿಂದ ಮುಂದಕ್ಕೆ (ಕೊನೆಯಿಂದ ಮೊದಲಿಗೆ) ಉಲ್ಟಾ ಜೋಡಿಸುತ್ತದೆ.
JS
let arr = [1, 2, 3, 4];
arr.reverse();
// ಫಲಿತಾಂಶ: [4, 3, 2, 1]
fill()
ಒಂದೇ ಮೌಲ್ಯದಿಂದ ತುಂಬಲು: Array ಯನ್ನು ಒಂದು ನಿರ್ದಿಷ್ಟ ಬೆಲೆಯಿಂದ (Value) ತುಂಬಲು ಇದನ್ನು ಬಳಸಲಾಗುತ್ತದೆ.
JS
let arr = new Array(3).fill("A");
// ಫಲಿತಾಂಶ: ["A", "A", "A"]

3. Sorting (ಕ್ರಮಬದ್ಧಗೊಳಿಸುವುದು)

• sort() - ಅಕ್ಷರಗಳನ್ನು (Alphabetical) ಕ್ರಮಬದ್ಧವಾಗಿ ಜೋಡಿಸಲು:

JS
let fruits = ["Banana", "Apple", "Mango"];
fruits.sort();
// ಫಲಿತಾಂಶ: ["Apple", "Banana", "Mango"]

• Number Sorting (ಸಂಖ್ಯೆಗಳನ್ನು ಕ್ರಮಬದ್ಧಗೊಳಿಸುವುದು):

ಸಾಮಾನ್ಯ sort() ಕೇವಲ ಅಕ್ಷರಗಳನ್ನು ಮಾತ್ರ ಸರಿಯಾಗಿ ಜೋಡಿಸುತ್ತದೆ. ಸಂಖ್ಯೆಗಳನ್ನು ಜೋಡಿಸುವಾಗ ಅದು ತಪ್ಪು ಲೆಕ್ಕ ಹಾಕುತ್ತದೆ (ಉದಾ: '2' ಎಂಬುದು '1' ಕ್ಕಿಂತ ದೊಡ್ಡದು ಎಂದು 25 ನ್ನು 100 ಕ್ಕಿಂತ ದೊಡ್ಡದು ಎನ್ನುತ್ತದೆ). ಇದನ್ನು ಸರಿಪಡಿಸಲು ನಾವು Compare Function ಅನ್ನು ಬಳಸಬೇಕು.

1. ಏರಿಕೆ ಕ್ರಮದಲ್ಲಿ (Ascending Order):
JS
let numbers = [40, 100, 1, 5, 25, 10];

numbers.sort(function(a, b) {
    return a - b;
});

console.log(numbers);
// ಫಲಿತಾಂಶ: [1, 5, 10, 25, 40, 100]
2. ಇಳಿಕೆ ಕ್ರಮದಲ್ಲಿ (Descending Order):
JS
let nums = [40, 100, 1, 5, 25, 10];

nums.sort(function(a, b) {
    return b - a;
});

console.log(nums);
// ಫಲಿತಾಂಶ: [100, 40, 25, 10, 5, 1]

*) ಈ ಪೂರ್ಣ function(a, b) ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ ಎಂಬುದನ್ನು ಸರಳವಾಗಿ ವಿವರಿಸುತ್ತೇನೆ.

sort() ವಿಧಾನವು ಪಟ್ಟಿಯಲ್ಲಿರುವ (Array) ಸಂಖ್ಯೆಗಳನ್ನು ಜೋಡಿಸಲು, ಪ್ರತಿ ಬಾರಿ ಎರಡು ಸಂಖ್ಯೆಗಳನ್ನು ತೆಗೆದುಕೊಂಡು ಅವೆರಡರ ನಡುವೆ ಹೋಲಿಕೆ (compare) ಮಾಡುತ್ತದೆ. ಆ ಎರಡು ಸಂಖ್ಯೆಗಳೇ ಇಲ್ಲಿ a ಮತ್ತು b.

return a - b; (ಏರಿಕೆ ಕ್ರಮಕ್ಕೆ) ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ?
ನೀವು return a - b; ಎಂದು ಬರೆದಾಗ, ಅದು a ಯಿಂದ b ಅನ್ನು ಕಳೆಯುತ್ತದೆ. ಈ ಕಳೆಯುವಿಕೆಯಿಂದ ಬರುವ ಉತ್ತರದ ಆಧಾರದ ಮೇಲೆ ಸಂಖ್ಯೆಗಳು ಜೋಡಣೆಯಾಗುತ್ತವೆ. ಇಲ್ಲಿ 3 ಸಾಧ್ಯತೆಗಳಿವೆ:

  • 1. ಉತ್ತರ ಮೈನಸ್ (Negative) ಬಂದರೆ:
    ಉದಾಹರಣೆಗೆ: a = 10, b = 25 ಆಗಿದ್ದರೆ, 10 - 25 = -15.
    ಮೈನಸ್ ಬಂದರೆ, a ಸಂಖ್ಯೆಯು b ಗಿಂತ ಚಿಕ್ಕದು ಎಂದರ್ಥ. ಆದ್ದರಿಂದ JavaScript a ಅನ್ನು ಮೊದಲೇ ಉಳಿಸಿಕೊಳ್ಳುತ್ತದೆ.
  • 2. ಉತ್ತರ ಪ್ಲಸ್ (Positive) ಬಂದರೆ:
    ಉದಾಹರಣೆಗೆ: a = 40, b = 10 ಆಗಿದ್ದರೆ, 40 - 10 = +30.
    ಪ್ಲಸ್ ಬಂದರೆ, a ಸಂಖ್ಯೆಯು b ಗಿಂತ ದೊಡ್ಡದು ಎಂದರ್ಥ. ಆದ್ದರಿಂದ JavaScript b (ಚಿಕ್ಕ ಸಂಖ್ಯೆ) ಅನ್ನು ಮೊದಲು ತಂದು, a ಅನ್ನು ಅದರ ನಂತರ ಇಡುತ್ತದೆ.
  • 3. ಉತ್ತರ ಸೊನ್ನೆ (0) ಬಂದರೆ:
    ಉದಾಹರಣೆಗೆ: a = 5, b = 5 ಆಗಿದ್ದರೆ, 5 - 5 = 0.
    ಎರಡೂ ಸಂಖ್ಯೆಗಳು ಸಮಾನ (equal) ಎಂದರ್ಥ. ಆಗ ಅವುಗಳ ಸ್ಥಾನ ಬದಲಾಗುವುದಿಲ್ಲ.
ಒಂದು ಸಣ್ಣ ಉದಾಹರಣೆಯೊಂದಿಗೆ ನೋಡೋಣ:
ನಮ್ಮ ಬಳಿ [40, 100, 1] ಎಂಬ ಪಟ್ಟಿ ಇದೆ ಅಂದುಕೊಳ್ಳಿ.

• ಹಂತ 1: ಮೊದಲು 40 ಮತ್ತು 100 ಅನ್ನು ತೆಗೆದುಕೊಳ್ಳುತ್ತದೆ. (a = 40, b = 100)
40 - 100 = -60 (ಮೈನಸ್ ಬಂತು). ಅಂದರೆ 40 ಚಿಕ್ಕದು, ಹಾಗಾಗಿ 40 ಮತ್ತು 100 ರ ಸ್ಥಾನ ಹಾಗೇ ಇರುತ್ತದೆ.

• ಹಂತ 2: ಈಗ 100 ಮತ್ತು 1 ಅನ್ನು ತೆಗೆದುಕೊಳ್ಳುತ್ತದೆ. (a = 100, b = 1)
100 - 1 = 99 (ಪ್ಲಸ್ ಬಂತು). ಅಂದರೆ 100 ದೊಡ್ಡದು. ಹಾಗಾಗಿ 1 ಅನ್ನು 100 ರ ಹಿಂದೆ ತರುತ್ತದೆ.

• ಈ ಪ್ರಕ್ರಿಯೆಯು ಪಟ್ಟಿಯ ಎಲ್ಲಾ ಸಂಖ್ಯೆಗಳು ಸರಿಯಾದ ಕ್ರಮದಲ್ಲಿ (ಚಿಕ್ಕದರಿಂದ ದೊಡ್ಡದಕ್ಕೆ) ಜೋಡಣೆಯಾಗುವವರೆಗೂ ನಡೆಯುತ್ತಲೇ ಇರುತ್ತದೆ.

ಇದೇ ಲೆಕ್ಕಾಚಾರವನ್ನು ಇಳಿಕೆ ಕ್ರಮಕ್ಕೆ (Descending) ಮಾಡಲು ನಾವು return b - a; ಎಂದು ಕಳೆಯುವಿಕೆಯನ್ನು ಉಲ್ಟಾ ಮಾಡುತ್ತೇವೆ, ಆಗ ದೊಡ್ಡ ಸಂಖ್ಯೆಗಳು ಮೊದಲು ಬರುತ್ತವೆ.

4. Pro Level (ಉನ್ನತ ಹಂತ)

ಇವುಗಳನ್ನು "Higher-Order Array Methods" ಎಂದು ಕರೆಯುತ್ತಾರೆ. ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ನಲ್ಲಿ ಇವುಗಳ ಬಳಕೆ ಅತಿ ಹೆಚ್ಚು.

map()
ಹೊಸ Array ರಚನೆ: ಪ್ರತಿಯೊಂದು item ನ ಮೇಲೆ ಒಂದು ಲೆಕ್ಕಾಚಾರ ಮಾಡಿ, ಹೊಸ Array ಕೊಡುತ್ತದೆ.
JS
let numbers = [1, 2, 3];
let doubled = numbers.map(num => num * 2);
// ಫಲಿತಾಂಶ: [2, 4, 6]
filter()
ಶರತ್ತಿಗೆ (Condition) ಹೊಂದುವ ಆಯ್ಕೆ: ನಮ್ಮ ಕಂಡೀಷನ್ ಪಾಸ್ ಮಾಡುವ items ಮಾತ್ರ ಇಟ್ಟುಕೊಂಡು ಹೊಸ Array ಕೊಡುತ್ತದೆ.
JS
let ages = [15, 22, 18, 12, 30];
let adults = ages.filter(age => age >= 18);
// ಫಲಿತಾಂಶ: [22, 18, 30]
reduce()
ಎಲ್ಲವನ್ನೂ ಸೇರಿಸಿ ಒಂದೇ ಮೌಲ್ಯ ಪಡೆಯಲು: Array ನಲ್ಲಿರುವ ಎಲ್ಲ ಸಂಖ್ಯೆಗಳನ್ನು ಸೇರಿಸಿ ಮೊತ್ತ (sum) ಕಂಡುಹಿಡಿಯಲು ಇದು ಬೆಸ್ಟ್.
JS
let prices = [10, 20, 30];
let total = prices.reduce((sum, current) => sum + current, 0);
// ಫಲಿತಾಂಶ: 60
find()
ಮೊದಲನೇ ಹೊಂದಾಣಿಕೆ (First match): ಕಂಡೀಷನ್ ಗೆ ಹೊಂದುವ ಮೊದಲನೇ item ಅನ್ನು ಮಾತ್ರ ಕೊಡುತ್ತದೆ.
JS
let numbers = [5, 12, 8, 130, 44];
let found = numbers.find(num => num > 10);
// ಫಲಿತಾಂಶ: 12 (130 ಕೂಡ 10ಕ್ಕಿಂತ ದೊಡ್ಡದು, ಆದರೆ 12 ಮೊದಲು ಸಿಕ್ಕಿತು)
findIndex()
ಇಂಡೆಕ್ಸ್ ನಂಬರ್ ಹುಡುಕಲು: ಇದು find() ನಂತೆಯೇ ಕೆಲಸ ಮಾಡುತ್ತದೆ, ಆದರೆ item ನ ಬೆಲೆಯನ್ನು ಕೊಡುವ ಬದಲು ಅದು ಎಷ್ಟನೇ ಸ್ಥಾನದಲ್ಲಿದೆ (Index Number) ಎಂದು ಹೇಳುತ್ತದೆ.
JS
let numbers = [10, 20, 30, 40];
let index = numbers.findIndex(num => num === 30);
// ಫಲಿತಾಂಶ: 2 (ಏಕೆಂದರೆ 30 ಎಂಬುದು 2ನೇ index ನಲ್ಲಿದೆ)
forEach()
ಪ್ರತಿಯೊಂದಕ್ಕೂ ಲೂಪ್ (Loop): Array ನಲ್ಲಿರುವ ಪ್ರತಿಯೊಂದು item ಗೂ ಒಂದು ಕೆಲಸ ಮಾಡಲು. (ಇದು ಹೊಸ Array ಕೊಡುವುದಿಲ್ಲ).
JS
let names = ["Ravi", "Kiran"];
names.forEach(name => console.log("Hello " + name));
// ಔಟ್ಪುಟ್: Hello Ravi, Hello Kiran
some()
ಕನಿಷ್ಠ ಒಂದಾದರೂ ಹೊಂದಾಣಿಕೆ ಇದೆಯೇ? (True/False): ಕನಿಷ್ಠ ಒಂದು item ಆದರೂ ನಮ್ಮ ಕಂಡೀಷನ್ ಪಾಸ್ ಮಾಡಿದರೆ ಸಾಕು, ಇದು true ಕೊಡುತ್ತದೆ.
JS
let marks = [35, 40, 20, 50];
let hasFailed = marks.some(mark => mark < 35);
// ಫಲಿತಾಂಶ: true (ಏಕೆಂದರೆ 20 ಎಂಬುದು 35ಕ್ಕಿಂತ ಕಡಿಮೆ ಇದೆ)
every()
ಎಲ್ಲವೂ ಹೊಂದಾಣಿಕೆ ಆಗುತ್ತವೆಯೇ? (True/False): Array ನಲ್ಲಿರುವ ಎಲ್ಲಾ items ಗಳು ನಮ್ಮ ಕಂಡೀಷನ್ ಪಾಸ್ ಮಾಡಿದರೆ ಮಾತ್ರ ಇದು true ಕೊಡುತ್ತದೆ.
JS
let ages = [20, 25, 30];
let allAdults = ages.every(age => age >= 18);
// ಫಲಿತಾಂಶ: true (ಎಲ್ಲರೂ 18ಕ್ಕಿಂತ ದೊಡ್ಡವರು)
flat()
Nested Array (Array ಒಳಗೆ Array) ಒಡೆಯಲು: Array ಒಳಗೆ ಮತ್ತೊಂದು Array ಇದ್ದರೆ, ಅವೆಲ್ಲವನ್ನೂ ತೆಗೆದು ಒಂದೇ ಸಮತಟ್ಟಾದ (flat) Array ಮಾಡುತ್ತದೆ.
JS
let nestedArr = [1, 2, [3, 4], [5, [6, 7]]];
let flatArr = nestedArr.flat(2); // 2 ಹಂತದ ವರೆಗೆ ಒಡೆಯುತ್ತದೆ
// ಫಲಿತಾಂಶ: [1, 2, 3, 4, 5, 6, 7]
Array.isArray()
ಇದು ನಿಜವಾಗಿಯೂ Array ಹೌದೋ ಅಲ್ಲವೋ?: ಒಂದು variable ನಲ್ಲಿರುವುದು Array ಆಗಿದೆಯೋ ಅಥವಾ ಬೇರೆ ಯಾವುದಾದರೂ ಡೇಟಾ ಆಗಿದೆಯೋ ಎಂದು ಪರೀಕ್ಷಿಸಲು ಬಳಸುತ್ತಾರೆ.
JS
let name = "Ravi";
let arr = [1, 2, 3];
console.log(Array.isArray(name)); // false
console.log(Array.isArray(arr));  // true

ಈಗ ನೀವು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ Array ನ ಪ್ರತಿಯೊಂದು ಮೂಲೆಯಿಂದ ಹಿಡಿದು Pro ಮಟ್ಟದವರೆಗಿನ ಎಲ್ಲಾ ಪ್ರಮುಖ Methods ಗಳನ್ನು ತಿಳಿದುಕೊಂಡಿದ್ದೀರಿ. map, filter, ಮತ್ತು reduce ನಂತಹ Methods ಗಳು ಪ್ರಾಜೆಕ್ಟ್ ಗಳಲ್ಲಿ ಪ್ರತಿದಿನ ಅತಿ ಹೆಚ್ಚು ಬಳಕೆಯಾಗುವ ಅಸ್ತ್ರಗಳಾಗಿವೆ!
