ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಟೆಂಪ್ಲೇಟ್ ಲಿಟರಲ್ಸ್

Template Literals ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ES6 (2015) ತಂದ ಅತ್ಯಂತ ಉಪಯುಕ್ತವಾದ ಹೊಸ ವೈಶಿಷ್ಟ್ಯವಾಗಿದೆ. ಇದು ಸ್ಟ್ರಿಂಗ್‌ಗಳನ್ನು (ಪಠ್ಯಗಳನ್ನು) ನಿರ್ವಹಿಸುವ ವಿಧಾನವನ್ನು ಸಂಪೂರ್ಣವಾಗಿ ಸುಲಭಗೊಳಿಸಿದೆ.

ಇದನ್ನು ಸಾಮಾನ್ಯ ಸಿಂಗಲ್ ಅಥವಾ ಡಬಲ್ ಕೋಟ್ಸ್ ಬದಲಿಗೆ ಬ್ಯಾಕ್‌ಟಿಕ್ಸ್ (` `) ಬಳಸಿ ಬರೆಯಲಾಗುತ್ತದೆ.

1. ಬ್ಯಾಕ್‌ಟಿಕ್ಸ್ ಬಳಕೆ (Syntax & Quotes)

ಟೆಂಪ್ಲೇಟ್ ಲಿಟರಲ್ಸ್ ಅನ್ನು ಬ್ಯಾಕ್‌ಟಿಕ್ಸ್ (`` ` ``) ಬಳಸಿ ಡಿಕ್ಲೇರ್ ಮಾಡಲಾಗುತ್ತದೆ. ಇದರ ಬಹುದೊಡ್ಡ ಲಾಭವೆಂದರೆ, ನಾವು ಸ್ಟ್ರಿಂಗ್ ಒಳಗೆ ಸಿಂಗಲ್ ಕೋಟ್ಸ್ (' ') ಮತ್ತು ಡಬಲ್ ಕೋಟ್ಸ್ (" ") ಎರಡನ್ನೂ ಯಾವುದೇ ಅಡೆತಡೆಯಿಲ್ಲದೆ (Escape characters ಇಲ್ಲದೆ) ಬಳಸಬಹುದು.

// ಹಳೆಯ ವಿಧಾನ (Old Method):
let oldStr = "He said \"Hello\"";

// ಹೊಸ ಟೆಂಪ್ಲೇಟ್ ಲಿಟರಲ್ಸ್:
let newStr = `He said "Hello" and 'Bye'`;
console.log(newStr);

2. ಮಲ್ಟಿ-ಲೈನ್ ಸ್ಟ್ರಿಂಗ್ಸ್ (Multi-line Strings)

// ❌ ಹಳೆಯ ವಿಧಾನ (Using \n):
let text1 = "ನಮಸ್ಕಾರ,\nಇದು ಮೊದಲ ಸಾಲು.\nಇದು ಎರಡನೇ ಸಾಲು.";

// ✅ ಟೆಂಪ್ಲೇಟ್ ಲಿಟರಲ್ಸ್ (Directly pressing Enter):
let text2 = `ನಮಸ್ಕಾರ,
ಇದು ಮೊದಲ ಸಾಲು.
ಇದು ಎರಡನೇ ಸಾಲು.`;

ಹಿಂದೆ ಸ್ಟ್ರಿಂಗ್‌ನಲ್ಲಿ ಹೊಸ ಸಾಲನ್ನು (New line) ಸೇರಿಸಬೇಕಾದರೆ \n ಬಳಸಬೇಕಿತ್ತು. ಆದರೆ ಬ್ಯಾಕ್‌ಟಿಕ್ಸ್ ಒಳಗೆ ನೀವು ಕೀಬೋರ್ಡ್‌ನಲ್ಲಿ ಎಂಟರ್ (Enter) ಒತ್ತಿ ನೇರವಾಗಿ ಮಲ್ಟಿ-ಲೈನ್ ಸ್ಟ್ರಿಂಗ್‌ಗಳನ್ನು ರಚಿಸಬಹುದು. ಕೋಡ್ ಓದಲು ಇದು ತುಂಬಾ ಸುಲಭ.

3. ಸ್ಟ್ರಿಂಗ್ ಇಂಟರ್‌ಪೋಲೇಶನ್ (String Interpolation)

ಇದು ಟೆಂಪ್ಲೇಟ್ ಲಿಟರಲ್ಸ್‌ನ ಅತ್ಯಂತ ಪ್ರಬಲವಾದ ವೈಶಿಷ್ಟ್ಯ! ಸ್ಟ್ರಿಂಗ್ ಮಧ್ಯದಲ್ಲಿ ವೇರಿಯೇಬಲ್‌ಗಳನ್ನು (Variables) ಅಥವಾ ಲೆಕ್ಕಾಚಾರಗಳನ್ನು (Expressions) ಸೇರಿಸಲು ${...} ಸಿಂಟ್ಯಾಕ್ಸ್ ಅನ್ನು ಬಳಸಲಾಗುತ್ತದೆ. ಇದರಿಂದ + ಆಪರೇಟರ್ ಬಳಸಿ ಸ್ಟ್ರಿಂಗ್ ಜೋಡಿಸುವ ಕಷ್ಟ ತಪ್ಪುತ್ತದೆ.

// ವೇರಿಯೇಬಲ್ ಸೇರಿಸುವುದು (Variables):
let name = "Ravi";
let age = 25;

// ಹಳೆಯ ರೀತಿ:
// "My name is " + name + " and I am " + age;

// ಹೊಸ ರೀತಿ:
let info = `My name is ${name} and I am ${age}`;
console.log(info);
// ಲೆಕ್ಕಾಚಾರಗಳನ್ನು ಸೇರಿಸುವುದು (Expressions):
let price = 100;
let tax = 18;

let totalMsg = `Total price is: ₹${price + tax}`;
console.log(totalMsg); // "Total price is: ₹118"

4. HTML ಟೆಂಪ್ಲೇಟ್‌ಗಳು (HTML Templates)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಬಳಸಿ ಡೈನಾಮಿಕ್ ಆಗಿ HTML ಕೋಡ್ ಅನ್ನು ರಚಿಸುವಾಗ ಟೆಂಪ್ಲೇಟ್ ಲಿಟರಲ್ಸ್ ಅತ್ಯಂತ ಸಹಕಾರಿಯಾಗಿದೆ. ಮಲ್ಟಿ-ಲೈನ್ ಬೆಂಬಲ ಮತ್ತು ಇಂಟರ್‌ಪೋಲೇಶನ್ ಇರುವುದರಿಂದ, ಪೂರ್ತಿ HTML ಬ್ಲಾಕ್ ಅನ್ನು ಒಂದೇ ವೇರಿಯೇಬಲ್ ಒಳಗೆ ಸುಂದರವಾಗಿ ವಿನ್ಯಾಸಗೊಳಿಸಬಹುದು.

let header = "Welcome";
let content = "This is a JS tutorial.";

let htmlCard = `
<div class="card">
<h2>${header}</h2>
<p>${content}</p>
</div>
`;

document.body.innerHTML = htmlCard;

5. ಟ್ಯಾಗ್ಡ್ ಟೆಂಪ್ಲೇಟ್‌ಗಳು (Tagged Templates)

ಇದು ಟೆಂಪ್ಲೇಟ್ ಲಿಟರಲ್ಸ್‌ನ ಒಂದು ಸುಧಾರಿತ (Advanced) ವೈಶಿಷ್ಟ್ಯವಾಗಿದೆ. ಟೆಂಪ್ಲೇಟ್ ಲಿಟರಲ್ ಅನ್ನು ಪಾರ್ಸ್ (Parse) ಮಾಡಲು ನೀವು ಒಂದು ಫಂಕ್ಷನ್ (Function) ಅನ್ನು ಬಳಸಬಹುದು. ಆ ಫಂಕ್ಷನ್‌ನ ಹೆಸರನ್ನು ಬ್ಯಾಕ್‌ಟಿಕ್ಸ್‌ಗೂ ಮುಂಚೆ ಬರೆಯಲಾಗುತ್ತದೆ. ಆ ಫಂಕ್ಷನ್ ಸ್ಟ್ರಿಂಗ್‌ನ ಭಾಗಗಳನ್ನು ಮತ್ತು ವೇರಿಯೇಬಲ್‌ಗಳನ್ನು ಪ್ರತ್ಯೇಕವಾಗಿ ಸ್ವೀಕರಿಸಿ ಕಾರ್ಯನಿರ್ವಹಿಸುತ್ತದೆ.

// ಟ್ಯಾಗ್ ಫಂಕ್ಷನ್ ರಚನೆ
function highlight(strings, ...values) {
  let str0 = strings[0]; // "Hi, I am "
  let str1 = strings[1]; // " and I live in "

  // values ಅರೇ ವೇರಿಯೇಬಲ್ ಮೌಲ್ಯಗಳನ್ನು ಹೊಂದಿರುತ್ತದೆ
  return `${str0}<b>${values[0]}</b>${str1}<b>${values[1]}</b>`;
}

let person = "John";
let city = "Bengaluru";

// ಫಂಕ್ಷನ್ ಅನ್ನು ಟ್ಯಾಗ್ ಆಗಿ ಬಳಸುವುದು (() ಇಲ್ಲದೆ)
let result = highlight`Hi, I am ${person} and I live in ${city}`;
console.log(result);
// Output: "Hi, I am <b>John</b> and I live in <b>Bengaluru</b>"

6. String.raw (Raw Strings)

String.raw ಎಂಬುದು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿರುವ ಒಂದು ಬಿಲ್ಟ್-ಇನ್ (built-in) ಟ್ಯಾಗ್ ಫಂಕ್ಷನ್ ಆಗಿದೆ. ಇದನ್ನು ಬಳಸಿದಾಗ ಸ್ಟ್ರಿಂಗ್‌ನಲ್ಲಿರುವ ಎಸ್ಕೇಪ್ ಅಕ್ಷರಗಳನ್ನು (ಉದಾಹರಣೆಗೆ \n, \t) ನಿರ್ಲಕ್ಷಿಸಲಾಗುತ್ತದೆ ಮತ್ತು ಸ್ಟ್ರಿಂಗ್ ಹೇಗಿದೆಯೋ ಹಾಗೆಯೇ (Raw ಆಗಿ) ಉಳಿಯುತ್ತದೆ. ಫೈಲ್ ಪಾತ್‌ಗಳನ್ನು (File paths) ಬರೆಯುವಾಗ ಇದು ತುಂಬಾ ಸಹಕಾರಿ.

// ಸಾಮಾನ್ಯ ಸ್ಟ್ರಿಂಗ್ ( \n ಹೊಸ ಲೈನ್ ಸೃಷ್ಟಿಸುತ್ತದೆ):
console.log(`C:\folder\newFile.txt`);
/* Output: C:\folder ewFile.txt */

// String.raw ಬಳಸುವುದು (ಬ್ಯಾಕ್‌ಸ್ಲ್ಯಾಶ್ ಹಾಗೇ ಇರುತ್ತದೆ):
console.log(String.raw`C:\folder\newFile.txt`);
// Output: C:\folder\newFile.txt
