ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಬಳಕೆದಾರರು ವೆಬ್‌ಪುಟದೊಂದಿಗೆ ಮಾಡುವ ಪ್ರತಿಯೊಂದು ಕ್ರಿಯೆಯನ್ನೂ (Events) ನಿಯಂತ್ರಿಸುವ ಕಲೆ.
ಮೌಸ್ ಕ್ಲಿಕ್, ಕೀಬೋರ್ಡ್ ಟೈಪಿಂಗ್, ಮತ್ತು ಫಾರ್ಮ್ ಸಬ್ಮಿಷನ್‌ಗಳನ್ನು Beginner ಹಂತದಿಂದ Pro ಹಂತದವರೆಗೆ ಆಳವಾಗಿ ಇಲ್ಲಿ ವಿವರಿಸಲಾಗಿದೆ.

Mouse Events | ಮೌಸ್ ಈವೆಂಟ್‌ಗಳು

ಬಳಕೆದಾರರು ಮೌಸ್ ಅಥವಾ ಟ್ರ್ಯಾಕ್‌ಪ್ಯಾಡ್ ಬಳಸಿ ಮಾಡುವ ಪ್ರತಿಯೊಂದು ಕ್ರಿಯೆ.

Level 1 ಮೂಲಭೂತ ಮೌಸ್ ಈವೆಂಟ್‌ಗಳು (Basic Clicks)

ಬಟನ್ ಕ್ಲಿಕ್ ಮಾಡುವುದು ಅಥವಾ ಡಬಲ್ ಕ್ಲಿಕ್ ಮಾಡುವುದು ಅತ್ಯಂತ ಸಾಮಾನ್ಯ ಮೌಸ್ ಈವೆಂಟ್‌ಗಳಾಗಿವೆ.

  • click: ಮೌಸ್‌ನ ಎಡ ಬಟನ್ ಅನ್ನು ಒತ್ತಿ ಬಿಟ್ಟಾಗ (Press & Release).
  • dblclick: ಮೌಸ್ ಅನ್ನು ವೇಗವಾಗಿ ಎರಡು ಬಾರಿ ಕ್ಲಿಕ್ ಮಾಡಿದಾಗ.
  • mousedown: ಮೌಸ್ ಬಟನ್ ಅನ್ನು ಒತ್ತಿದ ತಕ್ಷಣ (ಇನ್ನೂ ಬಿಟ್ಟಿಲ್ಲ).
  • mouseup: ಒತ್ತಿ ಹಿಡಿದ ಮೌಸ್ ಬಟನ್ ಅನ್ನು ಬಿಟ್ಟಾಗ.
  • ಗಮನಿಸಿ: ಒಂದು click ಈವೆಂಟ್ ಪೂರ್ಣಗೊಳ್ಳಲು, ಮೊದಲು mousedown ಆಗಿ ನಂತರ mouseup ಆಗಬೇಕು.
const btn = document.getElementById("myButton");

btn.addEventListener("click", () => console.log("ಕ್ಲಿಕ್ ಆಯ್ತು!"));
btn.addEventListener("dblclick", () => console.log("ಡಬಲ್ ಕ್ಲಿಕ್!"));
btn.addEventListener("mousedown", () => console.log("ಒತ್ತಿದ್ದೀರಿ.."));
btn.addEventListener("mouseup", () => console.log("ಬಿಟ್ಟಿರಿ."));

Level 2 ಹೋವರ್ (Hover) ಮತ್ತು ಚಲನೆ

ಬಳಕೆದಾರರು ಮೌಸ್ ಕರ್ಸರ್ ಅನ್ನು ಎಲಿಮೆಂಟ್‌ನ ಮೇಲೆ ತಂದಾಗ ಅಥವಾ ಹೊರತೆಗೆದಾಗ ನಿರ್ಧರಿಸಲು ಇವುಗಳನ್ನು ಬಳಸುತ್ತಾರೆ. ಇಲ್ಲಿ ಡೆವಲಪರ್‌ಗಳು ಬಹಳ ಗೊಂದಲಕ್ಕೊಳಗಾಗುತ್ತಾರೆ, ಆದ್ದರಿಂದ ಇದನ್ನು ಎಚ್ಚರಿಕೆಯಿಂದ ಗಮನಿಸಿ.

mouseover vs mouseenter

  • mouseover (Event Bubbling ಇದೆ): ಪೇರೆಂಟ್ ಎಲಿಮೆಂಟ್ ಮೇಲೆ ತಂದಾಗ ರನ್ ಆಗುತ್ತದೆ. ಅಷ್ಟೇ ಅಲ್ಲದೆ, ಪೇರೆಂಟ್ ಒಳಗಿರುವ ಮಗುವಿನ ಎಲಿಮೆಂಟ್ (Child) ಮೇಲೆ ಮೌಸ್ ಹೋದಾಗಲೂ ಇದು ಮತ್ತೆ ಮತ್ತೆ ರನ್ ಆಗುತ್ತದೆ. (ಕಿರಿಕಿರಿಯುಂಟುಮಾಡಬಹುದು).
  • mouseenter (Event Bubbling ಇಲ್ಲ): ಕೇವಲ ಪೇರೆಂಟ್ ಎಲಿಮೆಂಟ್ ಮೇಲೆ ಮೌಸ್ ಬಂದಾಗ ಒಮ್ಮೆ ಮಾತ್ರ ರನ್ ಆಗುತ್ತದೆ. ಒಳಗಿನ ಮಕ್ಕಳ ಮೇಲೆ ಮೌಸ್ ಹೋದರೂ ಇದು ಮತ್ತೆ ರನ್ ಆಗುವುದಿಲ್ಲ. (UI ಡಿಸೈನ್ ಮಾಡಲು ಡೆವಲಪರ್‌ಗಳು ಇದನ್ನೇ ಹೆಚ್ಚು ಬಳಸುತ್ತಾರೆ).

mouseout vs mouseleave

  • mouseout: ಒಳಗಿನ ಮಕ್ಕಳಿಂದ ಮೌಸ್ ಹೊರಬಂದರೂ ಪದೇ ಪದೇ ರನ್ ಆಗುತ್ತದೆ.
  • mouseleave: ಸಂಪೂರ್ಣವಾಗಿ ಮುಖ್ಯ (Parent) ಎಲಿಮೆಂಟ್‌ನಿಂದ ಹೊರಬಂದಾಗ ಮಾತ್ರ ಒಮ್ಮೆ ರನ್ ಆಗುತ್ತದೆ. (ಇದು ಬೆಸ್ಟ್).
const card = document.getElementById("profileCard");

// ಕಾರ್ಡ್ ಒಳಗೆ ಬಂದಾಗ ಒಮ್ಮೆ ರನ್ ಆಗುತ್ತದೆ
card.addEventListener("mouseenter", () => {
  card.style.boxShadow = "0px 10px 20px rgba(0,0,0,0.5)";
});

// ಕಾರ್ಡ್ ನಿಂದ ಹೊರಹೋದಾಗ ಒಮ್ಮೆ ರನ್ ಆಗುತ್ತದೆ
card.addEventListener("mouseleave", () => {
  card.style.boxShadow = "none";
});

Level 3 ಮೌಸ್ ಟ್ರ್ಯಾಕಿಂಗ್ ಮತ್ತು ರೈಟ್-ಕ್ಲಿಕ್ ಮೆನು

A. mousemove

ಮೌಸ್ ಕರ್ಸರ್ ಚಲಿಸುತ್ತಿರುವಾಗ ಪ್ರತಿ ಪಿಕ್ಸೆಲ್‌ಗೂ ರನ್ ಆಗುತ್ತದೆ. e.clientX ಮತ್ತು e.clientY ಬಳಸಿ ಮೌಸ್ ಎಲ್ಲಿದೆ ಎಂದು ಪತ್ತೆಹಚ್ಚಬಹುದು.

canvas.addEventListener("mousemove", (e) => {
  console.log(`ಮೌಸ್: X: ${e.clientX}, Y: ${e.clientY}`);
});

B. contextmenu & C. wheel

contextmenu: ಬ್ರೌಸರ್‌ನ ಸ್ವಂತ ರೈಟ್-ಕ್ಲಿಕ್ ಮೆನು ತಡೆದು (e.preventDefault()), ಸ್ವಂತ ಮೆನು ತೋರಿಸಲು.
wheel: e.deltaY ಬಳಸಿ ಸ್ಕ್ರಾಲ್ ವೀಲ್ ಕೆಳಗೆ/ಮೇಲೆ ತಿರುಗಿದೆಯಾ ಎಂದು ತಿಳಿಯಲು (Zoom In/Out).

Pro Level Drag and Drop & Performance

ಯಾವ ಬಟನ್ ಒತ್ತಲಾಗಿದೆ?

e.button: ಯಾವ ಬಟನ್ ಕ್ಲಿಕ್ ಆಗಿದೆ? (0=ಎಡ, 1=ಮಧ್ಯ, 2=ಬಲ).

e.buttons: ಮೌಸ್ ಚಲಿಸುವಾಗ (mousemove) ಯಾವ ಬಟನ್ ಒತ್ತಿ ಹಿಡಿಯಲಾಗಿದೆ? (1 ಎಂದರೆ ಎಡ ಬಟನ್ - Drag ಮಾಡುವಾಗ ಉಪಯುಕ್ತ).

ಸ್ವಂತ Drag and Drop ತರ್ಕ

  1. mousedown: ಡ್ರ್ಯಾಗ್ ಶುರು (Flag = true).
  2. mousemove: Flag true ಇದ್ದರೆ ಮೌಸ್ ಜೊತೆಗೆ ಸರಿಸು.
  3. mouseup: ಡ್ರ್ಯಾಗ್ ನಿಲ್ಲಿಸು (Flag = false).

Performance Warning

mousemove ಮತ್ತು wheel ಸೆಕೆಂಡಿಗೆ ನೂರಾರು ಬಾರಿ ರನ್ ಆಗುತ್ತವೆ. ವೆಬ್‌ಸೈಟ್ ಲ್ಯಾಗ್ ಆಗಬಹುದು.

Pro Solution:

Throttling ಅಥವಾ Debouncing. "ಪ್ರತಿ 50ms ಗೆ ಒಮ್ಮೆ ರನ್ ಆಗು" ಎಂದು setTimeout ಅಥವಾ requestAnimationFrame ಬಳಸಿ ಆಪ್ಟಿಮೈಸ್ ಮಾಡುವುದು.

Keyboard Events | ಕೀಬೋರ್ಡ್ ಈವೆಂಟ್‌ಗಳು

ಬಳಕೆದಾರರು ಕೀಬೋರ್ಡ್‌ನಲ್ಲಿ ಗುಂಡಿ ಒತ್ತಿದಾಗ ಅಥವಾ ಬಿಟ್ಟಾಗ ನಡೆಯುವ ಕ್ರಿಯೆಗಳು.

Level 1 ಪ್ರಮುಖ ಈವೆಂಟ್‌ಗಳು

  • keydown: ಕೀಬೋರ್ಡ್‌ನಲ್ಲಿ ಕೀ ಒತ್ತಿದ ತಕ್ಷಣ ರನ್ ಆಗುತ್ತದೆ. ಒತ್ತಿ ಹಿಡಿದರೆ ಪದೇ ಪದೇ ರನ್ ಆಗುತ್ತದೆ.
  • keyup: ಒತ್ತಿದ ಕೀ ಅನ್ನು ಬಿಟ್ಟಾಗ ರನ್ ಆಗುತ್ತದೆ. ಟೈಪಿಂಗ್ ಮುಗಿಸಿದ ನಂತರ ಕೆಲಸವಾಗಬೇಕಿದ್ದರೆ ಇದು ಬೆಸ್ಟ್.
  • keypress: ಹಳೆಯ ಬ್ರೌಸರ್‌ಗಳಲ್ಲಿತ್ತು (Deprecated). ಬಳಸಬೇಡಿ.
const myInput = document.getElementById("searchBox");

myInput.addEventListener("keydown", () => console.log("ಕೀ ಒತ್ತಿದ್ದೀರಿ!"));
myInput.addEventListener("keyup", () => console.log("ಕೀ ಬಿಟ್ಟಿರಿ!"));

Level 2 e.key vs e.code

ಡೆವಲಪರ್‌ಗಳು ಇವೆರಡರ ನಡುವೆ ಬಹಳ ಗೊಂದಲಕ್ಕೊಳಗಾಗುತ್ತಾರೆ.

e.key (ತೋರಿಸುವ ಮೌಲ್ಯ)

ನೀವು ನಿಖರವಾಗಿ ಯಾವ ಅಕ್ಷರ ಟೈಪ್ ಮಾಡಿದ್ದೀರಿ ಎಂದು ಹೇಳುತ್ತದೆ.

a ಒತ್ತಿದರೆ "a".
Shift + a ಒತ್ತಿದರೆ "A".

e.code (ಭೌತಿಕ ಬಟನ್)

ಭೌತಿಕ ಕೀಯ (Physical Button) ಹೆಸರು. ಭಾಷೆ ಬದಲಾದರೂ ಇದು ಬದಲಾಗುವುದಿಲ್ಲ.

'A' ಬಟನ್ ಒತ್ತಿದರೆ "KeyA" ಬರುತ್ತದೆ.

Pro Tip: ಪಾಸ್‌ವರ್ಡ್ ಅಥವಾ ಟೈಪಿಂಗ್‌ಗೆ e.key ಬಳಸಿ. ಗೇಮಿಂಗ್ (W,A,S,D) ಅಥವಾ ಶಾರ್ಟ್‌ಕಟ್‌ಗಳಿಗೆ e.code ಬಳಸಿ.

Level 3 Modifier Keys & Shortcuts

ಒಂದೇ ಬಾರಿಗೆ 2 ಕೀಗಳನ್ನು ಒತ್ತಿದಾಗ (ಉದಾ: Ctrl+S) ಪತ್ತೆಹಚ್ಚಲು: e.ctrlKey, e.shiftKey, e.altKey, e.metaKey ಗಳು true/false ನೀಡುತ್ತವೆ.

document.addEventListener("keydown", (e) => {
  // Ctrl + S ಒತ್ತಿದಾಗ
  if (e.ctrlKey && e.key === "s") {
    e.preventDefault(); // ಬ್ರೌಸರ್‌ನ Save ತಡೆಯುವುದು
    console.log("ನಮ್ಮ ಸ್ವಂತ ಸೇವ್ ಫಂಕ್ಷನ್ ರನ್ ಆಗುತ್ತಿದೆ!");
  }
});

Pro Level ತಂತ್ರಗಳು & Debouncing

  • Default ತಡೆಯುವುದು: ಫಾರ್ಮ್‌ನಲ್ಲಿ Enter ಒತ್ತಿದರೆ ಸಬ್ಮಿಟ್ ಆಗುವುದು, Space ಒತ್ತಿದರೆ ಪೇಜ್ ಸ್ಕ್ರಾಲ್ ಆಗುವುದನ್ನು e.preventDefault() ಬಳಸಿ ತಡೆಯಬಹುದು.
  • Global vs Local: window.addEventListener (ಇಡೀ ಪೇಜ್) vs inputBox.addEventListener (ಕೇವಲ ಇನ್‌ಪುಟ್).

Performance & Debouncing (ಮೆಮೊರಿ ಉಳಿಸುವಿಕೆ)

ನೀವು ಸರ್ಚ್ ಬಾಕ್ಸ್‌ನಲ್ಲಿ ಟೈಪ್ ಮಾಡುವಾಗ, ಪ್ರತಿಯೊಂದು ಅಕ್ಷರ ಟೈಪ್ ಮಾಡಿದಾಗಲೂ ಬ್ಯಾಕೆಂಡ್‌ಗೆ ರಿಕ್ವೆಸ್ಟ್ ಕಳಿಸಿದರೆ ಸರ್ವರ್ ಕ್ರ್ಯಾಶ್ ಆಗಬಹುದು! ಇದಕ್ಕೆ Debouncing ತಂತ್ರ ಬಳಸುತ್ತಾರೆ: "ಬಳಕೆದಾರ ಟೈಪ್ ಮಾಡುವುದನ್ನು ನಿಲ್ಲಿಸಿ ಅರ್ಧ ಸೆಕೆಂಡ್ (500ms) ಆದ ಮೇಲೆ ಮಾತ್ರ ಸರ್ಚ್ ಮಾಡು". 

let timeoutId;
const searchBox = document.getElementById("search");

searchBox.addEventListener("keyup", function(e) {
  clearTimeout(timeoutId); // ಹಳೆಯ ಟೈಮರ್ ಕ್ಯಾನ್ಸಲ್ ಮಾಡು
  
  // ಹೊಸ ಟೈಮರ್ ಶುರು ಮಾಡು (ಬಳಕೆದಾರ 500ms ಟೈಪ್ ಮಾಡದಿದ್ದರೆ ರನ್ ಆಗುತ್ತದೆ)
  timeoutId = setTimeout(() => {
    console.log("ಸರ್ವರ್‌ಗೆ ರಿಕ್ವೆಸ್ಟ್: " + e.target.value);
  }, 500); 
});

Form Events | ಫಾರ್ಮ್ ಈವೆಂಟ್‌ಗಳು

ಲಾಗಿನ್, ರಿಜಿಸ್ಟ್ರೇಷನ್, ಅಥವಾ ಸರ್ಚ್ ಮೂಲಕ ಬಳಕೆದಾರರಿಂದ ಡೇಟಾ ಪಡೆಯಲು.

Level 1 submit ಈವೆಂಟ್ (ಫಾರ್ಮ್ ಕಳುಹಿಸುವುದು)

ಬಳಕೆದಾರರು ಫಾರ್ಮ್ ಅನ್ನು ತುಂಬಿ "Submit" ಬಟನ್ ಒತ್ತಿದಾಗ ಅಥವಾ Enter ಕೀ ಒತ್ತಿದಾಗ ರನ್ ಆಗುತ್ತದೆ.

ಅತಿ ದೊಡ್ಡ ನಿಯಮ (Beginner Mistake):

submit ಈವೆಂಟ್ ಅನ್ನು ಯಾವಾಗಲೂ <form> ಟ್ಯಾಗ್‌ಗೆ ಹಾಕಬೇಕು, ಅದರ ಒಳಗಿರುವ <button> ಗಲ್ಲ!

e.preventDefault() ನ ಪ್ರಾಮುಖ್ಯತೆ:

ಸಬ್ಮಿಟ್ ಆದ ತಕ್ಷಣ ಇಡೀ ವೆಬ್‌ಪುಟ ರಿಫ್ರೆಶ್ (Reload) ಆಗುತ್ತದೆ. ಇದನ್ನು ತಡೆಯಲು ಕಡ್ಡಾಯವಾಗಿ ಇದನ್ನು ಬಳಸಬೇಕು.

<form id="loginForm">
  <input type="text" id="username" placeholder="ಹೆಸರು">
  <button type="submit">ಲಾಗಿನ್</button>
</form>
const form = document.getElementById("loginForm");
form.addEventListener("submit", function(e) {
  e.preventDefault(); // ಪೇಜ್ ರಿಫ್ರೆಶ್ ಆಗುವುದನ್ನು ತಡೆಯುವುದು
  const user = document.getElementById("username").value;
  console.log("ಲಾಗಿನ್ ಆದವರು: " + user);
});

Level 2 input ಮತ್ತು change ನಡುವಿನ ವ್ಯತ್ಯಾಸ

ಡೆವಲಪರ್‌ಗಳು ಅತಿ ಹೆಚ್ಚು ಗೊಂದಲಕ್ಕೊಳಗಾಗುವುದು ಇವೆರಡರ ನಡುವೆಯೇ. ಇವೆರಡೂ ಬಳಕೆದಾರರು ಟೈಪ್ ಮಾಡಿದ ಡೇಟಾವನ್ನು ಟ್ರ್ಯಾಕ್ ಮಾಡುತ್ತವೆ, ಆದರೆ ಅವು ರನ್ ಆಗುವ ಸಮಯ ಬೇರೆ ಬೇರೆ.

input ಈವೆಂಟ್

Real-time / ತಕ್ಷಣದ ಪ್ರತಿಕ್ರಿಯೆ

ಪ್ರತಿಯೊಂದು ಅಕ್ಷರ ಟೈಪ್ ಮಾಡಿದಾಗಲೂ (ಅಥವಾ ಅಳಿಸಿದಾಗಲೂ) ತಕ್ಷಣವೇ ರನ್ ಆಗುತ್ತದೆ.

ಬಳಕೆ: ಲೈವ್ ಸರ್ಚ್, ಪಾಸ್‌ವರ್ಡ್ ಸ್ಟ್ರೆಂತ್, ಅಕ್ಷರಗಳನ್ನು ಎಣಿಸಲು.

change ಈವೆಂಟ್

Loss of focus / ಕೆಲಸ ಮುಗಿದ ಮೇಲೆ

ಟೈಪ್ ಮಾಡುತ್ತಿರುವಾಗ ರನ್ ಆಗುವುದಿಲ್ಲ! ಟೈಪ್ ಮಾಡುವುದನ್ನು ಮುಗಿಸಿ ಆ ಬಾಕ್ಸ್‌ನಿಂದ ಹೊರಗೆ ಕ್ಲಿಕ್ ಮಾಡಿದಾಗ (Blur) ಮಾತ್ರ ಒಮ್ಮೆ ರನ್ ಆಗುತ್ತದೆ.

ಬಳಕೆ: ಈಮೇಲ್ ಟೈಪ್ ಮಾಡಿ ಮುಗಿಸಿದ ಮೇಲೆ ಸರಿಯಾಗಿದೆಯೇ ಎಂದು ಚೆಕ್ ಮಾಡಲು (Validation).

const inputField = document.getElementById("myInput");

// ಅಕ್ಷರ ಟೈಪ್ ಮಾಡಿದ ತಕ್ಷಣವೇ ಪ್ರಿಂಟ್ ಆಗುತ್ತಲೇ ಇರುತ್ತದೆ
inputField.addEventListener("input", (e) => console.log("Input: " + e.target.value));

// ಟೈಪ್ ಮಾಡಿ ಮುಗಿಸಿ, ಹೊರಗೆ ಕ್ಲಿಕ್ ಮಾಡಿದ ಮೇಲೆ ಮಾತ್ರ ಪ್ರಿಂಟ್ ಆಗುತ್ತದೆ
inputField.addEventListener("change", (e) => console.log("Change: " + e.target.value));

Level 3 FormData (The Modern Way)

ಒಂದು ಫಾರ್ಮ್‌ನಲ್ಲಿ 10 ಇನ್‌ಪುಟ್‌ಗಳಿದ್ದರೆ, ಹತ್ತು ಬಾರಿಯೂ getElementById('id').value ಎಂದು ಬರೆಯುವುದು ಹಳೆಯ ವಿಧಾನ. ವೃತ್ತಿಪರರು FormData Object ಅನ್ನು ಬಳಸುತ್ತಾರೆ. ಇದು ಫಾರ್ಮ್‌ನಲ್ಲಿರುವ ಎಲ್ಲಾ ಡೇಟಾವನ್ನು ಒಂದೇ ಬಾರಿಗೆ ಒಂದು ಪ್ಯಾಕೆಟ್ ತರಹ ಕಲೆಹಾಕುತ್ತದೆ. (ಇದಕ್ಕೆ ನಿಮ್ಮ HTML ನಲ್ಲಿ name ಆಟ್ರಿಬ್ಯೂಟ್ ಇರಲೇಬೇಕು). 

document.getElementById("registerForm").addEventListener("submit", function(e) {
  e.preventDefault();
  
  // ಫಾರ್ಮ್‌ನ ಎಲ್ಲಾ ಡೇಟಾವನ್ನು ಒಂದೇ ಬಾರಿಗೆ ಪ್ಯಾಕ್ ಮಾಡುವುದು
  const formData = new FormData(this); 

  // ಡೇಟಾವನ್ನು JSON ಆಬ್ಜೆಕ್ಟ್ ಮಾಡುವುದು
  const data = Object.fromEntries(formData);
  console.log("ಸಂಪೂರ್ಣ ಡೇಟಾ:", data); 
  // Output: {firstName: "ರಾಮ್", email: "ram@gmail.com"}
});

Expert Level Event Delegation on Forms

ದೊಡ್ಡ ಫಾರ್ಮ್‌ನಲ್ಲಿರುವ ಎಲ್ಲಾ ಇನ್‌ಪುಟ್‌ಗಳಿಗೂ input ಈವೆಂಟ್ ಹಾಕಬೇಕಾದರೆ, ಪ್ರತಿಯೊಂದಕ್ಕೂ ಪ್ರತ್ಯೇಕ ಲಿಸನರ್ ಹಾಕಬೇಡಿ. ಬದಲಾಗಿ ಇಡೀ <form> ಗೇ ಕೇವಲ ಒಂದು ಲಿಸನರ್ ಹಾಕಿ. 

const myForm = document.getElementById("registerForm");

// ಫಾರ್ಮ್ ಒಳಗಿನ ಯಾವುದೇ ಇನ್‌ಪುಟ್‌ನಲ್ಲಿ ಬದಲಾವಣೆಯಾದರೂ ಇದು ಹಿಡಿಯುತ್ತದೆ
myForm.addEventListener("input", function(e) {
  const inputName = e.target.name;
  const inputValue = e.target.value;
  console.log(`${inputName} ನಲ್ಲಿ ${inputValue} ಟೈಪ್ ಆಗುತ್ತಿದೆ.`);
});
