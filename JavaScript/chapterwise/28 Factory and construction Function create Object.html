JavaScript Objects: Factory & Constructor Functions

ಆರಂಭಿಕ ವಿದ್ಯಾರ್ಥಿಯಿಂದ (Beginner) ವೃತ್ತಿಪರ (Professional) ಮಟ್ಟದವರೆಗೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಆಬ್ಜೆಕ್ಟ್‌ಗಳನ್ನು ಸೃಷ್ಟಿಸುವ ಸುಧಾರಿತ ವಿಧಾನಗಳ ಸಂಪೂರ್ಣ ಕನ್ನಡಿಗರಿಗಾಗಿ ತಯಾರಿಸಿದ ಮಾರ್ಗದರ್ಶಿ.

೧. Factory Function (ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Factory Function ಎನ್ನುವುದನ್ನು ಅರ್ಥಮಾಡಿಕೊಳ್ಳುವುದು ಬಹಳ ಸುಲಭ. ನೀವು ಇದನ್ನು ಒಂದು "ಆಟಿಕೆ ತಯಾರಿಸುವ ಮಷಿನ್" (Toy Factory) ಎಂದು ಊಹಿಸಿಕೊಳ್ಳಿ.

ನೀವು ಮಷಿನ್‌ಗೆ "ನನಗೆ ಕೆಂಪು ಬಣ್ಣದ ಕಾರು ಬೇಕು" ಎಂದು ಹೇಳಿದರೆ, ಅದು ನಿಮಗೆ ಕೆಂಪು ಬಣ್ಣದ ಕಾರನ್ನು ತಯಾರಿಸಿ ಕೊಡುತ್ತದೆ. "ನೀಲಿ ಬಣ್ಣದ ಬೈಕ್ ಬೇಕು" ಅಂದರೆ ನೀಲಿ ಬೈಕ್ ಕೊಡುತ್ತದೆ.

ತಾಂತ್ರಿಕ ವ್ಯಾಖ್ಯಾನ (Technical Definition): Factory Function ಎಂದರೆ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್‌ಗಳನ್ನು (Objects) ಸೃಷ್ಟಿಸಿ, ನಮಗೆ ವಾಪಸ್ (return) ಕೊಡುವ ಒಂದು ಸಾಮಾನ್ಯ ಫಂಕ್ಷನ್. ಇದರಲ್ಲಿ new ಕೀವರ್ಡ್ ಬಳಸುವ ಅಗತ್ಯವಿರುವುದಿಲ್ಲ.

1. Beginner ಹಂತ (ಆರಂಭಿಕ ಹಂತ: ಆಟಿಕೆ ತಯಾರಿಸೋಣ!)

ಮೊದಲಿಗೆ ಒಂದು ಸರಳವಾದ ಆಟಿಕೆ ಮಷಿನ್ ಮಾಡೋಣ. ನಾವು ಆಟಿಕೆಯ ಹೆಸರು ಮತ್ತು ಬಣ್ಣವನ್ನು ಹೇಳಿದರೆ, ಅದು ನಮಗೆ ಹೊಸ ಆಟಿಕೆಯನ್ನು ಕೊಡಬೇಕು.

factory_beginner.js 
// ಇದು ನಮ್ಮ ಫ್ಯಾಕ್ಟರಿ (ಮಷಿನ್)
function createToy(name, color) {
  // ಇದು ಹೊಸ ಆಟಿಕೆಯನ್ನು (Object) ಬಾಕ್ಸ್‌ನಲ್ಲಿ ಹಾಕಿ ನಮಗೆ ಕೊಡುತ್ತದೆ
  return {
    toyName: name,
    toyColor: color
  };
}

// ಮಷಿನ್ ಬಳಸಿ ಆಟಿಕೆಗಳನ್ನು ತಯಾರಿಸೋಣ!
let car = createToy("ಕಾರು", "ಕೆಂಪು");
let ball = createToy("ಚೆಂಡು", "ಹಸಿರು");

console.log(car.toyName);  
console.log(ball.toyColor);
ಔಟ್‌ಪುಟ್ (Output):
ಕಾರು
ಹಸಿರು

ಉದಾಹರಣೆಯ ವಿವರಣೆ: ಇಲ್ಲಿ createToy ಎಂಬುದು ಒಂದು ಫ್ಯಾಕ್ಟರಿ. ಅದಕ್ಕೆ ನಾವು "ಕಾರು" ಮತ್ತು "ಕೆಂಪು" ಎಂದು ಹೇಳಿದ ತಕ್ಷಣ, ಅದು ಒಳಗಡೆ ಒಂದು ಹೊಸ ಬಾಕ್ಸ್ (Object {}) ಅನ್ನು ಸೃಷ್ಟಿಸಿ, ಅದರಲ್ಲಿ ಹೆಸರು ಮತ್ತು ಬಣ್ಣವನ್ನು ಹಾಕಿ ನಮಗೆ return ಮಾಡಿತು. ನಮಗೆ ಎಷ್ಟು ಬೇಕೋ ಅಷ್ಟು ಹೊಸ ಆಟಿಕೆಗಳನ್ನು ಈ ಮಷಿನ್‌ನಿಂದ ಪಡೆಯಬಹುದು!

2. Intermediate ಹಂತ (ಮಧ್ಯಮ ಹಂತ: ಮಾತನಾಡುವ ಆಟಿಕೆ!)

ಈಗ ನಮ್ಮ ಮಷಿನ್ ಅನ್ನು ಸ್ವಲ್ಪ ಅಪ್‌ಗ್ರೇಡ್ ಮಾಡೋಣ. ಬರೀ ಆಟಿಕೆ ಅಲ್ಲ, ಆ ಆಟಿಕೆ ಕೆಲಸ ಮಾಡುವಂತೆಯೂ (ಮಾತನಾಡುವಂತೆ) ಮಾಡೋಣ.

factory_intermediate.js 
function createRobot(robotName) {
  return {
    name: robotName,
    // ರೋಬೋಟ್ ಮಾತನಾಡುವ ಕೆಲಸ (Method)
    talk: function() {
      console.log("ಹಲೋ! ನನ್ನ ಹೆಸರು " + robotName);
    }
  };
}

// ಹೊಸ ರೋಬೋಟ್ ತಯಾರಿಸೋಣ
let chitti = createRobot("ಚಿಟ್ಟಿ");
let wallE = createRobot("ವಾಲ್-ಇ");

// ರೋಬೋಟ್ ಕೈಯಲ್ಲಿ ಮಾತನಾಡಿಸೋಣ
chitti.talk(); 
wallE.talk();
ಔಟ್‌ಪುಟ್ (Output):
ಹಲೋ! ನನ್ನ ಹೆಸರು ಚಿಟ್ಟಿ
ಹಲೋ! ನನ್ನ ಹೆಸರು ವಾಲ್-ಇ

ಉದಾಹರಣೆಯ ವಿವರಣೆ: ನಾವು ತಯಾರಿಸಿದ ಆಬ್ಜೆಕ್ಟ್ ಒಳಗೆ ಬರೀ ಹೆಸರು ಮಾತ್ರ ಇಲ್ಲ, talk ಎಂಬ ಒಂದು ಕೆಲಸ (ಫಂಕ್ಷನ್/method) ಕೂಡ ಇದೆ. ಚಿಟ್ಟಿ ಎಂಬ ರೋಬೋಟ್ ತಯಾರಾದ ಮೇಲೆ, ಅದಕ್ಕೆ chitti.talk() ಎಂದು ಹೇಳಿದಾಗ ಅದು ತನ್ನ ಹೆಸರನ್ನು ಹೇಳಿತು.

3. Pro ಹಂತ (ಪ್ರೊ ಹಂತ: ರಹಸ್ಯ ಕಾಪಾಡುವುದು - Data Privacy)

ಪ್ರೊಗ್ರಾಮರ್‌ಗಳು Factory Function ಅನ್ನು ಹೆಚ್ಚಾಗಿ ಬಳಸಲು ಕಾರಣವೇನೆಂದರೆ "ರಹಸ್ಯವನ್ನು ಕಾಪಾಡಲು" (Data Privacy / Closures).

ಒಂದು ಹುಂಡಿಯ (Piggy Bank) ಉದಾಹರಣೆ ತೆಗೆದುಕೊಳ್ಳೋಣ. ಹುಂಡಿಯಲ್ಲಿರುವ ಹಣವನ್ನು ಯಾರೂ ನೇರವಾಗಿ ಕದಿಯಬಾರದು, ಕೇವಲ ನಾಣ್ಯ ಹಾಕುವ ರಂಧ್ರದ ಮೂಲಕ ಮಾತ್ರ ಹಣ ಸೇರಿಸಬೇಕು ಅಲ್ಲವೇ? ಇದನ್ನೇ ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್‌ನಲ್ಲಿ ಹೇಗೆ ಮಾಡುವುದು ಎಂದು ನೋಡೋಣ.

factory_pro.js 
function createPiggyBank() {
  let secretMoney = 0; // ಇದು ರಹಸ್ಯ. ಯಾರೂ ಇದನ್ನು ಹೊರಗಿನಿಂದ ಮುಟ್ಟುವಂತಿಲ್ಲ!

  return {
    // ಹಣ ಸೇರಿಸಲು ಮಾತ್ರ ದಾರಿ
    addMoney: function(amount) {
      secretMoney = secretMoney + amount;
      console.log(amount + " ರೂ. ಜಮಾ ಆಯಿತು!");
    },
    // ಹಣ ಎಷ್ಟಿದೆ ಎಂದು ನೋಡಲು ಮಾತ್ರ ದಾರಿ
    checkBalance: function() {
      console.log("ಒಟ್ಟು ಹಣ: " + secretMoney + " ರೂ.");
    }
  };
}

let myBank = createPiggyBank();

myBank.addMoney(100); 
myBank.addMoney(50);  
myBank.checkBalance(); 

// ಯಾರಾದರೂ ಕದಿಯಲು ಪ್ರಯತ್ನಿಸಿದರೆ?
console.log(myBank.secretMoney);
ಔಟ್‌ಪುಟ್ (Output):
100 ರೂ. ಜಮಾ ಆಯಿತು!
50 ರೂ. ಜಮಾ ಆಯಿತು!
ಒಟ್ಟು ಹಣ: 150 ರೂ.
undefined (ಯಾರಿಗೂ ಸಿಗುವುದಿಲ್ಲ!)

ಉದಾಹರಣೆಯ ವಿವರಣೆ: ಇಲ್ಲಿ secretMoney ಎಂಬುದು ಫ್ಯಾಕ್ಟರಿಯ ಒಳಗಡೆಯೇ ಇರುವ ರಹಸ್ಯ. ನಾವು ವಾಪಸ್ (return) ಕೊಟ್ಟ ಆಬ್ಜೆಕ್ಟ್‌ನಲ್ಲಿ secretMoney ಅನ್ನು ಕೊಟ್ಟೇ ಇಲ್ಲ! ನಾವು ಕೊಟ್ಟಿದ್ದು ಕೇವಲ addMoney (ಹಣ ಹಾಕಲು) ಮತ್ತು checkBalance (ಹಣ ನೋಡಲು) ಮಾತ್ರ. ಇದರಿಂದಾಗಿ ನಿಮ್ಮ ಪ್ರೋಗ್ರಾಮ್‌ನಲ್ಲಿರುವ ಡೇಟಾ ಬಹಳ ಭದ್ರವಾಗಿರುತ್ತದೆ. ಇದನ್ನೇ ಪ್ರೊಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿ Encapsulation / Closure ಎನ್ನುತ್ತಾರೆ.

Factory Function ನ ಲಾಭಗಳೇನು? (Advantages)

  1. ಸುಲಭ ಮತ್ತು ಸರಳ: new ಅಥವಾ this ಎಂಬ ಗೊಂದಲಮಯ ಪದಗಳನ್ನು ಬಳಸುವ ಅಗತ್ಯವಿಲ್ಲ.
  2. ಭದ್ರತೆ (Privacy): ನಮ್ಮ ಡೇಟಾವನ್ನು ಬೇರೆಯವರು ಸುಲಭವಾಗಿ ಬದಲಾಯಿಸದಂತೆ ಬಚ್ಚಿಡಬಹುದು (Closures ಮೂಲಕ).
  3. ಮರುಬಳಕೆ (Reusability): ಒಂದೇ ಮಷಿನ್ ಬಳಸಿ ನೂರಾರು ಆಬ್ಜೆಕ್ಟ್‌ಗಳನ್ನು ಕ್ಷಣಾರ್ಧದಲ್ಲಿ ಮಾಡಬಹುದು.

ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್‌ನಲ್ಲಿ 'this' ಕೀವರ್ಡ್ ಬಳಸಬಹುದೇ?

ಹೌದು, ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್‌ನಲ್ಲಿ this ಕೀವರ್ಡ್ ಬಳಸಬಹುದು. ಆದರೆ, ಬಳಸದಿರುವುದೇ ಅತ್ಯುತ್ತಮ ಅಭ್ಯಾಸ (Best Practice)! ಏಕೆಂದರೆ, ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್ ಅನ್ನು ಬಳಸುವ ಮುಖ್ಯ ಉದ್ದೇಶವೇ this ಎಂಬ ಗೊಂದಲಮಯ ಕೀವರ್ಡ್‌ನಿಂದ ಪಾರಾಗುವುದು.

this ಕೀವರ್ಡ್ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಒಂದು "ಊಸರವಳ್ಳಿ" (Chameleon) ಇದ್ದಂತೆ. ಅದು ಎಲ್ಲಿ ಮತ್ತು ಹೇಗೆ ಕರೆಯಲ್ಪಡುತ್ತದೆ ಎಂಬುದರ ಮೇಲೆ ತನ್ನ ಅರ್ಥವನ್ನು ಬದಲಾಯಿಸುತ್ತಲೇ ಇರುತ್ತದೆ.

1. 'this' ಬಳಸಿದಾಗ ಆಗುವ ಅನಾಹುತ (The Problem)

ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್ ಒಳಗೆ this ಬಳಸಿದರೆ ಏನಾಗುತ್ತದೆ ಎಂದು ನೋಡೋಣ:

function createPlayer(name) {
  return {
    playerName: name,
    cheer: function() {
      // ಇಲ್ಲಿ this ಬಳಸಲಾಗಿದೆ
      console.log("ಕಪ್ ನಮ್ದೇ! ಎಂದರು " + this.playerName); 
    }
  };
}

let virat = createPlayer("ವಿರಾಟ್");

// ಇದು ಸರಿಯಾಗಿ ಕೆಲಸ ಮಾಡುತ್ತದೆ
virat.cheer(); // ಔಟ್‌ಪುಟ್: ಕಪ್ ನಮ್ದೇ! ಎಂದರು ವಿರಾಟ್

// ಆದರೆ ಇಲ್ಲಿ ನೋಡಿ (ಅನಾಹುತ ಶುರು!)
setTimeout(virat.cheer, 1000); 
// 1 ಸೆಕೆಂಡ್ ಆದಮೇಲೆ ಔಟ್‌ಪುಟ್: ಕಪ್ ನಮ್ದೇ! ಎಂದರು undefined

ಏನಾಯಿತು ಇಲ್ಲಿ? setTimeout ನಂತಹ ಬೇರೆ ಫಂಕ್ಷನ್‌ಗಳಿಗೆ ನಾವು ಈ cheer ಫಂಕ್ಷನ್ ಅನ್ನು ಕಳುಹಿಸಿದಾಗ, this ಕೀವರ್ಡ್ ತಾನು ಯಾವ ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ಸೇರಿದ್ದು (Context) ಎಂಬುದನ್ನು ಮರೆತುಬಿಡುತ್ತದೆ. ಆಗ this.playerName ಎನ್ನುವುದು ಸಿಗದೆ undefined ಎಂದು ಬರುತ್ತದೆ.

2. 'this' ಬಳಸದೆ ಇದ್ದಾಗ (The Solution - Pro Level)

ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್‌ನಲ್ಲಿ this ಬಳಸುವ ಬದಲು, ನೇರವಾಗಿ ಡೇಟಾವನ್ನು (Closures ಮೂಲಕ) ಬಳಸಬೇಕು.

function createPlayer(name) {
  // ಇಲ್ಲಿ name ಎಂಬುದು ಮಷಿನ್ ನೆನಪಿನಲ್ಲಿಟ್ಟುಕೊಂಡಿರುವ (Closure) ಡೇಟಾ
  return {
    cheer: function() {
      // ಇಲ್ಲಿ this ಬಳಸಿಲ್ಲ! ನೇರವಾಗಿ name ಬಳಸಿದ್ದೇವೆ
      console.log("ಕಪ್ ನಮ್ದೇ! ಎಂದರು " + name); 
    }
  };
}

let dhoni = createPlayer("ಧೋನಿ");
dhoni.cheer(); // ಔಟ್‌ಪುಟ್: ಕಪ್ ನಮ್ದೇ! ಎಂದರು ಧೋನಿ

// ಈಗ ಅನಾಹುತ ಆಗುವುದಿಲ್ಲ!
setTimeout(dhoni.cheer, 1000); 
// 1 ಸೆಕೆಂಡ್ ಆದಮೇಲೆ ಔಟ್‌ಪುಟ್: ಕಪ್ ನಮ್ದೇ! ಎಂದರು ಧೋನಿ

ಇಲ್ಲಿ ಏಕೆ ಸರಿಯಾಗಿ ಬಂತು? ನಾವು this ಬಳಸಲೇ ಇಲ್ಲ! ಫಂಕ್ಷನ್ ಒಳಗೆ ಇರುವ name ಅನ್ನು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ಕ್ಲೋಷರ್ (Closure) ಎಂಬ ಅದ್ಭುತ ಶಕ್ತಿಯು ತನ್ನ ನೆನಪಿನಲ್ಲಿ ಭದ್ರವಾಗಿ ಸೇವ್ ಮಾಡಿಟ್ಟುಕೊಂಡಿರುತ್ತದೆ. ನೀವು ಆ ಫಂಕ್ಷನ್ ಅನ್ನು ಎಲ್ಲೇ ತೆಗೆದುಕೊಂಡು ಹೋದರೂ ಅದು ತನ್ನ ಡೇಟಾವನ್ನು ಮರೆಯುವುದಿಲ್ಲ.

ಸಾರಾಂಶ: ನೀವು ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್ ಬರೆಯುತ್ತಿದ್ದೀರಿ ಎಂದರೆ this ಅನ್ನು ಮರೆತುಬಿಡಿ. ವೇರಿಯೇಬಲ್‌ಗಳನ್ನು ನೇರವಾಗಿ ಬಳಸಿ. ಇದರಿಂದ ನಿಮ್ಮ ಕೋಡ್ ಎಂದಿಗೂ ಗೊಂದಲ ಮಾಡುವುದಿಲ್ಲ ಮತ್ತು ಡೇಟಾ ಕಳೆದುಹೋಗುವುದಿಲ್ಲ!

೨. Constructor Function (ಕನ್‌ಸ್ಟ್ರಕ್ಟರ್ ಫಂಕ್ಷನ್)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ 'Constructor Function' ಎಂಬುದನ್ನು ನಾವು ಒಂದು ಆಟಿಕೆ ಕಾರ್ಖಾನೆ (Toy Factory) ಅಥವಾ ಬಿಸ್ಕತ್ ಅಚ್ಚು (Cookie Cutter) ಮೂಲಕ ಬಹಳ ಸುಲಭವಾಗಿ ಅರ್ಥಮಾಡಿಕೊಳ್ಳಬಹುದು.

ನೀವು ಮನೆಯಲ್ಲಿ ಬಿಸ್ಕತ್ ಮಾಡುವಾಗ ಒಂದು 'ಅಚ್ಚು' (mould) ಬಳಸುತ್ತೀರಿ, ಹೌದಲ್ವಾ? ಆ ಒಂದೇ ಅಚ್ಚನ್ನು ಬಳಸಿ ನೀವು ಬೇಕಾದಷ್ಟು ಬಿಸ್ಕತ್‌ಗಳನ್ನು ಮಾಡಬಹುದು. ಅದೇ ರೀತಿ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಒಂದೇ ರೀತಿಯ ಅನೇಕ 'ಆಬ್ಜೆಕ್ಟ್' (Objects) ಗಳನ್ನು ತಯಾರಿಸಲು ಬಳಸುವ 'ಅಚ್ಚು' ಅಥವಾ ಫ್ಯಾಕ್ಟರಿಯೇ ಈ Constructor Function.

1. ಬಿಗಿನರ್ ಲೆವೆಲ್ (Beginner): ನಮಗೇಕೆ ಇದು ಬೇಕು?

ಉದಾಹರಣೆಗೆ, ನೀವು 3 ಆಟದ ಕಾರುಗಳನ್ನು (Objects) ಮಾಡಬೇಕು. ಪ್ರತಿಯೊಂದಕ್ಕೂ ಹೆಸರು, ಬಣ್ಣ ಬರೆಯುತ್ತಾ ಕೂತರೆ ಬಹಳ ಸಮಯ ಹಿಡಿಯುತ್ತದೆ:

// ಅಚ್ಚು ಇಲ್ಲದೆ ಒಂದೊಂದೇ ಕಾರು ಮಾಡುವುದು (ತುಂಬಾ ಕಷ್ಟ)
let car1 = { name: "Maruti", color: "Red" };
let car2 = { name: "Tata", color: "Blue" };
let car3 = { name: "Mahindra", color: "Black" };

ಇದರ ಬದಲು ನಾವು ಒಂದು 'ಕಾರ್ ಫ್ಯಾಕ್ಟರಿ' (Constructor Function) ಯಂತ್ರವನ್ನು ಮಾಡಿದರೆ, ಕಾರಿನ ಬಣ್ಣ ಮತ್ತು ಹೆಸರು ಹೇಳಿದ ತಕ್ಷಣ ಅದು ಹೊಸ ಕಾರನ್ನು ರೆಡಿ ಮಾಡಿಕೊಡುತ್ತದೆ.

ಬರೆಯುವ ನಿಯಮಗಳು (Rules):

  • ಇದರ ಹೆಸರು ಯಾವಾಗಲೂ ದೊಡ್ಡ ಅಕ್ಷರದಿಂದ (Capital Letter) ಶುರುವಾಗಬೇಕು. (ಉದಾಹರಣೆಗೆ: Car, Robot). ಇದು ಕೇವಲ ಡೆವಲಪರ್‌ಗಳಿಗೆ ಇದು ಕನ್‌ಸ್ಟ್ರಕ್ಟರ್ ಎಂದು ತಿಳಿಸಲು ಮಾಡುವ ಕನ್ವೆನ್ಷನ್ (Convention).
  • this ಎಂಬ ಪದವನ್ನು ಬಳಸುತ್ತೇವೆ. (ಇದರ ಅರ್ಥ "ಈಗ ತಯಾರಾಗುತ್ತಿರುವ ಹೊಸ ಆಟಿಕೆ").
  • ಹೊಸ ವಸ್ತುವನ್ನು ಪಡೆಯಲು new ಎಂಬ ಪದವನ್ನು ಕಡ್ಡಾಯವಾಗಿ ಬಳಸುತ್ತೇವೆ.
ಉದಾಹರಣೆ 1: ಸರಳ ಕಾರು ಕಾರ್ಖಾನೆ
// 1. ಇದು ನಮ್ಮ ಫ್ಯಾಕ್ಟರಿ (Constructor Function)
function Car(ಹೆಸರು, ಬಣ್ಣ) {
    this.name = ಹೆಸರು;
    this.color = ಬಣ್ಣ;
}

// 2. new ಪದ ಬಳಸಿ ಕಾರ್ಖಾನೆಗೆ ಆರ್ಡರ್ ಕೊಡುವುದು!
let myFirstCar = new Car("Maruti", "Red");
let mySecondCar = new Car("Tata", "Blue");

console.log(myFirstCar.name); // ಔಟ್‌ಪುಟ್: Maruti
console.log(mySecondCar.color); // ಔಟ್‌ಪುಟ್: Blue

ಇಲ್ಲಿ new Car(...) ಎಂದು ಹೇಳಿದಾಗ, ಆ ಫ್ಯಾಕ್ಟರಿ ತಕ್ಷಣವೇ ಒಂದು ಹೊಸ ಕಾರನ್ನು ತಯಾರಿಸಿ ಕೊಡುತ್ತದೆ.

2. ಇಂಟರ್ಮೀಡಿಯೇಟ್ ಲೆವೆಲ್ (Intermediate): ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ಕೆಲಸ ಕೊಡುವುದು

ಕೇವಲ ಬಣ್ಣ, ಹೆಸರು ಮಾತ್ರವಲ್ಲದೆ ಆ ಹೊಸ ಕಾರು ಅಥವಾ ಆಟಿಕೆ ಕೆಲಸ ಮಾಡುವಂತೆ (Methods/Functions) ಮಾಡಬಹುದು. ಉದಾಹರಣೆಗೆ, ನಮ್ಮ ಕಾರು ಹಾರ್ನ್ (Horn) ಮಾಡುವಂತೆ ಮಾಡೋಣ.

function SmartCar(name, color) {
    this.name = name;
    this.color = color;
    
    // ಕಾರು ಹಾರ್ನ್ ಮಾಡುವ ಫಂಕ್ಷನ್
    this.soundHorn = function() {
        console.log(this.name + " ಕಾರು 'ಪೀಪ್ ಪೀಪ್' ಅನ್ನುತ್ತಿದೆ!");
    }
}

let tataCar = new SmartCar("Tata Nexon", "White");
tataCar.soundHorn(); 
// ಔಟ್‌ಪುಟ್: Tata Nexon ಕಾರು 'ಪೀಪ್ ಪೀಪ್' ಅನ್ನುತ್ತಿದೆ!

3. ಪ್ರೊ ಲೆವೆಲ್ (Pro Level): Prototype ಬಳಸಿ ಮೆಮೊರಿ ಉಳಿಸುವುದು

ಮೇಲಿನ SmartCar ಉದಾಹರಣೆಯಲ್ಲಿ ಒಂದು ಸಣ್ಣ ಸಮಸ್ಯೆಯಿದೆ. ನೀವು 1000 ಕಾರುಗಳನ್ನು ತಯಾರಿಸಿದರೆ, soundHorn ಎಂಬ ಫಂಕ್ಷನ್ ಕೂಡ 1000 ಬಾರಿ ಕಾಪಿ (Copy) ಆಗುತ್ತದೆ. ಪ್ರತಿಯೊಂದು ಆಬ್ಜೆಕ್ಟ್ ತನ್ನದೇ ಆದ ಫಂಕ್ಷನ್ ಮೆಮೊರಿ ತೆಗೆದುಕೊಳ್ಳುತ್ತದೆ. ಇದರಿಂದ ಕಂಪ್ಯೂಟರ್ ಮೆಮೊರಿ ವೇಸ್ಟ್ ಆಗುತ್ತದೆ.

ಇದನ್ನು ತಪ್ಪಿಸಲು ಪ್ರೊ (Pro) ಡೆವಲಪರ್‌ಗಳು prototype ಬಳಸುತ್ತಾರೆ. prototype ಅಂದರೆ ಊರಿನ ಲೈಬ್ರರಿ ಇದ್ದ ಹಾಗೆ. ಪ್ರತಿಯೊಬ್ಬರಿಗೂ ಒಂದೊಂದು ಪುಸ್ತಕ ಕೊಡುವ ಬದಲು, ಒಂದು ಪುಸ್ತಕವನ್ನು ಲೈಬ್ರರಿಯಲ್ಲಿ ಇಟ್ಟರೆ ಎಲ್ಲರೂ ಓದಬಹುದು!

function ProCar(name, color) {
    this.name = name;
    this.color = color;
    // ಇಲ್ಲಿ ಫಂಕ್ಷನ್ ಬರೆಯುವುದಿಲ್ಲ!
}

// ಎಲ್ಲ ಕಾರುಗಳಿಗೂ ಒಂದೇ ಜಾಗದಲ್ಲಿ (prototype ನಲ್ಲಿ) ನಿಯಮ ಬರೆಯುವುದು
ProCar.prototype.soundHorn = function() {
     console.log(this.name + " ಕಾರು 'ಪೀಪ್ ಪೀಪ್' ಅನ್ನುತ್ತಿದೆ!");
}

let carA = new ProCar("Mahindra Thar", "Black");
let carB = new ProCar("Kia", "Red");

// ಎರಡು ಕಾರುಗಳು ಲೈಬ್ರರಿಯಿಂದ (prototype) ಒಂದೇ ಫಂಕ್ಷನ್ ಅನ್ನು ಬಳಸುತ್ತವೆ
carA.soundHorn();
carB.soundHorn();

ಹೀಗೆ ಮಾಡುವುದರಿಂದ ಎಷ್ಟೇ ಕಾರುಗಳನ್ನು ತಯಾರಿಸಿದರೂ soundHorn ಫಂಕ್ಷನ್ ಮೆಮೊರಿಯಲ್ಲಿ ಒಂದೇ ಬಾರಿ ಸೇವ್ ಆಗಿರುತ್ತದೆ.

Advanced Concept: 'new' ಮರೆತರೆ ಏನಾಗುತ್ತದೆ?

ಒಂದು ವೇಳೆ ನೀವು ಕನ್‌ಸ್ಟ್ರಕ್ಟರ್ ಫಂಕ್ಷನ್ ಅನ್ನು ಕರೆಯುವಾಗ new ಕೀವರ್ಡ್ ಮರೆತರೆ, this ಗ್ಲೋಬಲ್ ವಿಂಡೋ ಆಬ್ಜೆಕ್ಟ್‌ಗೆ (Global Window Object) ಪಾಯಿಂಟ್ ಮಾಡುತ್ತದೆ. ಇದರಿಂದ ನಿಮ್ಮ ಡೇಟಾ ಗ್ಲೋಬಲ್ ಸ್ಕೋಪ್‌ಗೆ ಸೇರಿ ಅನಾಹುತವಾಗುತ್ತದೆ ಮತ್ತು ಫಂಕ್ಷನ್ undefined ಅನ್ನು ರಿಟರ್ನ್ ಮಾಡುತ್ತದೆ!

ಸಾರಾಂಶ (ನೆನಪಿಡಬೇಕಾದ ಮುಖ್ಯ ಅಂಶಗಳು):

  • Constructor Function: ಒಂದೇ ರೀತಿಯ ಅನೇಕ ಆಬ್ಜೆಕ್ಟ್ ಮಾಡಲು ಬಳಸುವ ಅಚ್ಚು.
  • Capital Letter: ಇದರ ಹೆಸರು ಯಾವಾಗಲೂ ದೊಡ್ಡ ಅಕ್ಷರದಿಂದ ಶುರುವಾಗಬೇಕು.
  • new Keyword: ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಹುಟ್ಟುಹಾಕಲು ಕಡ್ಡಾಯವಾಗಿ new ಬಳಸಬೇಕು.
  • this Keyword: ತಯಾರಾಗುತ್ತಿರುವ ಪ್ರಸ್ತುತ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಸೂಚಿಸುತ್ತದೆ.
  • prototype: ಮೆಮೊರಿ ಉಳಿಸಲು ಕಾಮನ್ ಆಗಿರುವ ಫಂಕ್ಷನ್‌ಗಳನ್ನು prototype ಒಳಗೆ ಹಾಕಬೇಕು.

೩. 'new' ಮತ್ತು 'this': ಒಂದು ಸರಳ ಪರಿಚಯ

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ನಾವು ಹೊಸ ವಸ್ತುಗಳನ್ನು (Objects) ತಯಾರಿಸುವಾಗ new ಮತ್ತು this ಕೀವರ್ಡ್‌ಗಳನ್ನು ಬಳಸುತ್ತೇವೆ. ಇವುಗಳ ಕೆಲಸವನ್ನು ಶಾಲೆಯ 'ಪುಸ್ತಕ'ದ ಉದಾಹರಣೆಯೊಂದಿಗೆ ಅರ್ಥಮಾಡಿಕೊಳ್ಳೋಣ.

🔹 new (ಹೊಸ ಖಾಲಿ ಪುಸ್ತಕ ತರುವುದು)

ದೈನಂದಿನ ಉದಾಹರಣೆ: ನೀವು ಅಂಗಡಿಗೆ ಹೋಗಿ "ನನಗೆ ಒಂದು ಹೊಸ ಪುಸ್ತಕ ಕೊಡಿ" ಎಂದು ಕೇಳಿದಾಗ, ಅಂಗಡಿಯವನು ನಿಮಗೆ ಒಂದು ಹೊಸ, ಖಾಲಿ ಪುಟಗಳಿರುವ ಪುಸ್ತಕವನ್ನು ಕೊಡುತ್ತಾನೆ.

ತಾಂತ್ರಿಕ ವಿವರಣೆ: ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ನೀವು new ಎಂದು ಬರೆದರೆ, ಕಂಪ್ಯೂಟರ್‌ನ ಮೆಮೊರಿಯಲ್ಲಿ ಒಂದು ಹೊಚ್ಚಹೊಸ "ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್ (Empty Object {})" ಸೃಷ್ಟಿಯಾಗುತ್ತದೆ.

🔹 this (ಪುಸ್ತಕದ ಮೇಲೆ 'ನನ್ನ' ಹೆಸರು ಬರೆಯುವುದು)

ದೈನಂದಿನ ಉದಾಹರಣೆ: ಮನೆಗೆ ತಂದ ಆ ಹೊಸ ಪುಸ್ತಕದ ಮೇಲಿರುವ ಚೀಟಿಯಲ್ಲಿ (Label) ನೀವು "ಈ ಪುಸ್ತಕ ನನ್ನದು, ನನ್ನ ಹೆಸರು ರಾಜು" ಎಂದು ಬರೆಯುತ್ತೀರಿ.

ತಾಂತ್ರಿಕ ವಿವರಣೆ: ಮೆಮೊರಿಯಲ್ಲಿ ಸೃಷ್ಟಿಯಾದ ಆ ಹೊಸ ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ಡೇಟಾವನ್ನು (Properties) ತುಂಬಲು this ಬಳಸಲಾಗುತ್ತದೆ. ಅಂದರೆ, ಆ ಆಬ್ಜೆಕ್ಟ್ ತನ್ನ ಸ್ವಂತ ಮಾಹಿತಿಯನ್ನು ಸೇರಿಸಿಕೊಳ್ಳಲು this ಅನ್ನು ಬಳಸುತ್ತದೆ.

ಕಂಪ್ಯೂಟರ್‌ನ ಒಳಗೆ ಏನಾಗುತ್ತದೆ? (The 4 Technical Steps)

ನೀವು new ಕೀವರ್ಡ್ ಬಳಸಿ ಫಂಕ್ಷನ್ ಅನ್ನು ಕರೆದಾಗ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಎಂಜಿನ್ (ಉದಾಹರಣೆಗೆ: V8 Engine) ಹಿನ್ನೆಲೆಯಲ್ಲಿ ಈ ಕೆಳಗಿನ 4 ಪ್ರಮುಖ ಕೆಲಸಗಳನ್ನು ಮಾಡುತ್ತದೆ:

  1. ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿ (Create Object): ಮೆಮೊರಿಯಲ್ಲಿ ಒಂದು ಹೊಸ ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್ ({}) ಸೃಷ್ಟಿಸುತ್ತದೆ. (ಇದೇ ನಿಮ್ಮ ಹೊಸ ಖಾಲಿ ಪುಸ್ತಕ)
  2. ಪ್ರೋಟೋಟೈಪ್ ಲಿಂಕ್ (Prototype Linking): ಆ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಕನ್‌ಸ್ಟ್ರಕ್ಟರ್ ಫಂಕ್ಷನ್‌ನ prototype ಗೆ ಲಿಂಕ್ ಮಾಡುತ್ತದೆ. (ಇದರಿಂದ ಮೆಮೊರಿ ಉಳಿಯುತ್ತದೆ).
  3. this ಬೈಂಡಿಂಗ್ (Binding this): ಫಂಕ್ಷನ್‌ನ ಒಳಗಿರುವ this ಪದವನ್ನು, ತಾನು ಈಗಷ್ಟೇ ಸೃಷ್ಟಿಸಿದ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ಪಾಯಿಂಟ್ (Bind) ಮಾಡುತ್ತದೆ. (ಪುಸ್ತಕದ ಮೇಲೆ ಹೆಸರು ಬರೆಯುವ ಹಂತ).
  4. ಆಟೋಮ್ಯಾಟಿಕ್ ರಿಟರ್ನ್ (Implicit Return): ನಾವು ಕೋಡ್‌ನಲ್ಲಿ return ಎಂದು ಬರೆಯದಿದ್ದರೂ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಆ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ತಾನಾಗಿಯೇ ನಮಗೆ ವಾಪಸ್ ಕೊಡುತ್ತದೆ.

ಎರಡನ್ನೂ ಒಟ್ಟುಗೂಡಿಸಿ ಕೋಡ್‌ನಲ್ಲಿ ನೋಡೋಣ

ಈಗ ಸರಳ ಕಥೆ ಮತ್ತು ತಾಂತ್ರಿಕ ಪ್ರಕ್ರಿಯೆ ಎರಡೂ ಕೋಡ್‌ನಲ್ಲಿ ಹೇಗೆ ಜೊತೆಯಾಗಿ ಕೆಲಸ ಮಾಡುತ್ತವೆ ಎಂದು ನೋಡಿ:

// ಇದು ನಮ್ಮ ಕನ್‌ಸ್ಟ್ರಕ್ಟರ್ ಫಂಕ್ಷನ್ (ಪುಸ್ತಕ ತಯಾರಿಸುವ ಫ್ಯಾಕ್ಟರಿ)
function Student(ಹೆಸರು, ತರಗತಿ) {
    
    // ತಾಂತ್ರಿಕವಾಗಿ: ನೀವು 'new' ಬಳಸಿದಾಗ ಹಿನ್ನೆಲೆಯಲ್ಲಿ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ 1ನೇ ಕೆಲಸ ಮಾಡುತ್ತದೆ:
    // this = {};  <-- (ಹೊಸ ಖಾಲಿ ಪುಸ್ತಕ / ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಯಾಯಿತು)
    
    // ಈಗ 'this' ಆ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ಡೇಟಾ ಸೇರಿಸುತ್ತಿದೆ:
    // (2ನೇ ಕೆಲಸ: ಪುಸ್ತಕದ ಮೇಲೆ ಹೆಸರು ಮತ್ತು ತರಗತಿ ಬರೆಯುವುದು)
    this.name = ಹೆಸರು;      
    this.grade = ತರಗತಿ;

    // ತಾಂತ್ರಿಕವಾಗಿ: ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ 4ನೇ ಕೆಲಸವಾಗಿ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ರಿಟರ್ನ್ ಮಾಡುತ್ತದೆ:
    // return this; <-- (ನಿಮ್ಮ ಕೈಗೆ ತುಂಬಿದ ಪುಸ್ತಕ ಸಿಕ್ಕಿತು)
}

// 'new' ಪದ ಬಳಸಿ ಹೊಸ Student ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸುತ್ತಿದ್ದೇವೆ
let student1 = new Student("ರಾಜು", "10ನೇ ತರಗತಿ"); 
let student2 = new Student("ರಮ್ಯ", "9ನೇ ತರಗತಿ");

// ಈಗ ಪ್ರಿಂಟ್ ಮಾಡಿ ನೋಡೋಣ
console.log(student1.name); // ಔಟ್‌ಪುಟ್: ರಾಜು
console.log(student2.name); // ಔಟ್‌ಪುಟ್: ರಮ್ಯ

ಅಂತಿಮ ಸಾರಾಂಶ (Conclusion)

  • new ಎಂದರೆ: "ಮೆಮೊರಿಯಲ್ಲಿ ಒಂದು ಹೊಸ, ಖಾಲಿ ಜಾಗವನ್ನು ({}) ಕೊಡು" ಎನ್ನುವ ಕಮಾಂಡ್.
  • this ಎಂದರೆ: "ಆ ಹೊಸ ಜಾಗದ ಒಳಗೆ ಈ ಕೆಳಗಿನ ಡೇಟಾವನ್ನು ತುಂಬು" ಎಂದು ಹೇಳುವ ಪಾಯಿಂಟರ್.

ಇವೆರಡೂ ಒಟ್ಟಿಗೆ ಸೇರಿದರೆ, ನಮಗೆ ಬೇಕಾದ ನಿಖರವಾದ ಮಾಹಿತಿಯನ್ನು ಹೊಂದಿರುವ ಒಂದು ಸಂಪೂರ್ಣ 'ಆಬ್ಜೆಕ್ಟ್' (Object) ತಯಾರಾಗುತ್ತದೆ.

೪. ವ್ಯತ್ಯಾಸಗಳು: Factory vs Constructor

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Factory Function ಮತ್ತು Constructor Function ಇವೆರಡರ ಮುಖ್ಯ ಕೆಲಸ ಒಂದೇ: "ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ (Object) ಗಳನ್ನು ಸೃಷ್ಟಿಸುವುದು".

ಆದರೆ ಅವು ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಸೃಷ್ಟಿಸುವ 'ವಿಧಾನ' ಬೇರೆ. ಇವೆರಡರ ವ್ಯತ್ಯಾಸವನ್ನು ಸರಳವಾಗಿ ಅರ್ಥಮಾಡಿಕೊಳ್ಳೋಣ.

ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್

ಇದನ್ನು ಒಂದು "ಸಾಮಾನ್ಯ ಕೆಲಸಗಾರ"ನಿಗೆ ಹೋಲಿಸಬಹುದು. ನೀವು ಒಂದು ಗೊಂಬೆ ಕೇಳಿದರೆ, ಆತನೇ ಒಂದು ಖಾಲಿ ಡಬ್ಬ ತಂದು, ಅದಕ್ಕೆ ಬಣ್ಣ ಬಳಿದು, ಕೊನೆಗೆ ಆ ಡಬ್ಬವನ್ನು ನಿಮ್ಮ ಕೈಗೆ ಕೊಡುತ್ತಾನೆ.

ಇಲ್ಲಿ ಯಾವುದೇ 'ಮ್ಯಾಜಿಕ್' ಇರುವುದಿಲ್ಲ. ನಾವೇ ಒಂದು ಆಬ್ಜೆಕ್ಟ್ ({}) ಅನ್ನು ಸೃಷ್ಟಿಸಿ, ಅದಕ್ಕೆ ಡೇಟಾ ತುಂಬಿ, ಕೊನೆಗೆ ನಾವೇ ಕಡ್ಡಾಯವಾಗಿ return ಮಾಡಬೇಕು.

// Factory Function ಉದಾಹರಣೆ
function createRobot(ಹೆಸರು, ಬಣ್ಣ) {
    // 1. ನಾವೇ ಒಂದು ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸುತ್ತಿದ್ದೇವೆ
    let robot = {}; 
    
    // 2. ಡೇಟಾ ತುಂಬುತ್ತಿದ್ದೇವೆ (this ಬಳಸುವಂತಿಲ್ಲ)
    robot.name = ಹೆಸರು;
    robot.color = ಬಣ್ಣ;
    
    // 3. ನಾವೇ ಕಡ್ಡಾಯವಾಗಿ ರಿಟರ್ನ್ ಮಾಡುತ್ತಿದ್ದೇವೆ
    return robot; 
}

// 'new' ಪದ ಇಲ್ಲದೆ ಕರೆಯುವುದು
let robot1 = createRobot("ಚಿಟ್ಟಿ", "ಬೆಳ್ಳಿ"); 
console.log(robot1.name); // ಔಟ್‌ಪುಟ್: ಚಿಟ್ಟಿ

ನೆನಪಿಡಬೇಕಾದದ್ದು: ಇಲ್ಲಿ new ಬೇಡ, this ಬೇಡ, ಆದರೆ return ಕಡ್ಡಾಯ!

ಕನ್‌ಸ್ಟ್ರಕ್ಟರ್ ಫಂಕ್ಷನ್

ಇದನ್ನು ಒಂದು "ಆಟೋಮ್ಯಾಟಿಕ್ ಮ್ಯಾಜಿಕ್ ಮಷಿನ್"ಗೆ ಹೋಲಿಸಬಹುದು. ನಾವು ಈ ಹಿಂದೆ ಕಲಿತಂತೆ, ಇಲ್ಲಿ new ಎಂಬ ಮ್ಯಾಜಿಕ್ ದಂಡವನ್ನು ಬಳಸುತ್ತೇವೆ. ಆ ಮ್ಯಾಜಿಕ್ ದಂಡವೇ ಗುಪ್ತವಾಗಿ ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸಿ, ಕೊನೆಗೆ ತಾನೇ ವಾಪಸ್ (return) ಕೊಡುತ್ತದೆ.


// Constructor Function ಉದಾಹರಣೆ (ಹೆಸರು ಕ್ಯಾಪಿಟಲ್ ಇರಬೇಕು)
function Robot(ಹೆಸರು, ಬಣ್ಣ) {
    // ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಗುಪ್ತವಾಗಿ ಸೃಷ್ಟಿಯಾಗಿದೆ: this = {}
    
    // ನಾವಿಲ್ಲಿ 'this' ಬಳಸುತ್ತೇವೆ
    this.name = ಹೆಸರು;
    this.color = ಬಣ್ಣ;
    
    // ರಿಟರ್ನ್ ಬರೆಯುವ ಅಗತ್ಯವಿಲ್ಲ! (ಗುಪ್ತವಾಗಿ: return this;)
}

// ಕಡ್ಡಾಯವಾಗಿ 'new' ಪದ ಬಳಸಬೇಕು
let robot2 = new Robot("ಟರ್ಮಿನೇಟರ್", "ಕಪ್ಪು");
console.log(robot2.name); // ಔಟ್‌ಪುಟ್: ಟರ್ಮಿನೇಟರ್

ನೆನಪಿಡಬೇಕಾದದ್ದು: ಇಲ್ಲಿ new ಮತ್ತು this ಕಡ್ಡಾಯ, ಆದರೆ return ಬರೆಯುವ ಅಗತ್ಯವಿಲ್ಲ!

ಮುಖಾಮುಖಿ ಹೋಲಿಕೆ (Comparison Table)

ವ್ಯತ್ಯಾಸ (Difference) ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್ (Factory Function) ಕನ್‌ಸ್ಟ್ರಕ್ಟರ್ ಫಂಕ್ಷನ್ (Constructor Function)
ಫಂಕ್ಷನ್ ಹೆಸರು ಸಣ್ಣ ಅಕ್ಷರದಿಂದ ಶುರುವಾಗುತ್ತದೆ (ಉದಾ: createRobot) ದೊಡ್ಡ ಅಕ್ಷರದಿಂದ ಶುರುವಾಗುತ್ತದೆ (ಉದಾ: Robot)
new ಕೀವರ್ಡ್ ಬಳಸುವುದಿಲ್ಲ ❌ ಕಡ್ಡಾಯವಾಗಿ ಬಳಸಬೇಕು ✅
this ಕೀವರ್ಡ್ ಬಳಸುವುದಿಲ್ಲ ❌ ಕಡ್ಡಾಯವಾಗಿ ಬಳಸಬೇಕು ✅
return ಕೀವರ್ಡ್ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ನಾವೇ ಕಡ್ಡಾಯವಾಗಿ return ಮಾಡಬೇಕು ಬರೆಯುವ ಅಗತ್ಯವಿಲ್ಲ (ತಾನೇ ರಿಟರ್ನ್ ಆಗುತ್ತದೆ)
ಯಾವುದು ಸುಲಭ? ಆರಂಭಿಕರಿಗೆ ಅರ್ಥಮಾಡಿಕೊಳ್ಳಲು ತುಂಬಾ ಸುಲಭ. ಸ್ವಲ್ಪ ಕನ್ಫ್ಯೂಸ್ ಆಗಬಹುದು, ಆದರೆ ಪ್ರೊ ಡೆವಲಪರ್‌ಗಳು ಇದನ್ನೇ ಹೆಚ್ಚು ಬಳಸುತ್ತಾರೆ.
ಸಾರಾಂಶ: ನಿಮಗೆ new ಮತ್ತು this ನ ತಲೆನೋವು ಬೇಡ, ಎಲ್ಲವನ್ನೂ ನಾನೇ ಮ್ಯಾನ್ಯುವಲ್ (Manual) ಆಗಿ ಬರೆಯುತ್ತೇನೆ ಎಂದರೆ Factory Function ಬಳಸಿ. ಇಲ್ಲ, "ಕಂಪ್ಯೂಟರ್ ತಾನೇ ಆಟೋಮ್ಯಾಟಿಕ್ ಆಗಿ ಆಬ್ಜೆಕ್ಟ್ ಕ್ರಿಯೇಟ್ ಮಾಡಿ ರಿಟರ್ನ್ ಮಾಡಲಿ, ನನ್ನ ಕೋಡ್ ಚಿಕ್ಕದಾಗಲಿ" ಎಂದರೆ Constructor Function ಬಳಸಿ!

Modern JavaScript Update (ES6 Classes)

ಇಂದು ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Constructor Function ಬರೆಯುವ ಬದಲು class ಕೀವರ್ಡ್ ಅನ್ನು ಹೆಚ್ಚಾಗಿ ಬಳಸುತ್ತಾರೆ. ಇದು ಕೇವಲ ಬರೆಯುವ ಶೈಲಿ (Syntactic Sugar) ಅಷ್ಟೇ, ಒಳಗಡೆ ಕೆಲಸ ಮಾಡುವುದು ಇದೇ ಕನ್‌ಸ್ಟ್ರಕ್ಟರ್ ನಿಯಮಗಳೇ!

೫. ಅಭ್ಯಾಸ (Practice Quiz)

1. Factory Function ನಲ್ಲಿ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ನಾವೇ 'return' ಮಾಡಬೇಕೇ?
  • A. ಹೌದು, ಕಡ್ಡಾಯವಾಗಿ ಮಾಡಬೇಕು.
  • B. ಇಲ್ಲ, ಅದು ತಾನಾಗಿಯೇ ರಿಟರ್ನ್ ಆಗುತ್ತದೆ.
ಸರಿಯಾದ ಉತ್ತರ: A. ಹೌದು, ಕಡ್ಡಾಯವಾಗಿ ಮಾಡಬೇಕು. Factory Function ನಲ್ಲಿ ಆಟೋಮ್ಯಾಟಿಕ್ ರಿಟರ್ನ್ ಇರುವುದಿಲ್ಲ.
2. Constructor Function ನ ಹೆಸರು ಯಾವ ಅಕ್ಷರದಿಂದ ಪ್ರಾರಂಭವಾಗಬೇಕು (Best Practice)?
  • A. ಸಣ್ಣ ಅಕ್ಷರದಿಂದ (lowercase)
  • B. ದೊಡ್ಡ ಅಕ್ಷರದಿಂದ (Capital Letter)
ಸರಿಯಾದ ಉತ್ತರ: B. ದೊಡ್ಡ ಅಕ್ಷರದಿಂದ. ಡೆವಲಪರ್‌ಗಳಿಗೆ ಇದೊಂದು ಕನ್‌ಸ್ಟ್ರಕ್ಟರ್ ಎಂದು ತಿಳಿಸಲು Capital letter ಬಳಸಲಾಗುತ್ತದೆ.

Original Notes — ಸಂಪೂರ್ಣ ಮೂಲ ನೋಟ್ಸ್

ಮಾಹಿತಿ ನಷ್ಟವಾಗದಿರಲು ನಿಮ್ಮ ಮೂಲ ನೋಟ್ಸ್‌ನ ಯಥಾವತ್ತಾದ ಪಠ್ಯವನ್ನು (Raw Text) ಇಲ್ಲಿ ಕಾಯ್ದಿರಿಸಲಾಗಿದೆ.

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Factory Function (ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್) ಎನ್ನುವುದನ್ನು ಅರ್ಥಮಾಡಿಕೊಳ್ಳುವುದು ಬಹಳ ಸುಲಭ. ನೀವು ಇದನ್ನು ಒಂದು "ಆಟಿಕೆ ತಯಾರಿಸುವ ಮಷಿನ್" (Toy Factory) ಎಂದು ಊಹಿಸಿಕೊಳ್ಳಿ.

ನೀವು ಮಷಿನ್‌ಗೆ "ನನಗೆ ಕೆಂಪು ಬಣ್ಣದ ಕಾರು ಬೇಕು" ಎಂದು ಹೇಳಿದರೆ, ಅದು ನಿಮಗೆ ಕೆಂಪು ಬಣ್ಣದ ಕಾರನ್ನು ತಯಾರಿಸಿ ಕೊಡುತ್ತದೆ. "ನೀಲಿ ಬಣ್ಣದ ಬೈಕ್ ಬೇಕು" ಅಂದರೆ ನೀಲಿ ಬೈಕ್ ಕೊಡುತ್ತದೆ.

ಪ್ರೋಗ್ರಾಮಿಂಗ್ ಭಾಷೆಯಲ್ಲಿ: Factory Function ಎಂದರೆ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್‌ಗಳನ್ನು (Objects) ತಯಾರಿಸಿ, ನಮಗೆ ವಾಪಸ್ (return) ಕೊಡುವ ಒಂದು ಫಂಕ್ಷನ್.

ಇದನ್ನು 3 ಹಂತಗಳಲ್ಲಿ (Beginner, Intermediate, Pro) ಉದಾಹರಣೆಯೊಂದಿಗೆ ಕಲಿಯೋಣ ಬನ್ನಿ!


1. Beginner ಹಂತ (ಆರಂಭಿಕ ಹಂತ: ಆಟಿಕೆ ತಯಾರಿಸೋಣ!)
ಮೊದಲಿಗೆ ಒಂದು ಸರಳವಾದ ಆಟಿಕೆ ಮಷಿನ್ ಮಾಡೋಣ. ನಾವು ಆಟಿಕೆಯ ಹೆಸರು ಮತ್ತು ಬಣ್ಣವನ್ನು ಹೇಳಿದರೆ, ಅದು ನಮಗೆ ಹೊಸ ಆಟಿಕೆಯನ್ನು ಕೊಡಬೇಕು.

ಉದಾಹರಣೆ ಕೋಡ್:


JavaScript
// ಇದು ನಮ್ಮ ಫ್ಯಾಕ್ಟರಿ (ಮಷಿನ್)
function createToy(name, color) {
  // ಇದು ಹೊಸ ಆಟಿಕೆಯನ್ನು (Object) ಬಾಕ್ಸ್‌ನಲ್ಲಿ ಹಾಕಿ ನಮಗೆ ಕೊಡುತ್ತದೆ
  return {
    toyName: name,
    toyColor: color
  };
}

// ಮಷಿನ್ ಬಳಸಿ ಆಟಿಕೆಗಳನ್ನು ತಯಾರಿಸೋಣ!
let car = createToy("ಕಾರು", "ಕೆಂಪು");
let ball = createToy("ಚೆಂಡು", "ಹಸಿರು");

console.log(car.toyName);  // ಔಟ್‌ಪುಟ್: ಕಾರು
console.log(ball.toyColor); // ಔಟ್‌ಪುಟ್: ಹಸಿರು
ಉದಾಹರಣೆಯ ವಿವರಣೆ: ಇಲ್ಲಿ createToy ಎಂಬುದು ಒಂದು ಫ್ಯಾಕ್ಟರಿ. ಅದಕ್ಕೆ ನಾವು "ಕಾರು" ಮತ್ತು "ಕೆಂಪು" ಎಂದು ಹೇಳಿದ ತಕ್ಷಣ, ಅದು ಒಳಗಡೆ ಒಂದು ಹೊಸ ಬಾಕ್ಸ್ (Object {}) ಅನ್ನು ಸೃಷ್ಟಿಸಿ, ಅದರಲ್ಲಿ ಹೆಸರು ಮತ್ತು ಬಣ್ಣವನ್ನು ಹಾಕಿ ನಮಗೆ return ಮಾಡಿತು. ನಮಗೆ ಎಷ್ಟು ಬೇಕೋ ಅಷ್ಟು ಹೊಸ ಆಟಿಕೆಗಳನ್ನು ಈ ಮಷಿನ್‌ನಿಂದ ಪಡೆಯಬಹುದು!


2. Intermediate ಹಂತ (ಮಧ್ಯಮ ಹಂತ: ಮಾತನಾಡುವ ಆಟಿಕೆ!)
ಈಗ ನಮ್ಮ ಮಷಿನ್ ಅನ್ನು ಸ್ವಲ್ಪ ಅಪ್‌ಗ್ರೇಡ್ ಮಾಡೋಣ. ಬರೀ ಆಟಿಕೆ ಅಲ್ಲ, ಆ ಆಟಿಕೆ ಕೆಲಸ ಮಾಡುವಂತೆಯೂ (ಮಾತನಾಡುವಂತೆ) ಮಾಡೋಣ.

ಉದಾಹರಣೆ ಕೋಡ್:


JavaScript
function createRobot(robotName) {
  return {
    name: robotName,
    // ರೋಬೋಟ್ ಮಾತನಾಡುವ ಕೆಲಸ (Method)
    talk: function() {
      console.log("ಹಲೋ! ನನ್ನ ಹೆಸರು " + robotName);
    }
  };
}

// ಹೊಸ ರೋಬೋಟ್ ತಯಾರಿಸೋಣ
let chitti = createRobot("ಚಿಟ್ಟಿ");
let wallE = createRobot("ವಾಲ್-ಇ");

// ರೋಬೋಟ್ ಕೈಯಲ್ಲಿ ಮಾತನಾಡಿಸೋಣ
chitti.talk(); // ಔಟ್‌ಪುಟ್: ಹಲೋ! ನನ್ನ ಹೆಸರು ಚಿಟ್ಟಿ
wallE.talk();  // ಔಟ್‌ಪುಟ್: ಹಲೋ! ನನ್ನ ಹೆಸರು ವಾಲ್-ಇ
ಉದಾಹರಣೆಯ ವಿವರಣೆ: ನಾವು ತಯಾರಿಸಿದ ಆಬ್ಜೆಕ್ಟ್ ಒಳಗೆ ಬರೀ ಹೆಸರು ಮಾತ್ರ ಇಲ್ಲ, talk ಎಂಬ ಒಂದು ಕೆಲಸ (ಫಂಕ್ಷನ್/method) ಕೂಡ ಇದೆ. ಚಿಟ್ಟಿ ಎಂಬ ರೋಬೋಟ್ ತಯಾರಾದ ಮೇಲೆ, ಅದಕ್ಕೆ chitti.talk() ಎಂದು ಹೇಳಿದಾಗ ಅದು ತನ್ನ ಹೆಸರನ್ನು ಹೇಳಿತು.


3. Pro ಹಂತ (ಪ್ರೊ ಹಂತ: ರಹಸ್ಯ ಕಾಪಾಡುವುದು - Data Privacy)
ಪ್ರೊಗ್ರಾಮರ್‌ಗಳು Factory Function ಅನ್ನು ಹೆಚ್ಚಾಗಿ ಬಳಸಲು ಕಾರಣವೇನೆಂದರೆ "ರಹಸ್ಯವನ್ನು ಕಾಪಾಡಲು" (Data Privacy / Closures).

ಒಂದು ಹುಂಡಿಯ (Piggy Bank) ಉದಾಹರಣೆ ತೆಗೆದುಕೊಳ್ಳೋಣ. ಹುಂಡಿಯಲ್ಲಿರುವ ಹಣವನ್ನು ಯಾರೂ ನೇರವಾಗಿ ಕದಿಯಬಾರದು, ಕೇವಲ ನಾಣ್ಯ ಹಾಕುವ ರಂಧ್ರದ ಮೂಲಕ ಮಾತ್ರ ಹಣ ಸೇರಿಸಬೇಕು ಅಲ್ಲವೇ? ಇದನ್ನೇ ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್‌ನಲ್ಲಿ ಹೇಗೆ ಮಾಡುವುದು ಎಂದು ನೋಡೋಣ.

ಉದಾಹರಣೆ ಕೋಡ್:


JavaScript
function createPiggyBank() {
  let secretMoney = 0; // ಇದು ರಹಸ್ಯ. ಯಾರೂ ಇದನ್ನು ಹೊರಗಿನಿಂದ ಮುಟ್ಟುವಂತಿಲ್ಲ!

  return {
    // ಹಣ ಸೇರಿಸಲು ಮಾತ್ರ ದಾರಿ
    addMoney: function(amount) {
      secretMoney = secretMoney + amount;
      console.log(amount + " ರೂ. ಜಮಾ ಆಯಿತು!");
    },
    // ಹಣ ಎಷ್ಟಿದೆ ಎಂದು ನೋಡಲು ಮಾತ್ರ ದಾರಿ
    checkBalance: function() {
      console.log("ಒಟ್ಟು ಹಣ: " + secretMoney + " ರೂ.");
    }
  };
}

let myBank = createPiggyBank();

myBank.addMoney(100); // ಔಟ್‌ಪುಟ್: 100 ರೂ. ಜಮಾ ಆಯಿತು!
myBank.addMoney(50);  // ಔಟ್‌ಪುಟ್: 50 ರೂ. ಜಮಾ ಆಯಿತು!
myBank.checkBalance(); // ಔಟ್‌ಪುಟ್: ಒಟ್ಟು ಹಣ: 150 ರೂ.

// ಯಾರಾದರೂ ಕದಿಯಲು ಪ್ರಯತ್ನಿಸಿದರೆ?
console.log(myBank.secretMoney); // ಔಟ್‌ಪುಟ್: undefined (ಯಾರಿಗೂ ಸಿಗುವುದಿಲ್ಲ!)
ಉದಾಹರಣೆಯ ವಿವರಣೆ: ಇಲ್ಲಿ secretMoney ಎಂಬುದು ಫ್ಯಾಕ್ಟರಿಯ ಒಳಗಡೆಯೇ ಇರುವ ರಹಸ್ಯ. ನಾವು ವಾಪಸ್ (return) ಕೊಟ್ಟ ಆಬ್ಜೆಕ್ಟ್‌ನಲ್ಲಿ secretMoney ಅನ್ನು ಕೊಟ್ಟೇ ಇಲ್ಲ! ನಾವು ಕೊಟ್ಟಿದ್ದು ಕೇವಲ addMoney (ಹಣ ಹಾಕಲು) ಮತ್ತು checkBalance (ಹಣ ನೋಡಲು) ಮಾತ್ರ. ಇದರಿಂದಾಗಿ ನಿಮ್ಮ ಪ್ರೋಗ್ರಾಮ್‌ನಲ್ಲಿರುವ ಡೇಟಾ ಬಹಳ ಭದ್ರವಾಗಿರುತ್ತದೆ. ಇದನ್ನೇ ಪ್ರೊಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿ Encapsulation / Closure ಎನ್ನುತ್ತಾರೆ.


ಒಟ್ಟಾರೆಯಾಗಿ Factory Function ನ ಲಾಭಗಳೇನು?
1.  ಸುಲಭ ಮತ್ತು ಸರಳ: new ಅಥವಾ this ಎಂಬ ಗೊಂದಲಮಯ ಪದಗಳನ್ನು ಬಳಸುವ ಅಗತ್ಯವಿಲ್ಲ.

2.  ಭದ್ರತೆ (Privacy): ನಮ್ಮ ಡೇಟಾವನ್ನು ಬೇರೆಯವರು ಸುಲಭವಾಗಿ ಬದಲಾಯಿಸದಂತೆ ಬಚ್ಚಿಡಬಹುದು.

3.  ಮರುಬಳಕೆ (Reusability): ಒಂದೇ ಮಷಿನ್ ಬಳಸಿ ನೂರಾರು ಆಬ್ಜೆಕ್ಟ್‌ಗಳನ್ನು ಕ್ಷಣಾರ್ಧದಲ್ಲಿ ಮಾಡಬಹುದು.
4.  ಹೌದು, ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್‌ನಲ್ಲಿ this ಕೀವರ್ಡ್ ಬಳಸಬಹುದು. ಆದರೆ, ಬಳಸದಿರುವುದೇ ಅತ್ಯುತ್ತಮ ಅಭ್ಯಾಸ (Best Practice)!
5.  
6.  ಏಕೆಂದರೆ, ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್ ಅನ್ನು ಬಳಸುವ ಮುಖ್ಯ ಉದ್ದೇಶವೇ this ಎಂಬ ಗೊಂದಲಮಯ ಕೀವರ್ಡ್‌ನಿಂದ ಪಾರಾಗುವುದು. this ಕೀವರ್ಡ್ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಒಂದು "ಊಸರವಳ್ಳಿ" (Chameleon) ಇದ್ದಂತೆ. ಅದು ಎಲ್ಲಿ ಮತ್ತು ಹೇಗೆ ಕರೆಯಲ್ಪಡುತ್ತದೆ ಎಂಬುದರ ಮೇಲೆ ತನ್ನ ಅರ್ಥವನ್ನು ಬದಲಾಯಿಸುತ್ತಲೇ ಇರುತ್ತದೆ.
7.  
8.  ಇದನ್ನು ಉದಾಹರಣೆಯೊಂದಿಗೆ ಅರ್ಥಮಾಡಿಕೊಳ್ಳೋಣ:
9.  

1. 10. this ಬಳಸಿದಾಗ ಆಗುವ ಅನಾಹುತ (The Problem)
11. ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್ ಒಳಗೆ this ಬಳಸಿದರೆ ಏನಾಗುತ್ತದೆ ಎಂದು ನೋಡೋಣ.
12.

32. ಏನಾಯಿತು ಇಲ್ಲಿ?setTimeout ನಂತಹ ಬೇರೆ ಫಂಕ್ಷನ್‌ಗಳಿಗೆ ನಾವು ಈ cheer ಫಂಕ್ಷನ್ ಅನ್ನು ಕಳುಹಿಸಿದಾಗ, this ಕೀವರ್ಡ್ ತಾನು ಯಾವ ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ಸೇರಿದ್ದು (Context) ಎಂಬುದನ್ನು ಮರೆತುಬಿಡುತ್ತದೆ. ಆಗ this.playerName ಎನ್ನುವುದು ಸಿಗದೆ undefined ಎಂದು ಬರುತ್ತದೆ.

34. 2. this ಬಳಸದೆ ಇದ್ದಾಗ (The Solution - Pro Level)
35. ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್‌ನಲ್ಲಿ this ಬಳಸುವ ಬದಲು, ನೇರವಾಗಿ ಡೇಟಾವನ್ನು (Closures ಮೂಲಕ) ಬಳಸಬೇಕು.

57. 
58. ಸಾರಾಂಶ: ನೀವು ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್ ಬರೆಯುತ್ತಿದ್ದೀರಿ ಎಂದರೆ this ಅನ್ನು ಮರೆತುಬಿಡಿ. ವೇರಿಯೇಬಲ್‌ಗಳನ್ನು ನೇರವಾಗಿ ಬಳಸಿ. ಇದರಿಂದ ನಿಮ್ಮ ಕೋಡ್ ಎಂದಿಗೂ ಗೊಂದಲ ಮಾಡುವುದಿಲ್ಲ ಮತ್ತು ಡೇಟಾ ಕಳೆದುಹೋಗುವುದಿಲ್ಲ!
ನಮಸ್ಕಾರ! ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ (JavaScript) ನಲ್ಲಿ 'Constructor Function' ಎಂಬುದನ್ನು ನಾವು ಒಂದು ಆಟಿಕೆ ಕಾರ್ಖಾನೆ (Toy Factory) ಅಥವಾ ಬಿಸ್ಕತ್ ಅಚ್ಚು (Cookie Cutter) ಮೂಲಕ ಬಹಳ ಸುಲಭವಾಗಿ ಅರ್ಥಮಾಡಿಕೊಳ್ಳಬಹುದು.

ನೀವು ಮನೆಯಲ್ಲಿ ಬಿಸ್ಕತ್ ಮಾಡುವಾಗ ಒಂದು 'ಅಚ್ಚು' (mould) ಬಳಸುತ್ತೀರಿ, ಹೌದಲ್ವಾ? ಆ ಒಂದೇ ಅಚ್ಚನ್ನು ಬಳಸಿ ನೀವು ಬೇಕಾದಷ್ಟು ಬಿಸ್ಕತ್‌ಗಳನ್ನು ಮಾಡಬಹುದು. ಅದೇ ರೀತಿ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಒಂದೇ ರೀತಿಯ ಅನೇಕ 'ಆಬ್ಜೆಕ್ಟ್' (Objects) ಗಳನ್ನು ತಯಾರಿಸಲು ಬಳಸುವ 'ಅಚ್ಚು' ಅಥವಾ ಫ್ಯಾಕ್ಟರಿಯೇ ಈ Constructor Function.

೧. ಬಿಗಿನರ್ ಲೆವೆಲ್ (Beginner): ನಮಗೇಕೆ ಇದು ಬೇಕು?
ಉದಾಹರಣೆಗೆ, ನೀವು 3 ಆಟದ ಕಾರುಗಳನ್ನು (Objects) ಮಾಡಬೇಕು. ಪ್ರತಿಯೊಂದಕ್ಕೂ ಹೆಸರು, ಬಣ್ಣ ಬರೆಯುತ್ತಾ ಕೂತರೆ ಬಹಳ ಸಮಯ ಹಿಡಿಯುತ್ತದೆ:


JavaScript
// ಅಚ್ಚು ಇಲ್ಲದೆ ಒಂದೊಂದೇ ಕಾರು ಮಾಡುವುದು (ತುಂಬಾ ಕಷ್ಟ)
let car1 = { name: "Maruti", color: "Red" };
let car2 = { name: "Tata", color: "Blue" };
let car3 = { name: "Mahindra", color: "Black" };
ಇದರ ಬದಲು ನಾವು ಒಂದು 'ಕಾರ್ ಫ್ಯಾಕ್ಟರಿ' (Constructor Function) ಯಂತ್ರವನ್ನು ಮಾಡಿದರೆ, ಕಾರಿನ ಬಣ್ಣ ಮತ್ತು ಹೆಸರು ಹೇಳಿದ ತಕ್ಷಣ ಅದು ಹೊಸ ಕಾರನ್ನು ರೆಡಿ ಮಾಡಿಕೊಡುತ್ತದೆ.

ಇದನ್ನು ಬರೆಯುವ ನಿಯಮಗಳು:


• ಇದರ ಹೆಸರು ಯಾವಾಗಲೂ ದೊಡ್ಡ ಅಕ್ಷರದಿಂದ (Capital Letter) ಶುರುವಾಗಬೇಕು. (ಉದಾಹರಣೆಗೆ: Car, Robot).

• this ಎಂಬ ಪದವನ್ನು ಬಳಸುತ್ತೇವೆ. (ಇದರ ಅರ್ಥ "ಈಗ ತಯಾರಾಗುತ್ತಿರುವ ಹೊಸ ಆಟಿಕೆ").

• ಹೊಸ ವಸ್ತುವನ್ನು ಪಡೆಯಲು new ಎಂಬ ಪದವನ್ನು ಬಳಸುತ್ತೇವೆ.

ಉದಾಹರಣೆ 1: ಸರಳ ಕಾರು ಕಾರ್ಖಾನೆ
JavaScript
// 1. ಇದು ನಮ್ಮ ಫ್ಯಾಕ್ಟರಿ (Constructor Function)
function Car(ಹೆಸರು, ಬಣ್ಣ) {
    this.name = ಹೆಸರು;
    this.color = ಬಣ್ಣ;
}

// 2. new ಪದ ಬಳಸಿ ಕಾರ್ಖಾನೆಗೆ ಆರ್ಡರ್ ಕೊಡುವುದು!
let myFirstCar = new Car("Maruti", "Red");
let mySecondCar = new Car("Tata", "Blue");

ಇಲ್ಲಿ new Car(...) ಎಂದು ಹೇಳಿದಾಗ, ಆ ಫ್ಯಾಕ್ಟರಿ ತಕ್ಷಣವೇ ಒಂದು ಹೊಸ ಕಾರನ್ನು ತಯಾರಿಸಿ ಕೊಡುತ್ತದೆ.


೨. ಇಂಟರ್ಮೀಡಿಯೇಟ್ ಲೆವೆಲ್ (Intermediate): ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ಕೆಲಸ ಕೊಡುವುದು
ಕೇವಲ ಬಣ್ಣ, ಹೆಸರು ಮಾತ್ರವಲ್ಲದೆ ಆ ಹೊಸ ಕಾರು ಅಥವಾ ಆಟಿಕೆ ಕೆಲಸ ಮಾಡುವಂತೆ (Methods/Functions) ಮಾಡಬಹುದು. ಉದಾಹರಣೆಗೆ, ನಮ್ಮ ಕಾರು ಹಾರ್ನ್ (Horn) ಮಾಡುವಂತೆ ಮಾಡೋಣ.


JavaScript
function SmartCar(name, color) {
    this.name = name;
    this.color = color;
    
    // ಕಾರು ಹಾರ್ನ್ ಮಾಡುವ ಫಂಕ್ಷನ್
    this.soundHorn = function() {
        console.log(this.name + " ಕಾರು 'ಪೀಪ್ ಪೀಪ್' ಅನ್ನುತ್ತಿದೆ!");
    }
}

let tataCar = new SmartCar("Tata Nexon", "White");
tataCar.soundHorn(); 
// ಔಟ್‌ಪುಟ್: Tata Nexon ಕಾರು 'ಪೀಪ್ ಪೀಪ್' ಅನ್ನುತ್ತಿದೆ!
೩. ಪ್ರೊ ಲೆವೆಲ್ (Pro Level): Prototype ಬಳಸಿ ಮೆಮೊರಿ ಉಳಿಸುವುದು
ಮೇಲಿನ SmartCar ಉದಾಹರಣೆಯಲ್ಲಿ ಒಂದು ಸಣ್ಣ ಸಮಸ್ಯೆಯಿದೆ. ನೀವು 1000 ಕಾರುಗಳನ್ನು ತಯಾರಿಸಿದರೆ, soundHorn ಎಂಬ ಫಂಕ್ಷನ್ ಕೂಡ 1000 ಬಾರಿ ಕಾಪಿ (Copy) ಆಗುತ್ತದೆ. ಇದರಿಂದ ಕಂಪ್ಯೂಟರ್ ಮೆಮೊರಿ ವೇಸ್ಟ್ ಆಗುತ್ತದೆ.

ಇದನ್ನು ತಪ್ಪಿಸಲು ಪ್ರೊ (Pro) ಡೆವಲಪರ್‌ಗಳು prototype ಬಳಸುತ್ತಾರೆ. prototype ಅಂದರೆ ಊರಿನ ಲೈಬ್ರರಿ ಇದ್ದ ಹಾಗೆ. ಪ್ರತಿಯೊಬ್ಬರಿಗೂ ಒಂದೊಂದು ಪುಸ್ತಕ ಕೊಡುವ ಬದಲು, ಒಂದು ಪುಸ್ತಕವನ್ನು ಲೈಬ್ರರಿಯಲ್ಲಿ ಇಟ್ಟರೆ ಎಲ್ಲರೂ ಓದಬಹುದು!


JavaScript
function ProCar(name, color) {
    this.name = name;
    this.color = color;
    // ಇಲ್ಲಿ ಫಂಕ್ಷನ್ ಬರೆಯುವುದಿಲ್ಲ!
}

// ಎಲ್ಲ ಕಾರುಗಳಿಗೂ ಒಂದೇ ಜಾಗದಲ್ಲಿ (prototype ನಲ್ಲಿ) ನಿಯಮ ಬರೆಯುವುದು
ProCar.prototype.soundHorn = function() {
     console.log(this.name + " ಕಾರು 'ಪೀಪ್ ಪೀಪ್' ಅನ್ನುತ್ತಿದೆ!");
}

let carA = new ProCar("Mahindra Thar", "Black");
let carB = new ProCar("Kia", "Red");

// ಎರಡು ಕಾರುಗಳು ಲೈಬ್ರರಿಯಿಂದ (prototype) ಒಂದೇ ಫಂಕ್ಷನ್ ಅನ್ನು ಬಳಸುತ್ತವೆ
carA.soundHorn();
carB.soundHorn();
ಹೀಗೆ ಮಾಡುವುದರಿಂದ ಎಷ್ಟೇ ಕಾರುಗಳನ್ನು ತಯಾರಿಸಿದರೂ soundHorn ಫಂಕ್ಷನ್ ಮೆಮೊರಿಯಲ್ಲಿ ಒಂದೇ ಬಾರಿ ಸೇವ್ ಆಗಿರುತ್ತದೆ.

ಸಾರಾಂಶ (ನೆನಪಿಡಬೇಕಾದ ಮುಖ್ಯ ಅಂಶಗಳು):


• Constructor Function: ಒಂದೇ ರೀತಿಯ ಅನೇಕ ಆಬ್ಜೆಕ್ಟ್ ಮಾಡಲು ಬಳಸುವ ಅಚ್ಚು.

• Capital Letter: ಇದರ ಹೆಸರು ಯಾವಾಗಲೂ ದೊಡ್ಡ ಅಕ್ಷರದಿಂದ ಶುರುವಾಗಬೇಕು.

• new Keyword: ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಹುಟ್ಟುಹಾಕಲು ಕಡ್ಡಾಯವಾಗಿ new ಬಳಸಬೇಕು.

• this Keyword: ತಯಾರಾಗುತ್ತಿರುವ ಪ್ರಸ್ತುತ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಸೂಚಿಸುತ್ತದೆ.

• prototype: ಮೆಮೊರಿ ಉಳಿಸಲು ಕಾಮನ್ ಆಗಿರುವ ಫಂಕ್ಷನ್‌ಗಳನ್ನು prototype ಒಳಗೆ ಹಾಕಬೇಕು.

ಖಂಡಿತ, ನೀವು ನೀಡಿದ ಮಾಹಿತಿಯನ್ನು ದೈನಂದಿನ ಉದಾಹರಣೆ ಮತ್ತು ತಾಂತ್ರಿಕ ವಿವರಣೆಗಳೆರಡನ್ನೂ ಸೇರಿಸಿ (Merge ಮಾಡಿ), ಯಾರಿಗಾದರೂ ಸುಲಭವಾಗಿ ಅರ್ಥವಾಗುವಂತೆ ಹಂತ-ಹಂತವಾಗಿ (Step-by-step) ಮತ್ತು ಕ್ರಮಬದ್ಧವಾಗಿ ಕೆಳಗೆ ವಿವರಿಸಿದ್ದೇನೆ.


೧. new ಮತ್ತು this: ಒಂದು ಸರಳ ಪರಿಚಯ
ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ನಾವು ಹೊಸ ವಸ್ತುಗಳನ್ನು (Objects) ತಯಾರಿಸುವಾಗ new ಮತ್ತು this ಕೀವರ್ಡ್‌ಗಳನ್ನು ಬಳಸುತ್ತೇವೆ. ಇವುಗಳ ಕೆಲಸವನ್ನು ಶಾಲೆಯ 'ಪುಸ್ತಕ'ದ ಉದಾಹರಣೆಯೊಂದಿಗೆ ಅರ್ಥಮಾಡಿಕೊಳ್ಳೋಣ.


🔹 new (ಹೊಸ ಖಾಲಿ ಪುಸ್ತಕ ತರುವುದು)
• ದೈನಂದಿನ ಉದಾಹರಣೆ: ನೀವು ಅಂಗಡಿಗೆ ಹೋಗಿ "ನನಗೆ ಒಂದು ಹೊಸ ಪುಸ್ತಕ ಕೊಡಿ" ಎಂದು ಕೇಳಿದಾಗ, ಅಂಗಡಿಯವನು ನಿಮಗೆ ಒಂದು ಹೊಸ, ಖಾಲಿ ಪುಟಗಳಿರುವ ಪುಸ್ತಕವನ್ನು ಕೊಡುತ್ತಾನೆ.

• ತಾಂತ್ರಿಕ ವಿವರಣೆ: ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ನೀವು new ಎಂದು ಬರೆದರೆ, ಕಂಪ್ಯೂಟರ್‌ನ ಮೆಮೊರಿಯಲ್ಲಿ ಒಂದು ಹೊಚ್ಚಹೊಸ "ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್ (Empty Object {})" ಸೃಷ್ಟಿಯಾಗುತ್ತದೆ.

🔹 this (ಪುಸ್ತಕದ ಮೇಲೆ 'ನನ್ನ' ಹೆಸರು ಬರೆಯುವುದು)
• ದೈನಂದಿನ ಉದಾಹರಣೆ: ಮನೆಗೆ ತಂದ ಆ ಹೊಸ ಪುಸ್ತಕದ ಮೇಲಿರುವ ಚೀಟಿಯಲ್ಲಿ (Label) ನೀವು "ಈ ಪುಸ್ತಕ ನನ್ನದು, ನನ್ನ ಹೆಸರು ರಾಜು" ಎಂದು ಬರೆಯುತ್ತೀರಿ.

• ತಾಂತ್ರಿಕ ವಿವರಣೆ: ಮೆಮೊರಿಯಲ್ಲಿ ಸೃಷ್ಟಿಯಾದ ಆ ಹೊಸ ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ಡೇಟಾವನ್ನು (Properties) ತುಂಬಲು this ಬಳಸಲಾಗುತ್ತದೆ. ಅಂದರೆ, ಆ ಆಬ್ಜೆಕ್ಟ್ ತನ್ನ ಸ್ವಂತ ಮಾಹಿತಿಯನ್ನು ಸೇರಿಸಿಕೊಳ್ಳಲು this ಅನ್ನು ಬಳಸುತ್ತದೆ.

೨. ಕಂಪ್ಯೂಟರ್‌ನ ಒಳಗೆ ಏನಾಗುತ್ತದೆ? (The 4 Technical Steps)
ನೀವು new ಕೀವರ್ಡ್ ಬಳಸಿ ಫಂಕ್ಷನ್ ಅನ್ನು ಕರೆದಾಗ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಎಂಜಿನ್ (ಉದಾಹರಣೆಗೆ: V8 Engine) ಹಿನ್ನೆಲೆಯಲ್ಲಿ ಈ ಕೆಳಗಿನ 4 ಪ್ರಮುಖ ಕೆಲಸಗಳನ್ನು ಮಾಡುತ್ತದೆ:


1.  ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿ (Create Object): ಮೆಮೊರಿಯಲ್ಲಿ ಒಂದು ಹೊಸ ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್ ({}) ಸೃಷ್ಟಿಸುತ್ತದೆ. (ಇದೇ ನಿಮ್ಮ ಹೊಸ ಖಾಲಿ ಪುಸ್ತಕ)

2.  ಪ್ರೋಟೋಟೈಪ್ ಲಿಂಕ್ (Prototype Linking): ಆ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಕನ್‌ಸ್ಟ್ರಕ್ಟರ್ ಫಂಕ್ಷನ್‌ನ prototype ಗೆ ಲಿಂಕ್ ಮಾಡುತ್ತದೆ. (ಇದರಿಂದ ಮೆಮೊರಿ ಉಳಿಯುತ್ತದೆ).

3.  this ಬೈಂಡಿಂಗ್ (Binding this): ಫಂಕ್ಷನ್‌ನ ಒಳಗಿರುವ this ಪದವನ್ನು, ತಾನು ಈಗಷ್ಟೇ ಸೃಷ್ಟಿಸಿದ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ಪಾಯಿಂಟ್ (Bind) ಮಾಡುತ್ತದೆ. (ಪುಸ್ತಕದ ಮೇಲೆ ಹೆಸರು ಬರೆಯುವ ಹಂತ).

4.  ಆಟೋಮ್ಯಾಟಿಕ್ ರಿಟರ್ನ್ (Implicit Return): ನಾವು ಕೋಡ್‌ನಲ್ಲಿ return ಎಂದು ಬರೆಯದಿದ್ದರೂ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಆ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ತಾನಾಗಿಯೇ ನಮಗೆ ವಾಪಸ್ ಕೊಡುತ್ತದೆ.

೩. ಎರಡನ್ನೂ ಒಟ್ಟುಗೂಡಿಸಿ ಕೋಡ್‌ನಲ್ಲಿ ನೋಡೋಣ
ಈಗ ಸರಳ ಕಥೆ ಮತ್ತು ತಾಂತ್ರಿಕ ಪ್ರಕ್ರಿಯೆ ಎರಡೂ ಕೋಡ್‌ನಲ್ಲಿ ಹೇಗೆ ಜೊತೆಯಾಗಿ ಕೆಲಸ ಮಾಡುತ್ತವೆ ಎಂದು ನೋಡಿ:


JavaScript
// ಇದು ನಮ್ಮ ಕನ್‌ಸ್ಟ್ರಕ್ಟರ್ ಫಂಕ್ಷನ್ (ಪುಸ್ತಕ ತಯಾರಿಸುವ ಫ್ಯಾಕ್ಟರಿ)
function Student(ಹೆಸರು, ತರಗತಿ) {
    
    // ತಾಂತ್ರಿಕವಾಗಿ: ನೀವು 'new' ಬಳಸಿದಾಗ ಹಿನ್ನೆಲೆಯಲ್ಲಿ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ 1ನೇ ಕೆಲಸ ಮಾಡುತ್ತದೆ:
    // this = {};  <-- (ಹೊಸ ಖಾಲಿ ಪುಸ್ತಕ / ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಯಾಯಿತು)
    
    // ಈಗ 'this' ಆ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ಡೇಟಾ ಸೇರಿಸುತ್ತಿದೆ:
    // (2ನೇ ಕೆಲಸ: ಪುಸ್ತಕದ ಮೇಲೆ ಹೆಸರು ಮತ್ತು ತರಗತಿ ಬರೆಯುವುದು)
    this.name = ಹೆಸರು;      
    this.grade = ತರಗತಿ;

    // ತಾಂತ್ರಿಕವಾಗಿ: ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ 4ನೇ ಕೆಲಸವಾಗಿ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ರಿಟರ್ನ್ ಮಾಡುತ್ತದೆ:
    // return this; <-- (ನಿಮ್ಮ ಕೈಗೆ ತುಂಬಿದ ಪುಸ್ತಕ ಸಿಕ್ಕಿತು)
}

// 'new' ಪದ ಬಳಸಿ ಹೊಸ Student ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸುತ್ತಿದ್ದೇವೆ
let student1 = new Student("ರಾಜು", "10ನೇ ತರಗತಿ"); 
let student2 = new Student("ರಮ್ಯ", "9ನೇ ತರಗತಿ");

// ಈಗ ಪ್ರಿಂಟ್ ಮಾಡಿ ನೋಡೋಣ
console.log(student1.name); // ಔಟ್‌ಪುಟ್: ರಾಜು
console.log(student2.name); // ಔಟ್‌ಪುಟ್: ರಮ್ಯ
೪. ಅಂತಿಮ ಸಾರಾಂಶ (Conclusion)
• new ಎಂದರೆ: "ಮೆಮೊರಿಯಲ್ಲಿ ಒಂದು ಹೊಸ, ಖಾಲಿ ಜಾಗವನ್ನು ({}) ಕೊಡು" ಎನ್ನುವ ಕಮಾಂಡ್.

• this ಎಂದರೆ: "ಆ ಹೊಸ ಜಾಗದ ಒಳಗೆ ಈ ಕೆಳಗಿನ ಡೇಟಾವನ್ನು ತುಂಬು" ಎಂದು ಹೇಳುವ ಪಾಯಿಂಟರ್.

ಇವೆರಡೂ ಒಟ್ಟಿಗೆ ಸೇರಿದರೆ, ನಮಗೆ ಬೇಕಾದ ನಿಖರವಾದ ಮಾಹಿತಿಯನ್ನು ಹೊಂದಿರುವ ಒಂದು ಸಂಪೂರ್ಣ 'ಆಬ್ಜೆಕ್ಟ್' (Object) ತಯಾರಾಗುತ್ತದೆ.
ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Factory Function ಮತ್ತು Constructor Function ಇವೆರಡರ ಮುಖ್ಯ ಕೆಲಸ ಒಂದೇ: "ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ (Object) ಗಳನ್ನು ಸೃಷ್ಟಿಸುವುದು".

ಆದರೆ ಅವು ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಸೃಷ್ಟಿಸುವ 'ವಿಧಾನ' ಬೇರೆ. ಇವೆರಡರ ವ್ಯತ್ಯಾಸವನ್ನು ಸರಳವಾಗಿ ಅರ್ಥಮಾಡಿಕೊಳ್ಳೋಣ.


೧. ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್ (Factory Function)
ಇದನ್ನು ಒಂದು "ಸಾಮಾನ್ಯ ಕೆಲಸಗಾರ"ನಿಗೆ ಹೋಲಿಸಬಹುದು. ನೀವು ಒಂದು ಗೊಂಬೆ ಕೇಳಿದರೆ, ಆತನೇ ಒಂದು ಖಾಲಿ ಡಬ್ಬ ತಂದು, ಅದಕ್ಕೆ ಬಣ್ಣ ಬಳಿದು, ಕೊನೆಗೆ ಆ ಡಬ್ಬವನ್ನು ನಿಮ್ಮ ಕೈಗೆ ಕೊಡುತ್ತಾನೆ.

ಇಲ್ಲಿ ಯಾವುದೇ 'ಮ್ಯಾಜಿಕ್' ಇರುವುದಿಲ್ಲ. ನಾವೇ ಒಂದು ಆಬ್ಜೆಕ್ಟ್ ({}) ಅನ್ನು ಸೃಷ್ಟಿಸಿ, ಅದಕ್ಕೆ ಡೇಟಾ ತುಂಬಿ, ಕೊನೆಗೆ ನಾವೇ ಕಡ್ಡಾಯವಾಗಿ return ಮಾಡಬೇಕು.


JavaScript
// Factory Function ಉದಾಹರಣೆ
function createRobot(ಹೆಸರು, ಬಣ್ಣ) {
    // 1. ನಾವೇ ಒಂದು ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸುತ್ತಿದ್ದೇವೆ
    let robot = {}; 
    
    // 2. ಡೇಟಾ ತುಂಬುತ್ತಿದ್ದೇವೆ (this ಬಳಸುವಂತಿಲ್ಲ)
    robot.name = ಹೆಸರು;
    robot.color = ಬಣ್ಣ;
    
    // 3. ನಾವೇ ಕಡ್ಡಾಯವಾಗಿ ರಿಟರ್ನ್ ಮಾಡುತ್ತಿದ್ದೇವೆ
    return robot; 
}

// 'new' ಪದ ಇಲ್ಲದೆ ಕರೆಯುವುದು
let robot1 = createRobot("ಚಿಟ್ಟಿ", "ಬೆಳ್ಳಿ"); 
console.log(robot1.name); // ಔಟ್‌ಪುಟ್: ಚಿಟ್ಟಿ
ನೆನಪಿಡಬೇಕಾದದ್ದು: ಇಲ್ಲಿ new ಬೇಡ, this ಬೇಡ, ಆದರೆ return ಕಡ್ಡಾಯ!


೨. ಕನ್‌ಸ್ಟ್ರಕ್ಟರ್ ಫಂಕ್ಷನ್ (Constructor Function)
ಇದನ್ನು ಒಂದು "ಆಟೋಮ್ಯಾಟಿಕ್ ಮ್ಯಾಜಿಕ್ ಮಷಿನ್"ಗೆ ಹೋಲಿಸಬಹುದು. ನಾವು ಈ ಹಿಂದೆ ಕಲಿತಂತೆ, ಇಲ್ಲಿ new ಎಂಬ ಮ್ಯಾಜಿಕ್ ದಂಡವನ್ನು ಬಳಸುತ್ತೇವೆ. ಆ ಮ್ಯಾಜಿಕ್ ದಂಡವೇ ಗುಪ್ತವಾಗಿ ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸಿ, ಕೊನೆಗೆ ತಾನೇ ವಾಪಸ್ (return) ಕೊಡುತ್ತದೆ.


JavaScript
// Constructor Function ಉದಾಹರಣೆ (ಹೆಸರು ಕ್ಯಾಪಿಟಲ್ ಇರಬೇಕು)
function Robot(ಹೆಸರು, ಬಣ್ಣ) {
    // ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಗುಪ್ತವಾಗಿ ಸೃಷ್ಟಿಯಾಗಿದೆ: this = {}
    
    // ನಾವಿಲ್ಲಿ 'this' ಬಳಸುತ್ತೇವೆ
    this.name = ಹೆಸರು;
    this.color = ಬಣ್ಣ;
    
    // ರಿಟರ್ನ್ ಬರೆಯುವ ಅಗತ್ಯವಿಲ್ಲ! (ಗುಪ್ತವಾಗಿ: return this;)
}

// ಕಡ್ಡಾಯವಾಗಿ 'new' ಪದ ಬಳಸಬೇಕು
let robot2 = new Robot("ಟರ್ಮಿನೇಟರ್", "ಕಪ್ಪು");
console.log(robot2.name); // ಔಟ್‌ಪುಟ್: ಟರ್ಮಿನೇಟರ್
ನೆನಪಿಡಬೇಕಾದದ್ದು: ಇಲ್ಲಿ new ಮತ್ತು this ಕಡ್ಡಾಯ, ಆದರೆ return ಬರೆಯುವ ಅಗತ್ಯವಿಲ್ಲ!


೩. ಮುಖ್ಯ ವ್ಯತ್ಯಾಸಗಳು (ಮುಖಾಮುಖಿ ಹೋಲಿಕೆ)
ವ್ಯತ್ಯಾಸ (Difference)   ಫ್ಯಾಕ್ಟರಿ ಫಂಕ್ಷನ್ (Factory Function)    ಕನ್‌ಸ್ಟ್ರಕ್ಟರ್ ಫಂಕ್ಷನ್ (Constructor Function)
ಫಂಕ್ಷನ್ ಹೆಸರು    ಸಣ್ಣ ಅಕ್ಷರದಿಂದ ಶುರುವಾಗುತ್ತದೆ (ಉದಾ: createRobot)   ದೊಡ್ಡ ಅಕ್ಷರದಿಂದ ಶುರುವಾಗುತ್ತದೆ (ಉದಾ: Robot)
new ಕೀವರ್ಡ್    ಬಳಸುವುದಿಲ್ಲ ಕಡ್ಡಾಯವಾಗಿ ಬಳಸಬೇಕು
this ಕೀವರ್ಡ್   ಬಳಸುವುದಿಲ್ಲ ಕಡ್ಡಾಯವಾಗಿ ಬಳಸಬೇಕು
return ಕೀವರ್ಡ್ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ನಾವೇ ಕಡ್ಡಾಯವಾಗಿ return ಮಾಡಬೇಕು    ಬರೆಯುವ ಅಗತ್ಯವಿಲ್ಲ (ತಾನೇ ರಿಟರ್ನ್ ಆಗುತ್ತದೆ)
ಯಾವುದು ಸುಲಭ?   ಆರಂಭಿಕರಿಗೆ ಅರ್ಥಮಾಡಿಕೊಳ್ಳಲು ತುಂಬಾ ಸುಲಭ.    ಸ್ವಲ್ಪ ಕನ್ಫ್ಯೂಸ್ ಆಗಬಹುದು, ಆದರೆ ಪ್ರೊ ಡೆವಲಪರ್‌ಗಳು ಇದನ್ನೇ ಹೆಚ್ಚು ಬಳಸುತ್ತಾರೆ.
ಸಾರಾಂಶ: ನಿಮಗೆ new ಮತ್ತು this ನ ತಲೆನೋವು ಬೇಡ, ಎಲ್ಲವನ್ನೂ ನಾನೇ ಮ್ಯಾನ್ಯುವಲ್ (Manual) ಆಗಿ ಬರೆಯುತ್ತೇನೆ ಎಂದರೆ Factory Function ಬಳಸಿ. ಇಲ್ಲ, "ಕಂಪ್ಯೂಟರ್ ತಾನೇ ಆಟೋಮ್ಯಾಟಿಕ್ ಆಗಿ ಆಬ್ಜೆಕ್ಟ್ ಕ್ರಿಯೇಟ್ ಮಾಡಿ ರಿಟರ್ನ್ ಮಾಡಲಿ, ನನ್ನ ಕೋಡ್ ಚಿಕ್ಕದಾಗಲಿ" ಎಂದರೆ Constructor Function ಬಳಸಿ!
