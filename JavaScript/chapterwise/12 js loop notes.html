ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಲೂಪ್‌ಗಳು (JS Loops)

ಮೆಮೊರಿ ನಿರ್ವಹಣೆ (Memory Management) ಮತ್ತು ಪ್ರೊಸೆಸರ್ ಹರಿವಿನ (Processor Flow) ಆಧಾರದ ಮೇಲೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಲೂಪ್‌ಗಳ ಸಂಪೂರ್ಣ ವಿಶ್ಲೇಷಣೆ. ಶೂನ್ಯದಿಂದ ವೃತ್ತಿಪರ (0 to Pro) ಹಂತದವರೆಗಿನ ವಿವರಣೆ.

0. ಶೂನ್ಯ ಹಂತ (The Absolute Basics)

ಲೂಪ್ ಎನ್ನುವುದು ಒಂದೇ ಕೆಲಸವನ್ನು ಪದೇ ಪದೇ ಮಾಡುವಂತೆ ಪ್ರೊಸೆಸರ್‌ಗೆ ನೀಡುವ ಸೂಚನೆ (DRY - Don't Repeat Yourself). ಪ್ರತಿ ಬಾರಿಯೂ ಕೋಡ್ ಬರೆಯುವ ಬದಲು, ನಿರ್ದಿಷ್ಟ ಷರತ್ತು ಪೂರ್ಣಗೊಳ್ಳುವವರೆಗೆ ಮೆಮೊರಿಯಲ್ಲಿರುವ ಒಂದೇ ಕೋಡ್ ಬ್ಲಾಕ್ ಅನ್ನು ಎಂಜಿನ್ ಮರುಬಳಕೆ ಮಾಡುತ್ತದೆ.

1. ಆರಂಭಿಕ ಹಂತ (Beginner Level)

ಇದು ಸಾಂಪ್ರದಾಯಿಕ for ಲೂಪ್ ಆಗಿದ್ದು, ನಿರ್ದಿಷ್ಟ ಸಂಖ್ಯೆಯ ಬಾರಿ ಕೋಡ್ ರನ್ ಮಾಡಲು ಬಳಸಲಾಗುತ್ತದೆ.

JavaScript 
for (let i = 1; i <= 5; i++) {
  console.log("ಸಂಖ್ಯೆ:", i);
}

ಕಾರ್ಯಾಚರಣೆಯ ಹರಿವು ಮತ್ತು ಮೆಮೊರಿ ವಿಶ್ಲೇಷಣೆ:

  • for: ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಎಂಜಿನ್‌ಗೆ (ಉದಾ: V8 ಎಂಜಿನ್) ಇದೊಂದು ಲೂಪ್ ಎಂದು ಸೂಚಿಸುವ ಕೀವರ್ಡ್. ಇದು ಇನ್ಸ್ಟ್ರಕ್ಷನ್ ಸೆಟ್‌ನಲ್ಲಿ (Instruction set) ಪುನರಾವರ್ತಿತ ಬ್ಲಾಕ್ ಅನ್ನು ಸೃಷ್ಟಿಸುತ್ತದೆ.
  • let i = 1 (Initialization): ಬ್ಲಾಕ್ ಸ್ಕೋಪ್ (Block-scoped) ವೇರಿಯೇಬಲ್ ಅನ್ನು ಘೋಷಿಸುವ ಕೀವರ್ಡ್. ಲೂಪ್ ಮುಗಿದ ತಕ್ಷಣ ಮೆಮೊರಿಯಿಂದ i ಅನ್ನು ಗಾರ್ಬೇಜ್ ಕಲೆಕ್ಟರ್ (Garbage Collector) ಅಳಿಸಿಹಾಕುತ್ತದೆ. ಇದು ಮೆಮೊರಿ ಸೋರಿಕೆಯನ್ನು (Memory leak) ತಡೆಯುತ್ತದೆ.
  • i <= 5 (Condition): ಇದು ಬೂಲಿಯನ್ (Boolean) ಷರತ್ತು. ಪ್ರತಿ ಲೂಪ್ ಆರಂಭಕ್ಕೂ ಮುನ್ನ ಇದು true ಅಥವಾ false ಎಂದು ಮೌಲ್ಯಮಾಪನ (Evaluate) ಆಗುತ್ತದೆ.
  • i++ (Increment): ಪೋಸ್ಟ್-ಇನ್ಕ್ರಿಮೆಂಟ್ ಆಪರೇಟರ್ (i = i + 1). ಲೂಪ್‌ನ ಒಳಗಿನ ಕೋಡ್ ಕಾರ್ಯಗತಗೊಂಡ ನಂತರವೇ ಪ್ರೊಸೆಸರ್ ರಿಜಿಸ್ಟರ್‌ನಲ್ಲಿ i ಯ ಮೌಲ್ಯ 1 ಹೆಚ್ಚಾಗುತ್ತದೆ.
  • { ... }: ಲೂಪ್‌ನ ಬ್ಲಾಕ್ (Block). ಷರತ್ತು true ಇದ್ದರೆ ಮಾತ್ರ ಇದರೊಳಗಿನ ಕೋಡ್ ರನ್ ಆಗುತ್ತದೆ.
  • console.log: ಬ್ರೌಸರ್ ಅಥವಾ Node.js ನ ಗ್ಲೋಬಲ್ ಆಬ್ಜೆಕ್ಟ್ ಮುಖಾಂತರ ಸ್ಟ್ಯಾಂಡರ್ಡ್ ಔಟ್‌ಪುಟ್ (stdout) ಗೆ ಡೇಟಾವನ್ನು ಮುದ್ರಿಸುತ್ತದೆ.

2. ಮಧ್ಯಮ ಹಂತ (Intermediate Level)

ಡೇಟಾ ಸ್ಟ್ರಕ್ಚರ್‌ಗಳ (ಉದಾಹರಣೆಗೆ ಅರೇ) ಮೇಲೆ ಲೂಪ್ ಮಾಡುವುದು ಮತ್ತು ಪ್ರೊಸೆಸರ್‌ನ ನಿಯಂತ್ರಣ ಹರಿವನ್ನು (Control flow) ಬದಲಾಯಿಸುವುದು.

JavaScript 
let fruits = ["Apple", "Banana", "Mango", "Orange", "Grapes", "Papaya"];

for (let i = 0; i < fruits.length; i++) {
  if (i === 2) { 
    continue; // 'Mango' ಅನ್ನು ಸ್ಕಿಪ್ ಮಾಡುತ್ತದೆ
  }
  if (i === 4) { 
    break;    // 'Grapes' ಬಂದಾಗ ಲೂಪ್ ನಿಲ್ಲುತ್ತದೆ
  }
  console.log(fruits[i]);
}

ತಾಂತ್ರಿಕ ವಿಶ್ಲೇಷಣೆ:

  • fruits.length: ಮೆಮೊರಿಯಲ್ಲಿ (Heap) ಇರುವ ಅರೇ ಆಬ್ಜೆಕ್ಟ್‌ನ ಆಂತರಿಕ ಪ್ರಾಪರ್ಟಿ. ಇದು ಅರೇಯಲ್ಲಿರುವ ಒಟ್ಟು ಅಂಶಗಳ ಸಂಖ್ಯೆಯನ್ನು ಹಿಂದಿರುಗಿಸುತ್ತದೆ.
  • fruits[i] (Bracket notation): ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಎಂಜಿನ್ ಮೆಮೊರಿಯಲ್ಲಿ fruits ಅರೇಯ i ನೇ ಇಂಡೆಕ್ಸ್‌ನ ಮೆಮೊರಿ ವಿಳಾಸವನ್ನು (Memory address pointer) ನೇರವಾಗಿ ಓದುತ್ತದೆ.
  • === (Strict Equality): ಪ್ರೊಸೆಸರ್ ಮಟ್ಟದಲ್ಲಿ ಮೌಲ್ಯ (Value) ಮತ್ತು ಮೆಮೊರಿಯ ಡೇಟಾ ಪ್ರಕಾರ (Data Type) ಎರಡನ್ನೂ ಬಿಟ್-ಬೈ-ಬಿಟ್ (Bit-by-bit) ಹೋಲಿಸುತ್ತದೆ.
  • continue: ಮೈಕ್ರೋ-ಪ್ರೊಸೆಸರ್ ಮಟ್ಟದಲ್ಲಿ ಇದೊಂದು 'Jump' ಇನ್ಸ್ಟ್ರಕ್ಷನ್. ಇದು ತಕ್ಷಣವೇ ಕೆಳಗಿನ ಕೋಡ್ ಅನ್ನು ನಿರ್ಲಕ್ಷಿಸಿ, ಮುಂದಿನ ಇಟರೇಶನ್‌ಗೆ (i++) ಜಿಗಿಯುತ್ತದೆ.
  • break: ಇದು ಸಂಪೂರ್ಣ ಲೂಪ್‌ನಿಂದ ಹೊರಬಂದು, ಲೂಪ್ ಬ್ಲಾಕ್‌ನ (}) ನಂತರದ ಮೆಮೊರಿ ವಿಳಾಸದಲ್ಲಿರುವ ಕೋಡ್‌ಗೆ ನಿಯಂತ್ರಣವನ್ನು ವರ್ಗಾಯಿಸುತ್ತದೆ.

3. ವೃತ್ತಿಪರ ಹಂತ (Pro Level)

ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ (ES6+) ಇಟರೇಟರ್ ಪ್ರೋಟೋಕಾಲ್ ಮತ್ತು ಮೆಮೊರಿ ಲುಕಪ್ ವ್ಯವಸ್ಥೆಗಳನ್ನು ಬಳಸುತ್ತದೆ.

ಅ. for...of ಲೂಪ್ (ಇಟರೇಟರ್ ಪ್ರೋಟೋಕಾಲ್ - Iterator Protocol)

ಅರೇ, ಸ್ಟ್ರಿಂಗ್, ಮ್ಯಾಪ್ (Map), ಸೆಟ್ (Set) ಗಳಂತಹ "ಇಟರೇಬಲ್" (Iterable) ಆಬ್ಜೆಕ್ಟ್‌ಗಳ ಮೇಲೆ ಕಾರ್ಯನಿರ್ವಹಿಸುತ್ತದೆ.

JavaScript 
let colors = ["Red", "Green", "Blue"];

for (let color of colors) {
  console.log(color);
}

ಹಿನ್ನೆಲೆಯಲ್ಲಿ ಏನಾಗುತ್ತದೆ (Under the hood):

  1. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಎಂಜಿನ್ colors ಅರೇಯ ಗುಪ್ತ [Symbol.iterator]() ಫಂಕ್ಷನ್ ಅನ್ನು ಕರೆಯುತ್ತದೆ.
  2. ಪ್ರತಿ ಹೆಜ್ಜೆಗೂ ಇಟರೇಟರ್‌ನ .next() ವಿಧಾನವನ್ನು (Method) ಕರೆಯುತ್ತದೆ.
  3. ಇದು ಮೆಮೊರಿಯಲ್ಲಿ ಒಂದು ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಸೃಷ್ಟಿಸುತ್ತದೆ: { value: "Red", done: false }.
  4. value ಅನ್ನು color ವೇರಿಯೇಬಲ್‌ಗೆ ಅಸೈನ್ ಮಾಡುತ್ತದೆ. done: true ಆಗುವವರೆಗೆ ಇದು ಸ್ವಯಂಚಾಲಿತವಾಗಿ ಮುಂದುವರಿಯುತ್ತದೆ. (ಇದು for ಲೂಪ್‌ಗಿಂತ ಹೆಚ್ಚು ಸುರಕ್ಷಿತ).

ಆ. for...in ಲೂಪ್ (ಆಬ್ಜೆಕ್ಟ್ ಎನ್ಯುಮರೇಶನ್ - Object Enumeration)

ಆಬ್ಜೆಕ್ಟ್‌ನ ಕೀಗಳನ್ನು (Keys/Properties) ಓದಲು ಬಳಸಲಾಗುತ್ತದೆ.

JavaScript 
let user = {
  name: "Raju",
  age: 25,
  city: "Bengaluru"
};

for (let key in user) {
  console.log(key + ": " + user[key]); 
}
ತಾಂತ್ರಿಕ ವಿಶ್ಲೇಷಣೆ ಮತ್ತು ಎಚ್ಚರಿಕೆ (Pro Tip):
  • ಡೈನಾಮಿಕ್ ಪ್ರಾಪರ್ಟಿ ಆಕ್ಸೆಸ್ (user[key]): ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಎಂಜಿನ್ (ಉದಾ: V8) ಆಬ್ಜೆಕ್ಟ್‌ನ ಹಿಡನ್ ಕ್ಲಾಸ್‌ಗಳು (Hidden Classes) ಅಥವಾ ಡಿಕ್ಷನರಿ ಮೋಡ್ (Hash table) ನಲ್ಲಿ ಆ ನಿರ್ದಿಷ್ಟ key ಯನ್ನು ಹುಡುಕಿ ಅದರ ಮೌಲ್ಯವನ್ನು ತರುತ್ತದೆ.
  • ಪ್ರೋಟೋಟೈಪ್ ಚೈನ್ ರಿಸ್ಕ್ (Prototype Chain Risk): for...in ಕೇವಲ ಆಬ್ಜೆಕ್ಟ್‌ನ ಸ್ವಂತ ಕೀಗಳನ್ನು ಮಾತ್ರವಲ್ಲದೆ, ಅದರ ಪ್ರೋಟೋಟೈಪ್ (Prototype) ನಿಂದ ಆನುವಂಶಿಕವಾಗಿ (Inherit) ಪಡೆದ ಕೀಗಳನ್ನೂ ಓದುತ್ತದೆ. ಆದ್ದರಿಂದ ವೃತ್ತಿಪರರು ಯಾವಾಗಲೂ Object.hasOwn(user, key) ಅನ್ನು ಬಳಸಿ ಪರಿಶೀಲಿಸುತ್ತಾರೆ.
  • ನಿಯಮ: ಅರೇಗಳ (Arrays) ಮೇಲೆ for...in ಅನ್ನು ಬಳಸಬಾರದು. ಏಕೆಂದರೆ ಅದು ಇಂಡೆಕ್ಸ್‌ಗಳನ್ನು ಸ್ಟ್ರಿಂಗ್‌ಗಳಾಗಿ ಪರಿವರ್ತಿಸುತ್ತದೆ ಮತ್ತು ಅರೇ ಕ್ರಮವನ್ನು (Order) ಖಚಿತಪಡಿಸುವುದಿಲ್ಲ. ಅರೇಗೆ ಯಾವಾಗಲೂ for...of ಅಥವಾ ಸಾಂಪ್ರದಾಯಿಕ for ಲೂಪ್ ಬಳಸಿ.

while ಲೂಪ್

JavaScript ನಲ್ಲಿ while ಲೂಪ್ ಅನ್ನು ಒಂದು ನಿರ್ದಿಷ್ಟ ಷರತ್ತು (condition) ನಿಜವಾಗಿರುವವರೆಗೆ (true) ಕೋಡ್‌ನ ಒಂದು ಭಾಗವನ್ನು ಮತ್ತೆ ಮತ್ತೆ ಕಾರ್ಯಗತಗೊಳಿಸಲು (execute) ಬಳಸಲಾಗುತ್ತದೆ.

ಹಂತ-ಹಂತವಾಗಿ ಇದು ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ?

  1. ಷರತ್ತು ಪರಿಶೀಲನೆ (Condition Check): ಲೂಪ್ ಪ್ರಾರಂಭವಾಗುವ ಮೊದಲು ಆವರಣದ () ಒಳಗಿನ ಷರತ್ತನ್ನು ಪರಿಶೀಲಿಸಲಾಗುತ್ತದೆ.
  2. ಕಾರ್ಯಗತಗೊಳಿಸುವಿಕೆ (Execution): ಷರತ್ತು true (ನಿಜ) ಆಗಿದ್ದರೆ ಮಾತ್ರ, ಆವರಣದ {} ಒಳಗಿನ ಕೋಡ್ ರನ್ ಆಗುತ್ತದೆ. false (ತಪ್ಪು) ಆಗಿದ್ದರೆ ಲೂಪ್ ಒಳಗೆ ಹೋಗುವುದಿಲ್ಲ.
  3. ಪುನರಾವರ್ತನೆ (Repetition): ಕೋಡ್ ಒಮ್ಮೆ ರನ್ ಆದ ನಂತರ, ಲೂಪ್ ಮತ್ತೆ ಹಿಂದಕ್ಕೆ ಹೋಗಿ ಷರತ್ತನ್ನು ಪರಿಶೀಲಿಸುತ್ತದೆ. ಇದು false ಆಗುವವರೆಗೆ ಈ ಪ್ರಕ್ರಿಯೆ ನಿರಂತರವಾಗಿ ಮುಂದುವರಿಯುತ್ತದೆ.
JavaScript (ಉದಾಹರಣೆ 1) 
let i = 1; // 1. ಪ್ರಾರಂಭದ ಮೌಲ್ಯ (Initialization)

while (i <= 5) { // 2. ಷರತ್ತು (i ಐದು ಅಥವಾ ಅದಕ್ಕಿಂತ ಕಡಿಮೆ ಇರಬೇಕು)
  console.log(i); // 3. ಕೋಡ್ ಕಾರ್ಯಗತಗೊಳ್ಳುತ್ತದೆ
  i++; // 4. ಪ್ರತಿ ಬಾರಿ i ನ ಬೆಲೆಯನ್ನು 1 ಹೆಚ್ಚಿಸುವುದು (Increment)
}
JavaScript (ಅರೇ ಜೊತೆ while ಲೂಪ್) 
let fruits = ["ಸೇಬು (Apple)", "ಬಾಳೆಹಣ್ಣು (Banana)", "ಮಾವಿನಹಣ್ಣು (Mango)"];
let index = 0; // ಅರೇಯ ಮೊದಲನೇ ಸ್ಥಾನ 0 ಯಿಂದ ಶುರುವಾಗುತ್ತದೆ

while (index < fruits.length) { // fruits.length ಇಲ್ಲಿ 3 ಆಗಿದೆ
  console.log(fruits[index]); 
  index++; 
}
ಎಚ್ಚರಿಕೆ (Infinite Loop):

while ಲೂಪ್ ಬಳಸುವಾಗ ವೇರಿಯೇಬಲ್ ಮೌಲ್ಯವನ್ನು ಹೆಚ್ಚಿಸಲು ಅಥವಾ ಕಡಿಮೆ ಮಾಡಲು (ಉದಾಹರಣೆಗೆ i++) ಎಂದಿಗೂ ಮರೆಯಬಾರದು. ಒಂದು ವೇಳೆ ಮರೆತರೆ, ಷರತ್ತು ಯಾವಾಗಲೂ true ಆಗಿಯೇ ಇರುತ್ತದೆ ಮತ್ತು ಲೂಪ್ ಎಂದಿಗೂ ನಿಲ್ಲದೆ ಬ್ರೌಸರ್ ಕ್ರ್ಯಾಶ್ ಆಗಬಹುದು.

do...while ಲೂಪ್

JavaScript ನಲ್ಲಿ do...while ಲೂಪ್ ಬಹುತೇಕ while ಲೂಪ್ ನಂತೆಯೇ ಕೆಲಸ ಮಾಡುತ್ತದೆ. ಆದರೆ ಒಂದು ಪ್ರಮುಖ ವ್ಯತ್ಯಾಸವಿದೆ: ಷರತ್ತು ಪರಿಶೀಲಿಸುವ ಮೊದಲೇ ಕೋಡ್ ಅನ್ನು ಕನಿಷ್ಠ ಒಮ್ಮೆಯಾದರೂ ಕಾರ್ಯಗತಗೊಳಿಸುತ್ತದೆ.

JavaScript (Syntax) 
let i = 1;

do {
  console.log(i); // ಮೊದಲು 1 ಪ್ರಿಂಟ್ ಆಗುತ್ತದೆ
  i++;            // i ಬೆಲೆ 2 ಆಗುತ್ತದೆ
} while (i <= 5); // ಕೊನೆಯಲ್ಲಿ 2 <= 5 ನಿಜವೇ ಎಂದು ಪರಿಶೀಲಿಸುತ್ತದೆ

while ಮತ್ತು do...while ನಡುವಿನ ಪ್ರಮುಖ ವ್ಯತ್ಯಾಸಗಳು

ವೈಶಿಷ್ಟ್ಯ (Feature) while ಲೂಪ್ do...while ಲೂಪ್
ಷರತ್ತು ಪರಿಶೀಲನೆ ಲೂಪ್ ಪ್ರಾರಂಭವಾಗುವ ಮೊದಲೇ ಪರಿಶೀಲಿಸುತ್ತದೆ. ಲೂಪ್ ಒಮ್ಮೆ ರನ್ ಆದ ನಂತರ (ಕೊನೆಯಲ್ಲಿ) ಪರಿಶೀಲಿಸುತ್ತದೆ.
ಲೂಪ್ ವಿಧ ಎಂಟ್ರಿ-ಕಂಟ್ರೋಲ್ಡ್ ಲೂಪ್ (Entry-controlled). ಎಕ್ಸಿಟ್-ಕಂಟ್ರೋಲ್ಡ್ ಲೂಪ್ (Exit-controlled).
ಷರತ್ತು ತಪ್ಪು (false) ಇದ್ದಾಗ ಕೋಡ್ ಒಮ್ಮೆಯೂ ರನ್ ಆಗುವುದಿಲ್ಲ. ಷರತ್ತು ತಪ್ಪಾಗಿದ್ದರೂ ಕೋಡ್ ಕನಿಷ್ಠ ಒಂದು ಬಾರಿ ರನ್ ಆಗುತ್ತದೆ.
JavaScript (ವ್ಯತ್ಯಾಸವನ್ನು ತೋರಿಸುವ ಉದಾಹರಣೆ) 
// 1. while ಲೂಪ್
let x = 10;
while (x < 5) { // 10 < 5 ಎಂಬುದು ಮೊದಲೇ false ಆಗಿದೆ
  console.log("ಇದು while ಲೂಪ್: " + x); 
}
// ಫಲಿತಾಂಶ: ಏನೂ ಪ್ರಿಂಟ್ ಆಗುವುದಿಲ್ಲ.

// 2. do...while ಲೂಪ್
let y = 10;
do {
  console.log("ಇದು do...while ಲೂಪ್: " + y); // ಇದು ಒಮ್ಮೆ ರನ್ ಆಗುತ್ತದೆ
  y++;
} while (y < 5); // 11 < 5 ತಪ್ಪು (false) ಆಗಿದ್ದರಿಂದ ಲೂಪ್ ಇಲ್ಲಿಗೆ ನಿಲ್ಲುತ್ತದೆ.
// ಫಲಿತಾಂಶ: "ಇದು do...while ಲೂಪ್: 10" ಎಂದು ಪ್ರಿಂಟ್ ಆಗುತ್ತದೆ.

ಯಾವಾಗ ಬಳಸಬೇಕು? (Use Cases)

User Input Validation: ಬಳಕೆದಾರರಿಂದ ಇನ್‌ಪುಟ್ ಪಡೆಯುವಾಗ ಬಳಕೆದಾರರು ಸರಿಯಾದ ಮಾಹಿತಿಯನ್ನು ನಮೂದಿಸುವವರೆಗೆ ಮತ್ತೆ ಮತ್ತೆ ಕೇಳಬೇಕಾದಾಗ ಇದನ್ನು ಬಳಸಲಾಗುತ್ತದೆ. ಉದಾಹರಣೆಗೆ: ಬಳಕೆದಾರರಿಗೆ "ದಯವಿಟ್ಟು ನಿಮ್ಮ ಪಾಸ್‌ವರ್ಡ್ ನಮೂದಿಸಿ" ಎಂದು ಕನಿಷ್ಠ ಒಮ್ಮೆ ಕೇಳಲೇಬೇಕು. ಅವರು ತಪ್ಪಾದ ಪಾಸ್‌ವರ್ಡ್ ನೀಡಿದರೆ ಮಾತ್ರ ಷರತ್ತು ಪರಿಶೀಲಿಸಿ ಮತ್ತೆ ಮತ್ತೆ ಕೇಳಬೇಕು.
