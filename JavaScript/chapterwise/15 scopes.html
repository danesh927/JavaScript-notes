JavaScript Local & Global Scope

ಶೂನ್ಯದಿಂದ ವೃತ್ತಿಪರ ಹಂತದವರೆಗೆ (Beginner to Pro) ಸಂಪೂರ್ಣ ಕೈಪಿಡಿ.

Concept ಎಂದರೇನು? (Definition)

Scope (ಸ್ಕೋಪ್) ಎಂದರೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ವೇರಿಯೇಬಲ್‌ಗಳ (Variables) "ವ್ಯಾಪ್ತಿ" ಅಥವಾ "ಗಡಿ". ಒಂದು ವೇರಿಯೇಬಲ್ ಅನ್ನು ನೀವು ಎಲ್ಲಿ ಬರೆಯುತ್ತೀರಿ ಎಂಬುದರ ಆಧಾರದ ಮೇಲೆ, ಆ ವೇರಿಯೇಬಲ್ ಅನ್ನು ಪ್ರೋಗ್ರಾಂನ ಯಾವ ಭಾಗದಲ್ಲಿ ಬಳಸಬಹುದು ಅಥವಾ ಬಳಸಬಾರದು ಎಂದು ನಿರ್ಧರಿಸುವ ನಿಯಮವೇ Scope.

ಈ Concept ಏಕೆ ಬೇಕು? (Why do we need it?)

ಭದ್ರತೆ (Security) ಮತ್ತು ಮೆಮೊರಿ ಉಳಿಸಲು! ಎಲ್ಲವನ್ನೂ ಒಂದೇ ಜಾಗದಲ್ಲಿ ಹಾಕಿದರೆ ಹೆಸರುಗಳು ಕ್ಲ್ಯಾಶ್ (Name collision) ಆಗುತ್ತವೆ. ಯಾವುದನ್ನು ಎಲ್ಲಿ ಮುಚ್ಚಿಡಬೇಕು ಎಂದು ನಿರ್ಧರಿಸಲು ಸ್ಕೋಪ್ ಸಹಾಯ ಮಾಡುತ್ತದೆ.

ನೈಜ ಜೀವನದ ಉದಾಹರಣೆ (Real-World Example)

ಸ್ಕೋಪ್ ಅನ್ನು ಅರ್ಥಮಾಡಿಕೊಳ್ಳಲು ಪಬ್ಲಿಕ್ ಪಾರ್ಕ್ (Public Park) ಮತ್ತು ನಿಮ್ಮ ಮನೆಯ ಬೆಡ್‌ರೂಮ್ (Bedroom) ಅನ್ನು ಕಲ್ಪಿಸಿಕೊಳ್ಳಿ.

🏞️
ಪಬ್ಲಿಕ್ ಪಾರ್ಕ್ (Global Scope)

ಪಾರ್ಕ್‌ನಲ್ಲಿರುವ ಬೆಂಚ್ ಅನ್ನು ಯಾರು ಬೇಕಾದರೂ ಬಳಸಬಹುದು. ಇದು ಎಲ್ಲರಿಗೂ ಮುಕ್ತವಾಗಿದೆ.

var parkBench = "Free";
🏠
ಮನೆಯ ಬೆಡ್‌ರೂಮ್ (Local Scope)

ನಿಮ್ಮ ಮನೆಯ ಒಳಗಿರುವ ಟಿವಿ ನಿಮ್ಮ ಮನೆಯವರಿಗೆ ಮಾತ್ರ. ಹೊರಗಿನವರಿಗೆ (ಪಾರ್ಕ್‌ನಲ್ಲಿರುವವರಿಗೆ) ಇದು ಸಿಗುವುದಿಲ್ಲ!

var myTV = "Private";

1. Global Scope (ಗ್ಲೋಬಲ್ ಸ್ಕೋಪ್)

ಯಾವುದೇ ಫಂಕ್ಷನ್ ಅಥವಾ ಬ್ಲಾಕ್‌ನ { } ಹೊರಗಡೆ ಸೃಷ್ಟಿಸುವ ವೇರಿಯೇಬಲ್‌ಗಳು Global Scope ನಲ್ಲಿರುತ್ತವೆ. ಇವುಗಳನ್ನು ಕೋಡ್‌ನ ಎಲ್ಲಿಂದಲಾದರೂ (ಯಾವ ಫಂಕ್ಷನ್ ಒಳಗಿಂದಲಾದರೂ) ಬಳಸಬಹುದು.

JavaScript - Global Scope 
// ಫಂಕ್ಷನ್ ಹೊರಗಿದೆ, ಹಾಗಾಗಿ ಇದು Global Variable (ಪಬ್ಲಿಕ್ ಪಾರ್ಕ್)
let playerName = "ವಿರಾಟ್"; 

function playGame() {
    // ಫಂಕ್ಷನ್ ಒಳಗಿಂದ Global Variable ಅನ್ನು ಕರೆಯುತ್ತಿದ್ದೇವೆ
    console.log(playerName + " ಆಡುತ್ತಿದ್ದಾರೆ!"); 
}

playGame(); // ಔಟ್‌ಪುಟ್: ವಿರಾಟ್ ಆಡುತ್ತಿದ್ದಾರೆ!
console.log(playerName); // ಔಟ್‌ಪುಟ್: ವಿರಾಟ್

2. Local / Function Scope (ಲೋಕಲ್ ಸ್ಕೋಪ್)

ಒಂದು ಫಂಕ್ಷನ್ function() { ... } ಒಳಗೆ ಸೃಷ್ಟಿಸಿದ ವೇರಿಯೇಬಲ್‌ಗಳು Local Scope ನಲ್ಲಿರುತ್ತವೆ. ಅಂದರೆ ಆ ಫಂಕ್ಷನ್ ಮುಗಿದ ತಕ್ಷಣ ಆ ವೇರಿಯೇಬಲ್ ಸಾಯುತ್ತದೆ. ಅದನ್ನು ಫಂಕ್ಷನ್‌ನ ಹೊರಗಡೆ ಬಳಸಲು ಸಾಧ್ಯವಿಲ್ಲ.

JavaScript - Local/Function Scope 
function secretMission() {
    // ಇದು Local Variable (ಮನೆಯ ಒಳಗಿನ ಟಿವಿ)
    let secretCode = 1234; 
    console.log("ಮಿಷನ್ ಒಳಗೆ: " + secretCode); // ಇದು ಕೆಲಸ ಮಾಡುತ್ತದೆ
}

secretMission();

// ಫಂಕ್ಷನ್ ಹೊರಗಡೆ ಕರೆಯಲು ಪ್ರಯತ್ನಿಸಿದರೆ...
console.log(secretCode); 
// ಔಟ್‌ಪುಟ್: ❌ ReferenceError: secretCode is not defined

3. Block Scope (ಬ್ಲಾಕ್ ಸ್ಕೋಪ್ - ES6 2015)

ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ (Modern JS) ನಲ್ಲಿ let ಮತ್ತು const ಕೀವರ್ಡ್‌ಗಳು ಬಂದ ಮೇಲೆ Block Scope ಪರಿಚಯವಾಯಿತು.

ಯಾವುದೇ { } (ಉದಾಹರಣೆಗೆ if, for ಲೂಪ್) ಒಳಗೆ let ಅಥವಾ const ಬಳಸಿದರೆ, ಅವು ಆ ಬ್ಲಾಕ್‌ನ ಹೊರಗೆ ಬರುವುದಿಲ್ಲ. (ಇದು ಬೀರುವಿನ ಒಳಗೆ ಲಾಕ್ ಮಾಡಿದ ಹಾಗೆ).

ಹಳೆಯ ವಿಧಾನದ ಎಚ್ಚರಿಕೆ (Deprecated Warning): var ಕೀವರ್ಡ್‌ಗೆ Block Scope ಅರ್ಥವಾಗುವುದಿಲ್ಲ! { } ಒಳಗೆ var ಬರೆದರೂ ಅದು ಹೊರಬಂದುಬಿಡುತ್ತದೆ. ಆದ್ದರಿಂದ ಆಧುನಿಕ ಕೋಡಿಂಗ್‌ನಲ್ಲಿ var ಬಳಸಬಾರದು.

4. Variable Shadowing (ಒಂದೇ ಹೆಸರಿನ ವೇರಿಯೇಬಲ್‌ಗಳಿದ್ದರೆ?)

ಸ್ಕೋಪ್ ಒಳಗಡೆ (Local) ಮತ್ತು ಹೊರಗಡೆ (Global) ಒಂದೇ ಹೆಸರಿನ ವೇರಿಯೇಬಲ್ ಇದ್ದರೆ ಏನಾಗುತ್ತದೆ? ಇದನ್ನು ತಾಂತ್ರಿಕವಾಗಿ "Variable Shadowing" (ವೇರಿಯೇಬಲ್ ಶ್ಯಾಡೋಯಿಂಗ್) ಎನ್ನುತ್ತಾರೆ.

ಸರಳ ಉದಾಹರಣೆ: ನಿಮ್ಮ ಕ್ಲಾಸ್‌ನಲ್ಲಿ ಇಬ್ಬರು "ರಾಜು" ಇದ್ದಾರೆ. ಒಬ್ಬ ನಿಮ್ಮ ಕ್ಲಾಸ್‌ನ ಒಳಗಿದ್ದಾನೆ (Local), ಮತ್ತೊಬ್ಬ ಪಕ್ಕದ ಕ್ಲಾಸ್‌ನಲ್ಲಿದ್ದಾನೆ (Global). ಮೇಷ್ಟ್ರು ಕ್ಲಾಸ್ ಒಳಗಡೆ ನಿಂತು "ರಾಜು ಬಾರೋ" ಅಂದರೆ, ಕ್ಲಾಸ್ ಒಳಗಿರುವ ರಾಜು (Local) ಬರುತ್ತಾನೆ ವಿನಹ ಪಕ್ಕದ ಕ್ಲಾಸ್‌ನವನಲ್ಲ!

ಅಂದರೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಒಳಗಿರುವ (Local) ವೇರಿಯೇಬಲ್‌ಗೆ ಮೊದಲ ಆದ್ಯತೆ ನೀಡುತ್ತದೆ. ಹೊರಗಿನ ವೇರಿಯೇಬಲ್ ತಾತ್ಕಾಲಿಕವಾಗಿ "ನೆರಳಿನಲ್ಲಿ (Shadow)" ಮರೆಯಾಗುತ್ತದೆ.
JavaScript - Variable Shadowing 
let heroName = "ದರ್ಶನ್"; // Global Variable

function movieSet() {
    // ಇದು ಒಳಗಿನ Local Variable. ಇದು ಹೊರಗಿನದನ್ನು "Shadow" ಮಾಡುತ್ತದೆ.
    let heroName = "ಸುದೀಪ್"; 
    
    console.log("ಫಿಲ್ಮ್ ಸೆಟ್ ಒಳಗೆ: " + heroName); // ಸುದೀಪ್ (Local ಗೆ ಆದ್ಯತೆ)
}

movieSet();

// ಫಂಕ್ಷನ್‌ನ ಹೊರಗಡೆ Global ವೇರಿಯೇಬಲ್ ಭದ್ರವಾಗಿರುತ್ತದೆ. ಬದಲಾಗಿರುವುದಿಲ್ಲ!
console.log("ಫಿಲ್ಮ್ ಸೆಟ್ ಹೊರಗಡೆ: " + heroName); // ದರ್ಶನ್

5. Lexical Scope (ವೃತ್ತಿಪರ ಹಂತ / Pro Level)

ಒಂದು ಫಂಕ್ಷನ್ ಒಳಗಡೆ ಮತ್ತೊಂದು ಫಂಕ್ಷನ್ ಇದ್ದರೆ (Nested Functions), ಒಳಗಿನ ಫಂಕ್ಷನ್ ತನ್ನ ಹೊರಗಿನ ಫಂಕ್ಷನ್‌ನ ವೇರಿಯೇಬಲ್‌ಗಳನ್ನು ಓದಬಲ್ಲದು. ಆದರೆ ಹೊರಗಿನ ಫಂಕ್ಷನ್‌ಗೆ ಒಳಗಿನ ವೇರಿಯೇಬಲ್ ಓದಲು ಸಾಧ್ಯವಿಲ್ಲ. ಇದನ್ನೇ Lexical Scope ಎನ್ನುತ್ತಾರೆ.

ಸರಳ ಉದಾಹರಣೆ: ಮಕ್ಕಳು (Inner function) ಅಪ್ಪ-ಅಮ್ಮನ (Outer function) ದುಡ್ಡನ್ನು ಬಳಸಬಹುದು, ಆದರೆ ಅಪ್ಪ-ಅಮ್ಮ ಮಕ್ಕಳ ಪಿಗ್ಗಿ-ಬ್ಯಾಂಕ್ (Piggy bank) ದುಡ್ಡನ್ನು ಬಳಸುವಂತಿಲ್ಲ!

JavaScript - Lexical Scope 
function outerFunction() {
    let outerMoney = 1000; // ಅಪ್ಪನ ದುಡ್ಡು

    function innerFunction() {
        let innerMoney = 50; // ಮಗುವಿನ ಪಿಗ್ಗಿ-ಬ್ಯಾಂಕ್
        console.log("ಮಗುವಿಗೆ ಸಿಗುವುದು: " + outerMoney); // 1000 (Lexical Scope ಇಂದ ಓದುತ್ತದೆ)
    }
    
    innerFunction();
}
outerFunction();

6. Execution Context ಮತ್ತು Call Stack (Internal Working)

ಇಂಟರ್ವ್ಯೂಗಳಲ್ಲಿ (Interviews) ಅತಿ ಹೆಚ್ಚು ಕೇಳಲಾಗುವ Technical ಕಾನ್ಸೆಪ್ಟ್. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಇಂಜಿನ್ (ಉದಾಹರಣೆಗೆ V8 Engine) ಮೆಮೊರಿಯಲ್ಲಿ Scope ಅನ್ನು ಹೇಗೆ ನಿಭಾಯಿಸುತ್ತದೆ?

1
Global Execution Context (GEC)

ನೀವು ಪ್ರೋಗ್ರಾಂ ರನ್ ಮಾಡಿದ ತಕ್ಷಣ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಒಂದು ದೊಡ್ಡ ಮೆಮೊರಿ ಜಾಗವನ್ನು ಸೃಷ್ಟಿಸುತ್ತದೆ (GEC). ಎಲ್ಲಾ Global ವೇರಿಯೇಬಲ್‌ಗಳು ಇಲ್ಲಿ ಸೇವ್ ಆಗುತ್ತವೆ.

2
Function Execution Context (FEC)

ನೀವು ಒಂದು ಫಂಕ್ಷನ್ playGame() ಕಾಲ್ ಮಾಡಿದಾಗ, Call Stack (ಕಾರ್ಯಗಳ ಸರದಿ) ಒಳಗೆ ಆ ಫಂಕ್ಷನ್‌ಗಾಗಿ ಒಂದು ಹೊಸ ಪ್ರತ್ಯೇಕ ಮೆಮೊರಿ ಬಾಕ್ಸ್ (FEC) ಸೃಷ್ಟಿಯಾಗುತ್ತದೆ. ಇದೇ Local Scope ನ ಜಾಗ.

3
Scope Chain & Cleanup

ಫಂಕ್ಷನ್ ಒಳಗೆ ವೇರಿಯೇಬಲ್ ಸಿಗದಿದ್ದರೆ, ಅದು Scope Chain ಮೂಲಕ ಹೊರಗಿನ GEC ಯಲ್ಲಿ ಹುಡುಕುತ್ತದೆ. ಫಂಕ್ಷನ್ ಕೆಲಸ ಮುಗಿದ ತಕ್ಷಣ, ಅದರ Local ಮೆಮೊರಿ (FEC) ಡಿಲೀಟ್ ಆಗುತ್ತದೆ (Garbage Collection). ಆದ್ದರಿಂದಲೇ Local ವೇರಿಯೇಬಲ್‌ಗಳು ಫಂಕ್ಷನ್ ಹೊರಗೆ ಸಿಗುವುದಿಲ್ಲ!
