JavaScript DOM "Single Element Selector"

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ DOM (Document Object Model) ನಲ್ಲಿ "Single Element Selector" ಎನ್ನುವುದು HTML ಪೇಜ್‌ನಿಂದ ಕೇವಲ ಒಂದು ನಿರ್ದಿಷ್ಟ ಎಲಿಮೆಂಟ್ ಅನ್ನು ಹಿಡಿದು (select), ಅದರ ಮೇಲೆ ನಮಗೆ ಬೇಕಾದ ಬದಲಾವಣೆಗಳನ್ನು (ಉದಾಹರಣೆಗೆ ಬಣ್ಣ, ಟೆಕ್ಸ್ಟ್ ಬದಲಾಯಿಸುವುದು, ಅನಿಮೇಷನ್ ಸೇರಿಸುವುದು) ಮಾಡಲು ಸಹಾಯ ಮಾಡುವ ವಿಧಾನವಾಗಿದೆ.

ಇದರ ಆಳವಾದ ಕಾರ್ಯವಿಧಾನ ಮತ್ತು ಉದಾಹರಣೆಗಳು ಇಲ್ಲಿವೆ:

1. document.getElementById('id_name') - ಆಳವಾದ ವಿವರಣೆ

ಇದು ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ?

ಇದು HTML ಎಲಿಮೆಂಟ್‌ಗಳಿಗೆ ನೀಡಲಾದ id ಅಟ್ರಿಬ್ಯೂಟ್ ಅನ್ನು ಮಾತ್ರ ಹುಡುಕುತ್ತದೆ. ನಿಯಮದ ಪ್ರಕಾರ, ಒಂದು HTML ಪೇಜ್‌ನಲ್ಲಿ ಒಂದು ನಿರ್ದಿಷ್ಟ id ಕೇವಲ ಒಂದೇ ಎಲಿಮೆಂಟ್‌ಗೆ ಇರಬೇಕು. ಹಾಗಾಗಿ ಬ್ರೌಸರ್ ಇಡೀ ಪೇಜ್ ಅನ್ನು ಹುಡುಕುವ ಬದಲು ನೇರವಾಗಿ ಆ ಎಲಿಮೆಂಟ್ ಅನ್ನು ಹಿಡಿಯುತ್ತದೆ. ಇದು ಅತ್ಯಂತ ವೇಗವಾದ (Fastest) ವಿಧಾನವಾಗಿದೆ.

Edge Case (ನೆನಪಿಡಬೇಕಾದ ಅಂಶ): ನೀವು ಹುಡುಕುತ್ತಿರುವ id ಪೇಜ್‌ನಲ್ಲಿ ಇಲ್ಲದಿದ್ದರೆ, ಇದು null (ಏನೂ ಇಲ್ಲ) ಎಂದು ಹಿಂದಿರುಗಿಸುತ್ತದೆ.

ಉದಾಹರಣೆ:

HTML 
<!-- HTML ಭಾಗ -->
<h1 id="heading">ಹಲೋ ಜಗತ್ತೇ!</h1>
<p id="description">ಇದು ಹಳೆಯ ಪ್ಯಾರಾಗ್ರಾಫ್.</p>
JavaScript 
// JavaScript ಭಾಗ

// 1. ಎಲಿಮೆಂಟ್ ಅನ್ನು ಆಯ್ಕೆ ಮಾಡುವುದು
const myHeading = document.getElementById('heading');
const myPara = document.getElementById('description');

// 2. ಅದರ ಮೇಲೆ ಬದಲಾವಣೆ ಮಾಡುವುದು (Manipulation)
myHeading.innerText = "ನಮಸ್ಕಾರ ಕನ್ನಡಿಗರೇ!"; // ಟೆಕ್ಸ್ಟ್ ಬದಲಾಯಿಸಿದೆ
myHeading.style.color = "red"; // ಅಕ್ಷರದ ಬಣ್ಣ ಕೆಂಪು ಮಾಡಿದೆ
myHeading.style.fontSize = "50px"; // ಅಕ್ಷರದ ಗಾತ್ರ ದೊಡ್ಡದು ಮಾಡಿದೆ

// 3. HTML ಅನ್ನು ಸಂಪೂರ್ಣವಾಗಿ ಬದಲಾಯಿಸುವುದು
myPara.innerHTML = "<strong>ಇದು ಹೊಸ, ದಪ್ಪವಾದ ಪ್ಯಾರಾಗ್ರಾಫ್.</strong>";

2. document.querySelector('css_selector') - ಆಳವಾದ ವಿವರಣೆ

ಇದು ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ?

ಇದು ಅತ್ಯಂತ ಶಕ್ತಿಶಾಲಿ (Powerful) ಸೆಲೆಕ್ಟರ್. CSS ನಲ್ಲಿ ನೀವು ಎಲಿಮೆಂಟ್‌ಗಳಿಗೆ ಸ್ಟೈಲ್ ಮಾಡಲು ಯಾವ ನಿಯಮಗಳನ್ನು ಬಳಸುತ್ತೀರೋ (ಉದಾಹರಣೆಗೆ: .class, #id, tag, [attribute]), ಅದೇ ನಿಯಮಗಳನ್ನು ಇಲ್ಲಿ ಬಳಸಬಹುದು.

ಮುಖ್ಯ ನಿಯಮ: ಇದು ಇಡೀ ಪೇಜ್ ಅನ್ನು ಮೇಲಿನಿಂದ ಕೆಳಗೆ ಹುಡುಕುತ್ತಾ ಬಂದು, ಸಿಗುವ ಮೊದಲನೇ ಎಲಿಮೆಂಟ್ ಅನ್ನು ಮಾತ್ರ ಆಯ್ಕೆ ಮಾಡಿ ಅಲ್ಲಿಗೆ ಹುಡುಕುವುದನ್ನು ನಿಲ್ಲಿಸುತ್ತದೆ.

ಉದಾಹರಣೆಗಳು:

HTML 
<!-- HTML ಭಾಗ -->
<div class="box">ಬಾಕ್ಸ್ 1</div>
<div class="box">ಬಾಕ್ಸ್ 2</div>
<a href="https://google.com">ಗೂಗಲ್</a>
<input type="password" placeholder="ಪಾಸ್‌ವರ್ಡ್ ಹಾಕಿ">
JavaScript 
// JavaScript ಭಾಗ

// A. Class ಮೂಲಕ ಆಯ್ಕೆ ಮಾಡುವುದು (ಕ್ಲಾಸ್ ಹೆಸರಿನ ಹಿಂದೆ ಚುಕ್ಕೆ '.' ಬಳಸಬೇಕು)
const firstBox = document.querySelector('.box');
firstBox.style.border = "2px solid blue";
// ಗಮನಿಸಿ: ಇದು 'ಬಾಕ್ಸ್ 1' ಗೆ ಮಾತ್ರ ನೀಲಿ ಬಾರ್ಡರ್ ಹಾಕುತ್ತದೆ, 'ಬಾಕ್ಸ್ 2' ಗಲ್ಲ. 
// ಏಕೆಂದರೆ ಇದು ಮೊದಲನೇ ಮ್ಯಾಚ್ ಅನ್ನು ಮಾತ್ರ ತೆಗೆದುಕೊಳ್ಳುತ್ತದೆ.

// B. ಟ್ಯಾಗ್ ಮೂಲಕ ಆಯ್ಕೆ (ನೇರವಾಗಿ ಟ್ಯಾಗ್ ಹೆಸರು)
const myLink = document.querySelector('a');
myLink.setAttribute('href', 'https://yahoo.com'); // ಲಿಂಕ್ ಅನ್ನು yahoo ಗೆ ಬದಲಾಯಿಸಿದೆ
myLink.innerText = "ಯಾಹೂ";

// C. Attribute (ಗುಣಲಕ್ಷಣ) ಮೂಲಕ ಸಂಕೀರ್ಣ ಆಯ್ಕೆ
// (ಪಾಸ್‌ವರ್ಡ್ ಇನ್ಪುಟ್ ಅನ್ನು ಮಾತ್ರ ಹುಡುಕಲು)
const passInput = document.querySelector('input[type="password"]');
passInput.style.backgroundColor = "lightgray";

3. ಎಲಿಮೆಂಟ್ ಅನ್ನು ಆಯ್ಕೆ ಮಾಡಿದ ಮೇಲೆ ಏನೇನು ಮಾಡಬಹುದು? (DOM Manipulation)

ಒಮ್ಮೆ ಎಲಿಮೆಂಟ್ ಅನ್ನು ಸೆಲೆಕ್ಟ್ ಮಾಡಿದ ಮೇಲೆ, ಅದಕ್ಕೆ ಜೀವ ತುಂಬಬಹುದು:

1. ಕ್ಲಾಸ್ ಸೇರಿಸುವುದು ಮತ್ತು ತೆಗೆಯುವುದು (classList):

const myButton = document.querySelector('#submit-btn');

myButton.classList.add('active'); // CSS ನಲ್ಲಿರುವ .active ಸ್ಟೈಲ್ ಅನ್ವಯವಾಗುತ್ತದೆ
myButton.classList.remove('hidden'); // .hidden ಸ್ಟೈಲ್ ತೆಗೆದುಹಾಕುತ್ತದೆ
myButton.classList.toggle('dark-mode'); // ಇದ್ದರೆ ತೆಗೆಯುತ್ತದೆ, ಇಲ್ಲದಿದ್ದರೆ ಸೇರಿಸುತ್ತದೆ

2. ಅಟ್ರಿಬ್ಯೂಟ್ (ಗುಣಲಕ್ಷಣ) ಓದುವುದು ಮತ್ತು ಬದಲಾಯಿಸುವುದು:

const myImage = document.querySelector('.profile-pic');

// ಇಮೇಜ್‌ನ src ಅಟ್ರಿಬ್ಯೂಟ್ ಬದಲಾಯಿಸುವುದು
myImage.src = "new-photo.jpg"; 
myImage.alt = "ಹೊಸ ಪ್ರೊಫೈಲ್ ಚಿತ್ರ";

4. Pro Level Tip: Null Check (ದೋಷ ತಡೆಯುವುದು)

ನಾವು ಹುಡುಕುತ್ತಿರುವ ಎಲಿಮೆಂಟ್ ಪೇಜ್‌ನಲ್ಲಿ ಇಲ್ಲದಿದ್ದರೆ, ಅದರ ಮೇಲೆ ಸ್ಟೈಲ್ ಅಪ್ಲೈ ಮಾಡಲು ಹೋದಾಗ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ TypeError ಕೊಡುತ್ತದೆ ಮತ್ತು ನಿಮ್ಮ ಮುಂದಿನ ಕೋಡ್ ಕೆಲಸ ಮಾಡುವುದಿಲ್ಲ. ಇದನ್ನು ತಡೆಯಲು ಯಾವಾಗಲೂ if ಕಂಡೀಶನ್ ಬಳಸಿ ದೃಢಪಡಿಸಿಕೊಳ್ಳಬೇಕು. 

// ಪೇಜ್‌ನಲ್ಲಿ .main-title ಎಂಬ ಕ್ಲಾಸ್ ಇದೆಯೋ ಇಲ್ಲವೋ ಎಂದು ಚೆಕ್ ಮಾಡುವುದು
const title = document.querySelector('.main-title');

if (title) { 
    // ಎಲಿಮೆಂಟ್ ಸಿಕ್ಕರೆ ಮಾತ್ರ ಈ ಕೋಡ್ ರನ್ ಆಗುತ್ತದೆ (ಇದು ಸೇಫ್ ವಿಧಾನ)
    title.style.color = "green";
} else {
    // ಎಲಿಮೆಂಟ್ ಸಿಗದಿದ್ದರೆ ಏನಾಗಬೇಕು?
    console.log("ಎಲಿಮೆಂಟ್ ಪೇಜ್‌ನಲ್ಲಿ ಕಂಡುಬಂದಿಲ್ಲ!");
}

CSS Pseudo-classes ಮತ್ತು querySelector

JavaScript ನ querySelector ನ ಅತಿ ದೊಡ್ಡ ಶಕ್ತಿ ಎಂದರೆ ಅದು CSS ನ Pseudo-classes (:first-child, :last-child, :nth-child) ಗಳನ್ನು ಕೂಡ ಬೆಂಬಲಿಸುತ್ತದೆ. ಒಂದು ಗುಂಪಿನಲ್ಲಿರುವ (ಉದಾಹರಣೆಗೆ ಒಂದು ಲಿಸ್ಟ್ ಅಥವಾ ಟೇಬಲ್‌ನಲ್ಲಿರುವ) ನಿರ್ದಿಷ್ಟವಾದ ಎಲಿಮೆಂಟ್ ಅನ್ನು ಮಾತ್ರ ಆಯ್ಕೆ ಮಾಡಲು ಇವು ಅತ್ಯಂತ ಉಪಯುಕ್ತವಾಗಿವೆ.

ಉದಾಹರಣೆಗೆ, ನಮ್ಮ ಬಳಿ ಈ ಕೆಳಗಿನ HTML ಇದೆ ಎಂದುಕೊಳ್ಳಿ:

<ul class="fruits-list">
    <li>ಸೇಬು (Apple) - 1</li>
    <li>ಬಾಳೆಹಣ್ಣು (Banana) - 2</li>
    <li>ಮಾವಿನಹಣ್ಣು (Mango) - 3</li>
    <li>ದ್ರಾಕ್ಷಿ (Grapes) - 4</li>
</ul>

ಈಗ ಇವುಗಳನ್ನು JavaScript ನಲ್ಲಿ ಒಂದೊಂದಾಗಿ ಹೇಗೆ ಆಯ್ಕೆ ಮಾಡುವುದು ಎಂದು ನೋಡೋಣ:

1. :first-child (ಮೊದಲನೇ ಎಲಿಮೆಂಟ್)

ಒಂದು ಗುಂಪಿನಲ್ಲಿರುವ (siblings) ಎಲಿಮೆಂಟ್‌ಗಳಲ್ಲಿ ಮೊದಲನೆಯದನ್ನು ಮಾತ್ರ ಆಯ್ಕೆ ಮಾಡಲು ಇದನ್ನು ಬಳಸಲಾಗುತ್ತದೆ.

// ಲಿಸ್ಟ್‌ನ ಮೊದಲನೇ <li> (ಸೇಬು) ಎಲಿಮೆಂಟ್ ಅನ್ನು ಆಯ್ಕೆ ಮಾಡುತ್ತದೆ
const firstFruit = document.querySelector('.fruits-list li:first-child');
firstFruit.style.color = "red"; 
// ಈಗ 'ಸೇಬು' ಮಾತ್ರ ಕೆಂಪು ಬಣ್ಣಕ್ಕೆ ತಿರುಗುತ್ತದೆ.

2. :last-child (ಕೊನೆಯ ಎಲಿಮೆಂಟ್)

ಒಂದು ಗುಂಪಿನಲ್ಲಿರುವ ಎಲಿಮೆಂಟ್‌ಗಳಲ್ಲಿ ಕೊನೆಯದನ್ನು ಮಾತ್ರ ಆಯ್ಕೆ ಮಾಡಲು ಇದನ್ನು ಬಳಸಲಾಗುತ್ತದೆ.

// ಲಿಸ್ಟ್‌ನ ಕೊನೆಯ <li> (ದ್ರಾಕ್ಷಿ) ಎಲಿಮೆಂಟ್ ಅನ್ನು ಆಯ್ಕೆ ಮಾಡುತ್ತದೆ
const lastFruit = document.querySelector('.fruits-list li:last-child');
lastFruit.style.fontWeight = "bold"; 
// ಈಗ 'ದ್ರಾಕ್ಷಿ' ದಪ್ಪ (bold) ಅಕ್ಷರಗಳಲ್ಲಿ ಕಾಣುತ್ತದೆ.

3. :nth-child(n) (ನಿರ್ದಿಷ್ಟ ಎಲಿಮೆಂಟ್)

ನಮಗೆ ಮೊದಲನೇ ಅಥವಾ ಕೊನೆಯ ಎಲಿಮೆಂಟ್ ಬೇಡ, ಬದಲಾಗಿ ಮಧ್ಯದಲ್ಲಿರುವ 2ನೇ ಅಥವಾ 3ನೇ ಎಲಿಮೆಂಟ್ ಬೇಕು ಎಂದಾಗ ಇದನ್ನು ಬಳಸುತ್ತೇವೆ. n ಜಾಗದಲ್ಲಿ ನಮಗೆ ಬೇಕಾದ ಸಂಖ್ಯೆಯನ್ನು ಹಾಕಬೇಕು.

// ಲಿಸ್ಟ್‌ನ 2ನೇ <li> (ಬಾಳೆಹಣ್ಣು) ಎಲಿಮೆಂಟ್ ಅನ್ನು ಆಯ್ಕೆ ಮಾಡುತ್ತದೆ
const secondFruit = document.querySelector('.fruits-list li:nth-child(2)');
secondFruit.style.backgroundColor = "yellow"; 
// ಈಗ 2ನೇ ಐಟಂ 'ಬಾಳೆಹಣ್ಣು' ನ ಹಿನ್ನೆಲೆ ಬಣ್ಣ ಹಳದಿಯಾಗುತ್ತದೆ.

// ಲಿಸ್ಟ್‌ನ 3ನೇ <li> (ಮಾವಿನಹಣ್ಣು) ಎಲಿಮೆಂಟ್ ಅನ್ನು ಆಯ್ಕೆ ಮಾಡುತ್ತದೆ
const thirdFruit = document.querySelector('.fruits-list li:nth-child(3)');

💡 Pro Tip: nth-child ಜೊತೆ even ಮತ್ತು odd ಬಳಕೆ

ನೀವು nth-child ಒಳಗೆ even (ಸಮ ಸಂಖ್ಯೆ) ಅಥವಾ odd (ಬೆಸ ಸಂಖ್ಯೆ) ಯನ್ನು ಕೂಡ ಬಳಸಬಹುದು.

ಆದರೆ ನೆನಪಿಡಿ: ನಾವು ಇಲ್ಲಿ ಕಲಿಯುತ್ತಿರುವುದು Single Element Selector (querySelector). ಇದು ಪೇಜ್‌ನಲ್ಲಿ ಮ್ಯಾಚ್ ಆಗುವ ಮೊದಲನೇ ಎಲಿಮೆಂಟ್ ಅನ್ನು ಮಾತ್ರ ಹಿಡಿಯುತ್ತದೆ.

  • ನೀವು document.querySelector('li:nth-child(even)') ಎಂದು ಕೊಟ್ಟರೆ, ಅದು ಪೇಜ್‌ನ ಮೊದಲನೇ ಸಮ ಸಂಖ್ಯೆಯ (ಅಂದರೆ 2ನೇ) ಐಟಂ ಅನ್ನು ಮಾತ್ರ ಹಿಡಿಯುತ್ತದೆ.
  • ಎಲ್ಲಾ ಸಮ/ಬೆಸ ಸಂಖ್ಯೆಯ ಐಟಂಗಳನ್ನು ಒಟ್ಟಿಗೆ ಹಿಡಿಯಲು ನಾವು Multiple Element Selector (querySelectorAll) ಅನ್ನು ಬಳಸಬೇಕಾಗುತ್ತದೆ.

innerHTML vs innerText vs textContent

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಒಂದು ಎಲಿಮೆಂಟ್ ಅನ್ನು ಆಯ್ಕೆ ಮಾಡಿದ ನಂತರ, ಆ ಎಲಿಮೆಂಟ್‌ನ ಒಳಗಿರುವ ವಿಷಯವನ್ನು (Content) ಓದಲು ಅಥವಾ ಬದಲಾಯಿಸಲು ನಾವು ಕೆಲವು ಮುಖ್ಯವಾದ ಪ್ರಾಪರ್ಟಿಗಳನ್ನು (Properties) ಬಳಸುತ್ತೇವೆ.

ಒಂದು ಉದಾಹರಣೆ HTML ನೊಂದಿಗೆ ಇವುಗಳ ಆಳವಾದ ವ್ಯತ್ಯಾಸವನ್ನು ನೋಡೋಣ:

<div id="demo-box">
    ನಮಸ್ಕಾರ, <span style="display: none;">ಗುಪ್ತ ಸಂದೇಶ</span> <strong>ಕನ್ನಡಿಗರೇ</strong>!
</div>

1. innerHTML

ಇದು ಎಲಿಮೆಂಟ್‌ನ ಒಳಗಿರುವ ಸಂಪೂರ್ಣ HTML ಕೋಡ್ ಅನ್ನು (ಟ್ಯಾಗ್‌ಗಳ ಸಮೇತ) ಹಿಡಿಯುತ್ತದೆ ಅಥವಾ ಹೊಸ HTML ಅನ್ನು ಸೇರಿಸಲು ಬಿಡುತ್ತದೆ.

  • ಬಳಕೆ: ನೀವು ಟೆಕ್ಸ್ಟ್ ಜೊತೆಗೆ ಹೊಸ HTML ಟ್ಯಾಗ್‌ಗಳನ್ನು (ಉದಾ: ಲಿಂಕ್, ಬೋಲ್ಡ್, ಹೊಸ ಲಿಸ್ಟ್) ಸೇರಿಸಬೇಕಾದರೆ ಇದನ್ನು ಬಳಸಿ.
const box = document.querySelector('#demo-box');

// ಓದುವುದು (Get): 
console.log(box.innerHTML); 
// ಔಟ್‌ಪುಟ್: ನಮಸ್ಕಾರ, <span style="display: none;">ಗುಪ್ತ ಸಂದೇಶ</span> <strong>ಕನ್ನಡಿಗರೇ</strong>!

// ಬದಲಾಯಿಸುವುದು (Set):
box.innerHTML = "<h1>ಹೊಸ ಸ್ವಾಗತ</h1><p>ಇದು ಹೊಸ ಪ್ಯಾರಾಗ್ರಾಫ್.</p>";
// ಈಗ ಹಳೆಯ ಕಂಟೆಂಟ್ ಹೋಗಿ, ಹೊಸ <h1> ಮತ್ತು <p> ಟ್ಯಾಗ್‌ಗಳು ಪೇಜ್‌ನಲ್ಲಿ ರೆಂಡರ್ ಆಗುತ್ತವೆ.

2. innerText

ಇದು ಕೇವಲ ಬಳಕೆದಾರನಿಗೆ ಪರದೆಯ ಮೇಲೆ ಕಾಣುವ (Visible) ಟೆಕ್ಸ್ಟ್ ಅನ್ನು ಮಾತ್ರ ಹಿಡಿಯುತ್ತದೆ. ಇದು HTML ಟ್ಯಾಗ್‌ಗಳನ್ನು ನಿರ್ಲಕ್ಷಿಸುತ್ತದೆ ಮತ್ತು CSS ಮೂಲಕ ಅಡಗಿಸಿಟ್ಟ (hidden) ಟೆಕ್ಸ್ಟ್ ಅನ್ನು ತೋರಿಸುವುದಿಲ್ಲ.

  • ಬಳಕೆ: ಕೇವಲ ಓದಲು ಸಾಧ್ಯವಾಗುವ ಟೆಕ್ಸ್ಟ್ ಬೇಕಿದ್ದಾಗ, ಅಥವಾ ನೀವು ನೀಡುವ ಟೆಕ್ಸ್ಟ್ HTML ಕೋಡ್ ಆಗಿ ಕನ್ವರ್ಟ್ ಆಗಬಾರದು ಎಂದಾಗ ಇದನ್ನು ಬಳಸಿ.
// ಓದುವುದು (Get):
console.log(box.innerText); 
// ಔಟ್‌ಪುಟ್: ನಮಸ್ಕಾರ, ಕನ್ನಡಿಗರೇ! 
// (ಇಲ್ಲಿ 'ಗುಪ್ತ ಸಂದೇಶ' ಕಾಣುವುದಿಲ್ಲ ಏಕೆಂದರೆ ಅದು display: none ನಲ್ಲಿದೆ)

// ಬದಲಾಯಿಸುವುದು (Set):
box.innerText = "<strong>ಹೊಸ ಟೆಕ್ಸ್ಟ್</strong>";
// ಇದು ದಪ್ಪಕ್ಷರದಲ್ಲಿ (Bold) ಬರುವುದಿಲ್ಲ. ಬದಲಾಗಿ <strong>ಹೊಸ ಟೆಕ್ಸ್ಟ್</strong> ಎಂಬುದು ಸಾಧಾರಣ ಟೆಕ್ಸ್ಟ್ ಆಗಿ ಪರದೆಯ ಮೇಲೆ ಹಾಗೆಯೇ ಕಾಣುತ್ತದೆ.

3. textContent

ಇದು innerText ಗೆ ಬಹಳ ಹತ್ತಿರವಾಗಿದೆ, ಆದರೆ ಇದು ಎಲಿಮೆಂಟ್‌ನ ಒಳಗಿರುವ ಪ್ರತಿಯೊಂದು ಟೆಕ್ಸ್ಟ್ ಅನ್ನು (ಅಡಗಿಸಿಟ್ಟದ್ದನ್ನು ಸೇರಿ) ಹಿಡಿಯುತ್ತದೆ. ಇದು innerText ಗಿಂತ ವೇಗವಾಗಿ ಕೆಲಸ ಮಾಡುತ್ತದೆ ಏಕೆಂದರೆ ಇದು CSS ಸ್ಟೈಲ್‌ಗಳನ್ನು ಲೆಕ್ಕಿಸುವುದಿಲ್ಲ.

// ಓದುವುದು (Get):
console.log(box.textContent); 
// ಔಟ್‌ಪುಟ್: ನಮಸ್ಕಾರ, ಗುಪ್ತ ಸಂದೇಶ ಕನ್ನಡಿಗರೇ! 
// (ಅಡಗಿಸಿಟ್ಟ ಟೆಕ್ಸ್ಟ್ ಕೂಡ ಇಲ್ಲಿ ಬರುತ್ತದೆ)

ಇತರ ಮುಖ್ಯವಾದ Properties

4. value (ಇನ್ಪುಟ್ ಫೀಲ್ಡ್‌ಗಳಿಗಾಗಿ)

innerHTML ಅಥವಾ innerText ಗಳು div, p, h1 ಗಳಿಗೆ ಕೆಲಸ ಮಾಡಿದರೆ, ಫಾರ್ಮ್ ಎಲಿಮೆಂಟ್‌ಗಳಾದ <input>, <textarea> ಗಳಲ್ಲಿ ಬಳಕೆದಾರರು ಟೈಪ್ ಮಾಡಿದ ಡೇಟಾವನ್ನು ಹಿಡಿಯಲು value ಪ್ರಾಪರ್ಟಿಯನ್ನು ಬಳಸಬೇಕು.

<input type="text" id="username" value="ರಾಜಕುಮಾರ್">
const inputField = document.querySelector('#username');

// ಓದುವುದು:
console.log(inputField.value); // ಔಟ್‌ಪುಟ್: ರಾಜಕುಮಾರ್

// ಬದಲಾಯಿಸುವುದು:
inputField.value = "ಪುನೀತ್"; // ಈಗ ಟೆಕ್ಸ್ಟ್ ಬಾಕ್ಸ್‌ನಲ್ಲಿ 'ಪುನೀತ್' ಎಂದು ಬದಲಾಗುತ್ತದೆ.
5. id ಮತ್ತು className

ಎಲಿಮೆಂಟ್‌ನ ಐಡಿ ಅಥವಾ ಕ್ಲಾಸ್ ಅನ್ನು ನೇರವಾಗಿ ಬದಲಾಯಿಸಲು ಇವು ಬಳಕೆಯಾಗುತ್ತವೆ.

const myPara = document.querySelector('p');

// ಹೊಸ ಐಡಿ ಕೊಡುವುದು
myPara.id = "new-id-name";

// ಕ್ಲಾಸ್ ಹೆಸರನ್ನು ಸಂಪೂರ್ಣವಾಗಿ ಬದಲಾಯಿಸುವುದು (ಹಳೆಯ ಕ್ಲಾಸ್ ಅಳಿಸಿಹೋಗುತ್ತದೆ)
myPara.className = "alert-text error";

ಮುಖ್ಯ ಸೂಚನೆ: innerHTML ಅನ್ನು ಬಳಸುವಾಗ ಎಚ್ಚರಿಕೆ ವಹಿಸಿ. ಬಳಕೆದಾರರು ಟೈಪ್ ಮಾಡಿದ ಡೇಟಾವನ್ನು (ಉದಾಹರಣೆಗೆ ಕಮೆಂಟ್ ಸೆಕ್ಷನ್) ನೇರವಾಗಿ innerHTML ಮೂಲಕ ಪೇಜ್‌ಗೆ ಸೇರಿಸಬಾರದು, ಇದು ಹ್ಯಾಕಿಂಗ್‌ಗೆ (Cross-Site Scripting ಅಥವಾ XSS) ದಾರಿ ಮಾಡಿಕೊಡುತ್ತದೆ. ಅಂತ ಕಡೆ ಕೇವಲ textContent ಅಥವಾ innerText ಬಳಸಿ.

classList vs className

JavaScript ನಲ್ಲಿ ಒಂದು ಎಲಿಮೆಂಟ್‌ಗೆ CSS ಕ್ಲಾಸ್‌ಗಳನ್ನು ಸೇರಿಸಲು ಅಥವಾ ತೆಗೆಯಲು className ಮತ್ತು classList ಎರಡನ್ನೂ ಬಳಸಬಹುದು. ಆದರೆ ಪ್ರೊಗ್ರಾಮರ್‌ಗಳು ಯಾವಾಗಲೂ classList ಅನ್ನು ಹೆಚ್ಚು ಇಷ್ಟಪಡುತ್ತಾರೆ. ಇದಕ್ಕೆ ಮುಖ್ಯ ಕಾರಣ "ಹಳೆಯ ಡೇಟಾ ಅಳಿಸಿಹೋಗದಿರುವುದು" (Avoids Overwriting).

1. className ನ ಸಮಸ್ಯೆ ಏನು?

className ಅನ್ನು ಬಳಸಿದಾಗ, ಎಲಿಮೆಂಟ್‌ನಲ್ಲಿ ಮೊದಲಿರುವ ಎಲ್ಲಾ ಕ್ಲಾಸ್‌ಗಳು ಸಂಪೂರ್ಣವಾಗಿ ಅಳಿಸಿಹೋಗುತ್ತವೆ ಮತ್ತು ನೀವು ನೀಡಿದ ಹೊಸ ಕ್ಲಾಸ್ ಮಾತ್ರ ಸೇರ್ಪಡೆಯಾಗುತ್ತದೆ.

<!-- HTML ನಲ್ಲಿ ಒಂದು ಬಟನ್ ಇದೆ, ಅದಕ್ಕೆ 2 ಕ್ಲಾಸ್‌ಗಳಿವೆ -->
<button id="myBtn" class="btn btn-primary">ಸಬ್ಮಿಟ್</button>
const button = document.querySelector('#myBtn');

// className ಬಳಸಿದರೆ:
button.className = "active"; 

// ಈಗ HTML ಏನಾಗುತ್ತದೆ? 
// <button id="myBtn" class="active">ಸಬ್ಮಿಟ್</button>
// ಗಮನಿಸಿ: ಹಳೆಯ 'btn' ಮತ್ತು 'btn-primary' ಕ್ಲಾಸ್‌ಗಳು ಡಿಲೀಟ್ ಆಗಿಹೋದವು! ಇದರಿಂದ ಬಟನ್‌ನ ಡಿಸೈನ್ ಹಾಳಾಗುತ್ತದೆ.

ಒಂದು ವೇಳೆ ಹಳೆಯದನ್ನು ಉಳಿಸಿಕೊಂಡು ಹೊಸದನ್ನು ಸೇರಿಸಬೇಕೆಂದರೆ button.className += " active"; ಎಂದು ಸ್ಪೇಸ್ ಕೊಟ್ಟು ಬರೆಯಬೇಕು. ಇದು ಕೋಡಿಂಗ್ ಮಾಡುವಾಗ ತಪ್ಪುಗಳಿಗೆ (Bugs) ದಾರಿ ಮಾಡಿಕೊಡುತ್ತದೆ.

2. classList ಏಕೆ ಉತ್ತಮ?

classList ಎನ್ನುವುದು ಎಲಿಮೆಂಟ್‌ನಲ್ಲಿರುವ ಕ್ಲಾಸ್‌ಗಳನ್ನು ಒಂದು ಪಟ್ಟಿಯಂತೆ (List) ನೋಡುತ್ತದೆ. ಇದರಿಂದ ನೀವು ಹಳೆಯ ಕ್ಲಾಸ್‌ಗಳಿಗೆ ಯಾವುದೇ ತೊಂದರೆ ಮಾಡದೆ, ಕೇವಲ ಒಂದೇ ಒಂದು ಕ್ಲಾಸ್‌ನ ಮೇಲೆ ನಿಖರವಾಗಿ ಕೆಲಸ ಮಾಡಬಹುದು. ಇದರಲ್ಲಿ 4 ಬಹಳ ಮುಖ್ಯವಾದ ವಿಧಾನಗಳಿವೆ (Methods):

A. classList.add()

ಹಳೆಯ ಕ್ಲಾಸ್‌ಗಳನ್ನು ಹಾಗೆಯೇ ಉಳಿಸಿಕೊಂಡು, ಹೊಸ ಕ್ಲಾಸ್ ಅನ್ನು ಮಾತ್ರ ಕೊನೆಗೆ ಸೇರಿಸುತ್ತದೆ.

button.classList.add('active');
B. classList.remove()

ನೀವು ಹೇಳಿದ ಕ್ಲಾಸ್ ಅನ್ನು ಮಾತ್ರ ಹುಡುಕಿ ತೆಗೆದುಹಾಕುತ್ತದೆ.

button.classList.remove('btn-primary');
C. classList.toggle()

ಕ್ಲಾಸ್ ಇಲ್ಲದಿದ್ದರೆ ಸೇರಿಸುತ್ತದೆ, ಇದ್ದರೆ ತೆಗೆಯುತ್ತದೆ. (On/Off ಸ್ವಿಚ್).

button.classList.toggle('dark-mode');
D. classList.contains()

ನಿರ್ದಿಷ್ಟ ಕ್ಲಾಸ್ ಇದೆಯೋ ಇಲ್ಲವೋ ಎಂದು ಚೆಕ್ ಮಾಡುತ್ತದೆ (true/false).

if(button.classList.contains('active'))
ಸಾರಾಂಶ:
  • className: ಗುಡಿಸಿಹಾಕಿ ಹೊಸದಾಗಿ ಬರೆಯುತ್ತದೆ (Overwrite).
  • classList: ನಿಖರವಾಗಿ ಬೇಕಾದ್ದನ್ನು ಮಾತ್ರ ಸೇರಿಸುತ್ತದೆ, ತೆಗೆಯುತ್ತದೆ ಅಥವಾ ಟಾಗಲ್ ಮಾಡುತ್ತದೆ. ಇದು ಕ್ಲೀನ್, ಸೇಫ್ ಮತ್ತು ಆಧುನಿಕ ವಿಧಾನ.

getAttribute ಮತ್ತು setAttribute

HTML ಎಲಿಮೆಂಟ್‌ಗಳು ತಮ್ಮದೇ ಆದ ಕೆಲವು ಗುಣಲಕ್ಷಣಗಳನ್ನು (Attributes) ಹೊಂದಿರುತ್ತವೆ. ಉದಾಹರಣೆಗೆ: ಇಮೇಜ್‌ಗಳಿಗೆ src ಮತ್ತು alt, ಲಿಂಕ್‌ಗಳಿಗೆ href, ಇನ್ಪುಟ್ ಫೀಲ್ಡ್‌ಗಳಿಗೆ type ಮತ್ತು placeholder ಇತ್ಯಾದಿ. ಈ ಗುಣಲಕ್ಷಣಗಳ ಮೌಲ್ಯವನ್ನು (value) ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಮೂಲಕ ಓದಲು (Read) ಮತ್ತು ಹೊಸದಾಗಿ ಸೇರಿಸಲು/ಬದಲಾಯಿಸಲು (Write/Update) getAttribute ಮತ್ತು setAttribute ವಿಧಾನಗಳನ್ನು ಬಳಸಲಾಗುತ್ತದೆ.

1. getAttribute('attribute_name') (ಓದಲು)

ಇದು ಒಂದು ಎಲಿಮೆಂಟ್‌ನಲ್ಲಿರುವ ನಿರ್ದಿಷ್ಟ ಅಟ್ರಿಬ್ಯೂಟ್‌ನ ಮೌಲ್ಯವನ್ನು (value) ನಮಗೆ ನೀಡುತ್ತದೆ. ಆ ಅಟ್ರಿಬ್ಯೂಟ್ ಇಲ್ಲದಿದ್ದರೆ ಅದು null ಎಂದು ಹಿಂದಿರುಗಿಸುತ್ತದೆ.

<a id="myLink" href="https://google.com" target="_blank">ಗೂಗಲ್</a>
const link = document.querySelector('#myLink');
const linkUrl = link.getAttribute('href');
console.log(linkUrl); // ಔಟ್‌ಪುಟ್: https://google.com

2. setAttribute('attribute_name', 'new_value') (ಬದಲಾಯಿಸಲು)

ಇದು ಒಂದು ಎಲಿಮೆಂಟ್‌ಗೆ ಹೊಸ ಅಟ್ರಿಬ್ಯೂಟ್ ಅನ್ನು ಸೇರಿಸುತ್ತದೆ. ಒಂದು ವೇಳೆ ಆ ಅಟ್ರಿಬ್ಯೂಟ್ ಈಗಾಗಲೇ ಇದ್ದರೆ, ಅದರ ಹಳೆಯ ಮೌಲ್ಯವನ್ನು ಅಳಿಸಿ, ನಾವು ನೀಡುವ ಹೊಸ ಮೌಲ್ಯಕ್ಕೆ ಬದಲಾಯಿಸುತ್ತದೆ (Overwrite ಮಾಡುತ್ತದೆ).

<img id="profile-pic" src="old-photo.jpg" alt="ಹಳೆಯ ಫೋಟೋ">
const image = document.querySelector('#profile-pic');
image.setAttribute('src', 'new-photo.jpg');
image.setAttribute('alt', 'ಹೊಸ ಪ್ರೊಫೈಲ್ ಚಿತ್ರ');
image.setAttribute('title', 'ಇದು ನನ್ನ ಹೊಸ ಫೋಟೋ');

3. ರಿಯಲ್-ವರ್ಲ್ಡ್ ಉದಾಹರಣೆ (ಪಾಸ್‌ವರ್ಡ್ ತೋರಿಸು/ಅಡಗಿಸು)

ವೆಬ್‌ಸೈಟ್‌ಗಳಲ್ಲಿ ಲಾಗಿನ್ ಆಗುವಾಗ "Show Password" (ಪಾಸ್‌ವರ್ಡ್ ತೋರಿಸು) ಎಂಬ ಆಯ್ಕೆ ಇರುತ್ತದೆ. ಇದು getAttribute ಮತ್ತು setAttribute ಬಳಸಿ ಮಾಡುವ ಅತ್ಯಂತ ಜನಪ್ರಿಯ ಟ್ರಿಕ್ ಆಗಿದೆ.

← Live Working Demo! 
const passwordBox = document.querySelector('#passInput');

// 1. ಮೊದಲು ಇನ್ಪುಟ್‌ನ type ಏನಿದೆ ಎಂದು ಚೆಕ್ ಮಾಡುವುದು
const currentType = passwordBox.getAttribute('type');

// 2. ಅದು 'password' ಆಗಿದ್ದರೆ, ಅದನ್ನು 'text' ಗೆ ಬದಲಾಯಿಸುವುದು
if (currentType === 'password') {
    passwordBox.setAttribute('type', 'text'); 
    // ಈಗ ಪಾಸ್‌ವರ್ಡ್ ಪರದೆಯ ಮೇಲೆ ಅಕ್ಷರಗಳ ರೂಪದಲ್ಲಿ ಕಾಣಿಸುತ್ತದೆ!
} else {
    passwordBox.setAttribute('type', 'password');
    // ಮತ್ತೆ ಚುಕ್ಕೆಗಳ (dots) ರೂಪದಲ್ಲಿ ಅಡಗಿಕೊಳ್ಳುತ್ತದೆ!
}
ಬೋನಸ್ ಟಿಪ್: removeAttribute('attribute_name') ಎಂಬ ಮತ್ತೊಂದು ಫಂಕ್ಷನ್ ಇದೆ. ಇದು ಎಲಿಮೆಂಟ್‌ನಿಂದ ಗುಣಲಕ್ಷಣವನ್ನು ಸಂಪೂರ್ಣವಾಗಿ ಡಿಲೀಟ್ ಮಾಡಲು ಸಹಾಯ ಮಾಡುತ್ತದೆ. (ಉದಾ: ಬಟನ್ ಮೇಲಿರುವ disabled ಅನ್ನು ತೆಗೆದುಹಾಕಲು button.removeAttribute('disabled') ಬಳಸಬಹುದು).
