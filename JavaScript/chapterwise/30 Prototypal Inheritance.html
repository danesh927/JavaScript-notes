ನೀವು ಒದಗಿಸಿದ ಮಾಹಿತಿಯು Prototypal Inheritance ಅನ್ನು ವಿವರಿಸಲು ಅತ್ಯುತ್ತಮವಾಗಿದೆ. ನಾನ್-ಐಟಿ ಹಂತದಿಂದ ಪ್ರೊಫೆಷನಲ್ ಹಂತದವರೆಗಿನ ನಿಮ್ಮ ಉದಾಹರಣೆಗಳು ಕಲಿಯುವವರಿಗೆ ವಿಷಯವನ್ನು ಬಹಳ ಸುಲಭವಾಗಿ ಅರ್ಥಮಾಡಿಸುತ್ತವೆ. ನೀವು ನೀಡಿದ ಪ್ರತಿಯೊಂದು ಅಂಶವನ್ನೂ ಉಳಿಸಿಕೊಂಡು, ಕೋಡ್ ಅನ್ನು ಸ್ಪಷ್ಟವಾಗಿ ಫಾರ್ಮ್ಯಾಟ್ ಮಾಡಿ, ತಾಂತ್ರಿಕವಾಗಿ ಇನ್ನಷ್ಟು ನಿಖರವಾಗಿಸಲು ಕೆಲವು ಪ್ರಮುಖ ಹೆಚ್ಚುವರಿ ಮಾಹಿತಿಗಳನ್ನು (ಉದಾಹರಣೆಗೆ: Object.prototype ಮತ್ತು ಆಧುನಿಕ getPrototypeOf ವಿಧಾನ) ಸೇರಿಸಿ ಈ ಕೆಳಗೆ ಕ್ರಮಬದ್ಧವಾಗಿ ತಯಾರಿಸಲಾಗಿದೆ.

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Prototypal Inheritance (ಪ್ರೋಟೋಟೈಪಲ್ ಇನ್ಹೆರಿಟೆನ್ಸ್)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Prototypal Inheritance ಎನ್ನುವುದು ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನ ಒಂದು ಅತಿ ಮುಖ್ಯವಾದ ಪರಿಕಲ್ಪನೆ. ಒಂದು ಆಬ್ಜೆಕ್ಟ್ (Object) ಮತ್ತೊಂದು ಆಬ್ಜೆಕ್ಟ್‌ನ ಗುಣಗಳನ್ನು (Properties) ಮತ್ತು ಕೆಲಸಗಳನ್ನು (Functions/Methods) ಎರವಲು ಪಡೆಯುವ ಅಥವಾ ವಂಶಪಾರಂಪರ್ಯವಾಗಿ ಪಡೆಯುವ ಪ್ರಕ್ರಿಯೆಯನ್ನೇ Inheritance ಎನ್ನುತ್ತಾರೆ.

ಇದನ್ನು ಯಾವುದೇ ಹಿನ್ನೆಲೆಯ ವ್ಯಕ್ತಿಗೂ ಸುಲಭವಾಗಿ ಅರ್ಥವಾಗುವಂತೆ "ನಾನ್-ಐಟಿ" ಹಂತದಿಂದ ಆರಂಭಿಸಿ, "ಪ್ರೊಫೆಷನಲ್" ಹಂತದವರೆಗೆ 4 ಹಂತಗಳಲ್ಲಿ ಸವಿವರವಾಗಿ ನೋಡೋಣ.

ಹಂತ 1: ನಾನ್-ಐಟಿ (Non-IT) ವಿದ್ಯಾರ್ಥಿಗಳಿಗೆ - "ತಲೆಮಾರುಗಳ ಆಸ್ತಿ" ಉದಾಹರಣೆ

Inheritance (ಇನ್ಹೆರಿಟೆನ್ಸ್) ಎಂದರೆ "ಆನುವಂಶಿಕತೆ" ಅಥವಾ ತಲೆಮಾರುಗಳಿಂದ ಬಂದ ಆಸ್ತಿ. ಒಂದು ಕುಟುಂಬವನ್ನು ಕಲ್ಪಿಸಿಕೊಳ್ಳಿ:

  • 1. ತಾತ (Grandfather): ಇವರ ಬಳಿ "10 ಎಕರೆ ಜಮೀನು" ಇದೆ.
  • 2. ಅಪ್ಪ (Father): ಇವರ ಬಳಿ ಸ್ವಂತ "1 ಮನೆ" ಇದೆ. ಜೊತೆಗೆ ತಾತನ 10 ಎಕರೆ ಜಮೀನಿನ ಹಕ್ಕೂ ಇದೆ.
  • 3. ಮಗ (Son): ಇವನ ಬಳಿ ಸ್ವಂತ "1 ಕಾರು" ಇದೆ.

ಈಗ ಮಗನ ಬಳಿ ಜಮೀನು ಇದೆಯಾ ಎಂದು ಕೇಳಿದರೆ, "ಹೌದು" ಎನ್ನುತ್ತೇವೆ. ಏಕೆಂದರೆ ಮಗನಿಗೆ ತನ್ನ ತಂದೆಯ ಮನೆ ಮತ್ತು ತಾತನ ಜಮೀನು ಎರಡನ್ನೂ ಬಳಸುವ ಹಕ್ಕಿರುತ್ತದೆ. ಅಂದರೆ, ಮಗ ತಾನು ಕಷ್ಟಪಟ್ಟು ಜಮೀನು ಖರೀದಿಸುವ ಬದಲು, ತನ್ನ ಮೂಲದಿಂದ (Parent/Grandparent) ಅದನ್ನು ಉಚಿತವಾಗಿ ಪಡೆದಿದ್ದಾನೆ (Inherit ಮಾಡಿದ್ದಾನೆ).

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲೂ ಆಬ್ಜೆಕ್ಟ್‌ಗಳು ಇದೇ ರೀತಿ ಕೆಲಸ ಮಾಡುತ್ತವೆ! ಒಂದು ಆಬ್ಜೆಕ್ಟ್ ತನ್ನ ಬಳಿ ಇಲ್ಲದ ಗುಣವನ್ನು, ತನ್ನ ಮೇಲಿರುವ Parent ಆಬ್ಜೆಕ್ಟ್‌ನಿಂದ ಪಡೆದುಕೊಳ್ಳುತ್ತದೆ.

ಹಂತ 2: ಬಿಗಿನರ್ಸ್ ಲೆವೆಲ್ - ಆಬ್ಜೆಕ್ಟ್‌ಗಳ ನಡುವೆ ನೇರ ಸಂಪರ್ಕ (__proto__)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಆಸ್ತಿಯನ್ನು (ಗುಣಗಳನ್ನು) ಹಸ್ತಾಂತರಿಸಲು ನಾವು __proto__ ಎಂಬ ಗುಪ್ತ ಕೊಂಡಿಯನ್ನು (Hidden Link) ಬಳಸುತ್ತೇವೆ.

ಉದಾಹರಣೆ 1: ಪ್ರಾಣಿ ಮತ್ತು ನಾಯಿ

// 1. ಇದು "parent" ಆಬ್ಜೆಕ್ಟ್ (ತಾತನ ಹಾಗೆ)
let animal = {
  eats: true,
  sleeps: true
};

// 2. ಇದು "child" ಆಬ್ಜೆಕ್ಟ್ (ಮಗನ ಹಾಗೆ)
let dog = {
  barks: true
};

// 3. ಈಗ 'dog' ಗೆ 'animal' ನ ಗುಣಗಳನ್ನು ವರ್ಗಾಯಿಸೋಣ (Inheritance)
dog.__proto__ = animal; 

// ಪರೀಕ್ಷೆ ಮಾಡೋಣ:
console.log(dog.barks); // true (ಇದು ನಾಯಿಯ ಸ್ವಂತ ಗುಣ)
console.log(dog.eats);  // true (ಇದು 'animal' ನಿಂದ Inherit ಆದ ಗುಣ!)

ಉದಾಹರಣೆಯ ವಿವರಣೆ: ಇಲ್ಲಿ dog ಆಬ್ಜೆಕ್ಟ್‌ನ ಒಳಗೆ eats ಎಂಬ ಗುಣವಿರಲಿಲ್ಲ. ಆದರೆ ನಾವು dog.__proto__ = animal ಎಂದು ಹೇಳಿದಾಗ, ನಾಯಿಗೆ ಪ್ರಾಣಿಯ ಆಸ್ತಿಯೆಲ್ಲಾ ಸಿಕ್ಕಿತು. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಮೊದಲು dog ಒಳಗೆ eats ಇದೆಯಾ ಎಂದು ಹುಡುಕುತ್ತದೆ, ಇಲ್ಲದಿದ್ದಾಗ ಗುಪ್ತ ಕನೆಕ್ಷನ್ ಮೂಲಕ animal ನ ಬಳಿ ಹೋಗಿ ಅದನ್ನು ತರುತ್ತದೆ.

ಹೆಚ್ಚುವರಿ ಮಾಹಿತಿ (Best Practice): ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ನೇರವಾಗಿ __proto__ ಅನ್ನು ಬಳಸುವುದು ಅಥವಾ ಬದಲಾಯಿಸುವುದು ಒಳ್ಳೆಯ ಅಭ್ಯಾಸವಲ್ಲ (ಇದು ಪರ್ಫಾರ್ಮೆನ್ಸ್ ಅನ್ನು ನಿಧಾನಗೊಳಿಸುತ್ತದೆ). ಇದರ ಬದಲಿಗೆ ಗುಣಗಳನ್ನು ಓದಲು Object.getPrototypeOf(dog) ಮತ್ತು ಸೆಟ್ ಮಾಡಲು Object.setPrototypeOf(dog, animal) ಅನ್ನು ಬಳಸಲಾಗುತ್ತದೆ. ಆದರೂ ಕಲಿಯುವ ಉದ್ದೇಶಕ್ಕೆ __proto__ ಅರ್ಥಮಾಡಿಕೊಳ್ಳಲು ಅತ್ಯುತ್ತಮವಾಗಿದೆ.

ಹಂತ 3: ಇಂಟರ್ಮೀಡಿಯೇಟ್ ಲೆವೆಲ್ - ಸರಿಯಾದ ವಿಧಾನ (Object.create)

ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿ __proto__ ಅನ್ನು ನೇರವಾಗಿ ಬರೆಯುವ ಬದಲು, ಆಸ್ತಿಯನ್ನು ಹಂಚಲು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ನಮಗೆ Object.create() ಎಂಬ ಹೊಸ ದಾರಿಯನ್ನು ನೀಡಿದೆ.

ಉದಾಹರಣೆ 2: ಉದ್ಯೋಗಿ ಮತ್ತು ಮ್ಯಾನೇಜರ್

// 1. ಮೂಲ ಆಬ್ಜೆಕ್ಟ್ (Parent)
const employee = {
  company: "Tech Corp",
  work: function() {
    console.log("ಕೆಲಸ ಮಾಡುತ್ತಿದ್ದೇನೆ...");
  }
};

// 2. Object.create() ಬಳಸಿ 'employee' ನಿಂದ ಹುಟ್ಟಿದ ಹೊಸ 'manager' ಆಬ್ಜೆಕ್ಟ್
const manager = Object.create(employee);

// ಮ್ಯಾನೇಜರ್‌ನ ಸ್ವಂತ ಗುಣಗಳು
manager.role = "Team Lead";
manager.giveOrders = function() {
  console.log("ಟೀಮ್‌ಗೆ ಸೂಚನೆ ನೀಡುತ್ತಿದ್ದೇನೆ...");
};

// ಪರೀಕ್ಷೆ:
manager.giveOrders();         // ಸ್ವಂತ ಕೆಲಸ: "ಟೀಮ್‌ಗೆ ಸೂಚನೆ ನೀಡುತ್ತಿದ್ದೇನೆ..."
manager.work();               // Inherited ಕೆಲಸ: "ಕೆಲಸ ಮಾಡುತ್ತಿದ್ದೇನೆ..."
console.log(manager.company); // Inherited ಗುಣ: "Tech Corp"

ಉದಾಹರಣೆಯ ವಿವರಣೆ: Object.create(employee) ಎಂದರೆ: "ನನಗೊಂದು ಹೊಸ ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್ (manager) ಕೊಡು, ಆದರೆ ಅದರ ಗುಪ್ತ ಕೊಂಡಿಯನ್ನು (Parent) ರಹಸ್ಯವಾಗಿ employee ಗೆ ಲಿಂಕ್ ಮಾಡು" ಎಂದು ಅರ್ಥ. ಇದರಿಂದ ಕೋಡ್ ಕ್ಲೀನ್ ಆಗಿರುತ್ತದೆ ಮತ್ತು ಸೇಫ್ ಆಗಿರುತ್ತದೆ.

ಹಂತ 4: ಪ್ರೊಫೆಷನಲ್ ಲೆವೆಲ್ - ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಮತ್ತು ಪ್ರೋಟೋಟೈಪ್ ಚೈನ್ (Prototype Chain)

ನೈಜ ಜಗತ್ತಿನ ಪ್ರಾಜೆಕ್ಟ್‌ಗಳಲ್ಲಿ, ನಾವು ಕೇವಲ ಒಂದು ಆಬ್ಜೆಕ್ಟ್‌ನಿಂದ ಇನ್ನೊಂದಕ್ಕೆ ಲಿಂಕ್ ಮಾಡುವುದಿಲ್ಲ. ಬದಲಿಗೆ, ನಾವು Constructor Functions ಬಳಸಿ ಸಾವಿರಾರು ಆಬ್ಜೆಕ್ಟ್‌ಗಳನ್ನು ಸೃಷ್ಟಿಸುತ್ತೇವೆ.

ಉದಾಹರಣೆ 3: ವಾಹನ ಮತ್ತು ಕಾರು (Classic Prototypal Inheritance)

// 1. Parent Constructor (ವಾಹನ)
function Vehicle(wheels) {
  this.wheels = wheels;
}

// ವಾಹನದ ಕೇಂದ್ರ ಕಚೇರಿಯಲ್ಲಿ (Prototype) ಒಂದು ಫಂಕ್ಷನ್ ಸೇರಿಸೋಣ
Vehicle.prototype.start = function() {
  console.log("ವಾಹನ ಸ್ಟಾರ್ಟ್ ಆಯಿತು. ಇದಕ್ಕೆ " + this.wheels + " ಗಾಲಿಗಳಿವೆ.");
};

// 2. Child Constructor (ಕಾರು)
function Car(brandName, wheels) {
  // Parent ನ ಗುಣಗಳನ್ನು ಇಲ್ಲಿಗೆ ಕರೆತರುವುದು (super() ಕಾಲ್ ಮಾಡಿದ ಹಾಗೆ)
  Vehicle.call(this, wheels); 
  this.brand = brandName;
}

// *** ಅತಿ ಮುಖ್ಯ ಹೆಜ್ಜೆ: ಇನ್ಹೆರಿಟೆನ್ಸ್ ಮಾಡುವುದು ***
// ಕಾರಿನ ಕೇಂದ್ರ ಕಚೇರಿಯನ್ನು, ವಾಹನದ ಕೇಂದ್ರ ಕಚೇರಿಗೆ ಲಿಂಕ್ ಮಾಡುವುದು
Car.prototype = Object.create(Vehicle.prototype);

// ಕಾರಿನ ಸ್ವಂತ ಫಂಕ್ಷನ್
Car.prototype.honk = function() {
  console.log(this.brand + " ಕಾರು ಹಾರ್ನ್ ಮಾಡುತ್ತಿದೆ: ಪಾಮ್ ಪಾಮ್!");
};

// ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಅಸ್ತಿತ್ವವನ್ನು ಮರಳಿ ಕಾರಿಗೆ ಸೆಟ್ ಮಾಡುವುದು (ಉತ್ತಮ ಅಭ್ಯಾಸ)
Car.prototype.constructor = Car;

// 3. ಈಗ ಹೊಸ ಕಾರು ಸೃಷ್ಟಿಸೋಣ
let myNewCar = new Car("Tata Safari", 4);

myNewCar.honk();  // ಔಟ್‌ಪುಟ್: Tata Safari ಕಾರು ಹಾರ್ನ್ ಮಾಡುತ್ತಿದೆ... (ಸ್ವಂತ ಗುಣ)
myNewCar.start(); // ಔಟ್‌ಪುಟ್: ವಾಹನ ಸ್ಟಾರ್ಟ್ ಆಯಿತು. ಇದಕ್ಕೆ 4 ಗಾಲಿಗಳಿವೆ. (Parent ಗುಣ)

ಉದಾಹರಣೆಯ ವಿವರಣೆ (Prototype Chain ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ?): ನೀವು myNewCar.start() ಎಂದು ಕರೆದಾಗ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಇಂಜಿನ್ ಮಾಡುವ ಹುಡುಕಾಟದ ಪ್ರಯಾಣವೇ Prototype Chain:

  1. ಮೊದಲು myNewCar ಒಳಗೆ start() ಇದೆಯಾ ನೋಡುತ್ತದೆ -> ಇಲ್ಲ.
  2. ಕಾರಿನ ಕೇಂದ್ರ ಕಚೇರಿಗೆ (Car.prototype) ಹೋಗುತ್ತದೆ -> ಅಲ್ಲಿ ಕೇವಲ honk() ಇದೆ, start() ಇಲ್ಲ.
  3. ಆಗ ಅದು ವಾಹನದ ಕೇಂದ್ರ ಕಚೇರಿಗೆ (Vehicle.prototype) ಹೋಗುತ್ತದೆ -> ಅಲ್ಲಿ start() ಸಿಗುತ್ತದೆ! ಅದನ್ನು ರನ್ ಮಾಡುತ್ತದೆ.
ಹೆಚ್ಚುವರಿ ಮಾಹಿತಿ (The End of the Chain): ಒಂದು ವೇಳೆ start() ವಾಹನದಲ್ಲಿಯೂ ಸಿಗದಿದ್ದರೆ, ಅದು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ಅಂತಿಮ ಪೋಷಕನಾದ Object.prototype ಗೆ ಹೋಗುತ್ತದೆ. ಅಲ್ಲಿಯೂ ಸಿಗದಿದ್ದರೆ, ಕೊನೆಗೆ null ಅನ್ನು ತಲುಪಿ undefined ಅಥವಾ Error ನೀಡುತ್ತದೆ. ಇದೆ Prototype Chain ನ ಅಂತ್ಯ.

ಹಂತ 5: ಮಾಡರ್ನ್ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ (ES6 Classes) - ದಿ ಸಿಂಟ್ಯಾಕ್ಟಿಕ್ ಶುಗರ್ (Syntactic Sugar)

ಹಂತ 4 ರಲ್ಲಿ ನೋಡಿದ ಕೋಡ್ ಬರೆಯಲು ಸ್ವಲ್ಪ ಕಷ್ಟವಲ್ಲವೇ? Vehicle.call(this) ಮತ್ತು Object.create ಇತ್ಯಾದಿ. ಇದನ್ನು ಸುಲಭ ಮಾಡಲು 2015 ರಲ್ಲಿ (ES6) ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ಗೆ class ಮತ್ತು extends ಎಂಬ ಕೀವರ್ಡ್‌ಗಳನ್ನು ತರಲಾಯಿತು.

ಇದು ಕೇವಲ ಮೇಲ್ನೋಟಕ್ಕೆ ಮಾತ್ರ ಹೊಸ ಡಿಸೈನ್ (ಇದನ್ನೇ ಸಿಂಟ್ಯಾಕ್ಟಿಕ್ ಶುಗರ್ ಎನ್ನುತ್ತಾರೆ), ಆದರೆ ಒಳಗೆ (Under the hood) ನಡೆಯುವುದು ಅದೇ ಹಂತ 4 ರ "ಪ್ರೋಟೋಟೈಪಲ್ ಇನ್ಹೆರಿಟೆನ್ಸ್"!

ಉದಾಹರಣೆ 4: ES6 ಕ್ಲಾಸ್ ಬಳಸಿ ಇನ್ಹೆರಿಟೆನ್ಸ್

// 1. Parent Class
class Animal {
  constructor(name) {
    this.name = name;
  }
  
  eat() {
    console.log(this.name + " ಊಟ ಮಾಡುತ್ತಿದೆ.");
  }
}

// 2. Child Class (extends ಕೀವರ್ಡ್ ಬಳಸಿ ಆಸ್ತಿ ಪಡೆಯುವುದು)
class Cat extends Animal {
  constructor(name, color) {
    super(name); // Parent (Animal) ನ ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಗೆ ಹೆಸರನ್ನು ಕಳುಹಿಸುತ್ತಿದೆ
    this.color = color;
  }
  
  meow() {
    console.log(this.name + " ಮಿಯಾವ್ ಎನ್ನುತ್ತಿದೆ.");
  }
}

// 3. ಹೊಸ ಬೆಕ್ಕಿನ ಸೃಷ್ಟಿ
let myCat = new Cat("ಟಾಮ್", "ಬೂದು");

myCat.meow(); // ಔಟ್‌ಪುಟ್: ಟಾಮ್ ಮಿಯಾವ್ ಎನ್ನುತ್ತಿದೆ. (ಸ್ವಂತ ಫಂಕ್ಷನ್)
myCat.eat();  // ಔಟ್‌ಪುಟ್: ಟಾಮ್ ಊಟ ಮಾಡುತ್ತಿದೆ. (Inherited ಫಂಕ್ಷನ್)

ಉದಾಹರಣೆಯ ವಿವರಣೆ: ಇಲ್ಲಿ extends ಎಂಬ ಪದವು Object.create(Parent.prototype) ಕೆಲಸವನ್ನು ತಾನಾಗಿಯೇ ಹಿನ್ನೆಲೆಯಲ್ಲಿ ಮಾಡುತ್ತದೆ. super() ಎಂಬುದು Parent.call(this) ನ ಕೆಲಸ ಮಾಡುತ್ತದೆ. ಇದು ಮಾಡರ್ನ್ ಡೆವಲಪರ್‌ಗಳು ಪ್ರತಿದಿನ ಬಳಸುವ ವಿಧಾನ.

"ಖಂಡಿತ, ನೀವು ನೀಡಿದ ಎರಡೂ ಮಾಹಿತಿಯನ್ನು ಒಟ್ಟುಗೂಡಿಸಿ, ಯಾವುದೇ ಅಂಶವನ್ನು ಬಿಡದೆ, ಅತ್ಯಂತ ಸುಲಭವಾಗಿ ಮತ್ತು ಕ್ರಮಬದ್ಧವಾಗಿ ಅರ್ಥವಾಗುವಂತೆ ಸಂಪೂರ್ಣ ಲೇಖನವನ್ನು ಕೆಳಗೆ ನೀಡಲಾಗಿದೆ."

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Prototypal Inheritance ಮತ್ತು Memory Management (Person ಮತ್ತು Customer ಉದಾಹರಣೆ)

ಈ ಕೋಡ್‌ನ ಮುಖ್ಯ ಉದ್ದೇಶ "ಮರುಬಳಕೆ" (Reusability). ಬರೆದ ಕೋಡ್ ಅನ್ನೇ ಮತ್ತೆ ಮತ್ತೆ ಬರೆಯದೆ, ಮೆಮೊರಿಯನ್ನು ಉಳಿಸುತ್ತಾ ಹೇಗೆ ಸಮರ್ಥವಾಗಿ ಪ್ರೋಗ್ರಾಮಿಂಗ್ ಮಾಡಬಹುದು ಎಂಬುದನ್ನು ಇದು ವಿವರಿಸುತ್ತದೆ.

ಸಾಮಾನ್ಯ ಉದಾಹರಣೆ (Analogy):

  • Person (ಮನುಷ್ಯ): ಪ್ರಪಂಚದಲ್ಲಿರುವ ಪ್ರತಿಯೊಬ್ಬ ಮನುಷ್ಯನಿಗೂ ಒಂದು 'ಹೆಸರು' ಇರುತ್ತದೆ.
  • Customer (ಗ್ರಾಹಕ): ನಿಮ್ಮ ಅಂಗಡಿಗೆ ಬರುವ ಗ್ರಾಹಕನೂ ಒಬ್ಬ ಮನುಷ್ಯನೇ ತಾನೇ? ಅವನಿಗೂ 'ಹೆಸರು' ಇರುತ್ತದೆ. ಆದರೆ ಜೊತೆಗೆ ಫೋನ್ ನಂಬರ್ ಮತ್ತು ಮೆಂಬರ್‌ಶಿಪ್ ಇರುತ್ತದೆ.

ಗ್ರಾಹಕನಿಗೆ ಮತ್ತೆ ಮೊದಲಿನಿಂದ "ನಿನ್ನ ಹೆಸರೇನು?" ಎಂದು ಪ್ರತ್ಯೇಕವಾಗಿ ಕೋಡ್ ಬರೆಯುವ ಬದಲು, ಆ ಕೆಲಸವನ್ನು Person ಗೆ ವಹಿಸುತ್ತೇವೆ. "ಮನುಷ್ಯನಿಗೆ ಏನೇನು ಗುಣಗಳಿವೆಯೋ, ಅದೆಲ್ಲವೂ ಗ್ರಾಹಕನಿಗೂ ಬರಲಿ, ಜೊತೆಗೆ ಗ್ರಾಹಕನಿಗೆ ತನ್ನದೇ ಆದ ಫೋನ್ ನಂಬರ್ ಇರಲಿ" ಎಂದು ಹೇಳುವುದೇ ಈ ಕೋಡ್‌ನ ಅರ್ಥ.

ಸಂಪೂರ್ಣ ಕೋಡ್ (The Complete Code)

// 1. ಪೋಷಕ ಕನ್ಸ್ಟ್ರಕ್ಟರ್ (Parent)
function Person(firstName, lastName) {
  this.firstName = firstName;
  this.lastName = lastName;
}

// ಪೋಷಕನ ಪ್ರೋಟೋಟೈಪ್‌ನಲ್ಲಿರುವ ಫಂಕ್ಷನ್ (ಕೇಂದ್ರ ಕಚೇರಿ)
Person.prototype.getFullName = function () {
  return `${this.firstName} ${this.lastName}`;
};

// 2. ಮಗು ಕನ್ಸ್ಟ್ರಕ್ಟರ್ (Child)
function Customer(firstName, lastName, phone, membership) {
  // ಹಂತ A: ಗುಣಗಳನ್ನು ಕದಿಯುವುದು (Borrowing Properties)
  Person.call(this, firstName, lastName);
  
  // ಗ್ರಾಹಕನ ಸ್ವಂತ ಗುಣಗಳು
  this.phone = phone;
  this.membership = membership;
}

// ಹಂತ B: ಕೆಲಸಗಳನ್ನು ಎರವಲು ಪಡೆಯುವುದು (Prototypal Linking)
Customer.prototype = Object.create(Person.prototype);

// ಹಂತ C: ಅಸ್ತಿತ್ವ ಕಾಪಾಡಿಕೊಳ್ಳುವುದು (Fixing the Constructor)
Customer.prototype.constructor = Customer;

// ಹಂತ D: ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿ (Instantiation)
const c1 = new Customer("Mani", "nair", 987654321, "Standard");
console.log(c1.getFullName()); // ಔಟ್‌ಪುಟ್: Mani nair
console.log(c1);

ಕೋಡ್‌ನ 4 ಪ್ರಮುಖ ಹಂತಗಳ ತಾಂತ್ರಿಕ ವಿಶ್ಲೇಷಣೆ (Deep Dive Analysis)

ಮೇಲ್ನೋಟಕ್ಕೆ ಇದು ಕೇವಲ 4 ಸಾಲಿನ ಕೋಡ್‌ನಂತೆ ಕಂಡರೂ, ಪ್ರತಿಯೊಂದು ಸಾಲಿನ ಹಿಂದೆಯೂ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ "ಮೆಮೊರಿ ಮ್ಯಾನೇಜ್ಮೆಂಟ್" ಮತ್ತು "ಆಬ್ಜೆಕ್ಟ್ ಲಿಂಕಿಂಗ್" ನ ದೊಡ್ಡ ವಿಜ್ಞಾನವೇ ಅಡಗಿದೆ.

ಹಂತ 1: ಗುಣಗಳನ್ನು ಕದಿಯುವುದು (Borrowing Properties) - Person.call(this) ಹಿಂದಿನ ರಹಸ್ಯ

ನಾವು Customer ಫಂಕ್ಷನ್ ಒಳಗೆ Person(firstName, lastName) ಎಂದು ನೇರವಾಗಿ ಕರೆಯಬಹುದಿತ್ತಲ್ಲವೇ? ಅದೇಕೆ .call(this) ಬಳಸಬೇಕು?

  • ನೇರವಾಗಿ ಕರೆದರೆ ಏನಾಗುತ್ತದೆ? (ತಪ್ಪು ವಿಧಾನ): ನೀವು new Customer(...) ಅಂದಾಗ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಒಂದು ಹೊಸ ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್ {} ಅನ್ನು ಸೃಷ್ಟಿಸಿ, ಅದಕ್ಕೆ this ಎಂದು ಹೆಸರಿಡುತ್ತದೆ. ಆದರೆ ನೀವು ಅದರೊಳಗೆ Person(...) ಎಂದು ಸಾಮಾನ್ಯ ಫಂಕ್ಷನ್ ತರಹ ಕರೆದರೆ, Person ಒಳಗಿನ this ಗ್ಲೋಬಲ್ ವಿಂಡೋಗೆ (Global window object) ಪಾಯಿಂಟ್ ಆಗುತ್ತದೆ. ಆಗ ಹೆಸರುಗಳು window.firstName = "Mani" ಎಂದು ಸೇವ್ ಆಗಿಬಿಡುತ್ತವೆ!
  • ಸರಿಯಾದ ವಿಧಾನ (call ಬಳಕೆ): Person.call(this, firstName, lastName); ಎಂದರೆ "ಹೇ Person ಕನ್ಸ್ಟ್ರಕ್ಟರ್, ಈಗ ತಾನೇ Customer ಗಾಗಿ ಒಂದು ಹೊಸ ಖಾಲಿ this {} ಸೃಷ್ಟಿಯಾಗಿದೆ. ದಯವಿಟ್ಟು ನಿನ್ನ ಒಳಗಿರುವ this ಬದಲು, ನನ್ನ ಈ this ಅನ್ನು ಬಳಸಿಕೊಂಡು ಅದಕ್ಕೆ firstName ಮತ್ತು lastName ಅನ್ನು ಅಂಟಿಸು" ಎಂದು ಕೇಳಿಕೊಂಡಂತೆ.

ಹಂತ 2: ಕೆಲಸಗಳನ್ನು ಎರವಲು ಪಡೆಯುವುದು - Object.create() ನ ಮ್ಯಾಜಿಕ್

ಕೇವಲ ಗುಣಗಳನ್ನು ಪಡೆದರೆ ಸಾಲದು, ನಮಗೆ Person ನ ಕೇಂದ್ರ ಕಚೇರಿಯಲ್ಲಿದ್ದ (Prototype) getFullName ಫಂಕ್ಷನ್ ಕೂಡ ಬೇಕು. ಇಲ್ಲಿ ಬಹಳ ಜನ ಮಾಡುವ ತಪ್ಪು ಎಂದರೆ, ಎರಡೂ ಕಚೇರಿಗಳನ್ನು ಒಂದೇ ಮಾಡುವುದು!

  • ತಪ್ಪು ವಿಧಾನ (Direct Assignment): Customer.prototype = Person.prototype; ಎಂದು ಮಾಡಿದರೆ Customer ಮತ್ತು Person ಇಬ್ಬರಿಗೂ ಒಂದೇ ಕೇಂದ್ರ ಕಚೇರಿ ಆಗಿಬಿಡುತ್ತದೆ. ನೀವು Customer ಗೆ ಹೊಸದಾಗಿ getMembership ಸೇರಿಸಿದರೆ, ಅದು ಸಾಮಾನ್ಯ Person ಗೂ ಸೇರ್ಪಡೆಯಾಗುತ್ತದೆ. ಇದು ತಪ್ಪು!
  • ಸರಿಯಾದ ವಿಧಾನ (Object.create): Customer.prototype = Object.create(Person.prototype); ಇದು ಒಂದು ಹೊಸ, ಖಾಲಿ ಕೇಂದ್ರ ಕಚೇರಿಯನ್ನು (Prototype object) ಸೃಷ್ಟಿಸುತ್ತದೆ. ಆದರೆ ಈ ಹೊಸ ಕಚೇರಿಯು Person.prototype ನೊಂದಿಗೆ ಗುಪ್ತ ಕೊಂಡಿಯನ್ನು (__proto__) ಹೊಂದಿರುತ್ತದೆ. ಇದರಿಂದ Customer ಗೆ ಏನಾದರೂ ಬೇಕಾದಾಗ ಅದು Person ನಿಂದ ಪಡೆದುಕೊಳ್ಳಬಲ್ಲದು, ಆದರೆ Customer ಗೆ ಸೇರಿಸಿದ್ದು Person ಗೆ ಹೋಗುವುದಿಲ್ಲ.

ಹಂತ 3: ಅಸ್ತಿತ್ವ ಕಾಪಾಡಿಕೊಳ್ಳುವುದು (Fixing the Constructor)

ನಾವು Object.create ಬಳಸಿದಾಗ, Customer ನ ಹಳೆಯ ಪ್ರೋಟೋಟೈಪ್ ಸಂಪೂರ್ಣವಾಗಿ ಅಳಿಸಿಹೋಗಿ ಹೊಸದು ಬರುತ್ತದೆ. ಪ್ರತಿಯೊಂದು ಪ್ರೋಟೋಟೈಪ್‌ನಲ್ಲೂ constructor ಎಂಬ ಪ್ರಾಪರ್ಟಿ ಇರುತ್ತದೆ. ನಾವು Person ನ ಪ್ರೋಟೋಟೈಪ್ ಅನ್ನು ಕಾಪಿ ಮಾಡಿದಾಗ, ಆ ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಕೂಡ 'Person' ಎಂದೇ ಕಾಪಿ ಆಗಿಬಿಡುತ್ತದೆ.

  • ಸರಿಪಡಿಸದಿದ್ದರೆ: console.log(c1.constructor.name); ಎಂದು ಚೆಕ್ ಮಾಡಿದರೆ "Person" ಎಂದು ಸುಳ್ಳು ಉತ್ತರ ಬರುತ್ತದೆ.
  • ಸರಿಪಡಿಸಿದ ನಂತರ: Customer.prototype.constructor = Customer; ಎಂದು ಬರೆದಾಗ, ತಾನು Person ನಿಂದ ಗುಣಗಳನ್ನು ಪಡೆದಿದ್ದರೂ, ತನ್ನ ಅصلಿ ಅಸ್ತಿತ್ವ (Identity) Customer ಆಗಿಯೇ ಉಳಿಯುವಂತೆ ಇದು ನೋಡಿಕೊಳ್ಳುತ್ತದೆ. ಭವಿಷ್ಯದಲ್ಲಿ ಕೋಡ್ ಬರೆಯುವ ಬೇರೆ ಡೆವಲಪರ್‌ಗಳಿಗೆ ಗೊಂದಲವಾಗುವುದಿಲ್ಲ.

ಹಂತ 4: ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿ ಮತ್ತು ಮೆಮೊರಿ (Instantiation & Memory Management)

ನೀವು const c1 = new Customer(...) ಸೃಷ್ಟಿಸಿದಾಗ, ಬ್ರೌಸರ್‌ನ ಮೆಮೊರಿಯಲ್ಲಿ (RAM) ಡೇಟಾ ಬಹಳ ಅಚ್ಚುಕಟ್ಟಾಗಿ ಸೇವ್ ಆಗುತ್ತದೆ. c1 ಆಬ್ಜೆಕ್ಟ್‌ನ ಒಳಗೆ ಕೇವಲ ಸ್ವಂತ ಗುಣಗಳು (Properties) ಮಾತ್ರ ಇರುತ್ತವೆ:

{
  "firstName": "Mani",
  "lastName": "nair",
  "phone": 987654321,
  "membership": "Standard"
}

ಗಮನಿಸಿ: ಇದರೊಳಗೆ getFullName ಫಂಕ್ಷನ್ ಇಲ್ಲ!

ಹಾಗಾದರೆ c1.getFullName() ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ?

  1. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಮೊದಲು c1 ಆಬ್ಜೆಕ್ಟ್ ಒಳಗೆ ಹುಡುಕುತ್ತದೆ (ಅಲ್ಲಿ ಕೇವಲ ಹೆಸರು, ಫೋನ್ ಇದೆ).
  2. ಅಲ್ಲಿ ಸಿಗದಿದ್ದಾಗ, ಅದು c1 ನ __proto__ (Customer.prototype) ಗೆ ಹೋಗುತ್ತದೆ. (ಅಲ್ಲಿ ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಮಾತ್ರ ಇದೆ).
  3. ಅಲ್ಲಿಯೂ ಸಿಗದಿದ್ದಾಗ, ಅದು ಅದರ ಮೇಲಿರುವ __proto__ (Person.prototype) ಗೆ ಹೋಗುತ್ತದೆ.
  4. ಅಲ್ಲಿ getFullName ಸಿಗುತ್ತದೆ! ಅದನ್ನು ಬಳಸಿಕೊಂಡು "Mani nair" ಎಂದು ರಿಟರ್ನ್ ಮಾಡುತ್ತದೆ.
ಇದರ ಅತಿದೊಡ್ಡ ಲಾಭ (Memory Advantage): ನಿಮ್ಮ ಬಳಿ 1,00,000 ಗ್ರಾಹಕರು (Customers) ಇದ್ದಾರೆ ಎಂದು ಕಲ್ಪಿಸಿಕೊಳ್ಳಿ. ಫಂಕ್ಷನ್ ಅನ್ನು ಪ್ರತಿಯೊಬ್ಬ ಗ್ರಾಹಕನ ಆಬ್ಜೆಕ್ಟ್ ಒಳಗೂ ಸೇವ್ ಮಾಡಿದ್ದರೆ ಮೆಮೊರಿ ಫುಲ್ ಆಗಿ ಬ್ರೌಸರ್ ಹ್ಯಾಂಗ್ ಆಗುತ್ತಿತ್ತು. ಆದರೆ ಪ್ರೋಟೋಟೈಪ್ ಇನ್ಹೆರಿಟೆನ್ಸ್‌ನಿಂದಾಗಿ getFullName ಫಂಕ್ಷನ್ ಮೆಮೊರಿಯಲ್ಲಿ ಕೇವಲ ಒಂದೇ ಒಂದು ಬಾರಿ ಸೇವ್ ಆಗಿರುತ್ತದೆ, 1 ಲಕ್ಷ ಆಬ್ಜೆಕ್ಟ್‌ಗಳು ಆ ಒಂದೇ ಫಂಕ್ಷನ್ ಅನ್ನು ಕೇವಲ ಲಿಂಕ್ (__proto__) ಮೂಲಕ ಬಳಸಿಕೊಳ್ಳುತ್ತವೆ!

ಸಾರಾಂಶ (Conclusion)

  • Prototypal Inheritance ಎಂದರೆ ಒಂದು ಆಬ್ಜೆಕ್ಟ್ ಮತ್ತೊಂದರಿಂದ ಗುಣಗಳನ್ನು ಮತ್ತು ಕೆಲಸಗಳನ್ನು (Functions) ಎರವಲು ಪಡೆಯುವುದು.
  • ಮೆಮೊರಿ ಉಳಿತಾಯ: ಒಂದೇ ಫಂಕ್ಷನ್ ಅನ್ನು ಸಾವಿರಾರು ಆಬ್ಜೆಕ್ಟ್‌ಗಳಿಗೆ ಹಂಚಬಹುದು; ಪ್ರತಿಯೊಂದು ಆಬ್ಜೆಕ್ಟ್‌ಗೂ ಪ್ರತ್ಯೇಕ ಕಾಪಿ ಸೃಷ್ಟಿಸುವ ಅಗತ್ಯವಿಲ್ಲ.
  • Prototype Chain: ಬ್ರೌಸರ್ __proto__ ಚೈನ್ ಮೂಲಕ ಒಂದು ಆಬ್ಜೆಕ್ಟ್‌ನಿಂದ ಇನ್ನೊಂದಕ್ಕೆ ಲಿಂಕ್ ಆಗುತ್ತಾ ಹೋಗಿ ಡೇಟಾವನ್ನು ಹುಡುಕುತ್ತದೆ. ಎಲ್ಲೂ ಸಿಗದಿದ್ದರೆ ಕೊನೆಯಲ್ಲಿ null ತಲುಪಿ undefined ಎಂದು ಹೇಳುತ್ತದೆ.

This, Call() ಮತ್ತು Bind() ರಹಸ್ಯಗಳು

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ this ಕೀವರ್ಡ್ ಒಂದು "ಚಾಮೆಲಿಯನ್" (ಊಸರವಳ್ಳಿ) ಇದ್ದ ಹಾಗೆ! ಅದು ಎಲ್ಲಿದೆ ಮತ್ತು ಅದನ್ನು ಯಾರು ಕರೆಯುತ್ತಿದ್ದಾರೆ ಎಂಬುದರ ಆಧಾರದ ಮೇಲೆ ಅದರ ಅರ್ಥ (ಬಣ್ಣ) ಬದಲಾಗುತ್ತದೆ.

Prototypal Inheritance ನಲ್ಲಿ this ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ ಮತ್ತು call() ಹಾಗೂ bind() ನಡುವಿನ ವ್ಯತ್ಯಾಸವೇನು ಎಂಬುದನ್ನು ಸರಳವಾಗಿ ಅರ್ಥಮಾಡಿಕೊಳ್ಳೋಣ.

1. Prototypal Inheritance ನಲ್ಲಿ this ಕೀವರ್ಡ್

ನಿಯಮ ಬಹಳ ಸರಳ: ಒಂದು ಫಂಕ್ಷನ್ ಎಲ್ಲಿಯೇ ಬರೆದಿರಲಿ, ಅದನ್ನು ಯಾರು ಕರೆಯುತ್ತಿದ್ದಾರೆಯೋ (Calling Object) this ಕೀವರ್ಡ್ ಅವರನ್ನು ಸೂಚಿಸುತ್ತದೆ.

ಇನ್ಹೆರಿಟೆನ್ಸ್‌ನಲ್ಲಿ, ಪೋಷಕ (Parent) ಆಬ್ಜೆಕ್ಟ್‌ನಲ್ಲಿ ಬರೆದ ಫಂಕ್ಷನ್ ಅನ್ನು ಮಗು (Child) ಆಬ್ಜೆಕ್ಟ್ ಬಳಸುವಾಗ, this ಕೀವರ್ಡ್ ಆಟೋಮ್ಯಾಟಿಕ್ ಆಗಿ ಮಗುವನ್ನೇ ಸೂಚಿಸುತ್ತದೆ.

ಸರಳ ಉದಾಹರಣೆ:

// ಪೋಷಕ ಆಬ್ಜೆಕ್ಟ್
const parent = {
  name: "ಅಪ್ಪ",
  introduce: function() {
    console.log("ನನ್ನ ಹೆಸರು: " + this.name);
  }
};

// ಪೋಷಕನಿಂದ ಹುಟ್ಟಿದ ಮಗು
const child = Object.create(parent);
child.name = "ಮಗ"; 

// ಪರೀಕ್ಷೆ
parent.introduce(); // ಔಟ್‌ಪುಟ್: ನನ್ನ ಹೆಸರು: ಅಪ್ಪ
child.introduce();  // ಔಟ್‌ಪುಟ್: ನನ್ನ ಹೆಸರು: ಮಗ

ಏನಾಯಿತು ಇಲ್ಲಿ? introduce ಫಂಕ್ಷನ್ ಇರುವುದು parent ಒಳಗೆ. ಆದರೆ ನಾವು child.introduce() ಎಂದು ಕರೆದಾಗ, this ಕೀವರ್ಡ್ child ಅನ್ನು ಸೂಚಿಸುತ್ತದೆ. ಹಾಗಾಗಿ ಅದು "ಅಪ್ಪ" ಎನ್ನುವ ಬದಲು "ಮಗ" ಎಂದು ಪ್ರಿಂಟ್ ಮಾಡುತ್ತದೆ.

2. call() ಮತ್ತು bind() ನಡುವಿನ ವ್ಯತ್ಯಾಸ

ಕೆಲವು ಬಾರಿ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ this ಅನ್ನು ತಪ್ಪಾಗಿ ಅರ್ಥಮಾಡಿಕೊಳ್ಳುತ್ತದೆ ಅಥವಾ ನಮಗೆ ಬೇಕಾದ ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ನಾವೇ ಬಲವಂತವಾಗಿ this ಅನ್ನು ಲಿಂಕ್ ಮಾಡಬೇಕಾಗುತ್ತದೆ. ಅದಕ್ಕಾಗಿ call() ಮತ್ತು bind() ಗಳನ್ನು ಬಳಸುತ್ತೇವೆ. ಇವೆರಡರ ಮುಖ್ಯ ಕೆಲಸ ಒಂದೇ: this ಏನಾಗಿರಬೇಕು ಎಂದು ಫಿಕ್ಸ್ ಮಾಡುವುದು. ಆದರೆ ಅವು ಕೆಲಸ ಮಾಡುವ ಸಮಯದಲ್ಲಿ ವ್ಯತ್ಯಾಸವಿದೆ.

A. call() - "ತಕ್ಷಣವೇ ಕೆಲಸ ಮಾಡು"

call() ಫಂಕ್ಷನ್ ಅನ್ನು ತಕ್ಷಣವೇ ರನ್ ಮಾಡುತ್ತದೆ ಮತ್ತು ಅದೇ ಸಮಯದಲ್ಲಿ this ಅನ್ನು ನಾವು ಹೇಳಿದ ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ಲಿಂಕ್ ಮಾಡುತ್ತದೆ.

  • ನಾನ್-ಐಟಿ ಉದಾಹರಣೆ: ನೀವು ಟ್ಯಾಕ್ಸಿ ಬುಕ್ ಮಾಡಿ, "ತಕ್ಷಣವೇ ಪ್ರಯಾಣ ಪ್ರಾರಂಭಿಸಿ" ಎಂದು ಹೇಳಿದ ಹಾಗೆ.

ಹಿಂದಿನ ಪಾಠದಲ್ಲಿ ನಾವು Customer ಒಳಗೆ Person.call(this, firstName, lastName) ಎಂದು ಬರೆದಿದ್ದೆವು. ಅಂದರೆ: "Person ಫಂಕ್ಷನ್ ಅನ್ನು ತಕ್ಷಣವೇ ರನ್ ಮಾಡು, ಆದರೆ ಅದರೊಳಗಿರುವ this ಮನುಷ್ಯನದ್ದಲ್ಲ, ಬದಲಿಗೆ ಈಗ ಹುಟ್ಟುತ್ತಿರುವ ಹೊಸ Customer ನದ್ದು" ಎಂದು ಹೇಳಿದಂತೆ.

const person1 = { name: "ಕಿರಣ್" };
const person2 = { name: "ರಾಜು" };

function sayHello(greeting) {
  console.log(greeting + ", " + this.name);
}

// sayHello ಫಂಕ್ಷನ್ ಅನ್ನು ತಕ್ಷಣ ರನ್ ಮಾಡುತ್ತಿದೆ, this = person1
sayHello.call(person1, "ನಮಸ್ಕಾರ"); // ಔಟ್‌ಪುಟ್: ನಮಸ್ಕಾರ, ಕಿರಣ್
sayHello.call(person2, "ಹಲೋ");    // ಔಟ್‌ಪುಟ್: ಹಲೋ, ರಾಜು

B. bind() - "ಭವಿಷ್ಯದಲ್ಲಿ ಕೆಲಸ ಮಾಡಲು ಸಿದ್ಧವಾಗಿರು"

bind() ಫಂಕ್ಷನ್ ಅನ್ನು ತಕ್ಷಣ ರನ್ ಮಾಡುವುದಿಲ್ಲ. ಬದಲಿಗೆ, this ಅನ್ನು ಫಿಕ್ಸ್ ಮಾಡಿ, ಹೊಸದೊಂದು ಕಾಪಿ (ಹೊಸ ಫಂಕ್ಷನ್) ಅನ್ನು ನಿಮಗೆ ಹಿಂದಿರುಗಿಸುತ್ತದೆ. ನೀವು ಆ ಹೊಸ ಫಂಕ್ಷನ್ ಅನ್ನು ಭವಿಷ್ಯದಲ್ಲಿ ಯಾವಾಗ ಬೇಕಾದರೂ ಬಳಸಬಹುದು.

  • ನಾನ್-ಐಟಿ ಉದಾಹರಣೆ: ಮುಂದಿನ ವಾರಕ್ಕೆ ವಿಮಾನದ ಟಿಕೆಟ್ ಬುಕ್ ಮಾಡಿಟ್ಟುಕೊಂಡ ಹಾಗೆ. ನೀವು ಟಿಕೆಟ್ ಈಗಲೇ ಫಿಕ್ಸ್ ಮಾಡಿದ್ದೀರಿ, ಆದರೆ ಪ್ರಯಾಣ ಭವಿಷ್ಯದಲ್ಲಿ ಮಾಡುತ್ತೀರಿ.
const user = {
  name: "ಸುಮನ್",
  greet: function() {
    console.log("ಸ್ವಾಗತ, " + this.name);
  }
};

// ಇಲ್ಲಿ ಫಂಕ್ಷನ್ ರನ್ ಆಗುವುದಿಲ್ಲ, ಬದಲಾಗಿ this=user ಎಂದು ಫಿಕ್ಸ್ ಆಗಿ ಹೊಸ ಫಂಕ್ಷನ್ ಬರುತ್ತದೆ
const boundFunction = user.greet.bind(user); 

// ಭವಿಷ್ಯದಲ್ಲಿ ನಮಗೆ ಯಾವಾಗ ಬೇಕೋ ಆಗ ಕರೆದುಕೊಳ್ಳಬಹುದು
setTimeout(boundFunction, 2000); // 2 ಸೆಕೆಂಡ್ ಆದ ಮೇಲೆ: ಸ್ವಾಗತ, ಸುಮನ್

3. ತ್ವರಿತ ಹೋಲಿಕೆ (Quick Comparison)

ವೈಶಿಷ್ಟ್ಯ call() bind()
ಕಾರ್ಯಗತಗೊಳ್ಳುವಿಕೆ (Execution) ಫಂಕ್ಷನ್ ಅನ್ನು ತಕ್ಷಣವೇ ರನ್ ಮಾಡುತ್ತದೆ. ತಕ್ಷಣ ರನ್ ಆಗುವುದಿಲ್ಲ. ಬದಲಿಗೆ ಹೊಸ ಫಂಕ್ಷನ್ ನೀಡುತ್ತದೆ.
ರಿಟರ್ನ್ (Returns) ಫಂಕ್ಷನ್ ರನ್ ಆದ ಮೇಲೆ ಬರುವ ಔಟ್‌ಪುಟ್. this ಫಿಕ್ಸ್ ಆಗಿರುವ ಹೊಸ ಫಂಕ್ಷನ್.
ಆರ್ಗ್ಯುಮೆಂಟ್ಸ್ ಕಳುಹಿಸುವುದು call(this, arg1, arg2) - ಬಿಡಿ ಬಿಡಿಯಾಗಿ ಕಳುಹಿಸಬೇಕು. bind(this, arg1, arg2) - ಬಿಡಿ ಬಿಡಿಯಾಗಿ ಕಳುಹಿಸಬಹುದು.
ಮುಖ್ಯ ಬಳಕೆ ಇನ್ಹೆರಿಟೆನ್ಸ್‌ನಲ್ಲಿ ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಕರೆಯಲು (Borrowing properties). Event Listeners ಮತ್ತು setTimeout ಗಳಲ್ಲಿ this ಕಳೆದುಹೋಗದಂತೆ ತಡೆಯಲು.
ನೆನಪಿರಲಿ: ಇನ್ನೊಂದು ಮೆಥಡ್ ಇದೆ, ಅದರ ಹೆಸರು apply(). ಇದು ಶೇ. ೧೦೦ ರಷ್ಟು call() ತರಹವೇ ತಕ್ಷಣ ಕೆಲಸ ಮಾಡುತ್ತದೆ, ಆದರೆ ಆರ್ಗ್ಯುಮೆಂಟ್‌ಗಳನ್ನು ಬಿಡಿ ಬಿಡಿಯಾಗಿ ಕೊಡುವ ಬದಲು ಒಂದೇ Array [arg1, arg2] ಒಳಗೆ ಹಾಕಿ ಕೊಡಬೇಕು ಅಷ್ಟೇ.
Bonus: Core Front-End Web APIs

1. DOM ಮ್ಯಾನಿಪುಲೇಷನ್ ಮತ್ತು ಎಲಿಮೆಂಟ್ ಆಯ್ಕೆ (DOM Architecture)

ಬ್ರೌಸರ್ HTML ಕೋಡ್ ಅನ್ನು ಓದಿ, ಅದನ್ನು ಒಂದು 'ಮರ'ದ (Tree structure) ರೂಪದಲ್ಲಿ ಮೆಮೊರಿಯಲ್ಲಿ ಜೋಡಿಸುತ್ತದೆ. ಇದನ್ನೇ DOM (Document Object Model) ಎನ್ನುತ್ತಾರೆ. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಬಳಸಿ ಈ ಮರದ ರೆಂಬೆ-ಕೊಂಬೆಗಳನ್ನು (Elements) ಹುಡುಕುವುದು ಮತ್ತು ಬದಲಾಯಿಸುವುದು ಫ್ರಂಟ್-ಎಂಡ್ ಡೆವಲಪ್ಮೆಂಟ್‌ನ ಅತಿ ಮುಖ್ಯ ಭಾಗ.

ಎಲಿಮೆಂಟ್ ಆಯ್ಕೆ ಮಾಡುವ ವಿಧಾನಗಳು (Selection Methods):

  • getElementById('id'): ಅತಿ ವೇಗವಾದ ಆಯ್ಕೆ. ಕೇವಲ ಒಂದು ನಿರ್ದಿಷ್ಟ ID ಯನ್ನು ಹುಡುಕುತ್ತದೆ.
  • querySelector('.class'): ಆಧುನಿಕ ವಿಧಾನ. CSS ಸೆಲೆಕ್ಟರ್‌ಗಳನ್ನು (class, id, tag) ಬಳಸಿ, ಸಿಗುವ ಮೊದಲ ಎಲಿಮೆಂಟ್ ಅನ್ನು ಮಾತ್ರ ತರುತ್ತದೆ.
  • querySelectorAll('.class'): ಸಿಗುವ ಎಲ್ಲಾ ಎಲಿಮೆಂಟ್‌ಗಳನ್ನು ಒಂದು NodeList (Array ಥರ) ರೂಪದಲ್ಲಿ ತರುತ್ತದೆ.
// 1. ಎಲಿಮೆಂಟ್ ಆಯ್ಕೆ
const title = document.getElementById('main-title');
const buttons = document.querySelectorAll('.btn-primary');

// 2. DOM ಮ್ಯಾನಿಪುಲೇಷನ್ (ಬದಲಾಯಿಸುವುದು)
title.textContent = "ಹೊಸ ಶೀರ್ಷಿಕೆ";
title.style.color = "blue"; // CSS ಬದಲಾವಣೆ

// 3. ಹೊಸ ಎಲಿಮೆಂಟ್ ಸೃಷ್ಟಿ ಮತ್ತು ಸೇರ್ಪಡೆ (Node Creation & Appending)
const newDiv = document.createElement('div');
newDiv.innerHTML = "ನಾನು ಹೊಸದಾಗಿ ಬಂದಿದ್ದೇನೆ!";
document.body.appendChild(newDiv);

2. ಇವೆಂಟ್ ಹ್ಯಾಂಡ್ಲಿಂಗ್ ಮತ್ತು ಡೆಲಿಗೇಶನ್ (Event Handling)

ವೆಬ್‌ಸೈಟ್‌ನಲ್ಲಿ ಯೂಸರ್ ಕ್ಲಿಕ್ ಮಾಡಿದಾಗ, ಮೌಸ್ ಮೂವ್ ಮಾಡಿದಾಗ ಅಥವಾ ಕೀಬೋರ್ಡ್ ಟೈಪ್ ಮಾಡಿದಾಗ ಏನು ನಡೆಯಬೇಕು ಎಂದು ನಿರ್ಧರಿಸುವುದೇ Event Handling.

Event Bubbling (ಇವೆಂಟ್ ಬಬ್ಲಿಂಗ್)

ಒಂದು ಮಗುವಿನ ಎಲಿಮೆಂಟ್ (ಉದಾ: ಬಟನ್) ಮೇಲೆ ಕ್ಲಿಕ್ ಮಾಡಿದರೆ, ಆ ಕ್ಲಿಕ್ ಇವೆಂಟ್ ನೀರಿನ ಗುಳ್ಳೆಯಂತೆ ಮೇಲಕ್ಕೆ (Parent element ಗಳಿಗೆ) ಹರಿಯುತ್ತದೆ.

Event Delegation (ಇವೆಂಟ್ ಡೆಲಿಗೇಶನ್)

ಪ್ರತಿಯೊಂದು ಲಿಸ್ಟ್ ಐಟಂಗೂ ಪ್ರತ್ಯೇಕ Event Listener ಹಾಕುವ ಬದಲು, ಅವುಗಳ Parent ಎಲಿಮೆಂಟ್‌ಗೆ ಒಂದೇ Listener ಹಾಕುವುದು. ಇದು ಮೆಮೊರಿಯನ್ನು ಉಳಿಸುತ್ತದೆ!

// Event Delegation ಉದಾಹರಣೆ
const listParent = document.getElementById('myList');

// ಪ್ರತ್ಯೇಕ 
  •  ಗಳಿಗೆ ಬರೆಯುವ ಬದಲು, ಒಂದೇ Parent ಗೆ ಬರೆಯುತ್ತಿದ್ದೇವೆ
listParent.addEventListener('click', function(event) {
  // ಯೂಸರ್ ಕ್ಲಿಕ್ ಮಾಡಿದ್ದು 
  •  ಹೌದೋ ಅಲ್ಲವೋ ಎಂದು ಚೆಕ್ ಮಾಡುವುದು
  if(event.target.tagName === 'LI') {
    console.log("ನೀವು ಕ್ಲಿಕ್ ಮಾಡಿದ್ದು: " + event.target.innerText);
  }
});

    • 3. ಕ್ಲೈಂಟ್ ಸೈಡ್ ಸ್ಟೋರೇಜ್ (Client-Side Storage APIs)

    ಬ್ರೌಸರ್‌ನಲ್ಲಿ (ಯೂಸರ್‌ನ ಕಂಪ್ಯೂಟರ್‌ನಲ್ಲಿ) ಡೇಟಾವನ್ನು ಉಳಿಸಿಡಲು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ನಮಗೆ ಸುಲಭವಾದ ಸ್ಟೋರೇಜ್ API ಗಳನ್ನು ನೀಡಿದೆ. ಲಾಗ್-ಇನ್ ಸ್ಟೇಟಸ್ ಅಥವಾ ಡಾರ್ಕ್-ಮೋಡ್ ಪ್ರಿಫರೆನ್ಸ್ ಸೇವ್ ಮಾಡಲು ಇದು ಅತ್ಯಗತ್ಯ.

    • 1

      localStorage (ಶಾಶ್ವತ ಸ್ಟೋರೇಜ್)

      ಬ್ರೌಸರ್ ಮುಚ್ಚಿದರೂ ಡೇಟಾ ಅಳಿಸಿಹೋಗುವುದಿಲ್ಲ. ಯೂಸರ್ ಸ್ವತಃ ಡಿಲೀಟ್ ಮಾಡುವವರೆಗೆ ಡೇಟಾ ಸೇವ್ ಆಗಿರುತ್ತದೆ.

    • 2

      sessionStorage (ತಾತ್ಕಾಲಿಕ ಸ್ಟೋರೇಜ್)

      ಆ ಟ್ಯಾಬ್ ಅಥವಾ ಬ್ರೌಸರ್ ಮುಚ್ಚಿದ ತಕ್ಷಣ ಡೇಟಾ ಆಟೋಮ್ಯಾಟಿಕ್ ಆಗಿ ಡಿಲೀಟ್ ಆಗುತ್ತದೆ.

    // ಡೇಟಾ ಸೇವ್ ಮಾಡುವುದು (Key-Value pair)
localStorage.setItem('theme', 'dark');

// ಸೇವ್ ಆದ ಡೇಟಾವನ್ನು ಓದುವುದು
const currentTheme = localStorage.getItem('theme');
console.log(currentTheme); // 'dark'

// ಡೇಟಾ ಡಿಲೀಟ್ ಮಾಡುವುದು
localStorage.removeItem('theme');

// Note: ಸ್ಟೋರೇಜ್ ಒಳಗೆ ಆಬ್ಜೆಕ್ಟ್ ಸೇವ್ ಮಾಡಬೇಕಾದರೆ JSON ಗೆ ಕನ್ವರ್ಟ್ ಮಾಡಬೇಕು!
const userSettings = { volume: 80, mode: "dark" };
localStorage.setItem('settings', JSON.stringify(userSettings));
