JavaScript Promise Masterclass
(ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಪ್ರಾಮಿಸ್)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ (JavaScript) Promise (ಪ್ರಾಮಿಸ್) ಎಂದರೆ ಭವಿಷ್ಯದಲ್ಲಿ (future) ಸಿಗುವ ಒಂದು ಮೌಲ್ಯ (value) ಅಥವಾ ಕಾರ್ಯಾಚರಣೆಯ ಅಂತಿಮ ಫಲಿತಾಂಶವನ್ನು ಪ್ರತಿನಿಧಿಸುವ ಒಂದು ವ್ಯವಸ್ಥೆಯಾಗಿದೆ. ಇದು ಅಸಿಂಕ್ರೊನಸ್ (asynchronous) ಕೆಲಸಗಳನ್ನು (ಉದಾಹರಣೆಗೆ: ಸರ್ವರ್‌ನಿಂದ ಡೇಟಾ ಪಡೆಯುವುದು, ಫೈಲ್ ಓದುವುದು) ನಿರ್ವಹಿಸಲು ಬಳಸಲ್ಪಡುತ್ತದೆ.

ಒಂದು Promise ಮೂರು ಹಂತಗಳಲ್ಲಿ (States) ಇರಬಹುದು:

  1. Pending (ಬಾಕಿ): ಕಾರ್ಯಾಚರಣೆ ಇನ್ನೂ ನಡೆಯುತ್ತಿದೆ, ಪೂರ್ಣಗೊಂಡಿಲ್ಲ.
  2. Fulfilled (ಯಶಸ್ವಿ): ಕಾರ್ಯಾಚರಣೆ ಯಶಸ್ವಿಯಾಗಿ ಪೂರ್ಣಗೊಂಡಿದೆ (resolve ಆಗಿದೆ).
  3. Rejected (ವಿಫಲ): ಕಾರ್ಯಾಚರಣೆ ವಿಫಲವಾಗಿದೆ ಅಥವಾ ದೋಷ (error) ಉಂಟಾಗಿದೆ (reject ಆಗಿದೆ).

Promise States Diagram

1. Promise ರಚನೆ ಮತ್ತು ಬೇಸಿಕ್ ಮೆಸೇಜ್ ಹ್ಯಾಂಡ್ಲಿಂಗ್ (Basic Message Handling)

ಪ್ರಾಮಿಸ್ ಅನ್ನು ರಚಿಸುವಾಗ resolve (ಯಶಸ್ಸು) ಮತ್ತು reject (ವಿಫಲ) ಎಂಬ ಎರಡು ಫಂಕ್ಷನ್‌ಗಳನ್ನು ಬಳಸಲಾಗುತ್ತದೆ. ಯಶಸ್ವಿಯಾದಾಗ .then() ಮೂಲಕ ಮೆಸೇಜ್ ಪಡೆಯುತ್ತೇವೆ ಮತ್ತು ವಿಫಲವಾದಾಗ .catch() ಮೂಲಕ ಎರೆರ್ ಮೆಸೇಜ್ (Error Message) ಪಡೆಯುತ್ತೇವೆ.

ಉದಾಹರಣೆ 1:

const myFirstPromise = new Promise((resolve, reject) => {
    let isTaskCompleted = true; // ಇದನ್ನು false ಮಾಡಿದರೆ ದೋಷ ಬರುತ್ತದೆ

    if (isTaskCompleted) {
        resolve("ಅಭಿನಂದನೆಗಳು! ನಿಮ್ಮ ಕೆಲಸ ಯಶಸ್ವಿಯಾಗಿ ಪೂರ್ಣಗೊಂಡಿದೆ."); // Success message
    } else {
        reject("ದೋಷ: ಕೆಲಸವನ್ನು ಪೂರ್ಣಗೊಳಿಸಲು ಸಾಧ್ಯವಾಗಲಿಲ್ಲ."); // Error message
    }
});

// ಪ್ರಾಮಿಸ್ ಅನ್ನು ಬಳಸುವುದು
myFirstPromise
    .then((message) => {
        // resolve ಆದಾಗ ಈ ಬ್ಲಾಕ್ ರನ್ ಆಗುತ್ತದೆ
        console.log("ಯಶಸ್ಸು:", message);
    })
    .catch((error) => {
        // reject ಆದಾಗ ಈ ಬ್ಲಾಕ್ ರನ್ ಆಗುತ್ತದೆ
        console.log("ವಿಫಲ:", error);
    });

ಔಟ್‌ಪುಟ್ (Output):

ಯಶಸ್ಸು: ಅಭಿನಂದನೆಗಳು! ನಿಮ್ಮ ಕೆಲಸ ಯಶಸ್ವಿಯಾಗಿ ಪೂರ್ಣಗೊಂಡಿದೆ.

ವಿವರಣೆ: ಇಲ್ಲಿ isTaskCompleted true ಆಗಿರುವುದರಿಂದ resolve ಕಾರ್ಯಗತಗೊಂಡು .then() ಒಳಗೆ ಇರುವ ಮೆಸೇಜ್ ಪ್ರಿಂಟ್ ಆಗಿದೆ. ಒಂದು ವೇಳೆ ಅದನ್ನು false ಮಾಡಿದರೆ .catch() ಒಳಗೆ ಇರುವ ದೋಷ (error) ಮೆಸೇಜ್ ಪ್ರಿಂಟ್ ಆಗುತ್ತದೆ.

2. Error Object ಮತ್ತು .finally() ಬಳಸಿ ದೋಷಗಳನ್ನು ನಿರ್ವಹಿಸುವುದು

ನೈಜ ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿ, ನಾವು ಕೇವಲ ಸ್ಟ್ರಿಂಗ್ ಮೆಸೇಜ್ ಬದಲಾಗಿ Error ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಪಾಸ್ ಮಾಡುತ್ತೇವೆ. ಇದರಿಂದ ದೋಷ ಎಲ್ಲಿದೆ ಮತ್ತು ಅದರ ನಿಖರವಾದ ಕಾರಣವೇನು (error.message) ಎಂದು ತಿಳಿಯಲು ಸುಲಭವಾಗುತ್ತದೆ. .finally() ಎಂಬುದು ಪ್ರಾಮಿಸ್ ಯಶಸ್ವಿಯಾಗಲಿ ಅಥವಾ ವಿಫಲವಾಗಲಿ, ಕೊನೆಯಲ್ಲಿ ಖಚಿತವಾಗಿ ರನ್ ಆಗುವ ಬ್ಲಾಕ್ ಆಗಿದೆ.

ಉದಾಹರಣೆ 2:

const fetchServerData = new Promise((resolve, reject) => {
    let serverConnected = false; 

    setTimeout(() => {
        if (serverConnected) {
            resolve({ user: "Kiran", role: "Developer" });
        } else {
            // ದೋಷದ ಸಂದೇಶವನ್ನು (Error object) ಕಳುಹಿಸುವುದು
            reject(new Error("ಸರ್ವರ್ ಸಂಪರ್ಕ ಕಡಿತಗೊಂಡಿದೆ (Network connection failed)."));
        }
    }, 2000); // 2 ಸೆಕೆಂಡ್‌ಗಳ ವಿಳಂಬ (delay)
});

fetchServerData
    .then((data) => {
        console.log("ಡೇಟಾ ಸಿಕ್ಕಿದೆ:", data);
    })
    .catch((error) => {
        // ದೋಷದ ನಿಖರವಾದ ಸಂದೇಶವನ್ನು ಮಾತ್ರ ಪಡೆಯಲು error.message ಬಳಸುತ್ತೇವೆ
        console.error("ದೋಷದ ವಿವರ (Error Message):", error.message);
    })
    .finally(() => {
        console.log("ಕಾರ್ಯಾಚರಣೆ ಮುಕ್ತಾಯವಾಗಿದೆ. (ಲೋಡಿಂಗ್ ಸ್ಪಿನ್ನರ್ ಅನ್ನು ನಿಲ್ಲಿಸಿ)");
    });

ಔಟ್‌ಪುಟ್ (Output): (2 ಸೆಕೆಂಡ್‌ಗಳ ನಂತರ)

ದೋಷದ ವಿವರ (Error Message): ಸರ್ವರ್ ಸಂಪರ್ಕ ಕಡಿತಗೊಂಡಿದೆ (Network connection failed).
ಕಾರ್ಯಾಚರಣೆ ಮುಕ್ತಾಯವಾಗಿದೆ. (ಲೋಡಿಂಗ್ ಸ್ಪಿನ್ನರ್ ಅನ್ನು ನಿಲ್ಲಿಸಿ)

ವಿವರಣೆ: ಇಲ್ಲಿ ಸರ್ವರ್ ಕನೆಕ್ಟ್ ಆಗಿಲ್ಲ (serverConnected = false), ಹಾಗಾಗಿ reject ಒಳಗೆ new Error() ಅನ್ನು ಕಳುಹಿಸಲಾಗಿದೆ. .catch() ಬ್ಲಾಕ್‌ನಲ್ಲಿ error.message ಅನ್ನು ಬಳಸುವುದರಿಂದ ನಮಗೆ ಕೇವಲ ದೋಷದ ಸಂದೇಶ ಸ್ಪಷ್ಟವಾಗಿ ಸಿಗುತ್ತದೆ. ಕೊನೆಯಲ್ಲಿ .finally() ಬ್ಲಾಕ್ ರನ್ ಆಗುತ್ತದೆ.

3. Promise Chaining (ಸರಣಿ ಪ್ರಾಮಿಸ್‌ಗಳು)

ಒಂದು ಕೆಲಸದ ಫಲಿತಾಂಶದ ಆಧಾರದ ಮೇಲೆ ಮತ್ತೊಂದು ಕೆಲಸವನ್ನು ಮಾಡಬೇಕಾದರೆ, ನಾವು .then() ಅನ್ನು ಒಂದರ ಹಿಂದೆ ಒಂದರಂತೆ ಜೋಡಿಸಬಹುದು. ಇದರ ಮಧ್ಯೆ ಎಲ್ಲಿಯೇ ದೋಷ ಬಂದರೂ ಕೊನೆಯಲ್ಲಿರುವ ಒಂದೇ .catch() ಅದನ್ನು ಹಿಡಿಯುತ್ತದೆ.

ಉದಾಹರಣೆ 3:

const calculateNumber = new Promise((resolve, reject) => {
    resolve(10);
});

calculateNumber
    .then((result) => {
        console.log("ಹಂತ 1ರ ಫಲಿತಾಂಶ:", result);
        return result * 2; // ಈ ಮೌಲ್ಯ ಮುಂದಿನ .then ಗೆ ಹೋಗುತ್ತದೆ
    })
    .then((newResult) => {
        console.log("ಹಂತ 2ರ ಫಲಿತಾಂಶ:", newResult);
        return newResult + 5;
    })
    .then((finalResult) => {
        console.log("ಅಂತಿಮ ಫಲಿತಾಂಶ:", finalResult);
        // ಬೇಕು ಎಂದಾದರೆ ಇಲ್ಲಿ ಕೃತಕ ದೋಷವನ್ನು ಸೃಷ್ಟಿಸಬಹುದು
        // throw new Error("ಲೆಕ್ಕಾಚಾರದಲ್ಲಿ ದೋಷವಿದೆ!");
    })
    .catch((error) => {
        console.log("ಯಾವುದೋ ಹಂತದಲ್ಲಿ ದೋಷ ಬಂದಿದೆ:", error.message);
    });

ಔಟ್‌ಪುಟ್ (Output):

ಹಂತ 1ರ ಫಲಿತಾಂಶ: 10
ಹಂತ 2ರ ಫಲಿತಾಂಶ: 20
ಅಂತಿಮ ಫಲಿತಾಂಶ: 25

ವಿವರಣೆ: ಮೊದಲ ಹಂತದಲ್ಲಿ 10 ಸಿಕ್ಕಿದೆ. ಆ 10 ಅನ್ನು 2 ರಿಂದ ಗುಣಿಸಿ ಮುಂದಿನ ಹಂತಕ್ಕೆ ಕಳುಹಿಸಲಾಗಿದೆ (20). ಕೊನೆಯ ಹಂತದಲ್ಲಿ 20 ಕ್ಕೆ 5 ಸೇರಿಸಿ 25 ಅನ್ನು ಪ್ರಿಂಟ್ ಮಾಡಲಾಗಿದೆ. ಈ ಸರಣಿಯಲ್ಲಿ ಎಲ್ಲಿಯಾದರೂ ದೋಷ ಸಂಭವಿಸಿದರೆ (ಉದಾಹರಣೆಗೆ throw new Error() ಬಳಸಿದರೆ), ನಡುವಿನ ಹಂತಗಳು ನಿಂತುಹೋಗಿ ನೇರವಾಗಿ .catch() ಬ್ಲಾಕ್ ಕಾರ್ಯಗತಗೊಳ್ಳುತ್ತದೆ.

4. Async/Await ಜೊತೆಗೆ Error ಮತ್ತು Message ನಿರ್ವಹಣೆ (ಆಧುನಿಕ ವಿಧಾನ)

ಇತ್ತೀಚಿನ ದಿನಗಳಲ್ಲಿ .then() ಮತ್ತು .catch() ಬದಲಾಗಿ ಹೆಚ್ಚು ಸುಲಭವಾಗಿ ಅರ್ಥವಾಗುವ async ಮತ್ತು await ಅನ್ನು try...catch ಬ್ಲಾಕ್‌ನೊಂದಿಗೆ ಬಳಸಲಾಗುತ್ತದೆ.

ಉದಾಹರಣೆ 4:

function getUserData() {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            reject(new Error("ಬಳಕೆದಾರನ ಮಾಹಿತಿ ಲಭ್ಯವಿಲ್ಲ (User not found)."));
        }, 1500);
    });
}

async function displayData() {
    try {
        console.log("ಡೇಟಾ ಪಡೆಯಲು ಪ್ರಯತ್ನಿಸುತ್ತಿದ್ದೇನೆ...");
        let data = await getUserData(); // ಪ್ರಾಮಿಸ್ ಪೂರ್ಣಗೊಳ್ಳುವವರೆಗೆ ಕಾಯುತ್ತದೆ
        console.log("ಯಶಸ್ಸು:", data);
    } catch (error) {
        // ಪ್ರಾಮಿಸ್ reject ಆದರೆ, ನೇರವಾಗಿ ಇಲ್ಲಿಗೆ ಬರುತ್ತದೆ
        console.log("ಕ್ಯಾಚ್ ಆದ ದೋಷ (Caught Error):", error.message);
    }
}

displayData();

ಔಟ್‌ಪುಟ್ (Output):

ಡೇಟಾ ಪಡೆಯಲು ಪ್ರಯತ್ನಿಸುತ್ತಿದ್ದೇನೆ...
(1.5 ಸೆಕೆಂಡ್‌ಗಳ ನಂತರ)
ಕ್ಯಾಚ್ ಆದ ದೋಷ (Caught Error): ಬಳಕೆದಾರನ ಮಾಹಿತಿ ಲಭ್ಯವಿಲ್ಲ (User not found).

ವಿವರಣೆ: async ಫಂಕ್ಷನ್ ಒಳಗೆ await ಬಳಸುವುದರಿಂದ ಕೋಡ್ ನೋಡಲು ಸಿಂಕ್ರೊನಸ್ (ಸಾಮಾನ್ಯ ಕೋಡ್) ರೀತಿಯಲ್ಲಿ ಕಾಣುತ್ತದೆ. try ಬ್ಲಾಕ್ ಒಳಗೆ ಯಶಸ್ವಿಯಾಗುವ ಕೆಲಸಗಳನ್ನು ಬರೆಯುತ್ತೇವೆ. getUserData ಪ್ರಾಮಿಸ್ ವಿಫಲವಾದರೆ (reject), ಅದು ತಕ್ಷಣವೇ catch ಬ್ಲಾಕ್‌ಗೆ ಜಿಗಿಯುತ್ತದೆ ಮತ್ತು ಅಲ್ಲಿ ನಾವು ದೋಷದ ಸಂದೇಶವನ್ನು (error.message) ಸುಲಭವಾಗಿ ನಿರ್ವಹಿಸಬಹುದು.

5. ಇವೆಂಟ್ ಲೂಪ್ ಮತ್ತು ಮೈಕ್ರೋಟಾಸ್ಕ್ ಕ್ಯೂ (Event Loop & Microtask Queue)

Promise.resolve() ತಕ್ಷಣವೇ ಡೇಟಾವನ್ನು ನೀಡಿದರೂ (Settled ಆದರೂ), ಅದರೊಳಗಿನ .then() ಕೋಡ್ ತಕ್ಷಣವೇ (Synchronous ಆಗಿ) ರನ್ ಆಗುವುದಿಲ್ಲ. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಅದನ್ನು 'ಮೈಕ್ರೋಟಾಸ್ಕ್ ಕ್ಯೂ' (Microtask Queue) ನಲ್ಲಿ ಇಡುತ್ತದೆ. ಉಳಿದ ಸಾಮಾನ್ಯ ಕೋಡ್ ರನ್ ಆದ ಮೇಲೆಯೇ ಇದು ರನ್ ಆಗುತ್ತದೆ.

ಉದಾಹರಣೆ 5:

console.log("1. ಮೊದಲು");
Promise.resolve("2. ಪ್ರಾಮಿಸ್ ಡೇಟಾ").then(res => console.log(res));
console.log("3. ಕೊನೆಯಲ್ಲಿ");

ಔಟ್‌ಪುಟ್:

1. ಮೊದಲು
3. ಕೊನೆಯಲ್ಲಿ
2. ಪ್ರಾಮಿಸ್ ಡೇಟಾ

6. Promise.allSettled() (ES2020 ವೈಶಿಷ್ಟ್ಯ)

ರಿಯಲ್-ಟೈಮ್ ಪ್ರಾಜೆಕ್ಟ್‌ಗಳಲ್ಲಿ ಒಂದಕ್ಕಿಂತ ಹೆಚ್ಚು ಪ್ರಾಮಿಸ್‌ಗಳಿದ್ದರೆ (ಉದಾಹರಣೆಗೆ 3 ಬೇರೆ ಬೇರೆ API ಗಳಿಂದ ಡೇಟಾ ತರುವಾಗ), ಅವುಗಳಲ್ಲಿ ಕೆಲವು ಯಶಸ್ವಿಯಾಗಿ, ಕೆಲವು ವಿಫಲವಾದರೂ, ಎಲ್ಲವೂ "ಸೆಟಲ್" (Settled) ಆಗುವವರೆಗೆ ಕಾಯಲು Promise.allSettled() ಅನ್ನು ಬಳಸಲಾಗುತ್ತದೆ.
