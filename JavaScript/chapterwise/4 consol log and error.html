ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ console.log() ಮತ್ತು Errors ಪತ್ತೆಹಚ್ಚುವ ವಿಧಾನ

ಸರಳ ಉದಾಹರಣೆಗಳಿಂದ ಹಿಡಿದು ಪ್ರೊಫೆಷನಲ್ (Pro) ಮಟ್ಟದವರೆಗೆ ಚಿತ್ರ ಸಹಿತ ಸಂಪೂರ್ಣ ವಿವರ

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ console.log() ಮತ್ತು Errors (ತಪ್ಪುಗಳನ್ನು) ಪತ್ತೆಹಚ್ಚುವ ವಿಧಾನ ಪ್ರತಿಯೊಬ್ಬ ಪ್ರೋಗ್ರಾಮರ್ ತಿಳಿದಿರಬೇಕಾದ ಅತ್ಯಂತ ಮೂಲಭೂತ ಹಾಗೂ ಪ್ರಮುಖ ಕೌಶಲ್ಯಗಳಾಗಿವೆ. ಸರಳ ಉದಾಹರಣೆಗಳಿಂದ ಹಿಡಿದು ಪ್ರೊಫೆಷನಲ್ (Pro) ಮಟ್ಟದವರೆಗೆ ಇದರ ಸಂಪೂರ್ಣ ವಿವರ ಇಲ್ಲಿದೆ:

ಭಾಗ 1: console.log() ಎಂದರೇನು? (ಸಂಪೂರ್ಣ ವಿವರ)

1. ದೈನಂದಿನ ಜೀವನದ ಉದಾಹರಣೆ (ಮಕ್ಕಳಿಗೆ / Beginner Level)

ನಿಮ್ಮ ಮನೆಯಲ್ಲಿ ಕುಕ್ಕರ್‌ನಲ್ಲಿ ಅಡುಗೆ ಮಾಡುವಾಗ ಒಳಗೆ ಅನ್ನ ಬೆಂದಿದೆಯೋ ಇಲ್ಲವೋ ಎಂದು ಹೊರಗಿನಿಂದ ಕಾಣಿಸುವುದಿಲ್ಲ. ಆದರೆ ಕುಕ್ಕರ್ ಮುಚ್ಚಳವು ಪಾರದರ್ಶಕ ಗಾಜಿನದ್ದಾಗಿದ್ದರೆ (Glass Lid) ಒಳಗೆ ಏನಾಗುತ್ತಿದೆ ಎಂದು ನೇರವಾಗಿ ಕಾಣಿಸುತ್ತದೆ ಅಲ್ಲವೇ?

ಅದೇ ರೀತಿ, ನಾವು ಬರೆಯುವ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಕೋಡ್ ಕಂಪ್ಯೂಟರ್ ಒಳಗೆ ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತಿದೆ, ವೇರಿಯೇಬಲ್ ಒಳಗೆ ಯಾವ ಬೆಲೆ ಇದೆ, ಲೆಕ್ಕಾಚಾರ ಸರಿಯಾಗಿ ನಡೆಯುತ್ತಿದೆಯೇ ಎಂದು ನೋಡಲು ಇರುವ ರಹಸ್ಯ ಕಿಟಕಿಯೇ (X-Ray / CCTV ಕ್ಯಾಮೆರಾ) console.log().

  • ಇದು ಯಾರಿಗಾಗಿ? ಇದು ವೆಬ್‌ಸೈಟ್ ಬಳಸುವ ಸಾಮಾನ್ಯ ಜನರಿಗೆ ಪರದೆಯ ಮೇಲೆ ಕಾಣಿಸುವುದಿಲ್ಲ. ಇದು ಕೇವಲ ಕೋಡ್ ಬರೆಯುವ ಡೆವಲಪರ್‌ಗಳಿಗೆ (Developers) ಪರೀಕ್ಷೆ ಮಾಡಲು ಮಾತ್ರ ಕಾಣಿಸುತ್ತದೆ.
  • ಬ್ರೌಸರ್‌‌ನಲ್ಲಿ ಇದನ್ನು ನೋಡುವುದು ಹೇಗೆ?
    1. ಕ್ರೋಮ್ (Chrome) ಬ್ರೌಸರ್ ಓಪನ್ ಮಾಡಿ.
    2. ಮೌಸ್‌‌ನಿಂದ Right Click ಮಾಡಿ ➔ Inspect ಮೇಲೆ ಕ್ಲಿಕ್ ಮಾಡಿ (ಅಥವಾ ಕೀಬೋರ್ಡ್‌ನಲ್ಲಿ F12 ಅಥವಾ Ctrl + Shift + I ಒತ್ತಿ).
    3. ಅಲ್ಲಿ ಮೇಲ್ಭಾಗದಲ್ಲಿ ಕಾಣುವ Console ಟ್ಯಾಬ್ ಕ್ಲಿಕ್ ಮಾಡಿದರೆ ನಿಮ್ಮ ಕೋಡ್‌ನ ಫಲಿತಾಂಶ ಕಾಣಿಸುತ್ತದೆ.
ಚಿತ್ರ 1: ಕುಕ್ಕರ್‌ನ ಗಾಜಿನ ಮುಚ್ಚಳದಂತೆ ಕೋಡ್ ಒಳಗಿನ ರಹಸ್ಯ ತೋರಿಸುವ console.log()

2. console.log() ಬಳಸುವ ವಿಧಾನಗಳು

JavaScript
// 1. ಸಾಮಾನ್ಯ ಸಂದೇಶ (String) ಪ್ರಿಂಟ್ ಮಾಡಲು:
console.log("ನಮಸ್ಕಾರ ಕರ್ನಾಟಕ!"); 

// 2. ಗಣಿತದ ಲೆಕ್ಕಾಚಾರ ಪರೀಕ್ಷಿಸಲು:
console.log(10 + 25); // Output: 35

// 3. ವೇರಿಯೇಬಲ್ ( ಡಬ್ಬಿಯ ) ಒಳಗೆ ಏನಿದೆ ಎಂದು ನೋಡಲು:
let score = 95;
console.log(score); // Output: 95

// 4. ಒಂದಕ್ಕಿಂತ ಹೆಚ್ಚು ಮಾಹಿತಿಗಳನ್ನು ಒಟ್ಟಿಗೆ ನೋಡಲು (Comma ಬಳಸಿ):
let userName = "ರವಿ";
console.log("ಬಳಕೆದಾರನ ಹೆಸರು:", userName, "ಅಂಕಗಳು:", score);
// Output: ಬಳಕೆದಾರನ ಹೆಸರು: ರವಿ ಅಂಕಗಳು: 95

3. Pro Level: ಕೇವಲ console.log ಮಾತ್ರವಲ್ಲ! ಇತರ ಪ್ರಮುಖ ವಿಧಗಳು

ಒಬ್ಬ ಪರಿಣಿತ ಡೆವಲಪರ್ ಕೇವಲ log ಮಾತ್ರ ಬಳಸುವುದಿಲ್ಲ, ಸಂದರ್ಭಕ್ಕೆ ತಕ್ಕಂತೆ ಇವುಗಳನ್ನೂ ಬಳಸುತ್ತಾರೆ:

  • console.warn() (ಎಚ್ಚರಿಕೆ): ಟ್ರಾಫಿಕ್ ಸಿಗ್ನಲ್‌ನ ಹಳದಿ ಲೈಟ್ ಇದ್ದಂತೆ. ಕೋಡ್ ನಿಲ್ಲುವುದಿಲ್ಲ, ಆದರೆ "ಮುಂದೆ ಸಮಸ್ಯೆಯಾಗಬಹುದು ಎಚ್ಚರ!" ಎಂದು ಹಳದಿ ಬಣ್ಣದಲ್ಲಿ ತೋರಿಸುತ್ತದೆ.
  • console.error() (ತಪ್ಪು/ಅಪಾಯ): ಟ್ರಾಫಿಕ್ ಸಿಗ್ನಲ್‌ನ ಕೆಂಪು ಲೈಟ್ ಇದ್ದಂತೆ. ಏನೋ ದೊಡ್ಡ ತಪ್ಪಾಗಿದೆ ಎಂದು ಕೆಂಪು ಬಣ್ಣದಲ್ಲಿ ತೋರಿಸುತ್ತದೆ.
  • console.table() (ಕೋಷ್ಟಕ): ನಿಮ್ಮ ಬಳಿ ದೊಡ್ಡ Array ಅಥವಾ Object ಇದ್ದರೆ, ಅದನ್ನು ಸುಂದರವಾದ ಟೇಬಲ್ (Table) ರೂಪದಲ್ಲಿ ತೋರಿಸುತ್ತದೆ.
  • console.time() & console.timeEnd(): ನಿಮ್ಮ ಕೋಡ್ ರನ್ ಆಗಲು ಎಷ್ಟು ಮಿಲಿಸೆಕೆಂಡ್ ಸಮಯ ತೆಗೆದುಕೊಂಡಿತು ಎಂದು ಸ್ಟಾಪ್‌ವಾಚ್ (Stopwatch) ಹಿಡಿದು ಅಳೆಯುತ್ತದೆ.
ಚಿತ್ರ 2: Console ನ ವಿವಿಧ ಬಣ್ಣಗಳು ಮತ್ತು ಕೆಲಸಗಳು (ಟ್ರಾಫಿಕ್ ಸಿಗ್ನಲ್ ಹೋಲಿಕೆ)
JavaScript
console.warn("ಇದು ಎಚ್ಚರಿಕೆಯ ಸಂದೇಶ! ಪಾಸ್‌ವರ್ಡ್ ಚಿಕ್ಕದಾಗಿದೆ.");
console.error("ಕ್ಷಮಿಸಿ! ಸರ್ವರ್ ಕನೆಕ್ಟ್ ಆಗುತ್ತಿಲ್ಲ.");

// Object ಅಥವಾ Array ಅನ್ನು ಟೇಬಲ್ ರೂಪದಲ್ಲಿ ನೋಡಲು:
let students = [
  { name: "ಅಪ್ಪು", marks: 98 },
  { name: "ರಾಮು", marks: 85 }
];
console.table(students);

ಭಾಗ 2: ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Errors (ತಪ್ಪುಗಳು) ಎಂದರೇನು ಮತ್ತು ಅವುಗಳನ್ನು ಪತ್ತೆಹಚ್ಚುವುದು (Identify) ಹೇಗೆ?

ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿ ನಾವು ಬರೆದ ಕೋಡ್‌ನಲ್ಲಿ ಏನಾದರೂ ತಪ್ಪಾದರೆ ಕಂಪ್ಯೂಟರ್‌ಗೆ ಮುಂದೆ ಏನು ಮಾಡಬೇಕೆಂದು ತಿಳಿಯದೆ ಕೆಲಸ ನಿಲ್ಲಿಸುತ್ತದೆ. ಇದನ್ನೇ Error (ಬಗ್ / Bug) ಎನ್ನುತ್ತಾರೆ. ಆ ತಪ್ಪನ್ನು ಹುಡುಕಿ ಸರಿಪಡಿಸುವ ಪ್ರಕ್ರಿಯೆಯನ್ನು Debugging ಎಂದು ಕರೆಯುತ್ತಾರೆ.

1. Error Message ಅನ್ನು ಓದಿ ತಪ್ಪು ಎಲ್ಲಿದೆ ಎಂದು ಕಂಡುಹಿಡಿಯುವುದು ಹೇಗೆ?

ಕನ್ಸೋಲ್‌ನಲ್ಲಿ ಕೆಂಪು ಬಣ್ಣದ Error ಬಂದಾಗ ಭಯಪಡಬೇಡಿ! ಆ ಒಂದೇ ಸಾಲಿನಲ್ಲಿ ಕಂಪ್ಯೂಟರ್ ನಿಮಗೆ 3 ಅತ್ಯಮೂಲ್ಯ ಸುಳಿವುಗಳನ್ನು ಕೊಟ್ಟಿರುತ್ತದೆ.

ಈ ಕೆಳಗಿನ Error ಗಮನಿಸಿ:

Uncaught ReferenceError: marks is not defined at script.js:12:13
ಚಿತ್ರ 3: ಕೆಂಪು Error Message ಅನ್ನು ಬಿಡಿಸಿ ಓದುವ X-Ray ವಿಧಾನ (3 ಸುಳಿವುಗಳು)

ಇದನ್ನು ಬಿಡಿಸಿ ಓದುವ ವಿಧಾನ:

  1. ಮೊದಲ ಭಾಗ (ReferenceError - ತಪ್ಪಿನ ಜಾತಿ): ಇದು ಯಾವ ರೀತಿಯ ತಪ್ಪು ಎಂದು ಹೇಳುತ್ತದೆ.
  2. ಎರಡನೇ ಭಾಗ (marks is not defined - ತಪ್ಪಿನ ಕಾರಣ): marks ಎಂಬ ವೇರಿಯೇಬಲ್ ಅನ್ನು ನೀವು ಎಲ್ಲಿಯೂ ಸೃಷ್ಟಿಸಿಲ್ಲ (ಅಥವಾ ಸ್ಪೆಲ್ಲಿಂಗ್ ತಪ್ಪಾಗಿದೆ) ಎಂದು ನೇರವಾಗಿ ಹೇಳುತ್ತಿದೆ!
  3. ಮೂರನೇ ಭಾಗ (at script.js:12:13 - ತಪ್ಪು ಇರುವ ವಿಳಾಸ):
    • script.js ➔ ಫೈಲ್ ಹೆಸರು.
    • 12 ➔ 12ನೇ ಸಾಲಿನಲ್ಲಿ (Line Number 12) ತಪ್ಪಾಗಿದೆ!
    • 13 ➔ ಆ ಸಾಲಿನ 13ನೇ ಅಕ್ಷರದ ಬಳಿ ತಪ್ಪಿದೆ.

2. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಬರುವ 4 ಪ್ರಮುಖ Error ವಿಧಗಳು

ಚಿತ್ರ 4: ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ 4 ಪ್ರಮುಖ Error ವಿಧಗಳ ಸಚಿತ್ರ ಹೋಲಿಕೆ

1. SyntaxError (ವ್ಯಾಕರಣದ ತಪ್ಪು)

  • ಮಕ್ಕಳ ಉದಾಹರಣೆ: ಕನ್ನಡದಲ್ಲಿ ವಾಕ್ಯ ಬರೆಯುವಾಗ ಪೂರ್ಣವಿರಾಮ ಅಥವಾ ಕಾಗುಣಿತ ತಪ್ಪು ಮಾಡಿದಂತೆ.
  • ಕೋಡಿಂಗ್‌ನಲ್ಲಿ: ಕೋಟ್ಸ್ (" ") ಮುಚ್ಚಲು ಮರೆತಾಗ, ಬ್ರಾಕೆಟ್ ) ಬಿಟ್ಟಾಗ ಅಥವಾ ಕೀವರ್ಡ್ ಸ್ಪೆಲ್ಲಿಂಗ್ ತಪ್ಪು ಬರೆದಾಗ ಇದು ಬರುತ್ತದೆ.
JavaScript
// ತಪ್ಪಾದ ಕೋಡ್ (ಕೊನೆಯಲ್ಲಿ ಬ್ರಾಕೆಟ್ ಹಾಕಿಲ್ಲ):
console.log("ನಮಸ್ಕಾರ"; 
// Error: Uncaught SyntaxError: missing ) after argument list

2. ReferenceError (ಇಲ್ಲದ ವಸ್ತುವನ್ನು ಕರೆದಾಗ)

  • ಮಕ್ಕಳ ಉದಾಹರಣೆ: ತರಗತಿಯಲ್ಲಿ ಇಲ್ಲದ ವಿದ್ಯಾರ್ಥಿಯ ಹೆಸರನ್ನು ಹಾಜರಾತಿಯಲ್ಲಿ ಕರೆದಂತೆ!
  • ಕೋಡಿಂಗ್‌ನಲ್ಲಿ: ವೇರಿಯೇಬಲ್ ಅನ್ನು ಡಿಕ್ಲೇರ್ ಮಾಡದೆಯೇ (let ಬಳಸದೆ) ಪ್ರಿಂಟ್ ಮಾಡಲು ಹೋದಾಗ ಅಥವಾ ಹೆಸರಿನ ಸ್ಪೆಲ್ಲಿಂಗ್ ತಪ್ಪು ಬರೆದಾಗ ಬರುತ್ತದೆ.
JavaScript
let userAge = 20;
console.log(userage); // ಇಲ್ಲಿ 'A' ಬದಲು ಸಣ್ಣ 'a' ಹಾಕಲಾಗಿದೆ!
// Error: Uncaught ReferenceError: userage is not defined

3. TypeError (ವಸ್ತುವಿನ ಗುಣಕ್ಕೆ ವಿರುದ್ಧವಾದ ಕೆಲಸ ಮಾಡಿದಾಗ)

  • ಮಕ್ಕಳ ಉದಾಹರಣೆ: ನೀರಿನ ಬಾಟಲಿಯಿಂದ ಇಸ್ತ್ರಿ (Iron) ಮಾಡಲು ಪ್ರಯತ್ನಿಸಿದಂತೆ!
  • ಕೋಡಿಂಗ್‌ನಲ್ಲಿ: const ವೇರಿಯೇಬಲ್ ಬೆಲೆಯನ್ನು ಬದಲಾಯಿಸಲು ಹೋದಾಗ, ಅಥವಾ Number ಮೇಲೆ String ಫಂಕ್ಷನ್ ಬಳಸಿದಾಗ, ಅಥವಾ null/undefined ಒಳಗಿನ ಮಾಹಿತಿ ಹುಡುಕಲು ಹೋದಾಗ ಬರುತ್ತದೆ.
JavaScript
const pi = 3.14;
pi = 5; // const ಬೆಲೆಯನ್ನು ಬದಲಾಯಿಸಲು ಸಾಧ್ಯವಿಲ್ಲ!
// Error: Uncaught TypeError: Assignment to constant variable.

4. Logical Error (ತಾರ್ಕಿಕ ತಪ್ಪು — ಸೈಲೆಂಟ್ ಕಿಲ್ಲರ್!)

  • ಮಕ್ಕಳ ಉದಾಹರಣೆ: ಚಹಾ ಮಾಡುವಾಗ ಸಕ್ಕರೆ ಬದಲು ಉಪ್ಪು ಹಾಕಿದಂತೆ! ನೋಡಲು ಚಹಾ ತರಹವೇ ಇರುತ್ತದೆ (ಯಾವುದೇ ಕೆಂಪು Error ಬರುವುದಿಲ್ಲ), ಆದರೆ ಕುಡಿದಾಗಲೇ ತಪ್ಪು ಗೊತ್ತಾಗುವುದು!
  • ಕೋಡಿಂಗ್‌ನಲ್ಲಿ: ಕಂಪ್ಯೂಟರ್‌ಗೆ ನಿಮ್ಮ ಸೂತ್ರ ತಪ್ಪು ಎಂದು ಗೊತ್ತಾಗುವುದಿಲ್ಲ, ಆದರೆ ಉತ್ತರ ಮಾತ್ರ ತಪ್ಪಾಗಿ ಬರುತ್ತದೆ. ಇದನ್ನು ಪತ್ತೆಹಚ್ಚಲು console.log() ಒಂದೇ ದಾರಿ!
JavaScript
// ಸರಾಸರಿ (Average) ಕಂಡುಹಿಡಿಯುವಾಗ ಬ್ರಾಕೆಟ್ ಹಾಕಲು ಮರೆತರೆ:
let m1 = 80;
let m2 = 100;
let average = m1 + m2 / 2; // ಮೊದಲು 100/2 = 50 ಆಗಿ, 80 + 50 = 130 ಆಗುತ್ತದೆ!
console.log("ಸರಾಸರಿ:", average); // Output: 130 (ಬರಬೇಕಾಗಿದ್ದು 90!)

3. ಯಾವುದೇ Error ಬಂದಾಗ ಪತ್ತೆಹಚ್ಚಲು 4 ಹಂತದ ಸೂತ್ರ (Debugging Steps)

ಚಿತ್ರ 5: ಯಾವುದೇ Error ಪತ್ತೆಹಚ್ಚಲು 4 ಹಂತದ ಸೂತ್ರ (Debugging Steps)
  1. ಹಂತ 1 (Console ನೋಡಿ): ಬ್ರೌಸರ್‌ನಲ್ಲಿ F12 ಒತ್ತಿ ಕನ್ಸೋಲ್‌ನಲ್ಲಿ ಕೆಂಪು ಸಾಲು ಬಂದಿದೆಯೇ ಎಂದು ನೋಡಿ.
  2. ಹಂತ 2 (Line Number ಗುರುತಿಸಿ): Error ಬಲಭಾಗದಲ್ಲಿರುವ ಫೈಲ್_ಹೆಸರು:ಸಾಲಿನ_ಸಂಖ್ಯೆ (ಉದಾ: app.js:24) ಮೇಲೆ ಕ್ಲಿಕ್ ಮಾಡಿ. ಅದು ನೇರವಾಗಿ ತಪ್ಪಿರುವ ಸಾಲಿಗೆ ಕರೆದುಕೊಂಡು ಹೋಗುತ್ತದೆ.
  3. ಹಂತ 3 (console.log ಟೆಸ್ಟ್ ಮಾಡಿ): ಒಂದು ವೇಳೆ ಕೆಂಪು Error ಬರದಿದ್ದರೂ ಫಲಿತಾಂಶ ತಪ್ಪಾಗಿದ್ದರೆ, ಪ್ರತಿ ಹಂತದಲ್ಲೂ ವೇರಿಯೇಬಲ್ ಬೆಲೆ ಏನಾಗುತ್ತಿದೆ ಎಂದು ನೋಡಲು ಸಾಲುಗಳ ನಡುವೆ console.log() ಹಾಕಿ ಚೆಕ್ ಮಾಡಿ.
  4. ಹಂತ 4 (typeof ಪರೀಕ್ಷಿಸಿ): ಕೆಲವೊಮ್ಮೆ ಸಂಖ್ಯೆ (10) ಬದಲು ಅಕ್ಷರ ("10") ಬಂದಿರುತ್ತದೆ. ಆಗ console.log(typeof variableName) ಬಳಸಿ ಡೇಟಾ ಟೈಪ್ ಪರೀಕ್ಷಿಸಿ.

4. ಮಾಸ್ಟರ್ ಸಾರಾಂಶ (Cheat Sheet)

ವಿಷಯ / Error ಹೆಸರು ಮಕ್ಕಳ ಭಾಷೆಯಲ್ಲಿ ಉದಾಹರಣೆ ಯಾವಾಗ ಬರುತ್ತದೆ / ಏನು ಕೆಲಸ ಮಾಡುತ್ತದೆ?
console.log() ಅಡುಗೆ ಕುಕ್ಕರ್‌ನ ಗಾಜಿನ ಮುಚ್ಚಳ ಕೋಡ್ ಒಳಗೆ ಏನಾಗುತ್ತಿದೆ ಎಂದು ಕನ್ಸೋಲ್‌ನಲ್ಲಿ ನೋಡಲು
console.table() ಶಾಲೆಯ ಹಾಜರಾತಿ ಪಟ್ಟಿ Object ಮತ್ತು Array ಡೇಟಾವನ್ನು ಟೇಬಲ್ ರೂಪದಲ್ಲಿ ನೋಡಲು
SyntaxError ಕನ್ನಡ ಕಾಗುಣಿತ / ವ್ಯಾಕರಣ ತಪ್ಪು ಬ್ರಾಕೆಟ್ (), ಕೋಟ್ಸ್ "", ಚಿಹ್ನೆಗಳನ್ನು ಮರೆತಾಗ
ReferenceError ಶಾಲೆಗೆ ಬರದ ಹುಡುಗನನ್ನು ಕರೆಯುವುದು ಡಿಕ್ಲೇರ್ ಮಾಡದ ವೇರಿಯೇಬಲ್ ಬಳಸಿದಾಗ ಅಥವಾ ಸ್ಪೆಲ್ಲಿಂಗ್ ತಪ್ಪಾದಾಗ
TypeError ಚಮಚದಿಂದ ಬೀಗ ತೆಗೆಯಲು ಪ್ರಯತ್ನಿಸುವುದು const ತಿದ್ದಲು ಹೋದಾಗ ಅಥವಾ ತಪ್ಪಾದ Data Type ಬಳಸಿದಾಗ
Logical Error ಚಹಾಕ್ಕೆ ಸಕ್ಕರೆ ಬದಲು ಉಪ್ಪು ಹಾಕುವುದು ಯಾವುದೇ Error ತೋರಿಸದೆ, ಫಲಿತಾಂಶ ಮಾತ್ರ ತಪ್ಪು ಬರುವುದು
