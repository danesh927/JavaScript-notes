JavaScript Prototypes (ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಪ್ರೋಟೋಟೈಪ್)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ 'ಪ್ರೋಟೋಟೈಪ್' (Prototype) ಎಂದರೆ ಒಂದು ಆಬ್ಜೆಕ್ಟ್ (Object) ಮತ್ತೊಂದು ಆಬ್ಜೆಕ್ಟ್‌ನಿಂದ ಗುಣಲಕ್ಷಣಗಳನ್ನು ಬಳುವಳಿಯಾಗಿ ಪಡೆಯುವ (Inheritance) ವ್ಯವಸ್ಥೆ.

ಐಟಿ ಹಿನ್ನೆಲೆ ಇಲ್ಲದವರಿಗೂ ಅರ್ಥವಾಗುವಂತೆ, ದಿನನಿತ್ಯದ ಉದಾಹರಣೆಗಳಿಂದ ಹಿಡಿದು ವೃತ್ತಿಪರ (Professional) ಮಟ್ಟದವರೆಗೆ ಇದನ್ನು ೪ ಹಂತಗಳಲ್ಲಿ ಕಲಿಯೋಣ.

ಭಾಗ 1: ಪ್ರೋಟೋಟೈಪ್‌ನ ೪ ಹಂತಗಳು

ಹಂತ 1

ಬಿಗಿನರ್ (ನೈಜ ಜೀವನದ ಉದಾಹರಣೆ - DNA ಮತ್ತು ವಂಶಾವಳಿ)

ಒಂದು ಮಗುವಿಗೆ ತಂದೆ-ತಾಯಿಯಿಂದ ಕೆಲವು ಗುಣಗಳು (ಉದಾಹರಣೆಗೆ: ಕಣ್ಣಿನ ಬಣ್ಣ, ಮುಖಚರ್ಯೆ) ಬಳುವಳಿಯಾಗಿ (Inheritance) ಬರುತ್ತವೆ. ತಂದೆಗೆ ಆತನ ತಂದೆಯಿಂದ (ಅಜ್ಜ) ಗುಣಗಳು ಬಂದಿರುತ್ತವೆ.

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಪ್ರತಿಯೊಂದು ಆಬ್ಜೆಕ್ಟ್ (Object) ಕೂಡ ಒಬ್ಬ "ತಂದೆ"ಯನ್ನು (Parent Object) ಹೊಂದಿರುತ್ತದೆ. ಆ ತಂದೆ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನೇ ನಾವು Prototype ಎಂದು ಕರೆಯುತ್ತೇವೆ. ಮಗುವಿನ ಬಳಿ ಯಾವುದಾದರೂ ವಸ್ತು ಅಥವಾ ಕೌಶಲ್ಯ ಇಲ್ಲದಿದ್ದರೆ, ಅದು ತಂದೆಯ ಬಳಿ ಕೇಳುತ್ತದೆ. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಕೂಡ ಇದೇ ರೀತಿ ಕೆಲಸ ಮಾಡುತ್ತದೆ.

ಚಿತ್ರ: ಗುಣಲಕ್ಷಣಗಳು ತಂದೆಯಿಂದ ಮಗುವಿಗೆ ಹರಿಯುವ ರೀತಿ (Inheritance)

ಹಂತ 2

ಇಂಟರ್ಮೀಡಿಯೇಟ್ (ಪ್ರೋಟೋಟೈಪ್ ಏಕೆ ಬೇಕು? ಮೆಮೊರಿ ಉಳಿತಾಯ)

ನೀವು ಒಂದು ಕಂಪನಿಗೆ ಸಾಫ್ಟ್‌ವೇರ್ ಮಾಡುತ್ತಿದ್ದೀರಿ. ಅದರಲ್ಲಿ 10,000 ಉದ್ಯೋಗಿಗಳಿದ್ದಾರೆ (Employees). ಪ್ರತಿಯೊಬ್ಬ ಉದ್ಯೋಗಿಗೂ ತಮ್ಮ ಸಂಬಳ ಲೆಕ್ಕಹಾಕಲು calculateSalary ಎಂಬ ಫಂಕ್ಷನ್ ಬೇಕು ಎಂದಿಟ್ಟುಕೊಳ್ಳಿ.

ನೀವು ಪ್ರತಿ 10,000 ಉದ್ಯೋಗಿಗಳ ಆಬ್ಜೆಕ್ಟ್ ಒಳಗೂ ಈ ಫಂಕ್ಷನ್ ಬರೆದರೆ, ಕಂಪ್ಯೂಟರ್‌ನ ಮೆಮೊರಿಯಲ್ಲಿ 10,000 ಫಂಕ್ಷನ್‌ಗಳು ಸೃಷ್ಟಿಯಾಗಿ ಜಾಗ ತುಂಬಿಹೋಗುತ್ತದೆ. ಇದಕ್ಕೆ ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನ ಪರಿಹಾರವೇ ಪ್ರೋಟೋಟೈಪ್.

calculateSalary ಫಂಕ್ಷನ್ ಅನ್ನು ಎಲ್ಲರಿಗೂ ಅನ್ವಯಿಸುವ ಒಂದು 'ಮುಖ್ಯ ಕಚೇರಿಯಲ್ಲಿ' (Prototype) ಇಡುತ್ತೇವೆ. 10,000 ಉದ್ಯೋಗಿಗಳು ತಮಗೆ ಬೇಕಾದಾಗ ಆ ಒಂದೇ ಫಂಕ್ಷನ್ ಅನ್ನು ಬಳಸಿಕೊಳ್ಳುತ್ತಾರೆ. ಇದರಿಂದ 9,999 ಫಂಕ್ಷನ್‌ಗಳ ಮೆಮೊರಿ ಉಳಿಯುತ್ತದೆ ಮತ್ತು ಪ್ರೋಗ್ರಾಮ್ ವೇಗವಾಗಿ ಕೆಲಸ ಮಾಡುತ್ತದೆ.

ಚಿತ್ರ: ಒಂದೇ ಫಂಕ್ಷನ್ ಅನ್ನು 10,000 ಆಬ್ಜೆಕ್ಟ್‌ಗಳು ಹಂಚಿಕೊಳ್ಳುವ ಮೂಲಕ ಮೆಮೊರಿ ಉಳಿತಾಯ

ಹಂತ 3

ಅಡ್ವಾನ್ಸ್ಡ್ (ಕೋಡ್‌ನಲ್ಲಿ ಪ್ರೋಟೋಟೈಪ್ ಬಳಕೆ)

ಹಿಂದಿನ ವಿಷಯದಲ್ಲಿ ನಾವು ಕನ್ಸ್ಟ್ರಕ್ಟರ್ (Constructor - ಅಚ್ಚು) ಬಗ್ಗೆ ತಿಳಿದಿದ್ದೆವು. ಈಗ ಆ ಅಚ್ಚಿನ ಪ್ರೋಟೋಟೈಪ್‌ಗೆ ಒಂದು ಫಂಕ್ಷನ್ ಸೇರಿಸಿ ನೋಡೋಣ.

JavaScript 
// 1. ಉದ್ಯೋಗಿಗಳನ್ನು ಸೃಷ್ಟಿಸುವ ಅಚ್ಚು (Constructor)
function Employee(name, age) {
  this.name = name;
  this.age = age;
}

// 2. ಪ್ರೋಟೋಟೈಪ್‌ಗೆ ಫಂಕ್ಷನ್ ಸೇರಿಸುವುದು (ಇದು ಕೇಂದ್ರ ಕಚೇರಿ ಇದ್ದಂತೆ)
Employee.prototype.sayHello = function() {
  console.log("ನಮಸ್ಕಾರ, ನನ್ನ ಹೆಸರು " + this.name);
};

// 3. ಹೊಸ ಉದ್ಯೋಗಿಗಳನ್ನು ಸೃಷ್ಟಿಸುವುದು
let emp1 = new Employee("ರಮೇಶ್", 30);
let emp2 = new Employee("ಸುರೇಶ್", 25);

// 4. ಪ್ರೋಟೋಟೈಪ್‌ನಲ್ಲಿರುವ ಫಂಕ್ಷನ್ ಬಳಸುವುದು
emp1.sayHello(); // ಔಟ್‌ಪುಟ್: ನಮಸ್ಕಾರ, ನನ್ನ ಹೆಸರು ರಮೇಶ್
emp2.sayHello(); // ಔಟ್‌ಪುಟ್: ನಮಸ್ಕಾರ, ನನ್ನ ಹೆಸರು ಸುರೇಶ್

ಇಲ್ಲಿ ಗಮನಿಸಿ: sayHello ಎಂಬ ಫಂಕ್ಷನ್ emp1 ಅಥವಾ emp2 ಆಬ್ಜೆಕ್ಟ್‌ನ ಒಳಗಿಲ್ಲ. ಬದಲಿಗೆ ಅವರ "ತಂದೆ"ಯ ಬಳಿ (Employee.prototype) ಇದೆ. emp1 ಗೆ ಆ ಫಂಕ್ಷನ್ ಬೇಕಾದಾಗ ಅದು ತನ್ನ ಪ್ರೋಟೋಟೈಪ್‌ನಿಂದ ಎರವಲು ಪಡೆಯುತ್ತದೆ. ಹಿಂದಿನ ಪಾಠದಲ್ಲಿ ಕಲಿತಂತೆ, ಇಲ್ಲಿರುವ this ಕೀವರ್ಡ್ ಆಯಾ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು (ರಮೇಶ್ ಅಥವಾ ಸುರೇಶ್) ಸೂಚಿಸುತ್ತದೆ.

ಹಂತ 4

ಪ್ರೊ (ಪ್ರೋಟೋಟೈಪ್ ಚೈನ್ - Prototype Chain)

ಒಂದು ಆಬ್ಜೆಕ್ಟ್‌ನಲ್ಲಿ ನೀವು ಯಾವುದಾದರೂ ಫಂಕ್ಷನ್ ಅಥವಾ ಗುಣವನ್ನು ಹುಡುಕಿದಾಗ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಅದನ್ನು ಹೇಗೆ ಹುಡುಕುತ್ತದೆ ಎಂಬುದೇ 'ಪ್ರೋಟೋಟೈಪ್ ಚೈನ್'. ಉದಾಹರಣೆಗೆ, ನೀವು emp1.toString() ಎಂದು ಟೈಪ್ ಮಾಡಿದರೆ ಏನಾಗುತ್ತದೆ?

  1. ಸ್ವಂತ ಆಬ್ಜೆಕ್ಟ್: ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಮೊದಲು emp1 ಆಬ್ಜೆಕ್ಟ್ ಒಳಗೆ toString ಇದೆಯೇ ಎಂದು ಹುಡುಕುತ್ತದೆ (ಇಲ್ಲಿ ಇಲ್ಲ).
  2. ತಂದೆ (Prototype): ನಂತರ, ಅದರ ತಂದೆಯ ಬಳಿ ಅಂದರೆ Employee.prototype ಒಳಗೆ ಹುಡುಕುತ್ತದೆ (ಇಲ್ಲೂ ಇಲ್ಲ).
  3. ಅಜ್ಜ (Grandparent): ನಂತರ, ತಂದೆಯ ತಂದೆಯ ಬಳಿ ಅಂದರೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ಅತಿ ದೊಡ್ಡ ಯಜಮಾನನಾದ Object.prototype ಬಳಿ ಹುಡುಕುತ್ತದೆ. toString ಎಂಬುದು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ಮೂಲ ಆಬ್ಜೆಕ್ಟ್‌ನಲ್ಲಿ (ಅಜ್ಜ) ಮೊದಲೇ ಬರೆದಿಡಲ್ಪಟ್ಟಿರುತ್ತದೆ. ಅಲ್ಲಿಂದ ಅದು ಕೆಲಸ ಮಾಡುತ್ತದೆ.
  4. ಅಂತ್ಯ (null): ಅಲ್ಲಿಯೂ ಸಿಗದಿದ್ದರೆ ಕೊನೆಗೆ ತುತ್ತತುದಿಗೆ ಹೋಗಿ null (ಏನೂ ಇಲ್ಲ / ದೋಷ) ಎಂದು ಹೇಳುತ್ತದೆ.

ಈ ಹುಡುಕಾಟದ ಸರಪಳಿಯನ್ನೇ Prototype Chain ಎನ್ನುತ್ತಾರೆ. ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ (ES6) ಇದೇ ಕೆಲಸವನ್ನು ಸುಲಭವಾಗಿ ಮಾಡಲು class ಮತ್ತು extends ಕೀವರ್ಡ್ ಬಳಸುತ್ತಾರೆ. ಆದರೆ ಅದರ ಹಿಂದಿರುವ ಅಸಲಿ ಯಂತ್ರವೇ (Engine) ಈ 'ಪ್ರೋಟೋಟೈಪ್'.

(ಹೆಚ್ಚುವರಿ ಮಾಹಿತಿ: ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಪ್ರೋಟೋಟೈಪ್ ಬದಲಿಗೆ ಬಳಸುವ class (ಕ್ಲಾಸ್) ಕಾನ್ಸೆಪ್ಟ್ ಅನ್ನು ಇದೇ ರೀತಿ ಉದಾಹರಣೆಯೊಂದಿಗೆ ವಿವರಿಸಬೇಕೇ? ಎಂದು ಮೂಲ ನೋಟ್ಸ್‌ನಲ್ಲಿ ಕೇಳಲಾಗಿದೆ - ಕ್ಲಾಸ್‌ಗಳು ಪ್ರೋಟೋಟೈಪ್‌ನ ಮೇಲಿರುವ ಕೇವಲ Syntax Sugar ಮಾತ್ರ!).

ಭಾಗ 2: Constructor Prototype ಮತ್ತು Object Prototype

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Constructor Prototype ಮತ್ತು Object Prototype ಎರಡೂ ಒಂದೇ ವ್ಯವಸ್ಥೆಯ ಎರಡು ವಿಭಿನ್ನ ಭಾಗಗಳಾಗಿವೆ. ಇವೆರಡರ ನಡುವಿನ ವ್ಯತ್ಯಾಸ ಮತ್ತು ಸಂಬಂಧವನ್ನು ಅರ್ಥಮಾಡಿಕೊಳ್ಳುವುದು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿ "ಪ್ರೊ" (Pro) ಮಟ್ಟಕ್ಕೇರಲು ಅತಿ ಮುಖ್ಯ ಹೆಜ್ಜೆಯಾಗಿದೆ.

ಪ್ರೋಟೋಟೈಪ್ (Prototype) ಎನ್ನುವುದು ಮೆಮೊರಿಯನ್ನು (RAM) ಉಳಿಸಲು ಮತ್ತು ಕೋಡ್ ಅನ್ನು ಮರುಬಳಕೆ ಮಾಡಲು ಇರುವ ಒಂದು ಅತ್ಯದ್ಭುತ ವ್ಯವಸ್ಥೆ. ಯಾವುದೇ ಹಿನ್ನೆಲೆಯ ವ್ಯಕ್ತಿಗೂ ಸುಲಭವಾಗಿ ಅರ್ಥವಾಗುವಂತೆ "ಕಾರ್ ಫ್ಯಾಕ್ಟರಿ ಮತ್ತು ಅದರ ಕೇಂದ್ರ ಕಚೇರಿ"ಯ ಉದಾಹರಣೆಯನ್ನು ತೆಗೆದುಕೊಂಡು ಇದನ್ನು 3 ಹಂತಗಳಲ್ಲಿ ವಿಂಗಡಿಸಿ ನೋಡೋಣ.

ಹಂತ 1

ನಾನ್-ಐಟಿ (Non-IT) ಹಿನ್ನೆಲೆಯವರಿಗಾಗಿ - "ಕಾರ್ ಫ್ಯಾಕ್ಟರಿ" ಉದಾಹರಣೆ

ನೀವು 'ಟಾಟಾ' ಕಂಪನಿಯ ಮಾಲೀಕರು ಎಂದು ಕಲ್ಪಿಸಿಕೊಳ್ಳಿ. ನೀವು ಪ್ರತಿದಿನ 10,000 ಕಾರುಗಳನ್ನು ತಯಾರಿಸುತ್ತೀರಿ. ಪ್ರತಿ ಕಾರಿಗೆ ಎರಡು ಮುಖ್ಯ ವಸ್ತುಗಳು ಬೇಕು:

  • ಭೌತಿಕ ವಸ್ತುಗಳು: ಕಾರಿನ ಮಾಡೆಲ್, ಬಣ್ಣ, ಟೈರ್ (ಇವು ಪ್ರತಿ ಕಾರಿಗೂ ಬೇರೆ ಬೇರೆ ಇರಬಹುದು).
  • ಸಾಫ್ಟ್‌ವೇರ್: "ಜಿಪಿಎಸ್ ಮ್ಯಾಪ್" ಮತ್ತು "ಮ್ಯೂಸಿಕ್ ಪ್ಲೇಯರ್" (ಇದು ಎಲ್ಲಾ 10,000 ಕಾರುಗಳಿಗೂ ಒಂದೇ ಆಗಿರುತ್ತದೆ).

ನೀವು ಈ ಮ್ಯಾಪ್ ಸಾಫ್ಟ್‌ವೇರ್ ಅನ್ನು ಪ್ರತಿಯೊಂದು ಕಾರಿನ ಮೆಮೊರಿ ಚಿಪ್‌ನಲ್ಲೂ ಇನ್‌ಸ್ಟಾಲ್ ಮಾಡುತ್ತಾ ಹೋದರೆ, ಕಾರಿನ ಮೆಮೊರಿ ಬೇಗ ಭರ್ತಿಯಾಗುತ್ತದೆ ಮತ್ತು ಕಂಪನಿಗೆ ನಷ್ಟವಾಗುತ್ತದೆ. ಇದಕ್ಕೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ನೀಡುವ ಪರಿಹಾರವೇ ಪ್ರೋಟೋಟೈಪ್. ಇದು ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ ಎಂದರೆ:

  • ಫ್ಯಾಕ್ಟರಿ (Constructor Function): ಇದು ಕಾರುಗಳನ್ನು ತಯಾರಿಸುವ ಅಚ್ಚು ಅಥವಾ ಫ್ಯಾಕ್ಟರಿ. ಇಲ್ಲಿ ಕಾರಿನ ಬಣ್ಣ ಮತ್ತು ಮಾಡೆಲ್‌ನಂತಹ ಬೇಸಿಕ್ (ಭೌತಿಕ) ವಿಷಯಗಳನ್ನು ಮಾತ್ರ ಫಿಕ್ಸ್ ಮಾಡಲಾಗುತ್ತದೆ.
  • ಕೇಂದ್ರ ಕಚೇರಿ (Constructor Prototype): ಇದು ಫ್ಯಾಕ್ಟರಿಯ ಕೇಂದ್ರ ಕಚೇರಿ (ಸರ್ವರ್). ಎಲ್ಲಾ ಕಾರುಗಳಿಗೂ ಬೇಕಾಗುವ ಸಾಮಾನ್ಯ ಸಾಫ್ಟ್‌ವೇರ್ (ಮ್ಯಾಪ್, ಮ್ಯೂಸಿಕ್ ಪ್ಲೇಯರ್) ಅನ್ನು ಪ್ರತಿಯೊಂದು ಕಾರಿಗೂ ಹಾಕುವ ಬದಲು, ಈ ಕೇಂದ್ರ ಕಚೇರಿಯಲ್ಲಿ ಒಂದೇ ಕಡೆ ಸ್ಟೋರ್ ಮಾಡಲಾಗುತ್ತದೆ. ಇಲ್ಲಿ ಹೊಸ ವೈಶಿಷ್ಟ್ಯ ಸೇರಿಸಿದರೆ, ಫ್ಯಾಕ್ಟರಿಯಿಂದ ಬಂದ ಎಲ್ಲಾ ಕಾರುಗಳಿಗೂ ಅದು ಉಚಿತವಾಗಿ ಅನ್ವಯವಾಗುತ್ತದೆ.
  • ಗುಪ್ತ ಕನೆಕ್ಷನ್ (Object Prototype / __proto__): ಫ್ಯಾಕ್ಟರಿಯಿಂದ ಸಿದ್ಧವಾಗಿ ಹೊರಬಂದ ಪ್ರತಿ ನಿಜವಾದ ಕಾರಿನ (Object) ಒಳಗೆ ಆ ಮ್ಯಾಪ್ ಸಾಫ್ಟ್‌ವೇರ್ ಇರುವುದಿಲ್ಲ. ಆದರೆ, ಆ ಕಾರಿಗೆ ತನ್ನ ಫ್ಯಾಕ್ಟರಿಯ ಕೇಂದ್ರ ಕಚೇರಿಯನ್ನು (Parent) ಸಂಪರ್ಕಿಸಲು ಒಂದು ಗುಪ್ತ ಇಂಟರ್ನೆಟ್ ಕನೆಕ್ಷನ್ ಅಥವಾ ಹೆಲ್ಪ್‌ಲೈನ್ ನಂಬರ್ ಇರುತ್ತದೆ. ಈ ಗುಪ್ತ ಕನೆಕ್ಷನ್ ಹೆಸರೇ __proto__. ಕಾರಿಗೆ ಸಹಾಯ ಬೇಕಾದಾಗಲೆಲ್ಲಾ (ಉದಾಹರಣೆಗೆ ಮ್ಯಾಪ್ ಓಪನ್ ಮಾಡಿದಾಗ) ಈ ಕನೆಕ್ಷನ್ ಮೂಲಕ ಕೇಂದ್ರ ಕಚೇರಿಯನ್ನು ಸಂಪರ್ಕಿಸಿ ಮ್ಯಾಪ್ ರನ್ ಮಾಡುತ್ತದೆ. ಇದರಿಂದ 10,000 ಕಾರುಗಳ ಮೆಮೊರಿ ಉಳಿಯುತ್ತದೆ!
ಹಂತ 2

ಬಿಗಿನರ್ಸ್ ಮತ್ತು ಐಟಿ ವಿದ್ಯಾರ್ಥಿಗಳಿಗೆ - ಕೋಡ್ ಮೂಲಕ ವಿವರಣೆ

ಡೆವಲಪರ್‌ಗಳು ಹೊಸ ಫಂಕ್ಷನ್‌ಗಳನ್ನು ಎಲ್ಲರಿಗೂ ಹಂಚಲು Constructor.prototype ಒಳಗೆ ಬರೆಯುತ್ತಾರೆ. ಮತ್ತು ಆಬ್ಜೆಕ್ಟ್ ತನಗೆ ಗೊತ್ತಿಲ್ಲದ ಕೆಲಸವನ್ನು ಮಾಡಲು __proto__ ಅನ್ನು ಬಳಸುತ್ತದೆ. ಇದನ್ನು ಕೋಡ್‌ಗೆ ಇಳಿಸೋಣ:

JavaScript 
// 1. Constructor Function (ಇದು ಫ್ಯಾಕ್ಟರಿ)
// ಇದು ಕೇವಲ ಬಣ್ಣ ಮತ್ತು ಮಾಡೆಲ್ ಅನ್ನು ಮಾತ್ರ ಕಾರಿಗೆ (this) ನೀಡುತ್ತದೆ.
function CarMaker(modelName, color) {
  this.model = modelName;
  this.color = color;
}

// 2. Constructor Prototype (ಇದು ಫ್ಯಾಕ್ಟರಿಯ ಕೇಂದ್ರ ಕಚೇರಿ)
// .prototype ಬಳಸಿ ನಾವು ಸಾಫ್ಟ್‌ವೇರ್ ಫಂಕ್ಷನ್ ಅನ್ನು ಮೆಮೊರಿಯಲ್ಲಿ ಒಂದೇ ಕಡೆ ಸೇವ್ ಮಾಡುತ್ತಿದ್ದೇವೆ.
CarMaker.prototype.startEngine = function() {
  console.log(this.model + " ಇಂಜಿನ್ ಸ್ಟಾರ್ಟ್ ಆಯಿತು!");
};

CarMaker.prototype.startMap = function() {
  console.log(this.model + " ಕಾರಿನಲ್ಲಿ ಜಿಪಿಎಸ್ ಮ್ಯಾಪ್ ಶುರುವಾಯಿತು!");
};

// 3. Object (ಇದು ಫ್ಯಾಕ್ಟರಿಯಿಂದ ತಯಾರಾದ ಹೊಸ ಕಾರುಗಳು)
let myCar = new CarMaker("Tata Nexon", "Red");
let yourCar = new CarMaker("Tata Harrier", "Black");

// 4. ಕಾರುಗಳು ಕೇಂದ್ರ ಕಚೇರಿಯ ಫಂಕ್ಷನ್ ಅನ್ನು ಬಳಸಿಕೊಳ್ಳುವುದು
// ಕಾರಿನ ಒಳಗೆ 'startEngine' ಅಥವಾ 'startMap' ಇಲ್ಲ. 
// ಆದರೂ ಅದು ತನ್ನ ಗುಪ್ತ ಕನೆಕ್ಷನ್ (__proto__) ಬಳಸಿ ಕೇಂದ್ರ ಕಚೇರಿಯಿಂದ ಅದನ್ನು ತಂದು ರನ್ ಮಾಡುತ್ತದೆ!
myCar.startEngine(); // ಔಟ್‌ಪುಟ್: Tata Nexon ಇಂಜಿನ್ ಸ್ಟಾರ್ಟ್ ಆಯಿತು!
yourCar.startMap();  // ಔಟ್‌ಪುಟ್: Tata Harrier ಕಾರಿನಲ್ಲಿ ಜಿಪಿಎಸ್ ಮ್ಯಾಪ್ ಶುರುವಾಯಿತು!

// ಆ ಗುಪ್ತ ಕನೆಕ್ಷನ್ ಅನ್ನು ನೋಡಬೇಕೆ?
console.log(myCar.__proto__); 
// ಇದು CarMaker.prototype (ಕೇಂದ್ರ ಕಚೇರಿ) ಗೆ ಲಿಂಕ್ ಆಗಿರುವುದನ್ನು ತೋರಿಸುತ್ತದೆ.

ಪ್ರಮುಖ ಅಂಶ: prototype ಎನ್ನುವ ಪ್ರಾಪರ್ಟಿ ಕೇವಲ Functions ಗಳಿಗೆ (ಇಲ್ಲಿ CarMaker) ಮಾತ್ರ ಇರುತ್ತದೆ. __proto__ ಎನ್ನುವುದು ಪ್ರತಿಯೊಂದು Object ನ (ಇಲ್ಲಿ myCar) ಒಳಗೂ ಇರುವ ಗುಪ್ತ ಪ್ರಾಪರ್ಟಿ.

ಹಂತ 3

ಪ್ರೊಫೆಷನಲ್ ಡೆವಲಪರ್‌ಗಳಿಗೆ - "The Under the Hood Mechanics"

ಬಿಗಿನರ್‌ಗಳಿಗೆ ಆಗುವ ಅತಿದೊಡ್ಡ ಗೊಂದಲವೆಂದರೆ "ಇವೆರಡೂ ಬೇರೆ ಬೇರೆಯೇ?" ಎಂಬುದು. ಉತ್ತರ: ಇಲ್ಲ, ಅವೆರಡೂ ಒಂದೇ ಜಾಗವನ್ನು (ಮೆಮೊರಿ ಲೊಕೇಶನ್) ಸೂಚಿಸುತ್ತವೆ. ನೋಡುವ ದೃಷ್ಟಿಕೋನ ಮಾತ್ರ ಬೇರೆ. ಪ್ರೊ ಲೆವೆಲ್‌ನಲ್ಲಿ ನೀವು ಅರ್ಥಮಾಡಿಕೊಳ್ಳಬೇಕಾದ 3 ಮುಖ್ಯ ನಿಯಮಗಳು ಇಲ್ಲಿವೆ:

ನಿಯಮ 1: ಮೆಮೊರಿ ಲಿಂಕಿಂಗ್ (The Truth of __proto__)

ಫ್ಯಾಕ್ಟರಿ ಕಡೆಯಿಂದ ನೋಡುವಾಗ ಇರುವ CarMaker.prototype ಮತ್ತು ಕಾರಿನ ಕಡೆಯಿಂದ ನೋಡುವಾಗ ಇರುವ myCar.__proto__ ಎರಡೂ ಮೆಮೊರಿಯಲ್ಲಿ ಒಂದೇ ಅಡ್ರೆಸ್ ಅನ್ನು ಪಾಯಿಂಟ್ (Point) ಮಾಡುತ್ತವೆ.

JavaScript
// ಕಾರಿನ ಗುಪ್ತ ಕನೆಕ್ಷನ್ ಮತ್ತು ಫ್ಯಾಕ್ಟರಿಯ ಕೇಂದ್ರ ಕಚೇರಿ ಎರಡೂ ಒಂದೇನಾ?
console.log(myCar.__proto__ === CarMaker.prototype); 
// ಔಟ್‌ಪುಟ್: true (ಹೌದು, ಎರಡೂ ಒಂದೇ!)

ನಿಯಮ 2: ರಿಯಲ್ ಟೈಮ್ ಅಪ್‌ಡೇಟ್ಸ್ (Dynamic Nature)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಪ್ರೋಟೋಟೈಪ್‌ಗಳು ಡೈನಾಮಿಕ್ ಆಗಿರುತ್ತವೆ. ಅಂದರೆ, ನೀವು ಕಾರುಗಳನ್ನು ಮಾರಾಟ ಮಾಡಿದ ನಂತರ (ಆಬ್ಜೆಕ್ಟ್ ಕ್ರಿಯೇಟ್ ಮಾಡಿದ ಮೇಲೆ), ಕೇಂದ್ರ ಕಚೇರಿಯಲ್ಲಿ ಹೊಸ ಫಂಕ್ಷನ್ ಸೇರಿಸಿದರೂ, ಹಳೆಯ ಕಾರುಗಳಿಗೂ ಆ ವೈಶಿಷ್ಟ್ಯ ತಕ್ಷಣವೇ ಸಿಗುತ್ತದೆ!

JavaScript
// ಹಳೆಯ ಕಾರುಗಳು ಆಲ್ರೆಡಿ ಕ್ರಿಯೇಟ್ ಆಗಿವೆ. ಈಗ ಕೇಂದ್ರ ಕಚೇರಿಗೆ ಹೊಸ ಫೀಚರ್ ಸೇರಿಸೋಣ:
CarMaker.prototype.playMusic = function() {
  console.log("ಮ್ಯೂಸಿಕ್ ಪ್ಲೇಯರ್ ಆನ್ ಆಯಿತು.");
};

// myCar ಮೊದಲೇ ಕ್ರಿಯೇಟ್ ಆಗಿದ್ದರೂ, ಹೊಸ ಫೀಚರ್ ಅದಕ್ಕೆ ತಾನಾಗಿಯೇ ಸಿಗುತ್ತದೆ!
myCar.playMusic(); // ಔಟ್‌ಪುಟ್: ಮ್ಯೂಸಿಕ್ ಪ್ಲೇಯರ್ ಆನ್ ಆಯಿತು.

ನಿಯಮ 3: ಪ್ರೋಟೋಟೈಪ್ ಚೈನ್ (The Prototype Chain)

ಒಂದು ವೇಳೆ ಕಾರಿನಲ್ಲಿ ಆ ಫಂಕ್ಷನ್ ಇಲ್ಲ, ಮತ್ತು ಕೇಂದ್ರ ಕಚೇರಿಯಲ್ಲೂ ಇಲ್ಲ ಎಂದಾದರೆ ಏನಾಗುತ್ತದೆ? ಆಗ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಸುಮ್ಮನೆ ಕೂರುವುದಿಲ್ಲ. ಅದು ಕೇಂದ್ರ ಕಚೇರಿಯ ಮೇಲಿರುವ "ಜಾಗತಿಕ ಕಚೇರಿಗೆ" (Global Object Prototype) ಹೋಗುತ್ತದೆ. ಇದರ ಹುಡುಕಾಟದ ದಾರಿ ಹೀಗಿರುತ್ತದೆ:

myCar -> myCar.__proto__ (CarMaker.prototype) -> Object.prototype -> null. ಎಲ್ಲೂ ಸಿಗದಿದ್ದಾಗ ಮಾತ್ರ ಅದು undefined ಅಥವಾ Error ನೀಡುತ್ತದೆ.

ಒಟ್ಟಾರೆಯಾಗಿ ಸಾರಾಂಶ (Summary)

ಈ ಇಡೀ ವ್ಯವಸ್ಥೆ 3 ಹಂತಗಳಲ್ಲಿ ನಡೆಯುತ್ತದೆ:

  1. ಸೃಷ್ಟಿ: ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಫಂಕ್ಷನ್ ಬರೆದ ತಕ್ಷಣ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಅದಕ್ಕೊಂದು prototype ಎಂಬ ಖಾಲಿ ಜಾಗವನ್ನು ಸೃಷ್ಟಿಸುತ್ತದೆ. ಮೆಮೊರಿ ಉಳಿಸಲು ಹಂಚಿಕೆಯಾಗುವ (Shared) ಫಂಕ್ಷನ್‌ಗಳನ್ನು ಇಲ್ಲಿ ಹಾಕಲಾಗುತ್ತದೆ.
  2. ಸಂಪರ್ಕ: new ಕೀವರ್ಡ್ ಬಳಸಿ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸಿದಾಗ, ಆಬ್ಜೆಕ್ಟ್‌ನ ಒಳಗೆ __proto__ ಎಂಬ ಗುಪ್ತ ಕೊಂಡಿ ಸೃಷ್ಟಿಯಾಗುತ್ತದೆ ಮತ್ತು ಅದು ಕನ್ಸ್ಟ್ರಕ್ಟರ್‌ನ prototype ಗೆ ಬೆಸೆದುಕೊಳ್ಳುತ್ತದೆ.
  3. ಹುಡುಕಾಟ: ಫಂಕ್ಷನ್ ಕರೆದಾಗ, ಆಬ್ಜೆಕ್ಟ್ ಮೊದಲು ತನ್ನೊಳಗೆ ಹುಡುಕುತ್ತದೆ. ಇಲ್ಲದಿದ್ದರೆ, ತನ್ನ __proto__ ಮೂಲಕ ಕನ್ಸ್ಟ್ರಕ್ಟರ್‌ನ prototype (ತನ್ನ Parent) ಗೆ ಹೋಗಿ ಅದನ್ನು ರನ್ ಮಾಡುತ್ತದೆ.

ಪ್ರಮುಖ ವ್ಯತ್ಯಾಸಗಳ ಹೋಲಿಕೆ ಪಟ್ಟಿ

ಗುಣಲಕ್ಷಣ Constructor Function / Prototype Object Prototype (__proto__)
ಉದಾಹರಣೆ ಅರ್ಥ ಕಾರು ತಯಾರಿಸುವ ಫ್ಯಾಕ್ಟರಿ ಮತ್ತು ಅದರ ಕೇಂದ್ರ ಕಚೇರಿ/ಸರ್ವರ್. ತಯಾರಾಗಿ ಹೊರಬಂದ ಕಾರು ಮತ್ತು ಅದರ ಹೆಲ್ಪ್‌ಲೈನ್ ಕನೆಕ್ಷನ್.
ಯಾರಿಗೆ ಇರುತ್ತದೆ? (ಎಲ್ಲಿರುತ್ತದೆ) ಕೇವಲ ಫಂಕ್ಷನ್‌ಗಳ (Constructor Functions) ಜೊತೆ ಇರುತ್ತದೆ. ಪ್ರತಿಯೊಂದು ಆಬ್ಜೆಕ್ಟ್‌ಗಳ ಒಳಗೆ (Instance) ಗುಪ್ತವಾಗಿ ಇರುತ್ತದೆ.
ಮುಖ್ಯ ಕೆಲಸವೇನು? ಆಬ್ಜೆಕ್ಟ್‌ಗಳಿಗೆ ಹಂಚಲು ಬೇಕಾದ ಸಾಮಾನ್ಯ ಗುಣ/ಫಂಕ್ಷನ್‌ಗಳನ್ನು ಸ್ಟೋರ್ ಮಾಡುವುದು. ತನ್ನಲ್ಲಿ ಇಲ್ಲದ ಫಂಕ್ಷನ್‌ಗಳನ್ನು ತನ್ನ ಮೂಲದಲ್ಲಿ (Parent) ಹುಡುಕಿಕೊಂಡು ಹೋಗುವುದು.
ಬಳಕೆದಾರರು ಯಾರು? ಡೆವಲಪರ್‌ಗಳು ಹೊಸ ವೈಶಿಷ್ಟ್ಯಗಳನ್ನು ಬರೆಯಲು/ಸೇರಿಸಲು ಬಳಸುತ್ತಾರೆ. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಇಂಜಿನ್ (ಬ್ರೌಸರ್) ರಹಸ್ಯವಾಗಿ ಇದನ್ನು ಬಳಸಿಕೊಳ್ಳುತ್ತದೆ.

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ 'ಪ್ರೋಟೋಟೈಪ್' (Prototype) ಎಂದರೆ ಒಂದು ಆಬ್ಜೆಕ್ಟ್ (Object) ಮತ್ತೊಂದು ಆಬ್ಜೆಕ್ಟ್‌ನಿಂದ ಗುಣಲಕ್ಷಣಗಳನ್ನು ಬಳುವಳಿಯಾಗಿ ಪಡೆಯುವ (Inheritance) ವ್ಯವಸ್ಥೆ. ಐಟಿ ಹಿನ್ನೆಲೆ ಇಲ್ಲದವರಿಗೂ ಅರ್ಥವಾಗುವಂತೆ, ದಿನನಿತ್ಯದ ಉದಾಹರಣೆಗಳಿಂದ ಹಿಡಿದು ವೃತ್ತಿಪರ (Professional) ಮಟ್ಟದವರೆಗೆ ಇದನ್ನು ೪ ಹಂತಗಳಲ್ಲಿ ಕಲಿಯೋಣ. ಹಂತ 1: ಬಿಗಿನರ್ (ನೈಜ ಜೀವನದ ಉದಾಹರಣೆ - DNA ಮತ್ತು ವಂಶಾವಳಿ) ಒಂದು ಮಗುವಿಗೆ ತಂದೆ-ತಾಯಿಯಿಂದ ಕೆಲವು ಗುಣಗಳು (ಉದಾಹರಣೆಗೆ: ಕಣ್ಣಿನ ಬಣ್ಣ, ಮುಖಚರ್ಯೆ) ಬಳುವಳಿಯಾಗಿ (Inheritance) ಬರುತ್ತವೆ. ತಂದೆಗೆ ಆತನ ತಂದೆಯಿಂದ (ಅಜ್ಜ) ಗುಣಗಳು ಬಂದಿರುತ್ತವೆ. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಪ್ರತಿಯೊಂದು ಆಬ್ಜೆಕ್ಟ್ (Object) ಕೂಡ ಒಬ್ಬ "ತಂದೆ"ಯನ್ನು (Parent Object) ಹೊಂದಿರುತ್ತದೆ. ಆ ತಂದೆ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನೇ ನಾವು Prototype ಎಂದು ಕರೆಯುತ್ತೇವೆ. ಮಗುವಿನ ಬಳಿ ಯಾವುದಾದರೂ ವಸ್ತು ಅಥವಾ ಕೌಶಲ್ಯ ಇಲ್ಲದಿದ್ದರೆ, ಅದು ತಂದೆಯ ಬಳಿ ಕೇಳುತ್ತದೆ. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಕೂಡ ಇದೇ ರೀತಿ ಕೆಲಸ ಮಾಡುತ್ತದೆ. ಹಂತ 2: ಇಂಟರ್ಮೀಡಿಯೇಟ್ (ಪ್ರೋಟೋಟೈಪ್ ಏಕೆ ಬೇಕು? ಮೆಮೊರಿ ಉಳಿತಾಯ) ನೀವು ಒಂದು ಕಂಪನಿಗೆ ಸಾಫ್ಟ್‌ವೇರ್ ಮಾಡುತ್ತಿದ್ದೀರಿ. ಅದರಲ್ಲಿ 10,000 ಉದ್ಯೋಗಿಗಳಿದ್ದಾರೆ (Employees). ಪ್ರತಿಯೊಬ್ಬ ಉದ್ಯೋಗಿಗೂ ತಮ್ಮ ಸಂಬಳ ಲೆಕ್ಕಹಾಕಲು calculateSalary ಎಂಬ ಫಂಕ್ಷನ್ ಬೇಕು ಎಂದಿಟ್ಟುಕೊಳ್ಳಿ. ನೀವು ಪ್ರತಿ 10,000 ಉದ್ಯೋಗಿಗಳ ಆಬ್ಜೆಕ್ಟ್ ಒಳಗೂ ಈ ಫಂಕ್ಷನ್ ಬರೆದರೆ, ಕಂಪ್ಯೂಟರ್‌ನ ಮೆಮೊರಿಯಲ್ಲಿ 10,000 ಫಂಕ್ಷನ್‌ಗಳು ಸೃಷ್ಟಿಯಾಗಿ ಜಾಗ ತುಂಬಿಹೋಗುತ್ತದೆ. ಇದಕ್ಕೆ ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನ ಪರಿಹಾರವೇ ಪ್ರೋಟೋಟೈಪ್. calculateSalary ಫಂಕ್ಷನ್ ಅನ್ನು ಎಲ್ಲರಿಗೂ ಅನ್ವಯಿಸುವ ಒಂದು 'ಮುಖ್ಯ ಕಚೇರಿಯಲ್ಲಿ' (Prototype) ಇಡುತ್ತೇವೆ. 10,000 ಉದ್ಯೋಗಿಗಳು ತಮಗೆ ಬೇಕಾದಾಗ ಆ ಒಂದೇ ಫಂಕ್ಷನ್ ಅನ್ನು ಬಳಸಿಕೊಳ್ಳುತ್ತಾರೆ. ಇದರಿಂದ 9,999 ಫಂಕ್ಷನ್‌ಗಳ ಮೆಮೊರಿ ಉಳಿಯುತ್ತದೆ ಮತ್ತು ಪ್ರೋಗ್ರಾಮ್ ವೇಗವಾಗಿ ಕೆಲಸ ಮಾಡುತ್ತದೆ. ಹಂತ 3: ಅಡ್ವಾನ್ಸ್ಡ್ (ಕೋಡ್‌ನಲ್ಲಿ ಪ್ರೋಟೋಟೈಪ್ ಬಳಕೆ) ಹಿಂದಿನ ವಿಷಯದಲ್ಲಿ ನಾವು ಕನ್ಸ್ಟ್ರಕ್ಟರ್ (Constructor - ಅಚ್ಚು) ಬಗ್ಗೆ ತಿಳಿದಿದ್ದೆವು. ಈಗ ಆ ಅಚ್ಚಿನ ಪ್ರೋಟೋಟೈಪ್‌ಗೆ ಒಂದು ಫಂಕ್ಷನ್ ಸೇರಿಸಿ ನೋಡೋಣ. // 1. ಉದ್ಯೋಗಿಗಳನ್ನು ಸೃಷ್ಟಿಸುವ ಅಚ್ಚು (Constructor) function Employee(name, age) { this.name = name; this.age = age; } // 2. ಪ್ರೋಟೋಟೈಪ್‌ಗೆ ಫಂಕ್ಷನ್ ಸೇರಿಸುವುದು (ಇದು ಕೇಂದ್ರ ಕಚೇರಿ ಇದ್ದಂತೆ) Employee.prototype.sayHello = function() { console.log("ನಮಸ್ಕಾರ, ನನ್ನ ಹೆಸರು " + this.name); }; // 3. ಹೊಸ ಉದ್ಯೋಗಿಗಳನ್ನು ಸೃಷ್ಟಿಸುವುದು let emp1 = new Employee("ರಮೇಶ್", 30); let emp2 = new Employee("ಸುರೇಶ್", 25); // 4. ಪ್ರೋಟೋಟೈಪ್‌ನಲ್ಲಿರುವ ಫಂಕ್ಷನ್ ಬಳಸುವುದು emp1.sayHello(); // ಔಟ್‌ಪುಟ್: ನಮಸ್ಕಾರ, ನನ್ನ ಹೆಸರು ರಮೇಶ್ emp2.sayHello(); // ಔಟ್‌ಪುಟ್: ನಮಸ್ಕಾರ, ನನ್ನ ಹೆಸರು ಸುರೇಶ್ ಇಲ್ಲಿ ಗಮನಿಸಿ: sayHello ಎಂಬ ಫಂಕ್ಷನ್ emp1 ಅಥವಾ emp2 ಆಬ್ಜೆಕ್ಟ್‌ನ ಒಳಗಿಲ್ಲ. ಬದಲಿಗೆ ಅವರ "ತಂದೆ"ಯ ಬಳಿ (Employee.prototype) ಇದೆ. emp1 ಗೆ ಆ ಫಂಕ್ಷನ್ ಬೇಕಾದಾಗ ಅದು ತನ್ನ ಪ್ರೋಟೋಟೈಪ್‌ನಿಂದ ಎರವಲು ಪಡೆಯುತ್ತದೆ. ಹಿಂದಿನ ಪಾಠದಲ್ಲಿ ಕಲಿತಂತೆ, ಇಲ್ಲಿರುವ this ಕೀವರ್ಡ್ ಆಯಾ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು (ರಮೇಶ್ ಅಥವಾ ಸುರೇಶ್) ಸೂಚಿಸುತ್ತದೆ. ಹಂತ 4: ಪ್ರೊ (ಪ್ರೋಟೋಟೈಪ್ ಚೈನ್ - Prototype Chain) ಒಂದು ಆಬ್ಜೆಕ್ಟ್‌ನಲ್ಲಿ ನೀವು ಯಾವುದಾದರೂ ಫಂಕ್ಷನ್ ಅಥವಾ ಗುಣವನ್ನು ಹುಡುಕಿದಾಗ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಅದನ್ನು ಹೇಗೆ ಹುಡುಕುತ್ತದೆ ಎಂಬುದೇ 'ಪ್ರೋಟೋಟೈಪ್ ಚೈನ್'. ಉದಾಹರಣೆಗೆ, ನೀವು emp1.toString() ಎಂದು ಟೈಪ್ ಮಾಡಿದರೆ ಏನಾಗುತ್ತದೆ? 1. ಸ್ವಂತ ಆಬ್ಜೆಕ್ಟ್: ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಮೊದಲು emp1 ಆಬ್ಜೆಕ್ಟ್ ಒಳಗೆ toString ಇದೆಯೇ ಎಂದು ಹುಡುಕುತ್ತದೆ (ಇಲ್ಲಿ ಇಲ್ಲ). 2. ತಂದೆ (Prototype): ನಂತರ, ಅದರ ತಂದೆಯ ಬಳಿ ಅಂದರೆ Employee.prototype ಒಳಗೆ ಹುಡುಕುತ್ತದೆ (ಇಲ್ಲೂ ಇಲ್ಲ). 3. ಅಜ್ಜ (Grandparent): ನಂತರ, ತಂದೆಯ ತಂದೆಯ ಬಳಿ ಅಂದರೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ಅತಿ ದೊಡ್ಡ ಯಜಮಾನನಾದ Object.prototype ಬಳಿ ಹುಡುಕುತ್ತದೆ. toString ಎಂಬುದು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ಮೂಲ ಆಬ್ಜೆಕ್ಟ್‌ನಲ್ಲಿ (ಅಜ್ಜ) ಮೊದಲೇ ಬರೆದಿಡಲ್ಪಟ್ಟಿರುತ್ತದೆ. ಅಲ್ಲಿಂದ ಅದು ಕೆಲಸ ಮಾಡುತ್ತದೆ. 4. ಅಂತ್ಯ (null): ಅಲ್ಲಿಯೂ ಸಿಗದಿದ್ದರೆ ಕೊನೆಗೆ ತುತ್ತತುದಿಗೆ ಹೋಗಿ null (ಏನೂ ಇಲ್ಲ / ದೋಷ) ಎಂದು ಹೇಳುತ್ತದೆ. ಈ ಹುಡುಕಾಟದ ಸರಪಳಿಯನ್ನೇ Prototype Chain ಎನ್ನುತ್ತಾರೆ. ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ (ES6) ಇದೇ ಕೆಲಸವನ್ನು ಸುಲಭವಾಗಿ ಮಾಡಲು class ಮತ್ತು extends ಕೀವರ್ಡ್ ಬಳಸುತ್ತಾರೆ. ಆದರೆ ಅದರ ಹಿಂದಿರುವ ಅಸಲಿ ಯಂತ್ರವೇ (Engine) ಈ 'ಪ್ರೋಟೋಟೈಪ್'. ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಪ್ರೋಟೋಟೈಪ್ ಬದಲಿಗೆ ಬಳಸುವ class (ಕ್ಲಾಸ್) ಕಾನ್ಸೆಪ್ಟ್ ಅನ್ನು ಇದೇ ರೀತಿ ಉದಾಹರಣೆಯೊಂದಿಗೆ ವಿವರಿಸಬೇಕೇ? ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Constructor Prototype ಮತ್ತು Object Prototype ಎರಡೂ ಒಂದೇ ವ್ಯವಸ್ಥೆಯ ಎರಡು ವಿಭಿನ್ನ ಭಾಗಗಳಾಗಿವೆ. ಇವೆರಡರ ನಡುವಿನ ವ್ಯತ್ಯಾಸ ಮತ್ತು ಸಂಬಂಧವನ್ನು ಅರ್ಥಮಾಡಿಕೊಳ್ಳುವುದು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿ "ಪ್ರೊ" (Pro) ಮಟ್ಟಕ್ಕೇರಲು ಅತಿ ಮುಖ್ಯ ಹೆಜ್ಜೆಯಾಗಿದೆ. ಪ್ರೋಟೋಟೈಪ್ (Prototype) ಎನ್ನುವುದು ಮೆಮೊರಿಯನ್ನು (RAM) ಉಳಿಸಲು ಮತ್ತು ಕೋಡ್ ಅನ್ನು ಮರುಬಳಕೆ ಮಾಡಲು ಇರುವ ಒಂದು ಅತ್ಯದ್ಭುತ ವ್ಯವಸ್ಥೆ. ಯಾವುದೇ ಹಿನ್ನೆಲೆಯ ವ್ಯಕ್ತಿಗೂ ಸುಲಭವಾಗಿ ಅರ್ಥವಾಗುವಂತೆ "ಕಾರ್ ಫ್ಯಾಕ್ಟರಿ ಮತ್ತು ಅದರ ಕೇಂದ್ರ ಕಚೇರಿ"ಯ ಉದಾಹರಣೆಯನ್ನು ತೆಗೆದುಕೊಂಡು ಇದನ್ನು 3 ಹಂತಗಳಲ್ಲಿ ವಿಂಗಡಿಸಿ ನೋಡೋಣ. ಹಂತ 1: ನಾನ್-ಐಟಿ (Non-IT) ಹಿನ್ನೆಲೆಯವರಿಗಾಗಿ - "ಕಾರ್ ಫ್ಯಾಕ್ಟರಿ" ಉದಾಹರಣೆ ನೀವು 'ಟಾಟಾ' ಕಂಪನಿಯ ಮಾಲೀಕರು ಎಂದು ಕಲ್ಪಿಸಿಕೊಳ್ಳಿ. ನೀವು ಪ್ರತಿದಿನ 10,000 ಕಾರುಗಳನ್ನು ತಯಾರಿಸುತ್ತೀರಿ. ಪ್ರತಿ ಕಾರಿಗೆ ಎರಡು ಮುಖ್ಯ ವಸ್ತುಗಳು ಬೇಕು: 1. ಭೌತಿಕ ವಸ್ತುಗಳು: ಕಾರಿನ ಮಾಡೆಲ್, ಬಣ್ಣ, ಟೈರ್ (ಇವು ಪ್ರತಿ ಕಾರಿಗೂ ಬೇರೆ ಬೇರೆ ಇರಬಹುದು). 2. ಸಾಫ್ಟ್‌ವೇರ್: "ಜಿಪಿಎಸ್ ಮ್ಯಾಪ್" ಮತ್ತು "ಮ್ಯೂಸಿಕ್ ಪ್ಲೇಯರ್" (ಇದು ಎಲ್ಲಾ 10,000 ಕಾರುಗಳಿಗೂ ಒಂದೇ ಆಗಿರುತ್ತದೆ). ನೀವು ಈ ಮ್ಯಾಪ್ ಸಾಫ್ಟ್‌ವೇರ್ ಅನ್ನು ಪ್ರತಿಯೊಂದು ಕಾರಿನ ಮೆಮೊರಿ ಚಿಪ್‌ನಲ್ಲೂ ಇನ್‌ಸ್ಟಾಲ್ ಮಾಡುತ್ತಾ ಹೋದರೆ, ಕಾರಿನ ಮೆಮೊರಿ ಬೇಗ ಭರ್ತಿಯಾಗುತ್ತದೆ ಮತ್ತು ಕಂಪನಿಗೆ ನಷ್ಟವಾಗುತ್ತದೆ. ಇದಕ್ಕೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ನೀಡುವ ಪರಿಹಾರವೇ ಪ್ರೋಟೋಟೈಪ್. ಇದು ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ ಎಂದರೆ: • ಫ್ಯಾಕ್ಟರಿ (Constructor Function): ಇದು ಕಾರುಗಳನ್ನು ತಯಾರಿಸುವ ಅಚ್ಚು ಅಥವಾ ಫ್ಯಾಕ್ಟರಿ. ಇಲ್ಲಿ ಕಾರಿನ ಬಣ್ಣ ಮತ್ತು ಮಾಡೆಲ್‌ನಂತಹ ಬೇಸಿಕ್ (ಭೌತಿಕ) ವಿಷಯಗಳನ್ನು ಮಾತ್ರ ಫಿಕ್ಸ್ ಮಾಡಲಾಗುತ್ತದೆ. • ಕೇಂದ್ರ ಕಚೇರಿ (Constructor Prototype): ಇದು ಫ್ಯಾಕ್ಟರಿಯ ಕೇಂದ್ರ ಕಚೇರಿ (ಸರ್ವರ್). ಎಲ್ಲಾ ಕಾರುಗಳಿಗೂ ಬೇಕಾಗುವ ಸಾಮಾನ್ಯ ಸಾಫ್ಟ್‌ವೇರ್ (ಮ್ಯಾಪ್, ಮ್ಯೂಸಿಕ್ ಪ್ಲೇಯರ್) ಅನ್ನು ಪ್ರತಿಯೊಂದು ಕಾರಿಗೂ ಹಾಕುವ ಬದಲು, ಈ ಕೇಂದ್ರ ಕಚೇರಿಯಲ್ಲಿ ಒಂದೇ ಕಡೆ ಸ್ಟೋರ್ ಮಾಡಲಾಗುತ್ತದೆ. ಇಲ್ಲಿ ಹೊಸ ವೈಶಿಷ್ಟ್ಯ ಸೇರಿಸಿದರೆ, ಫ್ಯಾಕ್ಟರಿಯಿಂದ ಬಂದ ಎಲ್ಲಾ ಕಾರುಗಳಿಗೂ ಅದು ಉಚಿತವಾಗಿ ಅನ್ವಯವಾಗುತ್ತದೆ. • ಗುಪ್ತ ಕನೆಕ್ಷನ್ (Object Prototype / __proto__): ಫ್ಯಾಕ್ಟರಿಯಿಂದ ಸಿದ್ಧವಾಗಿ ಹೊರಬಂದ ಪ್ರತಿ ನಿಜವಾದ ಕಾರಿನ (Object) ಒಳಗೆ ಆ ಮ್ಯಾಪ್ ಸಾಫ್ಟ್‌ವೇರ್ ಇರುವುದಿಲ್ಲ. ಆದರೆ, ಆ ಕಾರಿಗೆ ತನ್ನ ಫ್ಯಾಕ್ಟರಿಯ ಕೇಂದ್ರ ಕಚೇರಿಯನ್ನು (Parent) ಸಂಪರ್ಕಿಸಲು ಒಂದು ಗುಪ್ತ ಇಂಟರ್ನೆಟ್ ಕನೆಕ್ಷನ್ ಅಥವಾ ಹೆಲ್ಪ್‌ಲೈನ್ ನಂಬರ್ ಇರುತ್ತದೆ. ಈ ಗುಪ್ತ ಕನೆಕ್ಷನ್ ಹೆಸರೇ __proto__. ಕಾರಿಗೆ ಸಹಾಯ ಬೇಕಾದಾಗಲೆಲ್ಲಾ (ಉದಾಹರಣೆಗೆ ಮ್ಯಾಪ್ ಓಪನ್ ಮಾಡಿದಾಗ) ಈ ಕನೆಕ್ಷನ್ ಮೂಲಕ ಕೇಂದ್ರ ಕಚೇರಿಯನ್ನು ಸಂಪರ್ಕಿಸಿ ಮ್ಯಾಪ್ ರನ್ ಮಾಡುತ್ತದೆ. ಇದರಿಂದ 10,000 ಕಾರುಗಳ ಮೆಮೊರಿ ಉಳಿಯುತ್ತದೆ! ಹಂತ 2: ಬಿಗಿನರ್ಸ್ ಮತ್ತು ಐಟಿ ವಿದ್ಯಾರ್ಥಿಗಳಿಗೆ - ಕೋಡ್ ಮೂಲಕ ವಿವರಣೆ ಡೆವಲಪರ್‌ಗಳು ಹೊಸ ಫಂಕ್ಷನ್‌ಗಳನ್ನು ಎಲ್ಲರಿಗೂ ಹಂಚಲು Constructor.prototype ಒಳಗೆ ಬರೆಯುತ್ತಾರೆ. ಮತ್ತು ಆಬ್ಜೆಕ್ಟ್ ತನಗೆ ಗೊತ್ತಿಲ್ಲದ ಕೆಲಸವನ್ನು ಮಾಡಲು __proto__ ಅನ್ನು ಬಳಸುತ್ತದೆ. ಇದನ್ನು ಕೋಡ್‌ಗೆ ಇಳಿಸೋಣ: // 1. Constructor Function (ಇದು ಫ್ಯಾಕ್ಟರಿ) // ಇದು ಕೇವಲ ಬಣ್ಣ ಮತ್ತು ಮಾಡೆಲ್ ಅನ್ನು ಮಾತ್ರ ಕಾರಿಗೆ (this) ನೀಡುತ್ತದೆ. function CarMaker(modelName, color) { this.model = modelName; this.color = color; } // 2. Constructor Prototype (ಇದು ಫ್ಯಾಕ್ಟರಿಯ ಕೇಂದ್ರ ಕಚೇರಿ) // .prototype ಬಳಸಿ ನಾವು ಸಾಫ್ಟ್‌ವೇರ್ ಫಂಕ್ಷನ್ ಅನ್ನು ಮೆಮೊರಿಯಲ್ಲಿ ಒಂದೇ ಕಡೆ ಸೇವ್ ಮಾಡುತ್ತಿದ್ದೇವೆ. ಆಗ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಸುಮ್ಮನೆ ಕೂರುವುದಿಲ್ಲ. ಅದು ಕೇಂದ್ರ ಕಚೇರಿಯ ಮೇಲಿರುವ "ಜಾಗತಿಕ ಕಚೇರಿಗೆ" (Global Object Prototype) ಹೋಗುತ್ತದೆ. ಇದರ ಹುಡುಕಾಟದ ದಾರಿ ಹೀಗಿರುತ್ತದೆ:myCar -> myCar.__proto__ (CarMaker.prototype) -> Object.prototype -> null. ಎಲ್ಲೂ ಸಿಗದಿದ್ದಾಗ ಮಾತ್ರ ಅದು undefined ಅಥವಾ Error ನೀಡುತ್ತದೆ. ಒಟ್ಟಾರೆಯಾಗಿ ಸಾರಾಂಶ (Summary) ಈ ಇಡೀ ವ್ಯವಸ್ಥೆ 3 ಹಂತಗಳಲ್ಲಿ ನಡೆಯುತ್ತದೆ: 1. ಸೃಷ್ಟಿ: ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಫಂಕ್ಷನ್ ಬರೆದ ತಕ್ಷಣ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಅದಕ್ಕೊಂದು prototype ಎಂಬ ಖಾಲಿ ಜಾಗವನ್ನು ಸೃಷ್ಟಿಸುತ್ತದೆ. ಮೆಮೊರಿ ಉಳಿಸಲು ಹಂಚಿಕೆಯಾಗುವ (Shared) ಫಂಕ್ಷನ್‌ಗಳನ್ನು ಇಲ್ಲಿ ಹಾಕಲಾಗುತ್ತದೆ. 2. ಸಂಪರ್ಕ: new ಕೀವರ್ಡ್ ಬಳಸಿ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸಿದಾಗ, ಆಬ್ಜೆಕ್ಟ್‌ನ ಒಳಗೆ __proto__ ಎಂಬ ಗುಪ್ತ ಕೊಂಡಿ ಸೃಷ್ಟಿಯಾಗುತ್ತದೆ ಮತ್ತು ಅದು ಕನ್ಸ್ಟ್ರಕ್ಟರ್‌ನ prototype ಗೆ ಬೆಸೆದುಕೊಳ್ಳುತ್ತದೆ. 3. ಹುಡುಕಾಟ: ಫಂಕ್ಷನ್ ಕರೆದಾಗ, ಆಬ್ಜೆಕ್ಟ್ ಮೊದಲು ತನ್ನೊಳಗೆ ಹುಡುಕುತ್ತದೆ. ಇಲ್ಲದಿದ್ದರೆ, ತನ್ನ __proto__ ಮೂಲಕ ಕನ್ಸ್ಟ್ರಕ್ಟರ್‌ನ prototype (ತನ್ನ Parent) ಗೆ ಹೋಗಿ ಅದನ್ನು ರನ್ ಮಾಡುತ್ತದೆ. ಪ್ರಮುಖ ವ್ಯತ್ಯಾಸಗಳ ಹೋಲಿಕೆ ಪಟ್ಟಿ ಗುಣಲಕ್ಷಣ Constructor Function / Prototype Object Prototype (__proto__) ಉದಾಹರಣೆ ಅರ್ಥ ಕಾರು ತಯಾರಿಸುವ ಫ್ಯಾಕ್ಟರಿ ಮತ್ತು ಅದರ ಕೇಂದ್ರ ಕಚೇರಿ/ಸರ್ವರ್. ತಯಾರಾಗಿ ಹೊರಬಂದ ಕಾರು ಮತ್ತು ಅದರ ಹೆಲ್ಪ್‌ಲೈನ್ ಕನೆಕ್ಷನ್. ಯಾರಿಗೆ ಇರುತ್ತದೆ? (ಎಲ್ಲಿರುತ್ತದೆ) ಕೇವಲ ಫಂಕ್ಷನ್‌ಗಳ (Constructor Functions) ಜೊತೆ ಇರುತ್ತದೆ. ಪ್ರತಿಯೊಂದು ಆಬ್ಜೆಕ್ಟ್‌ಗಳ ಒಳಗೆ (Instance) ಗುಪ್ತವಾಗಿ ಇರುತ್ತದೆ. ಮುಖ್ಯ ಕೆಲಸವೇನು? ಆಬ್ಜೆಕ್ಟ್‌ಗಳಿಗೆ ಹಂಚಲು ಬೇಕಾದ ಸಾಮಾನ್ಯ ಗುಣ/ಫಂಕ್ಷನ್‌ಗಳನ್ನು ಸ್ಟೋರ್ ಮಾಡುವುದು. ತನ್ನಲ್ಲಿ ಇಲ್ಲದ ಫಂಕ್ಷನ್‌ಗಳನ್ನು ತನ್ನ ಮೂಲದಲ್ಲಿ (Parent) ಹುಡುಕಿಕೊಂಡು ಹೋಗುವುದು. ಬಳಕೆದಾರರು ಯಾರು? ಡೆವಲಪರ್‌ಗಳು ಹೊಸ ವೈಶಿಷ್ಟ್ಯಗಳನ್ನು ಬರೆಯಲು/ಸೇರಿಸಲು ಬಳಸುತ್ತಾರೆ. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಇಂಜಿನ್ (ಬ್ರೌಸರ್) ರಹಸ್ಯವಾಗಿ ಇದನ್ನು ಬಳಸಿಕೊಳ್ಳುತ್ತದೆ.
