ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಫಂಕ್ಷನ್‌ಗಳು

ಶೂನ್ಯದಿಂದ ವೃತ್ತಿಪರ ಹಂತದವರೆಗೆ (Beginner to Pro)

1. ಫಂಕ್ಷನ್ ಎಂದರೆ ಏನು? (Absolute Basic)

ಫಂಕ್ಷನ್ ಎಂದರೆ ಒಂದು "ಯಂತ್ರ (Machine)" ಇದ್ದಂತೆ. ನೀವು ಅದಕ್ಕೆ ಏನಾದರೂ ಕಚ್ಚಾವಸ್ತು (Input) ನೀಡಿದರೆ, ಅದು ಒಳಗೆ ಪ್ರಕ್ರಿಯೆ (Process) ನಡೆಸಿ ಒಂದು ಫಲಿತಾಂಶವನ್ನು (Output) ನೀಡುತ್ತದೆ.

ಸರಳ ಉದಾಹರಣೆ: ಮಿಕ್ಸಿ ಜಾರ್ (Mixer Grinder) ಒಂದು ಫಂಕ್ಷನ್. ಅದಕ್ಕೆ ಹಣ್ಣುಗಳನ್ನು (Input) ಹಾಕಿದರೆ, ಅದು ಜ್ಯೂಸ್ (Output) ಕೊಡುತ್ತದೆ. ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿ, ಒಂದೇ ಕೋಡ್ ಅನ್ನು ಮತ್ತೆ ಮತ್ತೆ ಬರೆಯುವುದನ್ನು ತಪ್ಪಿಸಲು (Reusability) ಫಂಕ್ಷನ್‌ಗಳನ್ನು ಬಳಸುತ್ತೇವೆ.

(ಉದಾಹರಣೆಗೆ: ವೆಬ್‌ಸೈಟ್‌ನಲ್ಲಿ "Buy Now" ಬಟನ್ ಅನ್ನು 10 ಕಡೆ ಹಾಕಿದ್ದರೂ, 10 ಬಾರಿ ಕೋಡ್ ಬರೆಯುವ ಬದಲು ಒಂದೇ ಫಂಕ್ಷನ್ ಬರೆದು 10 ಕಡೆ ಲಿಂಕ್ ಮಾಡಬಹುದು).

JavaScript - Basic Function 
// 1. ಫಂಕ್ಷನ್ ಅನ್ನು ರಚಿಸುವುದು (Declaration) - ಯಂತ್ರವನ್ನು ತರುವುದು
function sayHello() {
    console.log("ನಮಸ್ಕಾರ! ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಜಗತ್ತಿಗೆ ಸ್ವಾಗತ.");
}

// 2. ಫಂಕ್ಷನ್ ಅನ್ನು ಕರೆಯುವುದು (Calling / Invocation) - ಯಂತ್ರದ ಸ್ವಿಚ್ ಆನ್ ಮಾಡುವುದು
sayHello(); // ಔಟ್‌ಪುಟ್: ನಮಸ್ಕಾರ! ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಜಗತ್ತಿಗೆ ಸ್ವಾಗತ.

ಫಂಕ್ಷನ್ ಹಿನ್ನೆಲೆಯಲ್ಲಿ ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ? (Under the Hood)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಫಂಕ್ಷನ್ ಅನ್ನು ಕರೆಯುವಾಗ (Call ಮಾಡಿದಾಗ) ಪ್ರೊಸೆಸರ್ ಮತ್ತು ಮೆಮೊರಿಯಲ್ಲಿ 3 ಮುಖ್ಯ ಹಂತಗಳು ನಡೆಯುತ್ತವೆ:

1
ಮೆಮೊರಿ ಹಂಚಿಕೆ (Memory Allocation):

ನೀವು function name() {} ಎಂದು ಬರೆದ ತಕ್ಷಣ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಎಂಜಿನ್ (ಉದಾ: V8) RAM ನಲ್ಲಿ ಒಂದು ನಿರ್ದಿಷ್ಟ ಜಾಗವನ್ನು ಕಾಯ್ದಿರಿಸುತ್ತದೆ ಮತ್ತು ಸಂಪೂರ್ಣ ಕೋಡ್ ಬ್ಲಾಕ್ ಅನ್ನು ಆ ಮೆಮೊರಿಯಲ್ಲಿ ಸೇವ್ ಮಾಡುತ್ತದೆ.

2
ಕಾಲ್ ಸ್ಟ್ಯಾಕ್ (Call Stack): ಅರ್ಥಮಾಡಿಕೊಳ್ಳಲು ಸರಳ ಉದಾಹರಣೆ

ಉದಾಹರಣೆ: ಮದುವೆ ಮನೆಯಲ್ಲಿ ಪ್ಲೇಟ್‌ಗಳನ್ನು (ತಟ್ಟೆಗಳನ್ನು) ಒಂದರ ಮೇಲೊಂದು ಜೋಡಿಸಿಟ್ಟಿರುತ್ತಾರೆ. ನೀವು ಯಾವ ಪ್ಲೇಟ್ ಅನ್ನು ಮೊದಲು ತೆಗೆಯುತ್ತೀರಿ? ಅತ್ಯಂತ ಮೇಲಿರುವ ಪ್ಲೇಟ್! (Last In, First Out).
ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಕೂಡ ಇದೇ ರೀತಿ ಕೆಲಸ ಮಾಡುತ್ತದೆ. ನೀವು sayHello() ಎಂದು ಕರೆದಾಗ, ಈ ಫಂಕ್ಷನ್ "ಕಾಲ್ ಸ್ಟ್ಯಾಕ್" ಎಂಬ ಸರದಿಯ ಮೇಲ್ಭಾಗಕ್ಕೆ (Top) ಹೋಗಿ ಕೂರುತ್ತದೆ. ಪ್ರೊಸೆಸರ್ ತನ್ನ ಎಲ್ಲಾ ಬೇರೆ ಕೆಲಸ ಬಿಟ್ಟು ಮೊದಲು ಈ ಮೇಲಿರುವ ಫಂಕ್ಷನ್ ಒಳಗೆ ಪ್ರವೇಶಿಸುತ್ತದೆ (ಇದನ್ನೇ Execution Context ಎನ್ನುತ್ತಾರೆ).

3
ಗಾರ್ಬೇಜ್ ಕಲೆಕ್ಷನ್ (Execution & Cleanup):

ಕೋಡ್ ರನ್ ಆಗಿ ಮುಗಿದ ತಕ್ಷಣ, ಆ ಫಂಕ್ಷನ್ ಕಾಲ್ ಸ್ಟ್ಯಾಕ್‌ನಿಂದ ಹೊರಬರುತ್ತದೆ (ಪ್ಲೇಟ್ ತೆಗೆದಂತೆ). ಅದರ ಒಳಗಿದ್ದ ತಾತ್ಕಾಲಿಕ ವೇರಿಯೇಬಲ್‌ಗಳನ್ನು ಮೆಮೊರಿಯಿಂದ ಡಿಲೀಟ್ ಮಾಡಲಾಗುತ್ತದೆ (ಸ್ವಚ್ಛಗೊಳಿಸುವಿಕೆ).

ಪ್ಯಾರಾಮೀಟರ್ ಮತ್ತು ರಿಟರ್ನ್ (Parameters & Return)

ಸರಳ ಉದಾಹರಣೆ: ನೀವು ರೆಸ್ಟೋರೆಂಟ್‌ಗೆ ಹೋಗಿ "ಮಸಾಲ ದೋಸೆ" ಎಂದು ಆರ್ಡರ್ ಮಾಡುತ್ತೀರಿ (ಇದು Parameter / Input). ವೈಟರ್ ಒಳಗೆ ಹೋಗಿ ದೋಸೆ ತಂದು ನಿಮ್ಮ ಟೇಬಲ್ ಮೇಲೆ ಇಡುತ್ತಾನೆ (ಇದು Return / Output).

JavaScript - Parameters & Return 
// a ಮತ್ತು b ಎಂಬುದು Parameters (ಕಚ್ಚಾವಸ್ತು/ಆರ್ಡರ್)
function addNumbers(a, b) {
    let sum = a + b;
    return sum; // ಫಲಿತಾಂಶವನ್ನು ವಾಪಸ್ ಕಳುಹಿಸುತ್ತದೆ (ದೋಸೆ ತಂದುಕೊಟ್ಟಂತೆ)
}

// 10 ಮತ್ತು 5 ಎಂಬುದು Arguments (ನೈಜ ಮೌಲ್ಯಗಳು)
let result = addNumbers(10, 5);
console.log(result); // ಔಟ್‌ಪುಟ್: 15
ಗಮನಿಸಿ (Key Rule):

ಫಂಕ್ಷನ್ ಒಳಗೆ return ಕೀವರ್ಡ್ ನೋಡಿದ ತಕ್ಷಣ ಪ್ರೊಸೆಸರ್ ಆ ಫಂಕ್ಷನ್‌ನಿಂದ ತಕ್ಷಣವೇ ಹೊರಬರುತ್ತದೆ. return ನ ಕೆಳಗೆ ನೀವು ಯಾವುದೇ ಕೋಡ್ ಬರೆದರೂ ಅದು ರನ್ ಆಗುವುದಿಲ್ಲ!

ಮಧ್ಯಮ ಹಂತ (Intermediate Level)

ಫಂಕ್ಷನ್ ಎಕ್ಸ್‌ಪ್ರೆಶನ್ (Function Expression)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಫಂಕ್ಷನ್‌ಗಳನ್ನು ನಾವು ಕೇವಲ ಒಂದು 'ಮೌಲ್ಯ' (Value) ದಂತೆ ನೋಡಬಹುದು. ಅಂದರೆ ಒಂದು ವೇರಿಯೇಬಲ್‌ಗೆ ಫಂಕ್ಷನ್ ಅನ್ನು ಅಸೈನ್ ಮಾಡಬಹುದು.

JavaScript 
const multiply = function(x, y) { // ಇದಕ್ಕೆ ಹೆಸರಿಲ್ಲ (Anonymous Function)
    return x * y;
};

console.log(multiply(4, 3)); // ಔಟ್‌ಪುಟ್: 12

ವೃತ್ತಿಪರ ಹಂತ (Pro Concepts)

1. ಆರೋ ಫಂಕ್ಷನ್ಸ್ (Arrow Functions - ES6)

ಇದು ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಫಂಕ್ಷನ್ ಬರೆಯುವ ಅತ್ಯಂತ ಚಿಕ್ಕ ಮತ್ತು ಸ್ಮಾರ್ಟ್ ವಿಧಾನ. ಇದರಲ್ಲಿ function ಕೀವರ್ಡ್ ಬರೆಯುವ ಅಗತ್ಯವಿಲ್ಲ.

JavaScript - Arrow Function 
// ಸಾಂಪ್ರದಾಯಿಕ ವಿಧಾನ
let squareOld = function(n) { return n * n; };

// ಆರೋ ಫಂಕ್ಷನ್ (Pro Level)
const square = (n) => n * n; 

console.log(square(5)); // ಔಟ್‌ಪುಟ್: 25

2. ಕಾಲ್‌ಬ್ಯಾಕ್ ಫಂಕ್ಷನ್ (Callback Functions)

ಒಂದು ಫಂಕ್ಷನ್ ಅನ್ನು ಮತ್ತೊಂದು ಫಂಕ್ಷನ್‌ಗೆ "ಆರ್ಗ್ಯುಮೆಂಟ್" (Parameter) ಆಗಿ ಕಳುಹಿಸುವುದನ್ನು ಕಾಲ್‌ಬ್ಯಾಕ್ ಎನ್ನುತ್ತಾರೆ. ಇದು ಅಸಿಂಕ್ರೋನಸ್ (Asynchronous) ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ಗೆ ಬಹಳ ಮುಖ್ಯ.

function processUserInput(name, callbackFn) {
    console.log("ಬಳಕೆದಾರ: " + name);
    callbackFn(); // ಇಲ್ಲಿ ಮತ್ತೊಂದು ಫಂಕ್ಷನ್ ಕಾಲ್ ಆಗುತ್ತದೆ
}

function greet() {
    console.log("ಸ್ವಾಗತ!");
}

// greet ಫಂಕ್ಷನ್ ಅನ್ನು ಪಾಸ್ ಮಾಡುತ್ತಿದ್ದೇವೆ (ಕರೆಯುತ್ತಿಲ್ಲ, ಕೇವಲ ಹೆಸರು ಕೊಡುತ್ತಿದ್ದೇವೆ)
processUserInput("ರಾಜು", greet); 
// ಔಟ್‌ಪುಟ್:
// ಬಳಕೆದಾರ: ರಾಜು
// ಸ್ವಾಗತ!
Pro Tip - ಕ್ಲೋಜರ್ಸ್ (Closures):

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಫಂಕ್ಷನ್‌ಗಳು ತಮ್ಮ ಹುಟ್ಟಿದ ಜಾಗವನ್ನು ನೆನಪಿಟ್ಟುಕೊಳ್ಳುತ್ತವೆ! ಒಂದು ಫಂಕ್ಷನ್‌ನ ಒಳಗಿರುವ ಮತ್ತೊಂದು ಫಂಕ್ಷನ್, ತಾನು ಎಕ್ಸಿಕ್ಯೂಟ್ ಆದ ಮೇಲೂ ತನ್ನ ಪೇರೆಂಟ್ (Parent) ಫಂಕ್ಷನ್‌ನ ವೇರಿಯೇಬಲ್‌ಗಳನ್ನು ಮೆಮೊರಿಯಲ್ಲಿ ಭದ್ರವಾಗಿ ಇಟ್ಟುಕೊಳ್ಳುವ ಶಕ್ತಿಯನ್ನು Closure ಎನ್ನುತ್ತಾರೆ. ಇದು ಡೇಟಾ ಪ್ರೈವೆಸಿ (Data Privacy) ಕಾಪಾಡಲು ವೃತ್ತಿಪರರು ಬಳಸುವ ಪ್ರಮುಖ ತಂತ್ರ.

3. IIFE ಫಂಕ್ಷನ್ (ಹೊಸ ಸೇರ್ಪಡೆ)

IIFE (Immediately Invoked Function Expression): ಇದರ ಹೆಸರೇ ಹೇಳುವಂತೆ ಇದು "ತಕ್ಷಣವೇ ಕಾರ್ಯಗತಗೊಳ್ಳುವ (Invoke ಆಗುವ) ಫಂಕ್ಷನ್".

ಸಾಮಾನ್ಯವಾಗಿ ಫಂಕ್ಷನ್ ಬರೆದ ಮೇಲೆ ಅದನ್ನು ನಾವು ಬೇರೆ ಕಡೆ ಕರೆಯಬೇಕು (Call ಮಾಡಬೇಕು). ಆದರೆ, IIFE ಅನ್ನು ಬರೆದ ತಕ್ಷಣವೇ ಅದೇ ತಾನೇ ತಾನಾಗಿ ರನ್ ಆಗುತ್ತದೆ.

JavaScript - IIFE Syntax 
// ಸಾಧಾರಣ ಫಂಕ್ಷನ್ ಅನ್ನು ಬ್ರಾಕೆಟ್ ( ) ಒಳಗೆ ಹಾಕಿ, ಕೊನೆಯಲ್ಲಿ () ಸೇರಿಸಿದರೆ ಅದು IIFE ಆಗುತ್ತದೆ.

(function() {
    let secretMessage = "ಇದು ಗುಪ್ತ ಸಂದೇಶ!";
    console.log("ನಾನು ಹುಟ್ಟಿದ ತಕ್ಷಣ ರನ್ ಆದೆ!");
})(); // <-- ಈ ಕೊನೆಯ () ಫಂಕ್ಷನ್ ಅನ್ನು ತಕ್ಷಣ ಕಾಲ್ ಮಾಡುತ್ತದೆ!

// ಔಟ್‌ಪುಟ್: ನಾನು ಹುಟ್ಟಿದ ತಕ್ಷಣ ರನ್ ಆದೆ!
IIFE ಅನ್ನು ಏಕೆ ಬಳಸುತ್ತಾರೆ? (Why use IIFE?)
  • ಗ್ಲೋಬಲ್ ಮೆಮೊರಿ ಮಾಲಿನ್ಯ ತಪ್ಪಿಸಲು (Avoid Global Scope Pollution): ನೀವು ಸಾಮಾನ್ಯ ಫಂಕ್ಷನ್ ಅಥವಾ ವೇರಿಯೇಬಲ್ ಬರೆದರೆ ಅದು ಇಡೀ ವೆಬ್‌ಪೇಜ್‌ನ (Global) ಮೆಮೊರಿಯಲ್ಲಿ ಕೂರುತ್ತದೆ. ಒಂದೇ ಹೆಸರಿನ 2 ವೇರಿಯೇಬಲ್ ಇದ್ದರೆ ಬ್ರೌಸರ್ ಕ್ರ್ಯಾಶ್ ಅಥವಾ ಎರರ್ ಬರಬಹುದು.
  • IIFE ಒಳಗೆ ಬರೆದ ವೇರಿಯೇಬಲ್‌ಗಳು (ಉದಾ: secretMessage) ಆ ಫಂಕ್ಷನ್‌ನಿಂದ ಆಚೆ ಯಾರಿಗೂ ಸಿಗುವುದಿಲ್ಲ. ರನ್ ಆದ ತಕ್ಷಣ ಮೆಮೊರಿಯಿಂದ ಅಳಿಸಿಹೋಗುತ್ತದೆ. ಇದು Data Privacy ಕಾಪಾಡಲು ವೃತ್ತಿಪರರು (Pro Developers) ಬಳಸುವ ಅತ್ಯುತ್ತಮ ತಂತ್ರ.
