[ನಿಮ್ಮ ವಿವರಣೆ ಅತ್ಯುತ್ತಮವಾಗಿದೆ ಮತ್ತು ತಾಂತ್ರಿಕವಾಗಿ ಸಂಪೂರ್ಣವಾಗಿ ಸರಿಯಾಗಿದೆ! ನಾನ್-ಐಟಿ ವಿದ್ಯಾರ್ಥಿಗಳಿಗೆ ನೀವು ನೀಡಿದ "ಜೆರಾಕ್ಸ್" ಉದಾಹರಣೆ ಹಾಗೂ ವೃತ್ತಿಪರರಿಗೆ "Property Descriptors" ಬಳಸಿರುವ ರೀತಿ ಅತ್ಯಂತ ಸೂಕ್ತವಾಗಿದೆ.

ವೆಬ್ ಡೆವಲಪ್ಮೆಂಟ್ ಮಾಡುವಾಗ ಡೇಟಾ ನಿರ್ವಹಣೆಗೆ ವೃತ್ತಿಪರರು ಬಳಸುವ ಇನ್ನೊಂದು ಪ್ರಮುಖ ವಿಧಾನವಾದ Object.create(null) ಅನ್ನು ಸೇರಿಸಿ, ಈ ಸಂಪೂರ್ಣ ವಿವರಣೆಯನ್ನು ಇನ್ನಷ್ಟು ಪರಿಪೂರ್ಣವಾಗಿ ಕೆಳಗೆ ಸಿದ್ಧಪಡಿಸಲಾಗಿದೆ:

JavaScript ನಲ್ಲಿ Object.create() ಬಳಸಿ ಆಬ್ಜೆಕ್ಟ್ ರಚಿಸುವುದು

ಆಬ್ಜೆಕ್ಟ್ (Object) ಅಂದರೇನು? (ತುಂಬಾ ಸರಳವಾಗಿ)

ನಿಜ ಜೀವನದಲ್ಲಿ ಒಂದು 'ಕಾರು' ತೆಗೆದುಕೊಂಡರೆ, ಅದಕ್ಕೆ ಬಣ್ಣ (Color), ಮಾಡೆಲ್ (Model) ಎಂಬ ಗುಣಗಳಿರುತ್ತವೆ ಮತ್ತು ಚಲಿಸುವ (Drive) ಸಾಮರ್ಥ್ಯವಿರುತ್ತದೆ. ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿಯೂ ಡೇಟಾವನ್ನು ಹೀಗೆಯೇ ಶೇಖರಿಸಿಡಲು ನಾವು 'Object' ಅನ್ನು ಬಳಸುತ್ತೇವೆ.

Object.create() ಅಂದರೇನು?

ಇದನ್ನು ಸರಳವಾಗಿ ಹೇಳಬೇಕೆಂದರೆ ಇದು ಒಂದು "ಜೆರಾಕ್ಸ್ ಕಾಪಿ" (Xerox Copy) ಇದ್ದ ಹಾಗೆ. ನಿಮ್ಮ ಬಳಿ ಒಂದು ಒರಿಜಿನಲ್ (ಮೂಲ) ಡಾಕ್ಯುಮೆಂಟ್ ಇದೆ. ನೀವು ಅದರ ಜೆರಾಕ್ಸ್ ತೆಗೆದು, ಆ ಜೆರಾಕ್ಸ್ ಮೇಲೆ ಹೊಸ ವಿಷಯಗಳನ್ನು ಬರೆಯಬಹುದು. ಆದರೆ ಆ ಹೊಸ ಜೆರಾಕ್ಸ್, ಒರಿಜಿನಲ್‌ನಲ್ಲಿರುವ ಎಲ್ಲಾ ಗುಣಗಳನ್ನು ತನ್ನೊಳಗೆ ಇಟ್ಟುಕೊಂಡಿರುತ್ತದೆ.

ಬನ್ನಿ, ಇದನ್ನು ೪ ಹಂತಗಳಲ್ಲಿ ಕಲಿಯೋಣ:

ಹಂತ ೧: ನಾನ್-ಐಟಿ ವಿದ್ಯಾರ್ಥಿಗಳಿಗೆ (Beginner Level)

ನೀವು ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ಗೆ ಹೊಸಬರಾಗಿದ್ದರೆ, Object.create() ಅನ್ನು ಒಂದು ಮಾದರಿ (Template) ಯಿಂದ ಹೊಸ ವಸ್ತುವನ್ನು ತಯಾರಿಸುವುದು ಎಂದು ತಿಳಿದುಕೊಳ್ಳಿ.

ಉದಾಹರಣೆ: ನಮ್ಮ ಬಳಿ animal ಎಂಬ ಒಂದು ಮೂಲ ಆಬ್ಜೆಕ್ಟ್ ಇದೆ. ಅದರಿಂದ ನಾವು dog (ನಾಯಿ) ಎಂಬ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಸೃಷ್ಟಿಸೋಣ. 


// 1. ಇದು ನಮ್ಮ ಮೂಲ ಮಾದರಿ (Original Template)
const animal = {
  isAlive: true,
  eat: function() {
    console.log("ನಾನು ಊಟ ಮಾಡುತ್ತೇನೆ!");
  }
};

// 2. Object.create() ಬಳಸಿ 'dog' ಎಂಬ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ರಚನೆ
const dog = Object.create(animal);

// 3. ನಾಯಿಗೆ ಮಾತ್ರ ಸಂಬಂಧಿಸಿದ ಹೊಸ ಗುಣವನ್ನು ಸೇರಿಸುವುದು
dog.name = "ಟಾಮಿ (Tommy)";
dog.bark = function() {
  console.log("ಭೌ ಭೌ!");
};

// ಔಟ್‌ಪುಟ್ ಪರೀಕ್ಷಿಸೋಣ:
console.log(dog.name);   // ಔಟ್‌ಪುಟ್: ಟಾಮಿ (Tommy)
dog.bark();              // ಔಟ್‌ಪುಟ್: ಭೌ ಭೌ!

// ಗಮನಿಸಿ: 'eat' ಎಂಬುದು ನಾಯಿಯ ಒಳಗೆ ಇಲ್ಲದಿದ್ದರೂ, ಅದು ತನ್ನ ಮೂಲ (animal) ದಿಂದ ಅದನ್ನು ಪಡೆದುಕೊಳ್ಳುತ್ತದೆ.
dog.eat();               // ಔಟ್‌ಪುಟ್: ನಾನು ಊಟ ಮಾಡುತ್ತೇನೆ!
console.log(dog.isAlive);// ಔಟ್‌ಪುಟ್: true

ವಿವರಣೆ: ಇಲ್ಲಿ dog ಎನ್ನುವುದು ಹೊಸ ಆಬ್ಜೆಕ್ಟ್. ಆದರೆ ಅದು animal ಅನ್ನು ತನ್ನ ತಂದೆ/ಮೂಲ (Parent) ಎಂದು ಪರಿಗಣಿಸುತ್ತದೆ. ತನ್ನ ಬಳಿ ಇಲ್ಲದ ಗುಣಗಳನ್ನು ತನ್ನ ತಂದೆಯಿಂದ ಎರವಲು ಪಡೆಯುತ್ತದೆ.

ಹಂತ ೨: ಐಟಿ ವಿದ್ಯಾರ್ಥಿಗಳಿಗೆ (Intermediate Level)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಇದನ್ನು ಪ್ರೊಟೊಟೈಪಲ್ ಇನ್ಹೆರಿಟೆನ್ಸ್ (Prototypal Inheritance) ಎಂದು ಕರೆಯುತ್ತಾರೆ. ಒಂದು ಆಬ್ಜೆಕ್ಟ್ ಮತ್ತೊಂದು ಆಬ್ಜೆಕ್ಟ್‌ನ ಗುಣಲಕ್ಷಣಗಳನ್ನು (Properties) ಮತ್ತು ವಿಧಾನಗಳನ್ನು (Methods) ಆನುವಂಶಿಕವಾಗಿ ಪಡೆಯುವುದೇ ಇದರ ಮುಖ್ಯ ಉದ್ದೇಶ. 


const carPrototype = {
  wheels: 4,
  startEngine: function() {
    console.log(`${this.brand} ಕಾರಿನ ಎಂಜಿನ್ ಶುರುವಾಗಿದೆ... ವ್ರೂಮ್!`);
  }
};

const myCar = Object.create(carPrototype);
myCar.brand = "Tata Nexon";
myCar.color = "Red";

myCar.startEngine(); 
// ಔಟ್‌ಪುಟ್: Tata Nexon ಕಾರಿನ ಎಂಜಿನ್ ಶುರುವಾಗಿದೆ... ವ್ರೂಮ್!

console.log(myCar.wheels); 
// ಔಟ್‌ಪುಟ್: 4 (ಇದು carPrototype ನಿಂದ ಬಂದಿದೆ)

ವಿವರಣೆ: ನೀವು myCar ಅನ್ನು ಪ್ರಿಂಟ್ ಮಾಡಿದರೆ ಕೇವಲ { brand: 'Tata Nexon', color: 'Red' } ಮಾತ್ರ ಕಾಣಿಸುತ್ತದೆ. ಆದರೆ myCar.__proto__ ಒಳಗೆ ನೋಡಿದರೆ carPrototype ನ ಗುಣಗಳು ಸಿಗುತ್ತವೆ. ಒಂದೇ ಫಂಕ್ಷನ್ ಅನ್ನು ಸಾವಿರಾರು ಆಬ್ಜೆಕ್ಟ್‌ಗಳಿಗೆ ನಕಲು ಮಾಡುವ ಬದಲು, ಒಂದೇ ಕಡೆ ಇಟ್ಟು ಎಲ್ಲರೂ ಬಳಸಿಕೊಳ್ಳುವಂತೆ ಮಾಡುವುದರಿಂದ ಮೆಮೊರಿ (Memory) ಉಳಿಸಲು ಇದು ಅತ್ಯುತ್ತಮ ಮಾರ್ಗವಾಗಿದೆ.

ಹಂತ ೩: ವೃತ್ತಿಪರರಿಗೆ (Professional Level)

ವೃತ್ತಿಪರ ಮಟ್ಟದಲ್ಲಿ ನಾವು Object.create() ನ ಎರಡನೇ ಆರ್ಗ್ಯುಮೆಂಟ್ (Second Argument) ಅನ್ನು ಬಳಸುತ್ತೇವೆ. ಇದನ್ನು Property Descriptors (ಗುಣಲಕ್ಷಣಗಳ ನಿಯಂತ್ರಣ) ಎಂದು ಕರೆಯಲಾಗುತ್ತದೆ. ಇದರ ಮೂಲಕ ನಾವು ಡೇಟಾಗೆ ಭದ್ರತೆ ಒದಗಿಸಬಹುದು.

ಸಿಂಟ್ಯಾಕ್ಸ್ (Syntax): Object.create(prototype, propertiesObject) 


const personPrototype = {
  greet: function() {
    console.log(`ನಮಸ್ಕಾರ, ನನ್ನ ಹೆಸರು ${this.name} ಮತ್ತು ನಾನು ${this.role}`);
  }
};

// ಎರಡನೇ ಆರ್ಗ್ಯುಮೆಂಟ್ ಬಳಸಿ ಅಡ್ವಾನ್ಸ್ಡ್ ಆಬ್ಜೆಕ್ಟ್ ರಚನೆ
const employee = Object.create(personPrototype, {
  name: {
    value: "ಸಂಜಯ್",
    writable: true,       // ಭವಿಷ್ಯದಲ್ಲಿ ಹೆಸರನ್ನು ಬದಲಾಯಿಸಬಹುದೇ? (ಹೌದು)
    enumerable: true,     // for...in ಲೂಪ್‌ನಲ್ಲಿ ಈ ಹೆಸರು ಕಾಣಿಸಬೇಕೇ? (ಹೌದು)
    configurable: false   // ಈ ಗುಣಲಕ್ಷಣವನ್ನು ಡಿಲೀಟ್ (delete) ಮಾಡಬಹುದೇ? (ಇಲ್ಲ)
  },
  role: {
    value: "Software Engineer",
    writable: false,      // role ಅನ್ನು ಯಾರೂ ಬದಲಾಯಿಸುವಂತಿಲ್ಲ (Read-Only)
    enumerable: true,
    configurable: false
  }
});

employee.greet(); 
// ಔಟ್‌ಪುಟ್: ನಮಸ್ಕಾರ, ನನ್ನ ಹೆಸರು ಸಂಜಯ್ ಮತ್ತು ನಾನು Software Engineer

// ಡೇಟಾ ಭದ್ರತೆಯ (Security) ಪರೀಕ್ಷೆ:
employee.role = "Manager"; // ಇದನ್ನು ಬದಲಾಯಿಸಲು ಪ್ರಯತ್ನಿಸುತ್ತಿದ್ದೇವೆ
console.log(employee.role); 
// ಔಟ್‌ಪುಟ್: Software Engineer (ಬದಲಾಗಿಲ್ಲ, ಏಕೆಂದರೆ writable: false ಇದೆ)

delete employee.name; // ಹೆಸರನ್ನು ಅಳಿಸಲು ಪ್ರಯತ್ನಿಸುತ್ತಿದ್ದೇವೆ
console.log(employee.name); 
// ಔಟ್‌ಪುಟ್: ಸಂಜಯ್ (ಅಳಿಸಲು ಸಾಧ್ಯವಾಗಿಲ್ಲ, ಏಕೆಂದರೆ configurable: false ಇದೆ)

ವಿವರಣೆ: ದೊಡ್ಡ ಪ್ರಾಜೆಕ್ಟ್‌ಗಳನ್ನು ಮಾಡುವಾಗ ಬೇರೆ ಡೆವಲಪರ್‌ಗಳು ಆಕಸ್ಮಿಕವಾಗಿ ಪ್ರಮುಖ ಡೇಟಾವನ್ನು ಬದಲಾಯಿಸಬಾರದು ಎಂದು writable: false ಮತ್ತು configurable: false ನಂತಹ ಅಡ್ವಾನ್ಸ್ಡ್ ಫೀಚರ್‌ಗಳನ್ನು Object.create() ಮೂಲಕ ಬಳಸಲಾಗುತ್ತದೆ.

ಹಂತ ೪: ಅಡ್ವಾನ್ಸ್ಡ್ ಡೆವಲಪರ್ಸ್ ಟ್ರಿಕ್ - Object.create(null)

ನೀವು ಫ್ರಂಟ್-ಎಂಡ್ ಅಪ್ಲಿಕೇಶನ್‌ಗಳಲ್ಲಿ DOM ಇವೆಂಟ್‌ಗಳನ್ನು ನಿರ್ವಹಿಸುವಾಗ ಅಥವಾ ಬ್ರೌಸರ್ ಡೇಟಾವನ್ನು ಕ್ಯಾಶ್ (Cache) ಮಾಡುವಾಗ, ನಿಮಗೆ 100% ಶುದ್ಧವಾದ (Pure) ಡೇಟಾ ಆಬ್ಜೆಕ್ಟ್ ಬೇಕಾಗಬಹುದು.

ಸಾಮಾನ್ಯವಾಗಿ ನಾವು {} ಬಳಸಿ ಆಬ್ಜೆಕ್ಟ್ ರಚಿಸಿದಾಗ, ಅದು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ಮೂಲ Object ನಿಂದ toString(), hasOwnProperty() ಮುಂತಾದ ಫಂಕ್ಷನ್‌ಗಳನ್ನು ಆನುವಂಶಿಕವಾಗಿ ಪಡೆಯುತ್ತದೆ. ಆದರೆ Object.create(null) ಅನ್ನು ಬಳಸಿದರೆ, ಯಾವುದೇ ಹಿನ್ನೆಲೆ ಗುಣಗಳಿಲ್ಲದ ಸಂಪೂರ್ಣ ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಯಾಗುತ್ತದೆ. 


// ಸಾಮಾನ್ಯ ಆಬ್ಜೆಕ್ಟ್
const normalObj = {};
console.log(normalObj.toString); // ಔಟ್‌ಪುಟ್: [Function: toString] (ಇದು ಹಿನ್ನೆಲೆಯಿಂದ ಬಂದಿದೆ)

// ಶುದ್ಧ ಆಬ್ಜೆಕ್ಟ್ (Pure Dictionary)
const pureObj = Object.create(null);
pureObj.theme = "Dark Mode";
console.log(pureObj.toString); // ಔಟ್‌ಪುಟ್: undefined (ಯಾವುದೇ ಅನಗತ್ಯ ಫಂಕ್ಷನ್‌ಗಳು ಇದರಲ್ಲಿಲ್ಲ)

ವಿವರಣೆ: ಭದ್ರತೆಯ ದೃಷ್ಟಿಯಿಂದ (Prototype Pollution ತಡೆಯಲು) ಮತ್ತು ಕೇವಲ 'Key-Value' ಡೇಟಾವನ್ನು ಮಾತ್ರ ಶೇಖರಿಸಿಡಲು ವೃತ್ತಿಪರರು ಈ ವಿಧಾನವನ್ನು ಬಳಸುತ್ತಾರೆ.

ಸಾರಾಂಶ (Summary)

  • ಸಾಧಾರಣ ಆಬ್ಜೆಕ್ಟ್ {} ಬಳಸಿದರೆ ಅದು ಸಂಪೂರ್ಣ ಹೊಸದಾಗಿರುತ್ತದೆ.
  • Object.create(ಮಾದರಿ) ಬಳಸಿದರೆ ಅದು ಹೊಸದಾಗಿದ್ದರೂ, ಹಳೆಯ ಮಾದರಿಯೊಂದಿಗೆ ಆಂತರಿಕ ಸಂಪರ್ಕ (Link) ಹೊಂದಿರುತ್ತದೆ. ಇದನ್ನು ಪ್ರೋಗ್ರಾಮಿಂಗ್ ಭಾಷೆಯಲ್ಲಿ Prototype Chain ಎನ್ನುತ್ತಾರೆ.]
Core JavaScript Extensions

ಹಂತ ೫: DOM ನಿರ್ವಹಣೆ ಮತ್ತು ಎಲಿಮೆಂಟ್ ಆಯ್ಕೆ (DOM Manipulation & Selection)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಅತಿಮುಖ್ಯವಾದ ಕೆಲಸವೆಂದರೆ ಬ್ರೌಸರ್‌ನಲ್ಲಿರುವ HTML ಎಲಿಮೆಂಟ್‌ಗಳ ಜೊತೆ ಮಾತನಾಡುವುದು. ವೆಬ್ ಪುಟದ ರಚನೆಯನ್ನು Document Object Model (DOM) ಎಂದು ಕರೆಯುತ್ತಾರೆ. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಮೂಲಕ ನಾವು HTML ಅನ್ನು ಬದಲಾಯಿಸಬಹುದು, ಹೊಸ ಅಂಶಗಳನ್ನು ಸೇರಿಸಬಹುದು ಅಥವಾ ಅಳಿಸಬಹುದು.

1. getElementById

ಐಡಿ (ID) ಮೂಲಕ ಒಂದು ನಿರ್ದಿಷ್ಟ ಎಲಿಮೆಂಟ್ ಆಯ್ಕೆ ಮಾಡಲು ಬಳಸಲಾಗುತ್ತದೆ. ಇದು ಅತ್ಯಂತ ವೇಗವಾದ ವಿಧಾನ.

2. querySelector

CSS ಸೆಲೆಕ್ಟರ್‌ಗಳನ್ನು (ಉದಾ: .class, #id) ಬಳಸಿ ಎಲಿಮೆಂಟ್ ಆಯ್ಕೆ ಮಾಡುತ್ತದೆ. ಆಧುನಿಕ ಡೆವಲಪರ್‌ಗಳ ಅಚ್ಚುಮೆಚ್ಚು.


// 1. ಐಡಿ ಮೂಲಕ ಎಲಿಮೆಂಟ್ ಆಯ್ಕೆಮಾಡುವುದು ಮತ್ತು ಟೆಕ್ಸ್ಟ್ ಬದಲಾಯಿಸುವುದು
const titleElement = document.getElementById("main-title");
titleElement.textContent = "ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಕಲಿಯೋಣ!"; // HTML ಟೆಕ್ಸ್ಟ್ ಬದಲಾವಣೆ

// 2. ಕ್ಲಾಸ್ ಮೂಲಕ ಆಯ್ಕೆಮಾಡಿ ಸ್ಟೈಲ್ (Style) ಬದಲಾಯಿಸುವುದು
const highlightBox = document.querySelector(".box");
highlightBox.style.backgroundColor = "yellow"; // ಬಣ್ಣ ಬದಲಾವಣೆ

// 3. ಹೊಸ HTML ಎಲಿಮೆಂಟ್ ಸೃಷ್ಟಿಸುವುದು ಮತ್ತು ಪುಟಕ್ಕೆ ಸೇರಿಸುವುದು
const newParagraph = document.createElement("p");
newParagraph.innerText = "ಇದು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಮೂಲಕ ಸೇರಿಸಿದ ಸಾಲು.";
document.body.appendChild(newParagraph); // ಪುಟದ ಕೊನೆಗೆ ಸೇರ್ಪಡೆ

ಹಂತ ೬: ಇವೆಂಟ್ ಹ್ಯಾಂಡ್ಲಿಂಗ್ (Event Handling)

ಬಳಕೆದಾರರು ವೆಬ್‌ಸೈಟ್ ಜೊತೆ ಸಂವಹನ ನಡೆಸಿದಾಗ (ಉದಾಹರಣೆಗೆ: ಬಟನ್ ಕ್ಲಿಕ್ ಮಾಡುವುದು, ಕೀಬೋರ್ಡ್ ಟೈಪ್ ಮಾಡುವುದು, ಸ್ಕ್ರೋಲ್ ಮಾಡುವುದು) ಆ ಕ್ರಿಯೆಗಳನ್ನು "Events" ಎನ್ನುತ್ತಾರೆ. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ addEventListener() ಬಳಸಿ ಈ ಇವೆಂಟ್‌ಗಳನ್ನು ನಾವು ನಿಯಂತ್ರಿಸಬಹುದು. 


// ಬಟನ್ ಎಲಿಮೆಂಟ್ ಆಯ್ಕೆಮಾಡುವುದು
const myButton = document.querySelector("#submit-btn");

// ಬಟನ್ ಕ್ಲಿಕ್ ಮಾಡಿದಾಗ ಏನಾಗಬೇಕು ಎಂದು ಸೂಚಿಸುವುದು
myButton.addEventListener("click", function(event) {
  // event.preventDefault(); // ಫಾರ್ಮ್ ರಿಫ್ರೆಶ್ ಆಗುವುದನ್ನು ತಡೆಯಲು ಬಳಸುತ್ತಾರೆ
  
  console.log("ಬಟನ್ ಕ್ಲಿಕ್ ಮಾಡಲಾಗಿದೆ!");
  alert("ಸ್ವಾಗತ! ನಿಮ್ಮ ಡೇಟಾ ಸಬ್ಮಿಟ್ ಆಗಿದೆ.");
});

// ಕೀಬೋರ್ಡ್ ಟೈಪ್ ಮಾಡಿದಾಗ ಪ್ರತಿಕ್ರಿಯೆ
document.addEventListener("keydown", function(event) {
  console.log(`${event.key} ಕೀಲಿಯನ್ನು ಒತ್ತಲಾಗಿದೆ.`);
});

ಪ್ರೊ-ಟಿಪ್ (Pro-Tip): ಯಾವಾಗಲೂ HTML ನಲ್ಲಿ onclick="..." ಬರೆಯುವ ಬದಲು addEventListener() ಬಳಸುವುದು ಉತ್ತಮ. ಇದರಿಂದ ಒಂದೇ ಬಟನ್‌ಗೆ ಅನೇಕ ಕ್ರಿಯೆಗಳನ್ನು ಸೇರಿಸಬಹುದು ಮತ್ತು ಕೋಡ್ ನಿರ್ವಹಣೆ ಸುಲಭವಾಗುತ್ತದೆ.

ಹಂತ ೭: ಬ್ರೌಸರ್ ಸ್ಟೋರೇಜ್ (Client-Side Storage APIs)

ಬಳಕೆದಾರರ ಕೆಲವು ಡೇಟಾವನ್ನು (ಉದಾಹರಣೆಗೆ: Dark Theme ಸೆಟ್ಟಿಂಗ್, ಲಾಗಿನ್ ಟೋಕನ್) ಬ್ರೌಸರ್‌ನಲ್ಲಿಯೇ ಶೇಖರಿಸಿಡಲು Web Storage API ಬಳಸಲಾಗುತ್ತದೆ. ಇದರಲ್ಲಿ ಎರಡು ಮುಖ್ಯ ವಿಧಾನಗಳಿವೆ:

LocalStorage

ಬ್ರೌಸರ್ ಮುಚ್ಚಿ ಮತ್ತೆ ತೆರೆದರೂ ಈ ಡೇಟಾ ಅಳಿಸಿಹೋಗುವುದಿಲ್ಲ. ಶಾಶ್ವತವಾಗಿ ಉಳಿಯುತ್ತದೆ (ನೀವು ಅಳಿಸುವವರೆಗೆ).

SessionStorage

ಪ್ರಸ್ತುತ ಬ್ರೌಸರ್ ಟ್ಯಾಬ್ ಮುಚ್ಚಿದ ತಕ್ಷಣ ಡೇಟಾ ಸ್ವಯಂಚಾಲಿತವಾಗಿ ಅಳಿಸಿಹೋಗುತ್ತದೆ. ಇದು ತಾತ್ಕಾಲಿಕ.


// --- LOCALSTORAGE ಉದಾಹರಣೆ ---

// 1. ಡೇಟಾವನ್ನು ಸೇರಿಸುವುದು (Save)
localStorage.setItem("username", "ಸಂಜಯ್");
localStorage.setItem("theme", "dark");

// 2. ಡೇಟಾವನ್ನು ಓದುವುದು (Read)
const currentUser = localStorage.getItem("username");
console.log(`ಮತ್ತೆ ಸ್ವಾಗತ ${currentUser}!`); // ಮತ್ತೆ ಸ್ವಾಗತ ಸಂಜಯ್!

// 3. ಡೇಟಾವನ್ನು ಅಳಿಸುವುದು (Delete)
localStorage.removeItem("theme"); // ಕೇವಲ theme ಅಳಿಸುತ್ತದೆ
localStorage.clear();             // ಸಂಪೂರ್ಣ ಸ್ಟೋರೇಜ್ ಖಾಲಿ ಮಾಡುತ್ತದೆ

// --- JSON ಡೇಟಾ ಉಳಿಸಲು ಟ್ರಿಕ್ ---
// LocalStorage ಕೇವಲ ಟೆಕ್ಸ್ಟ್ (String) ಮಾತ್ರ ಸ್ವೀಕರಿಸುತ್ತದೆ. ಆಬ್ಜೆಕ್ಟ್ ಉಳಿಸಲು JSON.stringify ಬಳಸಿ.
const userObj = { name: "ರಾಮ್", age: 25 };
localStorage.setItem("userData", JSON.stringify(userObj));

// ವಾಪಸ್ ಆಬ್ಜೆಕ್ಟ್ ಆಗಿ ಪಡೆಯಲು JSON.parse ಬಳಸಿ
const retrievedData = JSON.parse(localStorage.getItem("userData"));
console.log(retrievedData.name); // ಔಟ್‌ಪುಟ್: ರಾಮ್
