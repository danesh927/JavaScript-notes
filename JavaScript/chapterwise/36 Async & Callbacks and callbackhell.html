JavaScript Masterclass: Async & Callbacks

ಪರಿವಿಡಿ (Contents)

ಅಸಿಂಕ್ರೋನಸ್ ಕಾಲ್ ಬ್ಯಾಕ್ 1. ಮಕ್ಕಳಿಗಾಗಿ ಸುಲಭವಾದ ಕಥೆ 2. ಕಾಲ್ ಬ್ಯಾಕ್ (Callback) ಎಂದರೇನು? 3. ಅಸಿಂಕ್ರೋನಸ್ ಕಾಲ್ ಬ್ಯಾಕ್ 4. ವೃತ್ತಿಪರ ಮಟ್ಟ (Error-First) ಕಾಲ್ ಬ್ಯಾಕ್‌ನ ತೊಂದರೆ (Callback Hell) Callback Hell ಉದಾಹರಣೆ ಕಾಲ್‌ಬ್ಯಾಕ್ ನರಕದಿಂದ ತಪ್ಪಿಸಿಕೊಳ್ಳುವುದು ಹೇಗೆ?

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಅಸಿಂಕ್ರೋನಸ್ ಕಾಲ್ ಬ್ಯಾಕ್ (Asynchronous Callbacks) ಎಂಬುದು ಬಹಳ ಮುಖ್ಯವಾದ ವಿಷಯ. ಇದನ್ನು ಸಣ್ಣ ಮಕ್ಕಳಿಂದ ಹಿಡಿದು ವೃತ್ತಿಪರರವರೆಗೆ (Professional) ಎಲ್ಲರಿಗೂ ಅರ್ಥವಾಗುವಂತೆ ಹಂತ ಹಂತವಾಗಿ ಕಲಿಯೋಣ.

1. ಮಕ್ಕಳಿಗಾಗಿ ಒಂದು ಸುಲಭವಾದ ಕಥೆ (ತಳಮಟ್ಟದ ವಿವರಣೆ)

ನೀವು ಒಂದು ಹೋಟೆಲ್‌ನಲ್ಲಿ ಪಿಜ್ಜಾ ಆರ್ಡರ್ ಮಾಡುತ್ತಿದ್ದೀರಾ ಎಂದು ಕಲ್ಪಿಸಿಕೊಳ್ಳಿ.

  • 1. ನೀವು ಕೌಂಟರ್‌ಗೆ ಹೋಗಿ, "ಒಂದು ಪಿಜ್ಜಾ ಕೊಡಿ" ಎಂದು ಹಣ ಕಟ್ಟುತ್ತೀರಿ.
  • 2.
  • 3. ಪಿಜ್ಜಾ ತಯಾರಾಗಲು 15 ನಿಮಿಷ ಬೇಕು.
  • 4.
  • 5. ಆ 15 ನಿಮಿಷ ನೀವು ಕೌಂಟರ್ ಮುಂದೆಯೇ, ಬೇರೆ ಯಾವ ಕೆಲಸವನ್ನೂ ಮಾಡದೆ ಸುಮ್ಮನೆ ನಿಂತುಕೊಳ್ಳುತ್ತೀರಾ? ಇಲ್ಲ!
  • 6.
  • 7. ಬದಲಾಗಿ, ಹೋಟೆಲ್‌ನವರು ನಿಮಗೆ ಒಂದು ಟೋಕನ್ (Token) ಕೊಡುತ್ತಾರೆ ಮತ್ತು "ಪಿಜ್ಜಾ ರೆಡಿಯಾದ ಮೇಲೆ ನಾವು ನಿಮ್ಮನ್ನು ಕರೆಯುತ್ತೇವೆ (Call back)" ಎಂದು ಹೇಳುತ್ತಾರೆ.
  • 8.
  • 9. ಆ ಸಮಯದಲ್ಲಿ ನೀವು ಕುಳಿತುಕೊಂಡು ಮೊಬೈಲ್‌ನಲ್ಲಿ ಗೇಮ್ ಆಡಬಹುದು ಅಥವಾ ಸ್ನೇಹಿತರ ಜೊತೆ ಮಾತನಾಡಬಹುದು.
  • 10.
  • 11. ಪಿಜ್ಜಾ ತಯಾರಾದ ತಕ್ಷಣ ಹೋಟೆಲ್‌ನವರು ನಿಮ್ಮನ್ನು ಕರೆಯುತ್ತಾರೆ. ಆಗ ನೀವು ಹೋಗಿ ಪಿಜ್ಜಾ ತರುತ್ತೀರಿ.
  • 12.

ಇದೇ ಅಸಿಂಕ್ರೋನಸ್ ಕಾಲ್ ಬ್ಯಾಕ್!

  • ಅಸಿಂಕ್ರೋನಸ್: ಒಂದು ಕೆಲಸ ಮುಗಿಯುವವರೆಗೆ ಕಾಯದೆ, ಬೇರೆ ಕೆಲಸ ಮಾಡುವುದು (ಗೇಮ್ ಆಡುವುದು).
  • ಕಾಲ್ ಬ್ಯಾಕ್: ಮೊದಲನೇ ಕೆಲಸ ಮುಗಿದ ತಕ್ಷಣ, ನಮಗೆ ವಾಪಸ್ ಬಂದು ವಿಷಯ ತಿಳಿಸುವುದು (ಪಿಜ್ಜಾ ರೆಡಿ ಎಂದು ಕರೆಯುವುದು).

2. ಕಾಲ್ ಬ್ಯಾಕ್ (Callback) ಎಂದರೇನು?

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಫಂಕ್ಷನ್ (Function) ಎಂದರೆ ಒಂದು ನಿರ್ದಿಷ್ಟ ಕೆಲಸ ಮಾಡುವ ಕೋಡ್. ಒಂದು ಫಂಕ್ಷನ್ ಅನ್ನು, ಇನ್ನೊಂದು ಫಂಕ್ಷನ್‌ಗೆ ಇನ್‌ಪುಟ್ (Input) ಆಗಿ ಕಳುಹಿಸಿದರೆ, ಆ ಕಳುಹಿಸಿದ ಫಂಕ್ಷನ್ ಅನ್ನು 'ಕಾಲ್ ಬ್ಯಾಕ್ ಫಂಕ್ಷನ್' ಎನ್ನುತ್ತಾರೆ.

ಉದಾಹರಣೆ 1: ಸಾಮಾನ್ಯ ಕಾಲ್ ಬ್ಯಾಕ್ (Synchronous Callback)

// ಇದು ಕಾಲ್ ಬ್ಯಾಕ್ ಫಂಕ್ಷನ್
function eatPizza() {
    console.log("ನಾನು ಪಿಜ್ಜಾ ತಿನ್ನುತ್ತಿದ್ದೇನೆ! 🍕");
}

// ಇದು ಮುಖ್ಯ ಫಂಕ್ಷನ್
function orderPizza(callbackFunction) {
    console.log("ಪಿಜ್ಜಾ ಆರ್ಡರ್ ಮಾಡಲಾಗಿದೆ...");
    // ಪಿಜ್ಜಾ ಬಂದ ಮೇಲೆ ಕಾಲ್ ಬ್ಯಾಕ್ ಫಂಕ್ಷನ್ ಕರೆಯಲಾಗುತ್ತದೆ
    callbackFunction(); 
}

// orderPizza ಫಂಕ್ಷನ್‌ಗೆ, eatPizza ಫಂಕ್ಷನ್ ಅನ್ನು ಇನ್‌ಪುಟ್ ಆಗಿ ಕಳುಹಿಸುತ್ತಿದ್ದೇವೆ
orderPizza(eatPizza);

ಔಟ್‌ಪುಟ್ (Output):

ಪಿಜ್ಜಾ ಆರ್ಡರ್ ಮಾಡಲಾಗಿದೆ...
ನಾನು ಪಿಜ್ಜಾ ತಿನ್ನುತ್ತಿದ್ದೇನೆ! 🍕

3. ಅಸಿಂಕ್ರೋನಸ್ ಕಾಲ್ ಬ್ಯಾಕ್ (Asynchronous Callback)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಒಂದೇ ಬಾರಿಗೆ ಒಂದೇ ಕೆಲಸ ಮಾಡುತ್ತದೆ. ಇಂಟರ್ನೆಟ್‌ನಿಂದ ಸಿನಿಮಾ ಡೌನ್‌ಲೋಡ್ ಮಾಡುವಂತಹ ದೊಡ್ಡ ಕೆಲಸ ಇದ್ದರೆ, ಅದು ಮುಗಿಯುವವರೆಗೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಸುಮ್ಮನೆ ಕೂರುವುದಿಲ್ಲ. ಆ ಕೆಲಸವನ್ನು ಹಿನ್ನೆಲೆಗೆ (Background) ಹಾಕಿ, ಬೇರೆ ಕೆಲಸಗಳನ್ನು ಮಾಡುತ್ತದೆ.

ಉದಾಹರಣೆ 2: ಅಸಿಂಕ್ರೋನಸ್ ಕಾಲ್ ಬ್ಯಾಕ್ (ಸಮಯ ತೆಗೆದುಕೊಳ್ಳುವ ಕೆಲಸ)

ಇಲ್ಲಿ ನಾವು setTimeout ಎಂಬ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಫಂಕ್ಷನ್ ಬಳಸುತ್ತೇವೆ. ಇದು ನಿರ್ದಿಷ್ಟ ಸಮಯದ ನಂತರ ಕೆಲಸ ಮಾಡುತ್ತದೆ.

function playGames() {
    console.log("ಪಿಜ್ಜಾ ಬಂತು! ಈಗ ಗೇಮ್ ನಿಲ್ಲಿಸಿ ಪಿಜ್ಜಾ ತಿನ್ನುತ್ತೇನೆ. 🍕");
}

function orderPizzaAsync(callback) {
    console.log("1. ಕೌಂಟರ್‌ನಲ್ಲಿ ಪಿಜ್ಜಾ ಆರ್ಡರ್ ಮಾಡಿದೆ.");
    
    // ಪಿಜ್ಜಾ ತಯಾರಾಗಲು 3 ಸೆಕೆಂಡ್ ಬೇಕು (3000 milliseconds)
    setTimeout(function() {
        console.log("3. ಹೋಟೆಲ್‌ನವರು: ನಿಮ್ಮ ಪಿಜ್ಜಾ ತಯಾರಾಗಿದೆ!");
        callback(); // 3 ಸೆಕೆಂಡ್ ನಂತರ ಕಾಲ್ ಬ್ಯಾಕ್ ಕರೆಯಲಾಗುತ್ತದೆ
    }, 3000); 
}

// ಪ್ರೋಗ್ರಾಮ್ ಇಲ್ಲಿಂದ ಶುರುವಾಗುತ್ತದೆ
orderPizzaAsync(playGames);
console.log("2. ನಾನು ಮೊಬೈಲ್‌ನಲ್ಲಿ ಗೇಮ್ ಆಡುತ್ತಾ ಕಾಯುತ್ತಿದ್ದೇನೆ... 📱");

ಔಟ್‌ಪುಟ್ ಬರುವ ಕ್ರಮವನ್ನು ಗಮನಿಸಿ:

  • 1. ಕೌಂಟರ್‌ನಲ್ಲಿ ಪಿಜ್ಜಾ ಆರ್ಡರ್ ಮಾಡಿದೆ.
  • 2.
  • 3. ನಾನು ಮೊಬೈಲ್‌ನಲ್ಲಿ ಗೇಮ್ ಆಡುತ್ತಾ ಕಾಯುತ್ತಿದ್ದೇನೆ... 📱 (ಗಮನಿಸಿ: ಇದು ಕಾಯದೆ ಪ್ರಿಂಟ್ ಆಗಿದೆ!)
  • 4. (3 ಸೆಕೆಂಡ್ ಕಾದ ನಂತರ)
  • 5.
  • 6. ಹೋಟೆಲ್‌ನವರು: ನಿಮ್ಮ ಪಿಜ್ಜಾ ತಯಾರಾಗಿದೆ!
  • 7.
  • 8. ಪಿಜ್ಜಾ ಬಂತು! ಈಗ ಗೇಮ್ ನಿಲ್ಲಿಸಿ ಪಿಜ್ಜಾ ತಿನ್ನುತ್ತೇನೆ. 🍕
  • 9.

4. ವೃತ್ತಿಪರ ಮಟ್ಟ (Professional Level)

ಸಾಫ್ಟ್‌ವೇರ್ ಕಂಪನಿಗಳಲ್ಲಿ ಕಾಲ್ ಬ್ಯಾಕ್ ಅನ್ನು ಇಂಟರ್ನೆಟ್‌ನಿಂದ ಡೇಟಾ (Data) ತರಲು ಅಥವಾ ಡೇಟಾಬೇಸ್ (Database) ಓದಲು ಬಳಸುತ್ತಾರೆ. ವೃತ್ತಿಪರರು "Error-First Callback" (ದೋಷ-ಮೊದಲು ಕಾಲ್ ಬ್ಯಾಕ್) ವಿಧಾನವನ್ನು ಬಳಸುತ್ತಾರೆ. ಅಂದರೆ, ಡೇಟಾ ತರುವಾಗ ಏನಾದರೂ ತಪ್ಪಾಗಿದೆಯೇ ಎಂದು ಮೊದಲು ಪರಿಶೀಲಿಸುವುದು.

ಉದಾಹರಣೆ 3: Error-First Callback (ವೃತ್ತಿಪರ ಮಟ್ಟ)

// ಸರ್ವರ್‌ನಿಂದ ಬಳಕೆದಾರನ ಮಾಹಿತಿಯನ್ನು ತರುವ ಫಂಕ್ಷನ್
function getUserData(userId, callback) {
    console.log("ಸರ್ವರ್‌ನಿಂದ ಡೇಟಾ ಹುಡುಕಲಾಗುತ್ತಿದೆ...");
    
    setTimeout(() => {
        let error = false; // ಇದನ್ನು true ಮಾಡಿದರೆ error ಬರುತ್ತದೆ
        
        if (error) {
            // ಏನಾದರೂ ತಪ್ಪಾದರೆ (ಉದಾಹರಣೆಗೆ ಇಂಟರ್ನೆಟ್ ಇಲ್ಲದಿದ್ದರೆ)
            callback("ಇಂಟರ್ನೆಟ್ ಸಂಪರ್ಕವಿಲ್ಲ! ಡೇಟಾ ಸಿಗಲಿಲ್ಲ.", null);
        } else {
            // ಡೇಟಾ ಸರಿಯಾಗಿ ಸಿಕ್ಕರೆ
            let user = { id: userId, name: "ರಾಮು", job: "ಸಾಫ್ಟ್‌ವೇರ್ ಎಂಜಿನಿಯರ್" };
            callback(null, user); // ಮೊದಲನೇ ಜಾಗದಲ್ಲಿ null (ಯಾವ ದೋಷವಿಲ್ಲ)
        }
    }, 2000);
}

// ಫಂಕ್ಷನ್ ಅನ್ನು ಕರೆಯುವುದು
getUserData(101, function(error, data) {
    if (error) {
        console.error("❌ ದೋಷ: " + error);
    } else {
        console.log("✅ ಡೇಟಾ ಯಶಸ್ವಿಯಾಗಿ ಸಿಕ್ಕಿತು:", data);
    }
});

ಕಾಲ್ ಬ್ಯಾಕ್‌ನ ತೊಂದರೆ (Callback Hell)

ಒಂದು ವೇಳೆ ನೀವು ಒಂದಾದ ಮೇಲೆ ಒಂದರಂತೆ 10 ಕೆಲಸಗಳನ್ನು ಕಾಲ್ ಬ್ಯಾಕ್ ಬಳಸಿ ಮಾಡಬೇಕಾದರೆ, ಕೋಡ್ ನೋಡಲು ತ್ರಿಭುಜದ (Pyramid) ಆಕಾರದಲ್ಲಿ ಬೆಳೆಯುತ್ತದೆ. ಅರ್ಥಮಾಡಿಕೊಳ್ಳಲು ತುಂಬಾ ಕಷ್ಟವಾಗುತ್ತದೆ. ಇದನ್ನು ಸಾಫ್ಟ್‌ವೇರ್ ಭಾಷೆಯಲ್ಲಿ Callback Hell (ಕಾಲ್ ಬ್ಯಾಕ್ ನರಕ) ಅಥವಾ Pyramid of Doom ಎನ್ನುತ್ತಾರೆ.

ಇದನ್ನು ಸರಿಪಡಿಸಲು ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Promises ಮತ್ತು Async/Await ಎಂಬ ಹೊಸ ವಿಧಾನಗಳನ್ನು ತರಲಾಗಿದೆ.

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Callback Hell (ಕಾಲ್‌ಬ್ಯಾಕ್ ನರಕ) ಎಂಬುದನ್ನು ಅರ್ಥಮಾಡಿಕೊಳ್ಳಲು ಒಂದು ಮೋಜಿನ "ನಿಧಿ ಶೋಧನೆ" (Treasure Hunt) ಕಥೆಯನ್ನು ಕಲ್ಪಿಸಿಕೊಳ್ಳಿ:

ನೀವು ಒಂದು ನಿಧಿಯನ್ನು ಹುಡುಕಲು ಹೋಗುತ್ತಿದ್ದೀರಿ. ಆದರೆ ಅದಕ್ಕೆ ಒಂದಾದ ಮೇಲೊಂದು ಕೆಲಸಗಳನ್ನು ಮಾಡಲೇಬೇಕು:

  1. ಮೊದಲು ಕಾಡಿಗೆ ಹೋಗಬೇಕು (ಇದಕ್ಕೆ 2 ಸೆಕೆಂಡ್ ಬೇಕು).
  2. ಕಾಡು ಸಿಕ್ಕ ಮೇಲೆ, ಅಲ್ಲಿರುವ ಗುಹೆಯನ್ನು ಹುಡುಕಬೇಕು (ಇದಕ್ಕೆ 2 ಸೆಕೆಂಡ್ ಬೇಕು).
  3. ಗುಹೆ ಸಿಕ್ಕ ಮೇಲೆ, ಒಳಗಿರುವ ಪೆಟ್ಟಿಗೆಯನ್ನು ಒಡೆಯಬೇಕು (ಇದಕ್ಕೆ 2 ಸೆಕೆಂಡ್ ಬೇಕು).
  4. ಪೆಟ್ಟಿಗೆ ಒಡೆದ ಮೇಲೆ, ನಿಧಿಯನ್ನು ಮನೆಗೆ ತರಬೇಕು.

ಇಲ್ಲಿ ಒಂದು ಕೆಲಸ ಮುಗಿಯದೆ ಮುಂದಿನ ಕೆಲಸ ಮಾಡುವಂತಿಲ್ಲ. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಇದನ್ನು ಕೋಡ್ ಮೂಲಕ ಬರೆದಾಗ ಏನಾಗುತ್ತದೆ ಎಂದು ನೋಡೋಣ.

1. ತಳಮಟ್ಟದ ಪರಿಕಲ್ಪನೆ (The Basics)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಒಂದು ಅಸಿಂಕ್ರೋನಸ್ (ಸಮಯ ತೆಗೆದುಕೊಳ್ಳುವ) ಕೆಲಸ ಮುಗಿದ ಮೇಲೆ ಇನ್ನೊಂದು ಕೆಲಸ ಶುರು ಮಾಡಲು ನಾವು Callback ಬಳಸುತ್ತೇವೆ. ಆದರೆ ೪-೫ ಕೆಲಸಗಳನ್ನು ಒಂದಾದ ಮೇಲೊಂದು ಮಾಡಬೇಕಾದಾಗ, ಒಂದರ ಒಳಗೆ ಇನ್ನೊಂದು ಕಾಲ್‌ಬ್ಯಾಕ್ ಫಂಕ್ಷನ್ ಬರೆಯುತ್ತಾ ಹೋಗುತ್ತೇವೆ.

ಹೀಗೆ ಒಂದರೊಳಗೊಂದು ಫಂಕ್ಷನ್‌ಗಳನ್ನು ಬರೆಯುತ್ತಾ ಹೋದಾಗ, ಕೋಡ್ ಬಲಭಾಗಕ್ಕೆ ಸರಿಯುತ್ತಾ ಹೋಗಿ ">" (V ಯ ಅಡ್ಡ ರೂಪ) ಆಕಾರದಲ್ಲಿ ಕಾಣುತ್ತದೆ. ಇದನ್ನು ನೋಡಲು ಮತ್ತು ಓದಲು ತುಂಬಾ ಕಷ್ಟವಾಗುತ್ತದೆ. ಇದನ್ನೇ ಪ್ರೋಗ್ರಾಮಿಂಗ್ ಜಗತ್ತಿನಲ್ಲಿ Callback Hell ಅಥವಾ Pyramid of Doom (ವಿನಾಶದ ಪಿರಮಿಡ್) ಎಂದು ಕರೆಯುತ್ತಾರೆ.

2. ಪ್ರಾಥಮಿಕ ಉದಾಹರಣೆ (Callback Hell ಹೇಗಿರುತ್ತದೆ?)

ಮೇಲೆ ಹೇಳಿದ ನಿಧಿ ಶೋಧನೆಯನ್ನೇ ಕೋಡ್‌ನಲ್ಲಿ ಬರೆಯೋಣ:

console.log("ನಿಧಿ ಶೋಧನೆ ಶುರುವಾಯಿತು...");

// 1. ಕಾಡಿಗೆ ಹೋಗುವುದು
setTimeout(function() {
    console.log("1. ಕಾಡು ಸಿಕ್ಕಿತು!");

    // 2. ಗುಹೆ ಹುಡುಕುವುದು (ಕಾಡು ಸಿಕ್ಕ ಮೇಲಷ್ಟೇ ಇದು ಶುರುವಾಗಬೇಕು)
    setTimeout(function() {
        console.log("2. ಗುಹೆ ಸಿಕ್ಕಿತು!");

        // 3. ಪೆಟ್ಟಿಗೆ ಒಡೆಯುವುದು
        setTimeout(function() {
            console.log("3. ಪೆಟ್ಟಿಗೆ ಒಡೆದೆವು!");

            // 4. ನಿಧಿ ಪಡೆಯುವುದು
            setTimeout(function() {
                console.log("4. ಯೆಸ್! ನಿಧಿ ಸಿಕ್ಕಿತು 💰");
            }, 1000);

        }, 2000);

    }, 2000);

}, 2000);

ಮೇಲಿನ ಕೋಡ್ ಅನ್ನು ಗಮನಿಸಿ. ಅದು ಮೆಟ್ಟಿಲುಗಳ ತರಹ ಬಲಕ್ಕೆ ಸರಿಯುತ್ತಾ ಹೋಗಿದೆ. ಸದ್ಯಕ್ಕೆ ಇದು ಕೇವಲ 4 ಕೆಲಸಗಳು. ಆದರೆ 20 ಕೆಲಸಗಳಿದ್ದರೆ? ಕೋಡ್ ಓದಲು ಅಸಾಧ್ಯವಾಗುತ್ತದೆ!

3. ವೃತ್ತಪರ ಮಟ್ಟ (Professional Level)

ಸಾಫ್ಟ್‌ವೇರ್ ಕಂಪನಿಗಳಲ್ಲಿ, ಡೇಟಾಬೇಸ್‌ನಿಂದ ಮಾಹಿತಿ ತರುವಾಗ Callback Hell ದೊಡ್ಡ ತಲೆನೋವಾಗುತ್ತದೆ. ವೃತ್ತಪರ ಕೋಡಿಂಗ್‌ನಲ್ಲಿ ನಾವು ಕೇವಲ ಯಶಸ್ಸನ್ನು ಮಾತ್ರವಲ್ಲ, ತಪ್ಪುಗಳನ್ನು (Errors) ಕೂಡ ನಿಭಾಯಿಸಬೇಕು.

ಒಂದು ಇ-ಕಾಮರ್ಸ್ ವೆಬ್‌ಸೈಟ್‌ನಲ್ಲಿ (ಉದಾಹರಣೆಗೆ: Amazon) ಒಂದು ಆರ್ಡರ್ ಮಾಡಬೇಕಾದರೆ ನಡೆಯುವ ಪ್ರಕ್ರಿಯೆ ನೋಡಿ:

// ವೃತ್ತಪರ Callback Hell ಉದಾಹರಣೆ (Error handling ಜೊತೆಗೆ)

getUserData(userId, function(error, user) {
    if (error) {
        console.log("ಯೂಸರ್ ಸಿಗಲಿಲ್ಲ");
    } else {
        checkBankBalance(user.account, function(error, balance) {
            if (error) {
                console.log("ಬ್ಯಾಂಕ್ ಸರ್ವರ್ ಡೌನ್");
            } else {
                deductMoney(balance, amount, function(error, paymentInfo) {
                    if (error) {
                        console.log("ಹಣ ಕಡಿತಗೊಳಿಸಲು ವಿಫಲವಾಗಿದೆ");
                    } else {
                        placeOrder(paymentInfo, function(error, orderId) {
                            if (error) {
                                console.log("ಆರ್ಡರ್ ಫೇಲ್ ಆಯಿತು");
                            } else {
                                console.log("ಆರ್ಡರ್ ಯಶಸ್ವಿಯಾಗಿದೆ! Order ID: " + orderId);
                            }
                        });
                    }
                });
            }
        });
    }
});

Callback Hell ನಿಂದ ಆಗುವ ತೊಂದರೆಗಳೇನು?

  1. ಓದಲು ಕಷ್ಟ (Poor Readability): ಕೋಡ್ ಎಲ್ಲಿ ಶುರುವಾಗುತ್ತದೆ, ಎಲ್ಲಿ ಮುಗಿಯುತ್ತದೆ ಎಂದು ಹುಡುಕುವುದು ಗೊಂದಲಮಯವಾಗುತ್ತದೆ.
  2. ದೋಷ ನಿವಾರಣೆ ಕಷ್ಟ (Hard to Debug): ಯಾವ ಹಂತದಲ್ಲಿ ತಪ್ಪು (Error) ನಡೆದಿದೆ ಎಂದು ಕಂಡುಹಿಡಿಯುವುದು ತುಂಬಾ ಕಷ್ಟ. ಪ್ರತಿ ಹಂತದಲ್ಲೂ if (error) ಬರೆಯುವುದು ಕಿರಿಕಿರಿಯಾಗುತ್ತದೆ.
  3. ನಿರ್ವಹಣೆ ಕಷ್ಟ (Hard to Maintain): ಭವಿಷ್ಯದಲ್ಲಿ ಹೊಸದಾಗಿ ಇನ್ನೊಂದು ಕೆಲಸವನ್ನು (ಉದಾಹರಣೆಗೆ: ಮೆಸೇಜ್ ಕಳುಹಿಸುವುದು) ಸೇರಿಸಬೇಕಾದರೆ ಇಡೀ ಕೋಡ್ ಅನ್ನು ಬದಲಾಯಿಸಬೇಕಾಗುತ್ತದೆ.

4. ಕಾಲ್‌ಬ್ಯಾಕ್ ನರಕದಿಂದ ತಪ್ಪಿಸಿಕೊಳ್ಳುವುದು ಹೇಗೆ? (The Solution)

ಈ Callback Hell ಸಮಸ್ಯೆಯನ್ನು ಬಗೆಹರಿಸಲು ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ (ES6 ಮತ್ತು ಮೇಲ್ಪಟ್ಟು) ಎರಡು ಅದ್ಭುತ ಅಸ್ತ್ರಗಳನ್ನು ಪರಿಚಯಿಸಿದೆ:

  • Promises (ಪ್ರಾಮಿಸ್‌ಗಳು)
  • Async / Await (ಅಸಿಂಕ್ / ಅವೇಟ್)

🔹 Promises ವಿಸ್ತೃತ ವಿವರಣೆ (ಹೆಚ್ಚುವರಿ ಮಾಹಿತಿ):

Promises ಬಳಸುವುದರಿಂದ .then() ಮತ್ತು .catch() ಮೂಲಕ ಮೇಲಿನ ನರಕವನ್ನು ಸರಪಳಿಯಂತೆ (Chain) ಸುಲಭವಾಗಿ ಬರೆಯಬಹುದು:

// Promises ಬಳಸಿ ಬರೆದಾಗ ಕೋಡ್ ಹೀಗಿರುತ್ತದೆ
processOrderPromise(userId, amount)
    .then(user => checkBankBalance(user.account))
    .then(balance => deductMoney(balance, amount))
    .then(paymentInfo => placeOrder(paymentInfo))
    .then(orderId => console.log("ಆರ್ಡರ್ ಯಶಸ್ವಿಯಾಗಿದೆ! Order ID: " + orderId))
    .catch(error => console.log("ತೊಂದರೆ ಉಂಟಾಗಿದೆ: " + error));

ಅದೇ ಆರ್ಡರ್ ಮಾಡುವ ಕೋಡ್ ಅನ್ನು Async/Await ಬಳಸಿ ಬರೆದರೆ ಎಷ್ಟೊಂದು ಸುಲಭವಾಗಿ ಕಾಣುತ್ತದೆ ನೋಡಿ:

// Callback Hell ಇಲ್ಲದ ಆಧುನಿಕ ಮತ್ತು ವೃತ್ತಪರ ವಿಧಾನ
async function processOrder(userId, amount) {
    try {
        const user = await getUserData(userId);
        const balance = await checkBankBalance(user.account);
        const paymentInfo = await deductMoney(balance, amount);
        const orderId = await placeOrder(paymentInfo);
        
        console.log("ಆರ್ಡರ್ ಯಶಸ್ವಿಯಾಗಿದೆ! Order ID: " + orderId);
    } catch (error) {
        // ಯಾವುದೇ ಹಂತದಲ್ಲಿ ತಪ್ಪಾದರೂ ಈ ಒಂದೇ ಜಾಗದಲ್ಲಿ ಹಿಡಿಯಬಹುದು
        console.log("ತೊಂದರೆ ಉಂಟಾಗಿದೆ: " + error);
    }
}

ಇದು ಮೇಲಿನಿಂದ ಕೆಳಕ್ಕೆ ಸರಾಗವಾಗಿ ಓದಲು (Synchronous ತರಹ) ಸಾಧ್ಯವಾಗುತ್ತದೆ. ಆದ್ದರಿಂದ ವೃತ್ತಪರರು Callback Hell ಅನ್ನು ತಪ್ಪಿಸಲು ಯಾವಾಗಲೂ Promises ಅಥವಾ Async/Await ಅನ್ನು ಬಳಸುತ್ತಾರೆ.
