ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ typeof ಪರೀಕ್ಷಕಗಳು ಮತ್ತು 6 ಅತ್ಯಂತ ಮುಖ್ಯವಾದ ಅಂಶಗಳು

ಸರಳ ಉದಾಹರಣೆಗಳಿಂದ ಪ್ರೊಫೆಷನಲ್ ಇಂಜಿನಿಯರ್ ಹಂತದವರೆಗೆ — ಚಿತ್ರ ಸಹಿತ ಸಂಪೂರ್ಣ ಕೈಪಿಡಿ

ಅಧ್ಯಾಯ 1: typeof ಮತ್ತು ಇತರ ಎಲ್ಲಾ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಪರೀಕ್ಷಕಗಳು (Type & Value Checking Methods)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌‌ನಲ್ಲಿ ಒಂದು ವೇರಿಯೇಬಲ್ ಒಳಗೆ ಯಾವ ರೀತಿಯ ಡೇಟಾ ಇದೆ ಎಂದು ಪರೀಕ್ಷಿಸಲು ಕೇವಲ typeof ಮಾತ್ರವಲ್ಲದೆ, ಸಂದರ್ಭಕ್ಕೆ ತಕ್ಕಂತೆ ಬಳಸುವ ಇನ್ನೂ 8ಕ್ಕೂ ಹೆಚ್ಚು ಪ್ರಮುಖ ವಿಧಾನಗಳಿವೆ (Type & Value Checking Methods).

ಮಕ್ಕಳಿಗೆ ಅರ್ಥವಾಗುವ ಸರಳ ಉದಾಹರಣೆಗಳಿಂದ ಹಿಡಿದು ಪ್ರೊಫೆಷನಲ್ (Pro) ಇಂಜಿನಿಯರ್‌ಗಳು ಬಳಸುವ ತಂತ್ರಗಳವರೆಗೆ ಸಂಪೂರ್ಣ ಮಾಹಿತಿ ಇಲ್ಲಿದೆ:

ಭಾಗ 1: typeof ಎಂದರೇನು ಮತ್ತು ಅದರ ಮಿತಿಗಳೇನು?

  • ಮಕ್ಕಳ ಉದಾಹರಣೆ: ಸೂಪರ್ ಮಾರ್ಕೆಟ್‌ನಲ್ಲಿರುವ ಬಾರ್‌ಕೋಡ್ ಸ್ಕ್ಯಾನರ್ (Barcode Scanner) ಇದ್ದಂತೆ! ಯಾವುದೇ ವಸ್ತುವನ್ನು ಅದರ ಮುಂದೆ ಹಿಡಿದರೆ ಅದು "ಇದು ಸೋಪು", "ಇದು ಬಿಸ್ಕೆಟ್" ಎಂದು ಹೇಳುತ್ತದೆ. ಹಾಗೆಯೇ typeof ಮುಂದೆ ಯಾವುದೇ ಡೇಟಾ ಇಟ್ಟರೆ ಅದರ ಜಾತಿ (Type) ಯಾವುದು ಎಂದು ಹೇಳುತ್ತದೆ.
JavaScript
console.log(typeof "ಕರ್ನಾಟಕ"); // "string"
console.log(typeof 100);       // "number"
console.log(typeof true);      // "boolean"
console.log(typeof undefined); // "undefined"
console.log(typeof 99999n);    // "bigint"
console.log(typeof Symbol());  // "symbol"
console.log(typeof function(){}); // "function"

typeof ನಲ್ಲಿರುವ 3 ದೊಡ್ಡ ಸಮಸ್ಯೆಗಳು (ಇದಕ್ಕಾಗಿಯೇ ಬೇರೆ ವಿಧಾನಗಳು ಬೇಕು!)

typeof ಸಾಮಾನ್ಯ ಡೇಟಾಗಳಿಗೆ (Primitive) ಸರಿಯಾಗಿ ಕೆಲಸ ಮಾಡುತ್ತದೆ, ಆದರೆ ಈ ಕೆಳಗಿನ 3 ಸಂದರ್ಭಗಳಲ್ಲಿ ದಾರಿ ತಪ್ಪಿಸುತ್ತದೆ:

  1. ಎಲ್ಲಾ ಸಂಕೀರ್ಣ ಡೇಟಾಗಳಿಗೂ "object" ಎಂದೇ ಹೇಳುತ್ತದೆ: ನೀವು Array [], Object {}, ಅಥವಾ new Date() ಕೊಟ್ಟರೂ ಎಲ್ಲವನ್ನೂ "object" ಎಂದೇ ತೋರಿಸುತ್ತದೆ!
  2. null ನ ಪ್ರಸಿದ್ಧ ತಪ್ಪು: typeof null ಎಂದರೆ "object" ಎಂದು ತೋರಿಸುತ್ತದೆ (ಇದು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ಹಳೆಯ ಬಗ್).
  3. NaN (Not a Number) ವಿಚಿತ್ರ: ಗಣಿತದಲ್ಲಿ ತಪ್ಪಾದಾಗ ಬರುವ NaN ಅನ್ನು ಪರೀಕ್ಷಿಸಿದರೆ typeof NaN ➔ "number" ಎಂದು ತೋರಿಸುತ್ತದೆ!
JavaScript
console.log(typeof [1, 2, 3]); // "object" (ಇದು Array ಆದರೂ object ಎನ್ನುತ್ತದೆ!)
console.log(typeof null);      // "object" (ತಪ್ಪು ಮಾಹಿತಿ)
console.log(typeof NaN);       // "number" (ಹೆಸರಲ್ಲೇ Not-a-Number ಎಂದಿದ್ದರೂ number ಎನ್ನುತ್ತದೆ!)
ಚಿತ್ರ 1: typeof ಬಾರ್‌ಕೋಡ್ ಸ್ಕ್ಯಾನರ್ ಮತ್ತು ಅದು ಗೊಂದಲಕ್ಕೀಡಾಗುವ 3 ಸಂದರ್ಭಗಳು

ಭಾಗ 2: typeof ತರಹವೇ ಇರುವ ಇತರ ಎಲ್ಲಾ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಪರೀಕ್ಷಕಗಳು

ಮೇಲಿನ ಸಮಸ್ಯೆಗಳನ್ನು ಬಗೆಹರಿಸಲು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಈ ಕೆಳಗಿನ ವಿಶೇಷ ಅಸ್ತ್ರಗಳನ್ನು ಬಳಸಲಾಗುತ್ತದೆ:

1. Array.isArray() — ಇದು ನಿಜವಾಗಿಯೂ Array ನಾ?

  • ಮಕ್ಕಳ ಉದಾಹರಣೆ: typeof ಗೆ ಶಾಲೆಯ ಬ್ಯಾಗ್ (Object) ಮತ್ತು ರೈಲಿನ ಡಬ್ಬಿ (Array) ಎರಡೂ ಒಂದೇ ತರಹ ಕಾಣಿಸುತ್ತವೆ. ಆದರೆ Array.isArray() ಕನ್ನಡಕ ಹಾಕಿಕೊಂಡು ನೋಡಿದರೆ ಅದು ನಿಜವಾಗಿಯೂ ರೈಲಿನ ಡಬ್ಬಿಯೇ (Array) ಎಂದು ಖಚಿತವಾಗಿ ಹೇಳುತ್ತದೆ.
JavaScript
let fruits = ["ಸೇಬು", "ಮಾವು"];
let student = { name: "ರವಿ" };

console.log(Array.isArray(fruits));  // true (ಹೌದು, ಇದು Array!)
console.log(Array.isArray(student)); // false (ಇಲ್ಲ, ಇದು Object)

2. instanceof ಆಪರೇಟರ್ — ವಸ್ತುವಿನ ಮೂಲ/ಕುಟುಂಬ ಪತ್ತೆಹಚ್ಚಲು (DNA ಟೆಸ್ಟ್!)

  • ಮಕ್ಕಳ ಉದಾಹರಣೆ: ಒಂದು ವಸ್ತು ಯಾವ ಕಾರ್ಖಾನೆಯಿಂದ (Class / Constructor) ತಯಾರಾಗಿ ಬಂದಿದೆ ಎಂದು ಪತ್ತೆಹಚ್ಚುವ DNA ಟೆಸ್ಟ್ ಇದ್ದಂತೆ!
  • ಪ್ರೊ ಗಳಿಗೆ: Non-Primitive ಡೇಟಾಗಳಾದ Array, Date, RegExp ಅಥವಾ ನಾವೇ ಸೃಷ್ಟಿಸಿದ Class ಗಳನ್ನು ಗುರುತಿಸಲು ಇದನ್ನು ಬಳಸುತ್ತಾರೆ.
JavaScript
let today = new Date();
let items = [10, 20];

console.log(today instanceof Date);   // true (ಇದು Date ಕಾರ್ಖಾನೆಯಿಂದ ಬಂದಿದೆ)
console.log(items instanceof Array);  // true (ಇದು Array ಜಾತಿಗೆ ಸೇರಿದೆ)
console.log(items instanceof Object); // true (ಎಲ್ಲಾ Array ಗಳೂ Object ಕುಟುಂಬವೇ!)

3. Number.isNaN() ಮತ್ತು isNaN() — ಇದು ಸಂಖ್ಯೆಯಲ್ಲವೇ?

ಯಾವುದಾದರೂ ತಪ್ಪಾದ ಲೆಕ್ಕಾಚಾರ ಮಾಡಿದಾಗ (ಉದಾ: "ಅಪ್ಪು" * 5) ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ NaN (Not a Number) ಕೊಡುತ್ತದೆ. ಅದನ್ನು ಹಿಡಿಯಲು ಇದನ್ನು ಬಳಸುತ್ತೇವೆ.

  • ಎಚ್ಚರಿಕೆ: ಹಳೆಯ isNaN() ಬದಲು ಹೊಸ Number.isNaN() ಬಳಸುವುದು ಹೆಚ್ಚು ನಿಖರ.
JavaScript
let badMath = "ಹತ್ತು" / 2; // ಉತ್ತರ: NaN

console.log(Number.isNaN(badMath)); // true (ಹೌದು, ಲೆಕ್ಕ ತಪ್ಪಾಗಿದೆ - ಇದು NaN)
console.log(Number.isNaN(100));     // false (ಇದು ಸರಿಯಾದ ಸಂಖ್ಯೆ)

4. Number.isInteger() ಮತ್ತು Number.isFinite() — ಸಂಖ್ಯೆಯ ಒಳಪರೀಕ್ಷೆ

ಸಂಖ್ಯೆಯು ಪೂರ್ಣ ಸಂಖ್ಯೆಯೇ ಅಥವಾ ದಶಮಾಂಶವೇ (Decimal) ಎಂದು ತಿಳಿಯಲು:

  • Number.isInteger(): ಚಿಲ್ಲರೆ ಇಲ್ಲದ ಪೂರ್ಣ ಸಂಖ್ಯೆಯೇ ಎಂದು ಪರೀಕ್ಷಿಸುತ್ತದೆ.
  • Number.isFinite(): ಇದು ಮಿತಿಯೊಳಗಿರುವ ಸಾಮಾನ್ಯ ಸಂಖ್ಯೆಯೇ (ಅನಂತ / Infinity ಅಲ್ಲವೇ) ಎಂದು ಪರೀಕ್ಷಿಸುತ್ತದೆ.
JavaScript
console.log(Number.isInteger(25));    // true (ಪೂರ್ಣ ಸಂಖ್ಯೆ)
console.log(Number.isInteger(25.50)); // false (ದಶಮಾಂಶ ಸಂಖ್ಯೆ)
console.log(Number.isFinite(10 / 0)); // false (ಏಕೆಂದರೆ 10/0 ಎಂದರೆ Infinity)

5. in ಆಪರೇಟರ್ & Object.hasOwn() — ಬ್ಯಾಗ್ ಒಳಗೆ ವಸ್ತು ಇದೆಯೇ?

  • ಮಕ್ಕಳ ಉದಾಹರಣೆ: ಸ್ಕೂಲ್ ಬ್ಯಾಗ್ ತೆರೆಯುವ ಮುನ್ನವೇ "ಈ ಬ್ಯಾಗ್ ಒಳಗೆ ನೀರಿನ ಬಾಟಲಿ ಇದೆಯೇ?" ಎಂದು ಹೊರಗಿನಿಂದಲೇ ಸ್ಕ್ಯಾನ್ ಮಾಡುವ ವಿಧಾನ.
  • ಕೋಡಿಂಗ್‌ನಲ್ಲಿ: ಒಂದು Object ಒಳಗೆ ನಿರ್ದಿಷ್ಟ Key (ಹೆಸರು) ಇದೆಯೇ ಎಂದು ಪರೀಕ್ಷಿಸಲು ಬಳಸುತ್ತಾರೆ.
JavaScript
let car = { brand: "Tata", price: 800000 };

console.log("brand" in car);             // true (brand ಎಂಬ ಮಾಹಿತಿ ಇದೆ)
console.log("color" in car);             // false (color ಇಲ್ಲ)
console.log(Object.hasOwn(car, "price")); // true (ಆಧುನಿಕ ಮತ್ತು ಸುರಕ್ಷಿತ ವಿಧಾನ)

6. Boolean() ಅಥವಾ !! — ಡಬ್ಬಿ ತುಂಬಿದೆಯೇ ಅಥವಾ ಖಾಲಿಯೇ? (Truthy vs Falsy)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌‌ನಲ್ಲಿ ಯಾವುದೇ ಮೌಲ್ಯವು ಅಸಲಿಯೇ (Truthy) ಅಥವಾ ಖಾಲಿಯೇ (Falsy) ಎಂದು ಪರೀಕ್ಷಿಸಲು ಇದನ್ನು ಬಳಸುತ್ತಾರೆ.

  • 6 Falsy ಮೌಲ್ಯಗಳು (ಇವು ಯಾವಾಗಲೂ false ಆಗಿರುತ್ತವೆ): false, 0, "" (ಖಾಲಿ ಸ್ಟ್ರಿಂಗ್), null, undefined, NaN. ಇವುಗಳನ್ನು ಬಿಟ್ಟು ಉಳಿದೆಲ್ಲವೂ true!
JavaScript
console.log(Boolean("ಕನ್ನಡ")); // true (ಒಳಗೆ ಅಕ್ಷರ ಇದೆ)
console.log(Boolean(""));      // false (ಖಾಲಿ ಸ್ಟ್ರಿಂಗ್)
console.log(Boolean(0));       // false (ಸೊನ್ನೆ)

// ಶಾರ್ಟ್‌ಕಟ್ ಆಗಿ ಎರಡು ಆಶ್ಚರ್ಯಕರ ಚಿಹ್ನೆ (!!) ಬಳಸುತ್ತಾರೆ:
console.log(!!"Hello"); // true

7. Object.is() — ಸೂಪರ್ ನಿಖರ ಹೋಲಿಕೆ (=== ಗಿಂತಲೂ ಸ್ಟ್ರಾಂಗ್!)

ಸಾಮಾನ್ಯವಾಗಿ ಎರಡು ಬೆಲೆಗಳನ್ನು ಹೋಲಿಸಲು === ಬಳಸುತ್ತೇವೆ. ಆದರೆ === ಎರಡು ಜಾಗದಲ್ಲಿ ಸೋಲುತ್ತದೆ: ಅದು NaN === NaN ಎಂದರೆ false ಎನ್ನುತ್ತದೆ ಮತ್ತು +0 === -0 ಎಂದರೆ true ಎನ್ನುತ್ತದೆ. ಇದನ್ನು ಸರಿಪಡಿಸಲು Object.is() ಬಳಸುತ್ತಾರೆ.

JavaScript
console.log(NaN === NaN);          // false (=== ಗೆ ಗುರುತಿಸಲು ಆಗುವುದಿಲ್ಲ)
console.log(Object.is(NaN, NaN));  // true (ನಿಖರವಾಗಿ ಗುರುತಿಸುತ್ತದೆ!)

console.log(+0 === -0);            // true
console.log(Object.is(+0, -0));    // false (+0 ಮತ್ತು -0 ಬೇರೆ ಎಂದು ಗುರುತಿಸುತ್ತದೆ)

8. Pro ರಹಸ್ಯ: Object.prototype.toString.call() — ಮಾಸ್ಟರ್ ಡಿಟೆಕ್ಟಿವ್!

  • ಇದೇ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ಬ್ರಹ್ಮಾಸ್ತ್ರ! ನೀವು ಇಂಟರ್ವ್ಯೂಗಳಲ್ಲಿ ಅಥವಾ ದೊಡ್ಡ ಸಾಫ್ಟ್‌ವೇರ್‌ಗಳಲ್ಲಿ ಯಾವುದೇ ಡೇಟಾದ 100% ಅಸಲಿ ಜಾತಿಯನ್ನು ಕಂಡುಹಿಡಿಯಬೇಕಾದರೆ ಇದನ್ನು ಬಳಸುತ್ತಾರೆ. typeof ಸೋಲುವ ಎಲ್ಲಾ ಜಾಗದಲ್ಲೂ ಇದು ಗೆಲ್ಲುತ್ತದೆ!
JavaScript
let checkType = (val) => Object.prototype.toString.call(val);

console.log(checkType(null));       // "[object Null]" (ನಿಖರವಾಗಿ ಹೇಳಿತು!)
console.log(checkType([1, 2, 3]));  // "[object Array]"
console.log(checkType({a: 1}));     // "[object Object]"
console.log(checkType(new Date())); // "[object Date]"
console.log(checkType(/abc/));      // "[object RegExp]"
ಚಿತ್ರ 2: ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ವಿಶೇಷ ಪರೀಕ್ಷಕಗಳ ಟೂಲ್‌ಕಿಟ್ (Special Inspector Toolkit)

ಭಾಗ 3: ಮಾಸ್ಟರ್ ಸಾರಾಂಶ ಕೋಷ್ಟಕ (Cheat Sheet)

ಪರೀಕ್ಷಕ (Method / Operator) ಮಕ್ಕಳ ಭಾಷೆಯಲ್ಲಿ ಉದಾಹರಣೆ ಯಾವಾಗ ಬಳಸಬೇಕು? ಉದಾಹರಣೆ ಮತ್ತು ಫಲಿತಾಂಶ
typeof ಸಾಮಾನ್ಯ ಬಾರ್‌ಕೋಡ್ ಸ್ಕ್ಯಾನರ್ ಮೂಲಭೂತ (Primitive) ಡೇಟಾ ಟೈಪ್ ತಿಳಿಯಲು typeof "ರವಿ" ➔ "string"
Array.isArray() ರೈಲು ಡಬ್ಬಿ ಪತ್ತೆದಾರ ಕೊಟ್ಟಿರುವ ಡೇಟಾ Array ಹೌದೋ ಅಲ್ಲವೋ ತಿಳಿಯಲು Array.isArray([1,2]) ➔ true
instanceof DNA / ಕುಟುಂಬ ಪರೀಕ್ಷೆ Object ಯಾವ Class/Constructor ಇಂದ ಬಂದಿದೆ ತಿಳಿಯಲು d instanceof Date ➔ true
Number.isNaN() ತಪ್ಪು ಲೆಕ್ಕ ಹಿಡಿಯುವ ಪೊಲೀಸ್ ಬೆಲೆಯು NaN (Not a Number) ಆಗಿದೆಯೇ ಪರೀಕ್ಷಿಸಲು Number.isNaN(0/0) ➔ true
Number.isInteger() ಪೂರ್ಣ ಸಂಖ್ಯೆ ಪರೀಕ್ಷಕ ಸಂಖ್ಯೆಯಲ್ಲಿ ದಶಮಾಂಶ (Decimal) ಇಲ್ಲವೇ ಎಂದು ನೋಡಲು Number.isInteger(10) ➔ true
in / Object.hasOwn() ಬ್ಯಾಗ್ ಒಳಗಿನ ವಸ್ತು ಸ್ಕ್ಯಾನರ್ Object ಒಳಗೆ ನಿರ್ದಿಷ್ಟ Key ಇದೆಯೇ ಎಂದು ನೋಡಲು "age" in user ➔ true
Boolean() / !! ಖಾಲಿ ಡಬ್ಬಿ ಪರೀಕ್ಷಕ ಮೌಲ್ಯವು Truthy ನಾ ಅಥವಾ Falsy ನಾ ಎಂದು ತಿಳಿಯಲು Boolean("") ➔ false
Object.is() ಸೂಕ್ಷ್ಮದರ್ಶಕ ಹೋಲಿಕೆ NaN ಮತ್ತು -0 ಸಮೇತ ನಿಖರವಾಗಿ ಹೋಲಿಸಲು Object.is(NaN, NaN) ➔ true
toString.call() ಮಾಸ್ಟರ್ ಡಿಟೆಕ್ಟಿವ್ (ಬ್ರಹ್ಮಾಸ್ತ್ರ) null, Array, Date ಸೇರಿದಂತೆ ಎಲ್ಲದರ ಅಸಲಿ ಟೈಪ್ ತಿಳಿಯಲು toString.call(null) ➔ "[object Null]"
ಅಧ್ಯಾಯ 2: ಪ್ರತಿಯೊಬ್ಬ ಡೆವಲಪರ್ ತಿಳಿದಿರಲೇಬೇಕಾದ ಇನ್ನೂ 6 ಅತ್ಯಂತ ಮುಖ್ಯವಾದ (Must-Know) ಅಂಶಗಳು

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಡೇಟಾ ಟೈಪ್‌ಗಳನ್ನು ಪರೀಕ್ಷಿಸುವ (typeof ಇತ್ಯಾದಿ) ನಂತರ, ಪ್ರತಿಯೊಬ್ಬ ಡೆವಲಪರ್ ಮತ್ತು ಸಂದರ್ಶನ ಎದುರಿಸುವವರು ತಿಳಿದಿರಲೇಬೇಕಾದ ಇನ್ನೂ 6 ಅತ್ಯಂತ ಮುಖ್ಯವಾದ (Must-Know) ಅಂಶಗಳಿವೆ. ಇವುಗಳಿಲ್ಲದೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಕಲಿಕೆ ಪೂರ್ಣವಾಗುವುದಿಲ್ಲ.

ಸರಳ ಉದಾಹರಣೆಗಳೊಂದಿಗೆ ಆ 6 ಮುಖ್ಯ ಅಂಶಗಳ ಸಂಪೂರ್ಣ ವಿವರ ಇಲ್ಲಿದೆ:

1. Explicit Type Conversion (ನಾವೇ ಖುದ್ದಾಗಿ ಡೇಟಾ ಟೈಪ್ ಬದಲಾಯಿಸುವುದು)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ತಾನಾಗಿಯೇ ಡೇಟಾ ಬದಲಾಯಿಸಿಕೊಂಡರೆ ಅದನ್ನು Type Coercion ಎನ್ನುತ್ತೇವೆ ಎಂದು ಹಿಂದೆ ನೋಡಿದ್ದೇವೆ. ಆದರೆ ಪ್ರೋಗ್ರಾಮಿಂಗ್ ಮಾಡುವಾಗ ನಾವೇ ಉದ್ದೇಶಪೂರ್ವಕವಾಗಿ ಒಂದು ಡೇಟಾ ಟೈಪ್ ಅನ್ನು ಇನ್ನೊಂದಕ್ಕೆ ಬದಲಾಯಿಸಬೇಕಾಗುತ್ತದೆ.

  • ಮಕ್ಕಳ ಉದಾಹರಣೆ: ಐಸ್ (ಮಂಜುಗಡ್ಡೆ) ಅನ್ನು ಬಿಸಿ ಮಾಡಿ ನೀರಾಗಿ ಬದಲಾಯಿಸಿದಂತೆ! ವಸ್ತು ಒಂದೇ, ಆದರೆ ನಾವು ಅದರ ರೂಪವನ್ನು ಬದಲಾಯಿಸುತ್ತೇವೆ.
  • ಕೋಡಿಂಗ್‌ನಲ್ಲಿ ಏಕೆ ಬೇಕು? HTML ಫಾರ್ಮ್‌‌ನಿಂದ (Input box) ಬಳಕೆದಾರರು ಅವರ ವಯಸ್ಸು ಅಥವಾ ಮೊಬೈಲ್ ನಂಬರ್ ಟೈಪ್ ಮಾಡಿದಾಗ ಅದು ಯಾವಾಗಲೂ String ("25") ರೂಪದಲ್ಲೇ ಬರುತ್ತದೆ! ಅದನ್ನು ಲೆಕ್ಕಾಚಾರಕ್ಕೆ ಬಳಸಲು Number (25) ಆಗಿ ಬದಲಾಯಿಸಲೇಬೇಕು.
ಚಿತ್ರ 3: Explicit Type Conversion — String ("150.75px") ಅನ್ನು Number ಆಗಿ ಬದಲಾಯಿಸುವ ವಿಧಾನಗಳು

A. ಯಾವುದೇ ಡೇಟಾವನ್ನು Number ಆಗಿ ಬದಲಾಯಿಸಲು 3 ವಿಧಾನಗಳು:

JavaScript
let priceText = "150.75";

// 1. Number() - ಪೂರ್ಣ ಅಥವಾ ದಶಮಾಂಶ ಸಂಖ್ಯೆಯಾಗಿ ಬದಲಾಯಿಸುತ್ತದೆ:
console.log(Number("150"));      // Output: 150 (Number)
console.log(Number("ಕನ್ನಡ"));    // Output: NaN (ಅಕ್ಷರವನ್ನು ಸಂಖ್ಯೆ ಮಾಡಲು ಸಾಧ್ಯವಿಲ್ಲ!)

// 2. parseInt() - ಕೇವಲ ಪೂರ್ಣ ಸಂಖ್ಯೆಯನ್ನು (Integer) ಮಾತ್ರ ತೆಗೆದುಕೊಳ್ಳುತ್ತದೆ:
console.log(parseInt("150.75")); // Output: 150 (.75 ಬಿಟ್ಟುಬಿಡುತ್ತದೆ)
console.log(parseInt("100px"));  // Output: 100 (ಪಕ್ಕದಲ್ಲಿರುವ 'px' ತೆಗೆದು ಸಂಖ್ಯೆ ಮಾತ್ರ ಕೊಡುತ್ತದೆ!)

// 3. parseFloat() - ದಶಮಾಂಶ ಸಂಖ್ಯೆ ಸಮೇತ ತೆಗೆದುಕೊಳ್ಳುತ್ತದೆ:
console.log(parseFloat("150.75px")); // Output: 150.75

B. ಯಾವುದೇ ಡೇಟಾವನ್ನು String (ಪದ) ಆಗಿ ಬದಲಾಯಿಸಲು:

JavaScript
let rollNo = 42;

console.log(String(rollNo));     // Output: "42"
console.log(rollNo.toString());  // Output: "42"

2. == vs === (ಸಾಮಾನ್ಯ ಹೋಲಿಕೆ vs ಕಟ್ಟುನಿಟ್ಟಿನ ಹೋಲಿಕೆ)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಇಂಟರ್ವ್ಯೂಗಳಲ್ಲಿ ಅತಿ ಹೆಚ್ಚು ಕೇಳುವ ಪ್ರಶ್ನೆ ಇದು!

  • == (Loose Equality - ಬಣ್ಣದ ಕನ್ನಡಕ ಹಾಕಿದ ಸೆಕ್ಯೂರಿಟಿ ಗಾರ್ಡ್): ಇವನು ಕೇವಲ ಒಳಗಿರುವ ಬೆಲೆಯನ್ನು (Value) ಮಾತ್ರ ನೋಡುತ್ತಾನೆ, ಅದರ ಜಾತಿ (Data Type) ಯಾವುದು ಎಂದು ನೋಡುವುದಿಲ್ಲ!
  • === (Strict Equality - ಕಟ್ಟುನಿಟ್ಟಾದ ಪೊಲೀಸ್ ಅಧಿಕಾರಿ): ಇವನು ಬೆಲೆಯನ್ನೂ (Value) ನೋಡುತ್ತಾನೆ ಮತ್ತು ಅದರ ಡೇಟಾ ಟೈಪ್ (Data Type) ಕೂಡ ಒಂದೇ ಆಗಿದೆಯೇ ಎಂದು ಚೆಕ್ ಮಾಡುತ್ತಾನೆ!
ಚಿತ್ರ 4: == (ಸಾಮಾನ್ಯ ಗಾರ್ಡ್) vs === (ಕಟ್ಟುನಿಟ್ಟಾದ ಪೊಲೀಸ್ ಅಧಿಕಾರಿ)
JavaScript
console.log(10 == "10");  // Output: true (ಬೆಲೆ ಒಂದೇ ಇದೆ ಎಂದು ಒಪ್ಪಿಕೊಳ್ಳುತ್ತದೆ!)
console.log(10 === "10"); // Output: false (ಒಂದು Number, ಇನ್ನೊಂದು String - ಆದ್ದರಿಂದ ತಪ್ಪು!)

console.log(0 == false);  // Output: true
console.log(0 === false); // Output: false
Pro ನಿಯಮ: ಕೋಡಿಂಗ್ ಮಾಡುವಾಗ ತಪ್ಪುಗಳಾಗಬಾರದು ಎಂದರೆ ಯಾವಾಗಲೂ === ಅನ್ನೇ ಬಳಸಿ!

3. Optional Chaining (?.) — ಕೋಡ್ ಕ್ರ್ಯಾಶ್ ಆಗದಂತೆ ತಡೆಯುವ ಏರ್‌ಬ್ಯಾಗ್ (Airbag)!

ನಾವು ದೊಡ್ಡ Object ಒಳಗಿನ ಮಾಹಿತಿಯನ್ನು ಹುಡುಕುವಾಗ, ಆ ಮಾಹಿತಿ ಇಲ್ಲದಿದ್ದರೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ತಕ್ಷಣ Uncaught TypeError: Cannot read properties of undefined ಎಂದು ಕೆಂಪು Error ಕೊಟ್ಟು ಇಡೀ ವೆಬ್‌‌ಸೈಟ್ ನಿಲ್ಲಿಸಿಬಿಡುತ್ತದೆ! ಇದನ್ನು ತಡೆಯಲು ಬಂದಿರುವ ಆಧುನಿಕ ಅಸ್ತ್ರವೇ ?. (Optional Chaining).

  • ಮಕ್ಕಳ ಉದಾಹರಣೆ: ಕಾರಿನಲ್ಲಿರುವ ಏರ್‌ಬ್ಯಾಗ್ (Airbag) ಇದ್ದಂತೆ! ಮುಂದೆ ದಾರಿ ಇಲ್ಲದಿದ್ದರೆ ಅಪಘಾತವಾಗಲು (Error ಬರಲು) ಬಿಡುವುದಿಲ್ಲ, ಬದಲಿಗೆ ಸುರಕ್ಷಿತವಾಗಿ ನಿಲ್ಲಿಸಿ undefined ಎಂದು ಹೇಳುತ್ತದೆ.
JavaScript
let student = {
  name: "ರವಿ"
  // ಇಲ್ಲಿ address ಎಂಬ ಮಾಹಿತಿಯೇ ಇಲ್ಲ!
};

// ಸಾಮಾನ್ಯ ರೀತಿ (ಇದು ಕೆಂಪು Error ಕೊಟ್ಟು ಕೋಡ್ ನಿಲ್ಲಿಸುತ್ತದೆ!):
// console.log(student.address.city); // ❌ Uncaught TypeError!

// Optional Chaining (?.) ಬಳಸಿದಾಗ (ಯಾವುದೇ Error ಬರುವುದಿಲ್ಲ!):
console.log(student?.address?.city);  // ✅ Output: undefined (ಕೋಡ್ ಸೇಫ್!)

4. Nullish Coalescing (??) vs OR (||) — ಡಿಫಾಲ್ಟ್ ಬೆಲೆ ಕೊಡುವ ಮ್ಯಾಜಿಕ್!

ಬಳಕೆದಾರರು ಯಾವುದೇ ಮಾಹಿತಿ ಕೊಡದಿದ್ದಾಗ (null ಅಥವಾ undefined ಇದ್ದಾಗ) ನಾವು ಒಂದು ಬದಲಿ ಬೆಲೆಯನ್ನು (Default Value) ಕೊಡಲು ?? ಬಳಸುತ್ತೇವೆ.

  • || (OR ಆಪರೇಟರ್ ಸಮಸ್ಯೆ): ವಿದ್ಯಾರ್ಥಿಯು ಪರೀಕ್ಷೆಯಲ್ಲಿ 0 ಅಂಕ ಪಡೆದಿದ್ದರೂ, || ಆಪರೇಟರ್ 0 ಯನ್ನು ಖಾಲಿ (Falsy) ಎಂದು ಭಾವಿಸಿ ಬದಲಿ ಬೆಲೆ ಕೊಟ್ಟುಬಿಡುತ್ತದೆ!
  • ?? (Nullish Coalescing ಪರಿಹಾರ): ಇದು ಕೇವಲ null ಅಥವಾ undefined ಇದ್ದಾಗ ಮಾತ್ರ ಬದಲಿ ಬೆಲೆ ಕೊಡುತ್ತದೆ. 0 ಅಥವಾ "" ಇದ್ದರೆ ಅದನ್ನೇ ಉಳಿಸಿಕೊಳ್ಳುತ್ತದೆ!
JavaScript
let marks = 0; // ವಿದ್ಯಾರ್ಥಿ 0 ಅಂಕ ಪಡೆದಿದ್ದಾನೆ

// ಹಳೆಯ || ಬಳಸಿದರೆ ಆಗುವ ತಪ್ಪು:
console.log(marks || "ಗೈರು ಹಾಜರಿ"); // Output: "ಗೈರು ಹಾಜರಿ" (0 ಅಂಕವನ್ನು ತಪ್ಪಾಗಿ ತಿಳಿಯಿತು!)

// ಹೊಸ ?? ಬಳಸಿದರೆ ಸರಿಯಾದ ಫಲಿತಾಂಶ:
console.log(marks ?? "ಗೈರು ಹಾಜರಿ"); // Output: 0 (ನಿಖರವಾಗಿ 0 ಎಂದೇ ತೋರಿಸುತ್ತದೆ!)
ಚಿತ್ರ 5: Optional Chaining (?.) ಏರ್‌ಬ್ಯಾಗ್ ಮತ್ತು Nullish Coalescing (??) ಸ್ಮಾರ್ಟ್ ಫಿಲ್ಟರ್

5. Object ರಕ್ಷಣಾ ಕವಚಗಳು: Object.freeze() vs Object.seal()

ನಾವು ಹಿಂದೆ ನೋಡಿದಂತೆ const ಬಳಸಿದರೂ Object ಒಳಗಿನ ಬೆಲೆಗಳನ್ನು ತಿದ್ದಬಹುದು. ಆದರೆ Object ಒಳಗಿನ ಮಾಹಿತಿಯನ್ನು ಯಾರೂ ಬದಲಾಯಿಸದಂತೆ ಬೀಗ ಹಾಕಲು ಇವೆರಡನ್ನು ಬಳಸುತ್ತಾರೆ:

  1. Object.freeze() (ಫ್ರಿಡ್ಜ್‌ನಲ್ಲಿಟ್ಟ ಕಲ್ಲು-ಐಸ್!): ಇದನ್ನು ಬಳಸಿದರೆ Object ಒಳಗೆ ಹೊಸದನ್ನು ಸೇರಿಸಲೂ ಆಗುವುದಿಲ್ಲ, ಇರುವುದನ್ನು ತೆಗೆಯಲೂ ಆಗುವುದಿಲ್ಲ ಮತ್ತು ಇರುವ ಬೆಲೆಯನ್ನು ತಿದ್ದಲೂ ಆಗುವುದಿಲ್ಲ! (100% ಲಾಕ್).
  2. Object.seal() (ಸೀಲ್ ಮಾಡಿದ ಲಕೋಟೆ!): ಇದರಲ್ಲಿ ಹೊಸ ಮಾಹಿತಿ ಸೇರಿಸಲು ಮತ್ತು ಇರುವುದನ್ನು ಅಳಿಸಲು ಸಾಧ್ಯವಿಲ್ಲ. ಆದರೆ ಇರುವ ಮಾಹಿತಿಯ ಬೆಲೆಯನ್ನು ಮಾತ್ರ ತಿದ್ದಬಹುದು!
JavaScript
let bankRules = { interest: 7 };
Object.freeze(bankRules); // ಸಂಪೂರ್ಣ ಬೀಗ ಹಾಕಲಾಗಿದೆ!

bankRules.interest = 10;  // ಬದಲಾಗುವುದಿಲ್ಲ!
console.log(bankRules.interest); // Output: 7

6. Shallow Copy vs Deep Copy (structuredClone()) — ಆಳವಾದ ಜೆರಾಕ್ಸ್!

ನಾವು ಹಿಂದೆ Spread Operator ({ ...user1 }) ಬಳಸಿ Object ಕಾಪಿ ಮಾಡುವುದನ್ನು ನೋಡಿದ್ದೇವೆ. ಆದರೆ ಒಂದು Object ಒಳಗೆ ಇನ್ನೊಂದು Object ಇದ್ದರೆ (Nested Object), Spread Operator ಕೇವಲ ಮೇಲ್ಪದರವನ್ನು ಮಾತ್ರ ಕಾಪಿ ಮಾಡುತ್ತದೆ (Shallow Copy). ಒಳಗಿರುವ Object ಅನ್ನು ಪೂರ್ತಿಯಾಗಿ ಹೊಸ ವಿಳಾಸದಲ್ಲಿ ಕಾಪಿ ಮಾಡಲು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ಹೊಸ ಬಿಲ್ಟ್-ಇನ್ ಫಂಕ್ಷನ್ structuredClone() ಬಳಸುತ್ತಾರೆ (Deep Copy).

  • ಮಕ್ಕಳ ಉದಾಹರಣೆ: ಬ್ಯಾಗ್ ಒಳಗೆ ಇನ್ನೊಂದು ಚಿಕ್ಕ ಪೌಚ್ ಇದೆ ಎಂದಿಟ್ಟುಕೊಳ್ಳಿ. ... ಬಳಸಿದರೆ ಕೇವಲ ಹೊರಗಿನ ಬ್ಯಾಗ್ ಮಾತ್ರ ಹೊಸದಾಗುತ್ತದೆ. ಆದರೆ structuredClone() ಬಳಸಿದರೆ ಬ್ಯಾಗ್ ಸಮೇತ ಒಳಗಿರುವ ಪೌಚ್ ಕೂಡ ಸಂಪೂರ್ಣ ಹೊಸದಾಗಿ ಜೆರಾಕ್ಸ್ ಆಗುತ್ತದೆ!
ಚಿತ್ರ 6: Object.freeze() ರಕ್ಷಣಾ ಕವಚ ಮತ್ತು structuredClone() ಆಳವಾದ ಜೆರಾಕ್ಸ್
JavaScript
let emp1 = {
  name: "ಕಿರಣ್",
  address: { city: "ಬೆಳಗಾವಿ" } // Object ಒಳಗೆ ಇನ್ನೊಂದು Object!
};

// Deep Copy ಮಾಡಲು ಆಧುನಿಕ ಬ್ರಹ್ಮಾಸ್ತ್ರ:
let emp2 = structuredClone(emp1);

emp2.address.city = "ಮೈಸೂರು";
console.log(emp1.address.city); // Output: "ಬೆಳಗಾವಿ" (emp1 100% ಸುರಕ್ಷಿತ!)

ಮಾಸ್ಟರ್ ಸಾರಾಂಶ ಕೋಷ್ಟಕ (Cheat Sheet)

ವಿಷಯ / ಆಪರೇಟರ್ ಮಕ್ಕಳ ಭಾಷೆಯಲ್ಲಿ ಉದಾಹರಣೆ ಮುಖ್ಯ ಉಪಯೋಗ ಕೋಡ್ ಉದಾಹರಣೆ
Number() / parseInt() ಮಂಜುಗಡ್ಡೆಯನ್ನು ನೀರಾಗಿಸುವುದು String ಅನ್ನು ಸಂಖ್ಯೆಯಾಗಿ ಪರಿವರ್ತಿಸಲು parseInt("50px") ➔ 50
=== (Strict Equality) ಕಟ್ಟುನಿಟ್ಟಾದ ಪೊಲೀಸ್ ಅಧಿಕಾರಿ ಬೆಲೆ ಮತ್ತು ಡೇಟಾ ಟೈಪ್ ಎರಡನ್ನೂ ಹೋಲಿಸಲು 5 === "5" ➔ false
?. (Optional Chaining) ಕಾರಿನ ಸುರಕ್ಷಾ ಏರ್‌ಬ್ಯಾಗ್ ಮಾಹಿತಿ ಇಲ್ಲದಿದ್ದಾಗ ಕೆಂಪು Error ಬರದಂತೆ ತಡೆಯಲು user?.address?.city
?? (Nullish Coalescing) ಸ್ಮಾರ್ಟ್ ಬದಲಿ ಆಟಗಾರ ಕೇವಲ null/undefined ಇದ್ದಾಗ ಬದಲಿ ಬೆಲೆ ಕೊಡಲು score ?? "NA"
Object.freeze() ಫ್ರಿಡ್ಜ್‌ನ ಕಲ್ಲು-ಐಸ್ (100% ಬೀಗ) Object ಅನ್ನು ಯಾರೂ ತಿದ್ದದಂತೆ ಸಂಪೂರ್ಣ ಲಾಕ್ ಮಾಡಲು Object.freeze(obj)
structuredClone() ಪೂರ್ಣ ಪ್ರಮಾಣದ 3D ಜೆರಾಕ್ಸ್ Object ಒಳಗಿನ Object ಅನ್ನೂ ಸೇಫ್ ಆಗಿ ಕಾಪಿ ಮಾಡಲು structuredClone(obj)
