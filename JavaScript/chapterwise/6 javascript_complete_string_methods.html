ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಸ್ಟ್ರಿಂಗ್ ವಿಧಾನಗಳು

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಸ್ಟ್ರಿಂಗ್ (String) ಎಂದರೆ ಪಠ್ಯವನ್ನು (Text) ನಿರ್ವಹಿಸಲು ಬಳಸುವ ಡೇಟಾ ಪ್ರಕಾರ. ಸ್ಟ್ರಿಂಗ್‌ಗಳನ್ನು ಮ್ಯಾನಿಪುಲೇಟ್ ಮಾಡಲು ಲಭ್ಯವಿರುವ ಎಲ್ಲಾ ಇನ್-ಬಿಲ್ಟ್ ವಿಧಾನಗಳ ಸಂಪೂರ್ಣ ತಾಂತ್ರಿಕ ವಿವರಣೆ ಇಲ್ಲಿದೆ.

1. ಮೂಲಭೂತ ಮತ್ತು ಅಕ್ಷರ ಪಡೆಯುವಿಕೆ (Basic & Character Access)

.length

ಇದು ಮೆಥಡ್ ಅಲ್ಲ, ಪ್ರಾಪರ್ಟಿ. ಸ್ಟ್ರಿಂಗ್‌ನಲ್ಲಿರುವ ಒಟ್ಟು ಅಕ್ಷರಗಳ ಸಂಖ್ಯೆಯನ್ನು (ಸ್ಪೇಸ್ ಸೇರಿ) ನೀಡುತ್ತದೆ.

let text = "Hello Kannada";
console.log(text.length); // Output: 13

.charAt(index)

ಸೂಚಿಸಿದ ಇಂಡೆಕ್ಸ್ (Index - 0 ಯಿಂದ ಆರಂಭ) ನಲ್ಲಿರುವ ನಿರ್ದಿಷ್ಟ ಅಕ್ಷರವನ್ನು ಹಿಂತಿರುಗಿಸುತ್ತದೆ.

let str = "JavaScript";
console.log(str.charAt(4)); // Output: "S"

.charCodeAt(index)

ಸೂಚಿಸಿದ ಇಂಡೆಕ್ಸ್‌ನಲ್ಲಿರುವ ಅಕ್ಷರದ ಯುನಿಕೋಡ್ (Unicode / UTF-16) ಮೌಲ್ಯವನ್ನು (0 ರಿಂದ 65535 ವರೆಗಿನ ಸಂಖ್ಯೆ) ನೀಡುತ್ತದೆ.

let str = "A";
console.log(str.charCodeAt(0)); // Output: 65

.at(index) ES2022

ಇದು `charAt()` ನಂತೆಯೇ ಕೆಲಸ ಮಾಡುತ್ತದೆ. ಆದರೆ ಇದರ ವಿಶೇಷತೆ ಎಂದರೆ ಇದು ಋಣಾತ್ಮಕ (negative) ಇಂಡೆಕ್ಸ್ ಅನ್ನು ಬೆಂಬಲಿಸುತ್ತದೆ. ಅಂದರೆ ಕೊನೆಯಿಂದ ಹಿಂದಕ್ಕೆ ಲೆಕ್ಕಾಚಾರ ಮಾಡಬಹುದು.

let str = "Hello";
console.log(str.at(-1)); // Output: "o" (ಕೊನೆಯ ಅಕ್ಷರ)
console.log(str.at(-2)); // Output: "l"

2. ಭಾಗಗಳನ್ನು ಬೇರ್ಪಡಿಸುವಿಕೆ (Extracting String Parts)

.slice(start, end)

ಸ್ಟ್ರಿಂಗ್‌ನ ಒಂದು ಭಾಗವನ್ನು ಕತ್ತರಿಸಿ, ಹೊಸ ಸ್ಟ್ರಿಂಗ್ ಆಗಿ ಹಿಂತಿರುಗಿಸುತ್ತದೆ. ಋಣಾತ್ಮಕ (negative) ಮೌಲ್ಯಗಳನ್ನು ಕೊಟ್ಟರೆ ಕೊನೆಯಿಂದ ಎಣಿಸುತ್ತದೆ.

let fruits = "Apple, Banana, Kiwi";
console.log(fruits.slice(7, 13)); // Output: "Banana"
console.log(fruits.slice(-4)); // Output: "Kiwi"

.substring(start, end)

`slice()` ನಂತೆಯೇ ಇರುತ್ತದೆ, ಆದರೆ ಇದು ಋಣಾತ್ಮಕ ಮೌಲ್ಯಗಳನ್ನು ಸ್ವೀಕರಿಸುವುದಿಲ್ಲ (ಅದನ್ನು 0 ಎಂದು ಪರಿಗಣಿಸುತ್ತದೆ).

let str = "Apple, Banana";
console.log(str.substring(7, 13)); // Output: "Banana"

.substr(start, length) Deprecated

ಎಲ್ಲಿಂದ ಶುರುಮಾಡಬೇಕು (start) ಮತ್ತು ಅಲ್ಲಿಂದ ಎಷ್ಟು ಅಕ್ಷರಗಳು ಬೇಕು (length) ಎಂಬುದನ್ನು ಆಧರಿಸಿ ಕತ್ತರಿಸುತ್ತದೆ. (ಗಮನಿಸಿ: ಇದು ಹಳೆಯ ವಿಧಾನ, ಈಗ `slice` ಬಳಸುವುದು ಉತ್ತಮ).

let str = "Apple, Banana";
console.log(str.substr(7, 6)); // Output: "Banana"

3. ಸ್ಟ್ರಿಂಗ್ ಮಾರ್ಪಡಿಸುವಿಕೆ (Modifying Strings)

.toUpperCase()

ಎಲ್ಲಾ ಅಕ್ಷರಗಳನ್ನು ದೊಡ್ಡಕ್ಷರಕ್ಕೆ (Capital letters) ಪರಿವರ್ತಿಸುತ್ತದೆ.

console.log("hello".toUpperCase()); // "HELLO"

.toLowerCase()

ಎಲ್ಲಾ ಅಕ್ಷರಗಳನ್ನು ಸಣ್ಣಕ್ಷರಕ್ಕೆ (Small letters) ಪರಿವರ್ತಿಸುತ್ತದೆ.

console.log("WORLD".toLowerCase()); // "world"

.concat(str1, str2...)

ಎರಡು ಅಥವಾ ಅದಕ್ಕಿಂತ ಹೆಚ್ಚು ಸ್ಟ್ರಿಂಗ್‌ಗಳನ್ನು ಒಟ್ಟಿಗೆ ಸೇರಿಸುತ್ತದೆ. (+ ಆಪರೇಟರ್ ಬದಲಿಗೆ ಬಳಸಬಹುದು).

let text1 = "Hello";
let text2 = "World";
console.log(text1.concat(" ", text2)); // "Hello World"

.trim() / .trimStart() / .trimEnd()

ಸ್ಟ್ರಿಂಗ್‌ನ ಆರಂಭ ಮತ್ತು ಅಂತ್ಯದಲ್ಲಿರುವ ಅನಗತ್ಯ ಖಾಲಿ ಜಾಗಗಳನ್ನು (Whitespace) ತೆಗೆದುಹಾಕುತ್ತದೆ.

let str = " Hello ";
console.log(str.trim()); // "Hello"
console.log(str.trimStart()); // "Hello "

.padStart() / .padEnd()

ಸ್ಟ್ರಿಂಗ್ ಅನ್ನು ನಿರ್ದಿಷ್ಟ ಉದ್ದಕ್ಕೆ ತರಲು ಆರಂಭದಲ್ಲಿ ಅಥವಾ ಕೊನೆಯಲ್ಲಿ ಅಕ್ಷರಗಳನ್ನು ಸೇರಿಸುತ್ತದೆ.

let num = "5";
console.log(num.padStart(3, "0")); // "005"
console.log(num.padEnd(3, "x")); // "5xx"

.repeat(count)

ಸ್ಟ್ರಿಂಗ್ ಅನ್ನು ನಿರ್ದಿಷ್ಟ ಬಾರಿ ಪುನರಾವರ್ತಿಸಿ ಹೊಸ ಸ್ಟ್ರಿಂಗ್ ನೀಡುತ್ತದೆ.

console.log("Ha!".repeat(3)); // "Ha!Ha!Ha!"

4. ಹುಡುಕುವಿಕೆ (Searching in Strings)

.indexOf() / .lastIndexOf()

ಪದವು ಮೊದಲ ಬಾರಿ ಅಥವಾ ಕೊನೆಯದಾಗಿ ಸಿಗುವ ಇಂಡೆಕ್ಸ್ ಅನ್ನು ನೀಡುತ್ತದೆ. ಪದ ಸಿಗದಿದ್ದರೆ `-1` ನೀಡುತ್ತದೆ.

let str = "locate please locate";
console.log(str.indexOf("locate")); // Output: 0
console.log(str.lastIndexOf("locate")); // Output: 14

.includes(searchString)

ಸ್ಟ್ರಿಂಗ್‌ನಲ್ಲಿ ನಿರ್ದಿಷ್ಟ ಪದವಿದೆಯೇ ಎಂದು ಪರಿಶೀಲಿಸುತ್ತದೆ ಮತ್ತು `true` ಅಥವಾ `false` ನೀಡುತ್ತದೆ.

let str = "Hello world";
console.log(str.includes("world")); // Output: true

.startsWith() / .endsWith()

ಸ್ಟ್ರಿಂಗ್ ನಿರ್ದಿಷ್ಟ ಪದದಿಂದ ಆರಂಭವಾಗುತ್ತದೆಯೇ ಅಥವಾ ಕೊನೆಗೊಳ್ಳುತ್ತದೆಯೇ ಎಂದು ಪರಿಶೀಲಿಸುತ್ತದೆ.

let str = "John Doe";
console.log(str.startsWith("John")); // Output: true
console.log(str.endsWith("oe")); // Output: true

5. ಬದಲಾಯಿಸುವ ಮತ್ತು ವಿಭಜಿಸುವಿಕೆ (Replacing & Splitting)

.replace(search, replaceWith)

ಮೊದಲ ಬಾರಿಗೆ ಮ್ಯಾಚ್ ಆಗುವ ಪದವನ್ನು ಮಾತ್ರ ಹೊಸ ಪದದಿಂದ ಬದಲಾಯಿಸುತ್ತದೆ.

let str = "I love cats. cats are nice.";
console.log(str.replace("cats", "dogs"));
// Output: "I love dogs. cats are nice."

.replaceAll(search, replaceWith) ES2021

ಸ್ಟ್ರಿಂಗ್‌ನಲ್ಲಿರುವ ಎಲ್ಲಾ ಮ್ಯಾಚ್‌ಗಳನ್ನು ಬದಲಾಯಿಸುತ್ತದೆ.

let str = "I love cats. cats are nice.";
console.log(str.replaceAll("cats", "dogs"));
// Output: "I love dogs. dogs are nice."

.split(separator)

ನಿರ್ದಿಷ್ಟ ಅಕ್ಷರದ ಆಧಾರದ ಮೇಲೆ ಸ್ಟ್ರಿಂಗ್ ಅನ್ನು ತುಂಡರಿಸಿ, ಅರೇ (Array) ಆಗಿ ಪರಿವರ್ತಿಸುತ್ತದೆ.

let str = "a,b,c,d";
console.log(str.split(",")); // Output: ["a", "b", "c", "d"]
console.log(str.split("")); // Output: ["a", ",", "b", ",", "c", ",", "d"]

6. Regex ಆಧಾರಿತ ವಿಧಾನಗಳು (Regular Expressions)

.search(regex)

Regex ಮೂಲಕ ಹುಡುಕಿ, ಪದವು ಇರುವ ಮೊದಲ ಇಂಡೆಕ್ಸ್ ಅನ್ನು ನೀಡುತ್ತದೆ.

let str = "Visit W3Schools!";
console.log(str.search(/w3schools/i)); // Output: 6 (i = case insensitive)

.match(regex)

ಮ್ಯಾಚ್ ಆಗುವ ಡೇಟಾವನ್ನು ಅರೇ ರೂಪದಲ್ಲಿ ಹಿಂತಿರುಗಿಸುತ್ತದೆ. Regex ನಲ್ಲಿ `g` (global) ಫ್ಲ್ಯಾಗ್ ಬಳಸಿದರೆ ಎಲ್ಲಾ ಮ್ಯಾಚ್‌ಗಳನ್ನು ನೀಡುತ್ತದೆ.

let str = "ain, ain, and ain";
console.log(str.match(/ain/g)); // Output: ["ain", "ain", "ain"]

.matchAll(regex)

ಎಲ್ಲಾ ಮ್ಯಾಚ್ ಆಗುವ ಡೇಟಾಗಳನ್ನು ವಿಸ್ತೃತ ಮಾಹಿತಿಯೊಂದಿಗೆ ಐಟರೇಟರ್ (iterator) ರೂಪದಲ್ಲಿ ನೀಡುತ್ತದೆ. (Regex `g` ಫ್ಲ್ಯಾಗ್ ಕಡ್ಡಾಯ).

let str = "test1 test2";
let results = [...str.matchAll(/test\d/g)];
console.log(results[0][0]); // Output: "test1"

7. ಸುಧಾರಿತ ವಿಧಾನಗಳು (Advanced Methods)

Template Literals (ಬ್ಯಾಕ್‌ಟಿಕ್ಸ್)

ಇದು ಮೆಥಡ್ ಅಲ್ಲದಿದ್ದರೂ, ಸ್ಟ್ರಿಂಗ್ ಒಳಗೆ ವೇರಿಯೇಬಲ್‌ಗಳನ್ನು ನೇರವಾಗಿ ಬಳಸಲು (Interpolation) ಅನುಮತಿಸುತ್ತದೆ.

let name = "Ravi";
let msg = `Hello ${name}, Welcome!`; // "Hello Ravi, Welcome!"

.localeCompare(compareString)

ಭಾಷೆಯ ನಿಯಮಗಳ ಆಧಾರದ ಮೇಲೆ ಸ್ಟ್ರಿಂಗ್‌ಗಳನ್ನು ಆಲ್ಫಾಬೆಟಿಕಲ್ ಆಗಿ ವಿಂಗಡಿಸಲು ಬಳಸಲಾಗುತ್ತದೆ.

console.log("apple".localeCompare("banana")); // Output: -1 (apple ಮೊದಲು ಬರುತ್ತದೆ)

String.fromCharCode()

ಯುನಿಕೋಡ್ ಸಂಖ್ಯೆಗಳನ್ನು ಅಕ್ಷರಗಳಾಗಿ ಪರಿವರ್ತಿಸುತ್ತದೆ.

String.fromCharCode(65); // "A"

String.fromCodePoint()

`fromCharCode` ನಂತೆಯೇ, ಆದರೆ ಎಮೋಜಿಗಳನ್ನು ಬೆಂಬಲಿಸುತ್ತದೆ.

String.fromCodePoint(128512); // "😀"

.normalize()

ಸ್ಟ್ರಿಂಗ್‌ನ ಯುನಿಕೋಡ್ ರೂಪವನ್ನು ಸಾಮಾನ್ಯೀಕರಿಸುತ್ತದೆ.

"\u00F1".normalize(); // "ñ"

String.raw``

ಎಸ್ಕೇಪ್ ಅಕ್ಷರಗಳನ್ನು (ಉದಾ: \n) ನಿರ್ಲಕ್ಷಿಸುತ್ತದೆ.

String.raw`Hi\nThere`; // "Hi\nThere"

.toString() / .valueOf()

ಸ್ಟ್ರಿಂಗ್‌ನ ಮೂಲ (primitive) ಮೌಲ್ಯವನ್ನು ನೀಡುತ್ತವೆ.

let s = new String("A");
s.valueOf(); // "A"

.isWellFormed() / .toWellFormed() ES2024

ಸರಿಯಾದ ಯುನಿಕೋಡ್ ಇದೆಯೇ ಎಂದು ಪರೀಕ್ಷಿಸಿ/ಸರಿಪಡಿಸುತ್ತದೆ.

"abc".isWellFormed(); // true
