1. ಪೀಠಿಕೆ: ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ `this` ಎಂದರೆ ಏನು?

ಮೂಲ ನೋಟ್ಸ್‌ನಿಂದ (From Original Notes)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ this ಕೀವರ್ಡ್ ಒಂದು "ಬಣ್ಣ ಬದಲಿಸುವ ಓತಿಕ್ಯಾತ" (Chameleon) ಇದ್ದಂತೆ. ನೀವು ಅದನ್ನು ಎಲ್ಲಿ ಮತ್ತು ಹೇಗೆ ಬಳಸುತ್ತೀರಿ ಎಂಬುದರ ಆಧಾರದ ಮೇಲೆ ಅದರ ಅರ್ಥ ಸಂಪೂರ್ಣವಾಗಿ ಬದಲಾಗುತ್ತದೆ. ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿ this ಎಂದರೆ ಸಾಮಾನ್ಯವಾಗಿ "ನನ್ನ" (My) ಅಥವಾ "ನಾನು ಸೇರಿರುವ ವಸ್ತು" ಎಂದರ್ಥ.

ಹೆಚ್ಚುವರಿ ಮಾಹಿತಿ (Technical Concept)

ಅದು ಏಕೆ ಅಗತ್ಯ (Why is it needed)?
ಒಂದೇ ಫಂಕ್ಷನ್ ಅನ್ನು ಬೇರೆ ಬೇರೆ ಆಬ್ಜೆಕ್ಟ್‌ಗಳಿಗೆ ಮರುಬಳಕೆ (Reuse) ಮಾಡಲು this ಸಹಾಯ ಮಾಡುತ್ತದೆ. ಪ್ರತಿ ಬಾರಿಯೂ ಆಬ್ಜೆಕ್ಟ್‌ನ ಹೆಸರನ್ನು ಹಾರ್ಡ್‌ಕೋಡ್ (Hardcode) ಮಾಡುವ ಬದಲು, this ಬಳಸುವುದರಿಂದ ಕೋಡ್ ಚಲನಶೀಲ (Dynamic) ಆಗುತ್ತದೆ.

this ಕೀವರ್ಡ್ ಪ್ರಮುಖವಾಗಿ ಮೂರು ಬೇರೆ ಬೇರೆ ಸನ್ನಿವೇಶಗಳಲ್ಲಿ ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ ಎಂಬುದನ್ನು ನಾವು ಈಗ ಹಂತ-ಹಂತವಾಗಿ ನೋಡೋಣ.

2. Object ಒಳಗೆ `this` (ಆಬ್ಜೆಕ್ಟ್ ಮೆಥಡ್)

ಹಂತ 1 - ಸರಳ ವಿವರಣೆ (Simple)

ಒಂದು ಆಬ್ಜೆಕ್ಟ್‌ನ (ವಸ್ತುವಿನ) ಒಳಗಿರುವ ಫಂಕ್ಷನ್‌ನಲ್ಲಿ this ಅನ್ನು ಬಳಸಿದರೆ, ಅದು ಅದೇ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಸೂಚಿಸುತ್ತದೆ. ಅಂದರೆ, "ನಾನು ಸೇರಿರುವ ಆಬ್ಜೆಕ್ಟ್‌ನ ಗುಣಲಕ್ಷಣ" ಎಂದರ್ಥ.

JavaScript - Object Method 
let employee = {
  name: "ರಮೇಶ್",
  salary: 50000,
  
  showDetails: function() {
    // ಇಲ್ಲಿ 'this' ಎಂದರೆ 'employee' ಆಬ್ಜೆಕ್ಟ್
    console.log("ನನ್ನ ಹೆಸರು " + this.name + ", ನನ್ನ ಸಂಬಳ " + this.salary);
  }
};

employee.showDetails(); 
// ಔಟ್‌ಪುಟ್: ನನ್ನ ಹೆಸರು ರಮೇಶ್, ನನ್ನ ಸಂಬಳ 50000
ಹಂತ 2 - Technical Explanation

ಇಲ್ಲಿ showDetails ಎಂಬುದು employee ಆಬ್ಜೆಕ್ಟ್‌ನ ಮೆಥಡ್ (Method - ಆಬ್ಜೆಕ್ಟ್ ಒಳಗೆ ಇರುವ ಫಂಕ್ಷನ್). ಹಾಗಾಗಿ this.name ಎಂದರೆ, ನೇರವಾಗಿ employee.name (ರಮೇಶ್) ಎಂದೇ ಅರ್ಥ. ಫಂಕ್ಷನ್ ಅನ್ನು ಯಾರು ಕರೆಯುತ್ತಿದ್ದಾರೆ (Who is calling the function?) ಎಂಬುದರ ಮೇಲೆ this ನಿರ್ಧಾರವಾಗುತ್ತದೆ. ಇಲ್ಲಿ ಫಂಕ್ಷನ್ ಕರೆಯುತ್ತಿರುವುದು employee.

3. ಸಾಮಾನ್ಯ Function ಒಳಗೆ `this` (Standalone Function)

ಹಂತ 1 - ಸರಳ ವಿವರಣೆ (Simple)

ಒಂದು ಫಂಕ್ಷನ್ ಯಾವುದೇ ಆಬ್ಜೆಕ್ಟ್ ಒಳಗೆ ಇಲ್ಲದೆ, ಪ್ರೋಗ್ರಾಮ್‌ನಲ್ಲಿ ಸುಮ್ಮನೆ ಹೊರಗಡೆ ಇದ್ದರೆ, ಆಗ this ಅನ್ನು ಬಳಸಿದರೆ ಅದು "ಗ್ಲೋಬಲ್ ಆಬ್ಜೆಕ್ಟ್" (Global Object) ಅನ್ನು ಸೂಚಿಸುತ್ತದೆ. ಬ್ರೌಸರ್‌ನಲ್ಲಿ ರನ್ ಮಾಡುವಾಗ ಈ ಗ್ಲೋಬಲ್ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು Window ಎನ್ನುತ್ತಾರೆ.

JavaScript - Standalone Function 
function sayHello() {
  // ಈ ಫಂಕ್ಷನ್ ಯಾವುದೇ ಆಬ್ಜೆಕ್ಟ್ ಒಳಗೆ ಇಲ್ಲ.
  console.log(this); 
}

sayHello(); 
// ಔಟ್‌ಪುಟ್: Window {...} (ಇಡೀ ಬ್ರೌಸರ್‌ನ ವಿಂಡೋ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಪ್ರಿಂಟ್ ಮಾಡುತ್ತದೆ)
ಹಂತ 2 - Technical Explanation

ಇಲ್ಲಿ sayHello ಫಂಕ್ಷನ್‌ಗೆ ಯಾವುದೇ ನಿರ್ದಿಷ್ಟ ಯಜಮಾನ (Object) ಇಲ್ಲ. ಆದ್ದರಿಂದ this ಎನ್ನುವುದು ಇಡೀ ಪ್ರೋಗ್ರಾಮ್‌ನ ಅತಿ ದೊಡ್ಡ ಯಜಮಾನನಾದ (Global Object / Window Object) ಕಡೆ ಬೆರಳು ತೋರಿಸುತ್ತದೆ.

ವೃತ್ತಿಪರ ಎಚ್ಚರಿಕೆ (Strict Mode Warning)

ಒಂದು ವೇಳೆ ನಿಮ್ಮ ಕೋಡ್‌ನ ಆರಂಭದಲ್ಲಿ "use strict"; ಎಂದು ಬರೆದಿದ್ದರೆ, ಸಾಮಾನ್ಯ ಫಂಕ್ಷನ್ ಒಳಗೆ this ಗ್ಲೋಬಲ್ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಸೂಚಿಸುವುದಿಲ್ಲ. ಬದಲಾಗಿ ಅದು undefined ಆಗಿರುತ್ತದೆ. ಇದು ತಪ್ಪುಗಳನ್ನು ತಡೆಯಲು JavaScript ನಲ್ಲಿರುವ ಭದ್ರತಾ ವ್ಯವಸ್ಥೆ.

4. Constructor ಒಳಗೆ `this` (ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸುವಾಗ)

ಹಂತ 1 - ಸರಳ ವಿವರಣೆ (Simple)

ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಎಂದರೆ ಒಂದೇ ರೀತಿಯ ಹಲವು ಆಬ್ಜೆಕ್ಟ್‌ಗಳನ್ನು ಸೃಷ್ಟಿಸಲು ಬಳಸುವ ಒಂದು 'ಅಚ್ಚು' (Blueprint). ನೀವು new ಕೀವರ್ಡ್ ಬಳಸಿ ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಮೂಲಕ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸುವಾಗ, this ಎಂದರೆ "ಹೊಸದಾಗಿ ಹುಟ್ಟುತ್ತಿರುವ ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್" ಎಂದರ್ಥ.

JavaScript - Constructor Function 
// ಇದು ಕನ್ಸ್ಟ್ರಕ್ಟರ್ (ಅಚ್ಚು)
function BankAccount(accountHolderName, depositAmount) {
  // ಇಲ್ಲಿ 'this' ಎಂದರೆ, ಹೊಸದಾಗಿ ಹುಟ್ಟುವ ಖಾಲಿ ಅಕೌಂಟ್ ಡಬ್ಬಿ
  this.name = accountHolderName;
  this.balance = depositAmount;
}

// 'new' ಕೀವರ್ಡ್ ಬಳಸಿ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸುತ್ತಿದ್ದೇವೆ
let customer1 = new BankAccount("ಸೀತಾ", 10000);
let customer2 = new BankAccount("ರಾಮು", 5000);

console.log(customer1.name);
// ಔಟ್‌ಪುಟ್: ಸೀತಾ
console.log(customer2.balance);
// ಔಟ್‌ಪುಟ್: 5000
ಹಂತ 2 - Technical Explanation

ನೀವು new BankAccount("ಸೀತಾ", 10000) ಎಂದಾಗ, ಕಂಪ್ಯೂಟರ್‌ನ ಆಂತರಿಕ ಕಾರ್ಯವಿಧಾನ (Internal Working) ಹೀಗಿರುತ್ತದೆ:

  1. ಮೊದಲು ಒಂದು ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್ {} ಸೃಷ್ಟಿಯಾಗುತ್ತದೆ.
  2. ಆ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನೇ ಇಲ್ಲಿ this ಎಂದು ಕರೆಯಲಾಗುತ್ತದೆ.
  3. ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಆ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ಹೆಸರು ಮತ್ತು ಹಣವನ್ನು ತುಂಬುತ್ತದೆ.
  4. ಕೊನೆಯಲ್ಲಿ ಆ ಆಬ್ಜೆಕ್ಟ್ ತಾನಾಗಿಯೇ return ಆಗುತ್ತದೆ.

`this` ನ ಬಣ್ಣ ಬದಲಾವಣೆ (Visual Diagram)

5. ನೈಜ ಉದಾಹರಣೆಯ ಸಂಪೂರ್ಣ ವಿಶ್ಲೇಷಣೆ (`person` ಆಬ್ಜೆಕ್ಟ್)

ಈಗ ಮೇಲಿನ ನಿಯಮಗಳನ್ನು ಇಟ್ಟುಕೊಂಡು, ಈ ಕೆಳಗಿನ ಕೋಡ್ ಹೇಗೆ ಹಂತ-ಹಂತವಾಗಿ ಕೆಲಸ ಮಾಡುತ್ತದೆ ಎಂದು ನೋಡೋಣ.

JavaScript - Complex Object 
const person = {
  firstName: "Mani",
  age: 25,
  
  getBirthDay: function () {
    return 2020 - this.age;
  },
  
  hobbies: ["eat", "sleep", "code"],
  
  getHobbies: function () {
    this.hobbies.forEach(function (hobby) {
      console.log(this.firstName, hobby);
    }, this); // ಗಮನಿಸಿ: ಇಲ್ಲಿ ಕೊನೆಯಲ್ಲಿ 'this' ಕಳುಹಿಸಲಾಗಿದೆ
  }
};

ಕೋಡ್‌ನ ಹಂತ-ಹಂತದ ವಿವರಣೆ (Execution Flow)

ಹಂತ 1: ಆಬ್ಜೆಕ್ಟ್ ಮತ್ತು ಗುಣಲಕ್ಷಣಗಳು (Object & Properties)

  • const person = {: ಇಲ್ಲಿ person ಎಂಬ ಹೆಸರಿನ ಒಂದು ಮುಖ್ಯ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಸೃಷ್ಟಿಸಲಾಗುತ್ತಿದೆ.
  • firstName: "Mani" & age: 25: ಇವು ಆಬ್ಜೆಕ್ಟ್‌ನ ಗುಣಲಕ್ಷಣಗಳು (Properties). ಈ ವ್ಯಕ್ತಿಯ ಹೆಸರು ಮಣಿ ಮತ್ತು ವಯಸ್ಸು 25.

ಹಂತ 2: ಮೊದಲ ಮೆಥಡ್ (getBirthDay)

  • getBirthDay: function () {: ವ್ಯಕ್ತಿಯು ಹುಟ್ಟಿದ ವರ್ಷವನ್ನು ಲೆಕ್ಕಹಾಕಲು ಬರೆದ ಫಂಕ್ಷನ್ ಇದು.
  • return 2020 - this.age;: ಇಲ್ಲಿ this ಎಂದರೆ person ಆಬ್ಜೆಕ್ಟ್. ಹಾಗಾಗಿ this.age ಎಂದರೆ 25. ಇದು 2020 ರಿಂದ 25 ನ್ನು ಕಳೆದು 1995 ಎಂಬ ಉತ್ತರವನ್ನು ನೀಡುತ್ತದೆ.

ಹಂತ 3: ಅರೇ (Array - ಪಟ್ಟಿ)

  • hobbies: ["eat", "sleep", "code"]: ಇದು ಹವ್ಯಾಸಗಳ ಪಟ್ಟಿ (Array). ಒಂದಕ್ಕಿಂತ ಹೆಚ್ಚು ಮಾಹಿತಿಗಳನ್ನು ಒಂದೇ ಕಡೆ ಸೇರಿಸಿಡಲು ಅರೇ ಬಳಸುತ್ತೇವೆ.

ಹಂತ 4: ಎರಡನೇ ಮೆಥಡ್ ಮತ್ತು ಲೂಪ್ (getHobbies)

  • getHobbies: function () {: ಹವ್ಯಾಸಗಳನ್ನು ಸ್ಕ್ರೀನ್ ಮೇಲೆ ತೋರಿಸಲು ಬರೆದ ಫಂಕ್ಷನ್.
  • this.hobbies.forEach(...): this.hobbies ಎಂದರೆ ಮೇಲಿರುವ ಹವ್ಯಾಸಗಳ ಪಟ್ಟಿ. forEach ಎಂದರೆ ಆ ಪಟ್ಟಿಯಲ್ಲಿರುವ ಪ್ರತಿಯೊಂದು ಹವ್ಯಾಸಕ್ಕೂ ಲೂಪ್ ಮಾಡು (ಒಂದೊಂದಾಗಿ ಹೋಗು) ಎಂದು ಅರ್ಥ.
ಹಂತ 3 - Professional Explanation (ಪ್ರೊ-ಲೆವೆಲ್ ರಹಸ್ಯ)

Callback ದೋಷವನ್ನು ತಪ್ಪಿಸುವುದು (Fixing Context Loss):

  • forEach ನ ಒಳಗೆ ಮತ್ತೊಂದು ಫಂಕ್ಷನ್ ಇದೆ: function (hobby) { console.log(this.firstName, hobby); }. ಇದನ್ನು ಕಾಲ್-ಬ್ಯಾಕ್ (Callback) ಫಂಕ್ಷನ್ ಎನ್ನುತ್ತಾರೆ.
  • ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ನಿಯಮದ ಪ್ರಕಾರ, ಈ ರೀತಿ ಒಂದು ಫಂಕ್ಷನ್‌ನ ಒಳಗೆ ಮತ್ತೊಂದು ಫಂಕ್ಷನ್ ಬರೆದಾಗ, ಆ ಒಳಗಿನ ಫಂಕ್ಷನ್‌ಗೆ ತನ್ನ ಮೂಲ 'ಯಜಮಾನ' (ಆಬ್ಜೆಕ್ಟ್) ಯಾರು ಎಂಬುದು ಮರೆತುಹೋಗುತ್ತದೆ. ಅದು ನೇರವಾಗಿ 'Global Window' ಅನ್ನು ನೋಡಲು ಶುರುಮಾಡುತ್ತದೆ. ಆಗ this.firstName ಎಂಬುದು ಸಿಗದೆ ದೋಷ (Error / undefined) ಉಂಟಾಗುತ್ತದೆ.
  • ಪರಿಹಾರ: ಈ ಗೊಂದಲವನ್ನು ತಪ್ಪಿಸಲು, forEach ನ ಬ್ರಾಕೆಟ್ ಮುಗಿಯುವ ಮುನ್ನ ಕಾಮಾ (,) ಹಾಕಿ this ಅನ್ನು ಎರಡನೇ ಆರ್ಗ್ಯುಮೆಂಟ್ ಆಗಿ }, this); ಕಳುಹಿಸಲಾಗಿದೆ.
  • ಇದು "ಮರೆಯಬೇಡ, ನಿನ್ನ ಯಜಮಾನ person ಆಬ್ಜೆಕ್ಟ್" ಎಂದು ಒಳಗಿನ ಫಂಕ್ಷನ್‌ಗೆ ನೆನಪಿಸುವ ಕೆಲಸ ಮಾಡುತ್ತದೆ. ಇದರಿಂದ ಕೋಡ್ ಸರಿಯಾಗಿ "Mani eat", "Mani sleep", "Mani code" ಎಂದು ಔಟ್‌ಪುಟ್ ನೀಡುತ್ತದೆ.
ಆಧುನಿಕ ತಂತ್ರಜ್ಞಾನ (Modern ES6 JavaScript)

6. Arrow Functions ಮತ್ತು `this` (Modern Approach)

ಮೂಲ ನೋಟ್ಸ್‌ನಲ್ಲಿ ವಿವರಿಸಿರುವ Callback ದೋಷವನ್ನು ಸರಿಪಡಿಸಲು ಆಧುನಿಕ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Arrow Functions () => {} ಅನ್ನು ಬಳಸಲಾಗುತ್ತದೆ.

Arrow ಫಂಕ್ಷನ್‌ಗಳಿಗೆ ತಮ್ಮದೇ ಆದ this ಇರುವುದಿಲ್ಲ. ಅವು ತಮ್ಮ ಸುತ್ತಮುತ್ತಲಿನ (Lexical scope) this ಅನ್ನು ಹಾಗೆಯೇ ಉಳಿಸಿಕೊಳ್ಳುತ್ತವೆ. ಹಾಗಾಗಿ ನೀವು ಎರಡನೇ ಆರ್ಗ್ಯುಮೆಂಟ್ ಆಗಿ this ಕಳುಹಿಸುವ ಅವಶ್ಯಕತೆ ಇರುವುದಿಲ್ಲ.

Modern JS - Arrow Function 
getHobbies: function () {
  // Arrow function ಬಳಸಿರುವುದರಿಂದ 'this' ದೋಷ ಉಂಟಾಗುವುದಿಲ್ಲ
  this.hobbies.forEach((hobby) => {
    console.log(this.firstName, hobby);
  }); 
}

ಮೂಲ ನೋಟ್ಸ್‌ನ ಕೊನೆಯ ಪ್ರಶ್ನೆ:
"ಈಗ ನಿಮಗೆ ಕಾಲ್-ಬ್ಯಾಕ್ ಫಂಕ್ಷನ್ (Callback function) ಮತ್ತು ಅದರಲ್ಲಿ ಉಂಟಾಗುವ ಗೊಂದಲಗಳ ಬಗ್ಗೆ ಸ್ಪಷ್ಟ ಚಿತ್ರಣ ಸಿಕ್ಕಿತೇ ಅಥವಾ ಬೇರೆ ಯಾವುದಾದರೂ ಲೂಪ್ ಉದಾಹರಣೆ (ಉದಾಹರಣೆಗೆ for-loop) ಮೂಲಕ ವಿವರಿಸಬೇಕೇ?"

ಉತ್ತರ: ಸಾಮಾನ್ಯ for-loop (ಉದಾ: for(let i=0; i<length; i++)) ಬಳಸಿದರೆ ಅದು ಪ್ರತ್ಯೇಕ ಫಂಕ್ಷನ್ ಸೃಷ್ಟಿಸುವುದಿಲ್ಲ, ಆದ್ದರಿಂದ ಅಲ್ಲಿ this ನ ಗೊಂದಲ ಇರುವುದಿಲ್ಲ. this ನ ಗೊಂದಲ ಕೇವಲ Callback ಫಂಕ್ಷನ್‌ಗಳಲ್ಲಿ (forEach, map, setTimeout) ಮಾತ್ರ ಬರುತ್ತದೆ.

7. Original Notes Archive (ಮೂಲ ನೋಟ್ಸ್)

ಯಾವುದೇ ಬದಲಾವಣೆ ಇಲ್ಲದೆ ಮೂಲ ಬರಹವನ್ನು ಇಲ್ಲಿ ಸಂರಕ್ಷಿಸಲಾಗಿದೆ (Zero Content Loss).

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ this ಕೀವರ್ಡ್ ಮತ್ತು ಅದರ ಬಳಕೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ this ಕೀವರ್ಡ್ ಒಂದು "ಬಣ್ಣ ಬದಲಿಸುವ ಓತಿಕ್ಯಾತ" (Chameleon) ಇದ್ದಂತೆ. ನೀವು ಅದನ್ನು ಎಲ್ಲಿ ಮತ್ತು ಹೇಗೆ ಬಳಸುತ್ತೀರಿ ಎಂಬುದರ ಆಧಾರದ ಮೇಲೆ ಅದರ ಅರ್ಥ ಸಂಪೂರ್ಣವಾಗಿ ಬದಲಾಗುತ್ತದೆ. ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿ this ಎಂದರೆ ಸಾಮಾನ್ಯವಾಗಿ "ನನ್ನ" (My) ಅಥವಾ "ನಾನು ಸೇರಿರುವ ವಸ್ತು" ಎಂದರ್ಥ. this ಕೀವರ್ಡ್ ಪ್ರಮುಖವಾಗಿ ಮೂರು ಬೇರೆ ಬೇರೆ ಸನ್ನಿವೇಶಗಳಲ್ಲಿ ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ ಎಂಬುದನ್ನು ಮೊದಲು ತಿಳಿದುಕೊಳ್ಳೋಣ. 1. Object ಒಳಗೆ this (ಆಬ್ಜೆಕ್ಟ್ ಮೆಥಡ್) ಒಂದು ಆಬ್ಜೆಕ್ಟ್‌ನ (ವಸ್ತುವಿನ) ಒಳಗಿರುವ ಫಂಕ್ಷನ್‌ನಲ್ಲಿ this ಅನ್ನು ಬಳಸಿದರೆ, ಅದು ಅದೇ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಸೂಚಿಸುತ್ತದೆ. ಅಂದರೆ, "ನಾನು ಸೇರಿರುವ ಆಬ್ಜೆಕ್ಟ್‌ನ ಗುಣಲಕ್ಷಣ" ಎಂದರ್ಥ. JavaScript let employee = { name: "ರಮೇಶ್", salary: 50000, showDetails: function() { // ಇಲ್ಲಿ 'this' ಎಂದರೆ 'employee' ಆಬ್ಜೆಕ್ಟ್ console.log("ನನ್ನ ಹೆಸರು " + this.name + ", ನನ್ನ ಸಂಬಳ " + this.salary); } }; employee.showDetails(); // ಔಟ್‌ಪುಟ್: ನನ್ನ ಹೆಸರು ರಮೇಶ್, ನನ್ನ ಸಂಬಳ 50000 ಇಲ್ಲಿ showDetails ಎಂಬುದು employee ಆಬ್ಜೆಕ್ಟ್‌ನ ಸ್ವತ್ತು. ಹಾಗಾಗಿ this.name ಎಂದರೆ, ನೇರವಾಗಿ employee.name (ರಮೇಶ್) ಎಂದೇ ಅರ್ಥ. 2. ಸಾಮಾನ್ಯ Function ಒಳಗೆ this (Standalone Function) ಒಂದು ಫಂಕ್ಷನ್ ಯಾವುದೇ ಆಬ್ಜೆಕ್ಟ್ ಒಳಗೆ ಇಲ್ಲದೆ, ಪ್ರೋಗ್ರಾಮ್‌ನಲ್ಲಿ ಸುಮ್ಮನೆ ಹೊರಗಡೆ ಇದ್ದರೆ, ಆಗ this ಅನ್ನು ಬಳಸಿದರೆ ಅದು "ಗ್ಲೋಬಲ್ ಆಬ್ಜೆಕ್ಟ್" (Global Object) ಅನ್ನು ಸೂಚಿಸುತ್ತದೆ. ಬ್ರೌಸರ್‌ನಲ್ಲಿ ರನ್ ಮಾಡುವಾಗ ಈ ಗ್ಲೋಬಲ್ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು 'Window' ಎನ್ನುತ್ತಾರೆ. JavaScript function sayHello() { // ಈ ಫಂಕ್ಷನ್ ಯಾವುದೇ ಆಬ್ಜೆಕ್ಟ್ ಒಳಗೆ ಇಲ್ಲ. console.log(this); } sayHello(); // ಔಟ್‌ಪುಟ್: Window {...} (ಇಡೀ ಬ್ರೌಸರ್‌ನ ವಿಂಡೋ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಪ್ರಿಂಟ್ ಮಾಡುತ್ತದೆ) ಇಲ್ಲಿ sayHello ಫಂಕ್ಷನ್‌ಗೆ ಯಾವುದೇ ನಿರ್ದಿಷ್ಟ ಯಜಮಾನ (Object) ಇಲ್ಲ. ಆದ್ದರಿಂದ this ಎನ್ನುವುದು ಇಡೀ ಪ್ರೋಗ್ರಾಮ್‌ನ ಅತಿ ದೊಡ್ಡ ಯಜಮಾನನಾದ (Global Object) ಕಡೆ ಬೆರಳು ತೋರಿಸುತ್ತದೆ. 3. Constructor ಒಳಗೆ this (ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸುವಾಗ) ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಎಂದರೆ ಒಂದೇ ರೀತಿಯ ಹಲವು ಆಬ್ಜೆಕ್ಟ್‌ಗಳನ್ನು ಸೃಷ್ಟಿಸಲು ಬಳಸುವ ಒಂದು 'ಅಚ್ಚು' (Blueprint). ನೀವು new ಕೀವರ್ಡ್ ಬಳಸಿ ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಮೂಲಕ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸುವಾಗ, this ಎಂದರೆ "ಹೊಸದಾಗಿ ಹುಟ್ಟುತ್ತಿರುವ ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್" ಎಂದರ್ಥ. JavaScript // ಇದು ಕನ್ಸ್ಟ್ರಕ್ಟರ್ (ಅಚ್ಚು) function BankAccount(accountHolderName, depositAmount) { // ಇಲ್ಲಿ 'this' ಎಂದರೆ, ಹೊಸದಾಗಿ ಹುಟ್ಟುವ ಖಾಲಿ ಅಕೌಂಟ್ ಡಬ್ಬಿ this.name = accountHolderName; this.balance = depositAmount; } // 'new' ಕೀವರ್ಡ್ ಬಳಸಿ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಸೃಷ್ಟಿಸುತ್ತಿದ್ದೇವೆ let customer1 = new BankAccount("ಸೀತಾ", 10000); let customer2 = new BankAccount("ರಾಮು", 5000); console.log(customer1.name); // ಔಟ್‌ಪುಟ್: ಸೀತಾ console.log(customer2.balance); // ಔಟ್‌ಪುಟ್: 5000 ನೀವು new BankAccount("ಸೀತಾ", 10000) ಎಂದಾಗ, ಕಂಪ್ಯೂಟರ್ ಮೊದಲು ಒಂದು ಖಾಲಿ ಆಬ್ಜೆಕ್ಟ್ {} ಸೃಷ್ಟಿಸುತ್ತದೆ. ಆ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನೇ ಇಲ್ಲಿ this ಎಂದು ಕರೆಯಲಾಗುತ್ತಿದೆ. ಕನ್ಸ್ಟ್ರಕ್ಟರ್ ಆ ಹೊಸ ಆಬ್ಜೆಕ್ಟ್‌ಗೆ ಹೆಸರು ಮತ್ತು ಹಣವನ್ನು ತುಂಬುತ್ತದೆ. ನೈಜ ಉದಾಹರಣೆಯ ಸಂಪೂರ್ಣ ವಿಶ್ಲೇಷಣೆ (person ಆಬ್ಜೆಕ್ಟ್) ಈಗ ಮೇಲಿನ ನಿಯಮಗಳನ್ನು ಇಟ್ಟುಕೊಂಡು, ಈ ಕೆಳಗಿನ ಕೋಡ್ ಹೇಗೆ ಹಂತ-ಹಂತವಾಗಿ ಕೆಲಸ ಮಾಡುತ್ತದೆ ಎಂದು ನೋಡೋಣ. JavaScript const person = { firstName: "Mani", age: 25, getBirthDay: function () { return 2020 - this.age; }, hobbies: ["eat", "sleep", "code"], getHobbies: function () { this.hobbies.forEach(function (hobby) { console.log(this.firstName, hobby); }, this); } }; ಹಂತ 1: ಆಬ್ಜೆಕ್ಟ್ ಮತ್ತು ಗುಣಲಕ್ಷಣಗಳು (Object & Properties) - const person = {: ಇಲ್ಲಿ person ಎಂಬ ಹೆಸರಿನ ಒಂದು ಮುಖ್ಯ ಆಬ್ಜೆಕ್ಟ್ ಅನ್ನು ಸೃಷ್ಟಿಸಲಾಗುತ್ತಿದೆ. - firstName: "Mani" & age: 25: ಇವು ಆಬ್ಜೆಕ್ಟ್‌ನ ಗುಣಲಕ್ಷಣಗಳು (Properties). ಈ ವ್ಯಕ್ತಿಯ ಹೆಸರು ಮಣಿ ಮತ್ತು ವಯಸ್ಸು 25. ಹಂತ 2: ಮೊದಲ ಮೆಥಡ್ (getBirthDay) - getBirthDay: function () {: ವ್ಯಕ್ತಿಯು ಹುಟ್ಟಿದ ವರ್ಷವನ್ನು ಲೆಕ್ಕಹಾಕಲು ಬರೆದ ಫಂಕ್ಷನ್ ಇದು. - return 2020 - this.age;: ಇಲ್ಲಿ this ಎಂದರೆ person ಆಬ್ಜೆಕ್ಟ್. ಹಾಗಾಗಿ this.age ಎಂದರೆ 25. ಇದು 2020 ರಿಂದ 25 ನ್ನು ಕಳೆದು 1995 ಎಂಬ ಉತ್ತರವನ್ನು ನೀಡುತ್ತದೆ. ಹಂತ 3: ಅರೇ (Array - ಪಟ್ಟಿ) - hobbies: ["eat", "sleep", "code"]: ಇದು ಹವ್ಯಾಸಗಳ ಪಟ್ಟಿ (Array). ಒಂದಕ್ಕಿಂತ ಹೆಚ್ಚು ಮಾಹಿತಿಗಳನ್ನು ಒಂದೇ ಕಡೆ ಸೇರಿಸಿಡಲು ಅರೇ ಬಳಸುತ್ತೇವೆ. ಹಂತ 4: ಎರಡನೇ ಮೆಥಡ್ ಮತ್ತು ಲೂಪ್ (getHobbies) - getHobbies: function () {: ಹವ್ಯಾಸಗಳನ್ನು ಸ್ಕ್ರೀನ್ ಮೇಲೆ ತೋರಿಸಲು ಬರೆದ ಫಂಕ್ಷನ್. - this.hobbies.forEach(...): this.hobbies ಎಂದರೆ ಮೇಲಿರುವ ಹವ್ಯಾಸಗಳ ಪಟ್ಟಿ. forEach ಎಂದರೆ ಆ ಪಟ್ಟಿಯಲ್ಲಿರುವ ಪ್ರತಿಯೊಂದು ಹವ್ಯಾಸಕ್ಕೂ ಲೂಪ್ ಮಾಡು (ಒಂದೊಂದಾಗಿ ಹೋಗು) ಎಂದು ಅರ್ಥ. ಹಂತ 5: ಪ್ರೊ-ಲೆವೆಲ್ ರಹಸ್ಯ (Callback ದೋಷವನ್ನು ತಪ್ಪಿಸುವುದು) - forEach ನ ಒಳಗೆ ಮತ್ತೊಂದು ಫಂಕ್ಷನ್ ಇದೆ: function (hobby) { console.log(this.firstName, hobby); }. ಇದನ್ನು ಕಾಲ್-ಬ್ಯಾಕ್ (Callback) ಫಂಕ್ಷನ್ ಎನ್ನುತ್ತಾರೆ. - ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ನಿಯಮದ ಪ್ರಕಾರ, ಈ ರೀತಿ ಒಂದು ಫಂಕ್ಷನ್‌ನ ಒಳಗೆ ಮತ್ತೊಂದು ಫಂಕ್ಷನ್ ಬರೆದಾಗ, ಆ ಒಳಗಿನ ಫಂಕ್ಷನ್‌ಗೆ ತನ್ನ ಮೂಲ 'ಯಜಮಾನ' (ಆಬ್ಜೆಕ್ಟ್) ಯಾರು ಎಂಬುದು ಮರೆತುಹೋಗುತ್ತದೆ. ಅದು ನೇರವಾಗಿ 'Global Window' ಅನ್ನು ನೋಡಲು ಶುರುಮಾಡುತ್ತದೆ. ಆಗ this.firstName ಎಂಬುದು ಸಿಗದೆ ದೋಷ (Error) ಉಂಟಾಗುತ್ತದೆ. - ಪರಿಹಾರ (}, this);): ಈ ಗೊಂದಲವನ್ನು ತಪ್ಪಿಸಲು, forEach ನ ಬ್ರಾಕೆಟ್ ಮುಗಿಯುವ ಮುನ್ನ ಕಾಮಾ (,) ಹಾಕಿ this ಅನ್ನು ಎರಡನೇ ಆರ್ಗ್ಯುಮೆಂಟ್ ಆಗಿ ಕಳುಹಿಸಲಾಗಿದೆ. ಇದು "ಮರೆಯಬೇಡ, ನಿನ್ನ ಯಜಮಾನ person ಆಬ್ಜೆಕ್ಟ್" ಎಂದು ಒಳಗಿನ ಫಂಕ್ಷನ್‌ಗೆ ನೆನಪಿಸುವ ಕೆಲಸ ಮಾಡುತ್ತದೆ. ಇದರಿಂದ ಕೋಡ್ ಸರಿಯಾಗಿ "Mani eat", "Mani sleep", "Mani code" ಎಂದು ಔಟ್‌ಪುಟ್ ನೀಡುತ್ತದೆ. ಈ ರೀತಿಯಲ್ಲಿ this ಕೀವರ್ಡ್ ಆಬ್ಜೆಕ್ಟ್‌ಗಳ ಒಳಗೆ, ಹೊರಗೆ ಮತ್ತು ಲೂಪ್‌ಗಳ ಒಳಗೆ ತನ್ನ ವರ್ತನೆಯನ್ನು ಬದಲಾಯಿಸಿಕೊಳ್ಳುತ್ತಾ ಪ್ರೋಗ್ರಾಮ್ ಸರಿಯಾಗಿ ಕೆಲಸ ಮಾಡುವಂತೆ ನೋಡಿಕೊಳ್ಳುತ್ತದೆ. ಈಗ ನಿಮಗೆ ಕಾಲ್-ಬ್ಯಾಕ್ ಫಂಕ್ಷನ್ (Callback function) ಮತ್ತು ಅದರಲ್ಲಿ ಉಂಟಾಗುವ ಗೊಂದಲಗಳ ಬಗ್ಗೆ ಸ್ಪಷ್ಟ ಚಿತ್ರಣ ಸಿಕ್ಕಿತೇ ಅಥವಾ ಬೇರೆ ಯಾವುದಾದರೂ ಲೂಪ್ ಉದಾಹರಣೆ (ಉದಾಹರಣೆಗೆ for-loop) ಮೂಲಕ ವಿವರಿಸಬೇಕೇ?
