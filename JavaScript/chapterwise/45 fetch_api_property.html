ಪೀಠಿಕೆ ಭಾಗ 1: ಪಾರ್ಸೆಲ್ ಮೇಲಿರುವ ಚೀಟಿಗಳು ಭಾಗ 2: ಪಾರ್ಸೆಲ್ ಬಿಡಿಸುವ ವಿಧಾನಗಳು ಭಾಗ 3: ಸಂಪೂರ್ಣ ಉದಾಹರಣೆ

Fetch API: ಸಂಪೂರ್ಣ ಮಾರ್ಗದರ್ಶಿ (ಉದಾಹರಣೆಗಳೊಂದಿಗೆ)

Fetch API ಎಂದರೆ ಏನು?

ಇಂಟರ್ನೆಟ್‌ನಿಂದ ಅಥವಾ ಬೇರೆ ಫೈಲ್‌ನಿಂದ ಮಾಹಿತಿಯನ್ನು (Data) ತಂದುಕೊಡುವ 'ಸಪ್ಲೈಯರ್' ಅಥವಾ ಡೆಲಿವರಿ ವ್ಯವಸ್ಥೆಯೇ Fetch. ನೀವು ಆರ್ಡರ್ (Request) ಮಾಡಿದರೆ, ಅದು ನಿಮಗೆ ಪಾರ್ಸೆಲ್ (Response) ತಂದುಕೊಡುತ್ತದೆ.

ಭಾಗ 1: ಪಾರ್ಸೆಲ್ ಮೇಲಿರುವ ಚೀಟಿಗಳು (Properties of Response Object)

ಡೆಲಿವರಿ ಬಾಯ್ ತಂದ ಪಾರ್ಸೆಲ್ (Response) ಬಾಕ್ಸ್‌ನ ಮೇಲೆ ಕೆಲವು ಪ್ರಮುಖ ಮಾಹಿತಿಗಳಿರುತ್ತವೆ. ಪ್ರತಿಯೊಂದಕ್ಕೂ ಉದಾಹರಣೆ ಮತ್ತು ವಿವರಣೆ ಇಲ್ಲಿದೆ:

1. response.ok (ಆರ್ಡರ್ ಯಶಸ್ವಿಯಾಗಿದೆಯೇ?)

ವಿವರಣೆ: ಫೈಲ್ ಅಥವಾ ಡೇಟಾ ಯಶಸ್ವಿಯಾಗಿ ಸಿಕ್ಕರೆ ಇದು true ಆಗಿರುತ್ತದೆ. ಸಿಗದೆ ಹೋದರೆ (ಉದಾ: ಫೈಲ್ ಡಿಲೀಟ್ ಆಗಿದ್ದರೆ) false ಆಗಿರುತ್ತದೆ.

if (response.ok === true) {
    console.log("ಫೈಲ್ ಯಶಸ್ವಿಯಾಗಿ ಸಿಕ್ಕಿದೆ!");
}

2. response.status (ಇಂಟರ್ನೆಟ್‌ನ ರಹಸ್ಯ ಕೋಡ್)

ವಿವರಣೆ: ಇದು ಸರ್ವರ್ ನೀಡುವ ನಂಬರ್ ಕೋಡ್. 200 ಎಂದರೆ ಯಶಸ್ವಿ, 404 ಎಂದರೆ ಫೈಲ್ ಸಿಗಲಿಲ್ಲ, 500 ಎಂದರೆ ಸರ್ವರ್ ತೊಂದರೆ.

console.log(response.status); // ಔಟ್‌ಪುಟ್: 200 (ಎಲ್ಲವೂ ಸರಿಯಿದೆ ಎಂದರ್ಥ)

3. response.statusText (ಕೋಡ್‌ನ ಅರ್ಥ)

ವಿವರಣೆ: ಮೇಲಿನ ನಂಬರ್ ಕೋಡ್‌ನ ಅರ್ಥವನ್ನು ಇಂಗ್ಲಿಷ್‌ನಲ್ಲಿ ಹೇಳುತ್ತದೆ.

console.log(response.statusText); // ಔಟ್‌ಪುಟ್: "OK" (200 ಇದ್ದರೆ), ಅಥವಾ "Not Found" (404 ಇದ್ದರೆ)

4. response.url (ನಿಖರವಾದ ವಿಳಾಸ)

ವಿವರಣೆ: ನೀವು ಆರ್ಡರ್ ಮಾಡಿದ ಫೈಲ್ ಇಂಟರ್ನೆಟ್‌ನ ಯಾವ ವಿಳಾಸದಿಂದ (URL) ಬಂದಿದೆ ಎಂದು ಹೇಳುತ್ತದೆ.

console.log(response.url); // ಔಟ್‌ಪುಟ್: "https://example.com/student.json"

5. response.redirected (ಬೇರೆ ಕಡೆಯಿಂದ ಬಂದಿದೆಯೇ?)

ವಿವರಣೆ: ಹೋಟೆಲ್‌ನ ಒಂದು ಬ್ರಾಂಚ್ ಮುಚ್ಚಿದ್ದು, ಅವರು ಆಟೋಮ್ಯಾಟಿಕ್ ಆಗಿ ಪಕ್ಕದ ಬ್ರಾಂಚ್‌ನಿಂದ ಊಟ ಕಳುಹಿಸಿದರೆ ಇದು true ಆಗಿರುತ್ತದೆ. ನೇರವಾಗಿ ಅದೇ ವಿಳಾಸದಿಂದ ಬಂದರೆ false.

console.log(response.redirected); // ಔಟ್‌ಪುಟ್: false (ನೇರವಾಗಿ ಬಂದಿದೆ)

6. response.headers (ರಸೀದಿ / ಬಿಲ್)

ವಿವರಣೆ: ಪಾರ್ಸೆಲ್ ಒಳಗೆ ಇರುವ ಡೇಟಾ ಯಾವ ಪ್ರಕಾರದ್ದು (ಉದಾಹರಣೆಗೆ ಅದು ಟೆಕ್ಸ್ಟ್, ಅಥವಾ ಇಮೇಜ್?) ಎಂಬ ಮಾಹಿತಿಯನ್ನು ಇದು ಹೊಂದಿರುತ್ತದೆ.

console.log(response.headers.get("Content-Type")); // ಔಟ್‌ಪುಟ್: "application/json"

7. response.type (ಯಾವ ರೀತಿಯ ಪಾರ್ಸೆಲ್?)

ವಿವರಣೆ: ಡೇಟಾ ನಿಮ್ಮದೇ ವೆಬ್‌ಸೈಟ್‌ನಿಂದ ಬಂದರೆ basic ಎಂದು, ಬೇರೆ ಕಡೆಯ ವೆಬ್‌ಸೈಟ್‌ನಿಂದ (ಉದಾ: Google ನಿಂದ) ಬಂದರೆ cors ಎಂದು ತೋರಿಸುತ್ತದೆ.

console.log(response.type); // ಔಟ್‌ಪುಟ್: "basic"

8. response.bodyUsed (ಪಾರ್ಸೆಲ್ ಓದಿದ್ದೇವಾ?)

ವಿವರಣೆ: ಪಾರ್ಸೆಲ್ ಅನ್ನು ಈಗಾಗಲೇ ಒಡೆದು ಓದಲಾಗಿದೆಯೇ? ಓದಿದ್ದರೆ true, ಓದಿಲ್ಲ ಎಂದರೆ false. ನೆನಪಿಡಿ, ಒಂದು ಪಾರ್ಸೆಲ್ ಅನ್ನು ಒಮ್ಮೆ ಮಾತ್ರ ಓದಲು (Extract ಮಾಡಲು) ಸಾಧ್ಯ.

console.log(response.bodyUsed); // ಪಾರ್ಸೆಲ್ ಒಡೆಯುವ ಮುನ್ನ ಔಟ್‌ಪುಟ್: false

9. response.body (ಕಚ್ಚಾ ಮಾಹಿತಿ)

ವಿವರಣೆ: ಇದು ಪಾರ್ಸೆಲ್ ಒಳಗಿರುವ ಕಚ್ಚಾ (Raw ReadableStream) ಡೇಟಾ. ಇದನ್ನು ನಾವು ನೇರವಾಗಿ ಓದಲು ಸಾಧ್ಯವಿಲ್ಲ, ಬದಲಾಗಿ ಕೆಳಗಿನ ವಿಧಾನಗಳನ್ನು (Methods) ಬಳಸಬೇಕು.

ಭಾಗ 2: ಪಾರ್ಸೆಲ್ ಬಿಡಿಸುವ ವಿಧಾನಗಳು (Methods to read Data)

ಬಾಕ್ಸ್ ಮೇಲಿನ ಮಾಹಿತಿಯನ್ನು ನೋಡಿದ ನಂತರ, ಒಳಗಿನ ಮಾಹಿತಿಯನ್ನು ನಮ್ಮ ಭಾಷೆಗೆ ತಕ್ಕಂತೆ ಬಿಡಿಸಿ ಓದಲು 5 ಮುಖ್ಯ ವಿಧಾನಗಳಿವೆ:

  • 1. .json():

    ವ್ಯವಸ್ಥಿತ ಮಾಹಿತಿ (ವಿದ್ಯಾರ್ಥಿಗಳ ಹೆಸರು, ಊರು) ಅಥವಾ ಡೇಟಾಬೇಸ್ ಮಾಹಿತಿಯನ್ನು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಆಬ್ಜೆಕ್ಟ್ ಆಗಿ ಓದಲು. (ಅತಿ ಹೆಚ್ಚು ಬಳಕೆಯಾಗುವುದು).

    let data = await response.json();
console.log(data.name); // ಉದಾ: "ರಾಮ"

  • 2. .text():

    ಕೇವಲ ಸಾಧಾರಣ ಪಠ್ಯವಾಗಿ (Plain Text / ಕಥೆ, ಲೇಖನ) ಓದಲು.

    let textData = await response.text();
console.log(textData); // ಪೂರ್ತಿ ಲೇಖನ ಪ್ರಿಂಟ್ ಆಗುತ್ತದೆ

  • 3. .blob():

    ಚಿತ್ರಗಳು (Images), ಆಡಿಯೋ ಅಥವಾ ವಿಡಿಯೋ ಫೈಲ್‌ಗಳನ್ನು ಓದಲು (Blob = Binary Large Object).

    let imageBlob = await response.blob();
let imageUrl = URL.createObjectURL(imageBlob);
document.getElementById('myImage').src = imageUrl;

  • 4. .formData():

    ವೆಬ್‌ಸೈಟ್‌ನಲ್ಲಿ ಯಾರಾದರೂ ತುಂಬಿದ ಅರ್ಜಿಯ (Form) ಮಾಹಿತಿಯನ್ನು ಓದಲು.

    let formData = await response.formData();
console.log(formData.get("username")); // ಫಾರ್ಮ್‌ನ ಹೆಸರು ಓದಲು

  • 5. .arrayBuffer():

    ವೃತ್ತಿಪರ ಗೇಮಿಂಗ್ ಅಥವಾ ವಿಡಿಯೋ ಎಡಿಟಿಂಗ್‌ಗೆ ಬೇಕಾದ ಕಚ್ಚಾ ಬೈನರಿ ಡೇಟಾವನ್ನು ಓದಲು.

    let buffer = await response.arrayBuffer();
console.log(buffer.byteLength); // ಕಚ್ಚಾ ಡೇಟಾದ ಅಳತೆ (bytes ನಲ್ಲಿ)

ಭಾಗ 3: ಸಂಪೂರ್ಣ ಉದಾಹರಣೆ (ಎಲ್ಲಾ Properties ಜೊತೆಗೆ)

ಇಲ್ಲಿ Arrow function ಇಲ್ಲದೆ, ಸಾಂಪ್ರದಾಯಿಕ function() {} ಮತ್ತು async / await ಬಳಸಿ ಬರೆದ ವೃತ್ತಿಪರ ಕೋಡ್ ಇಲ್ಲಿದೆ. ಇದು student.json ಎಂಬ ಫೈಲ್ ಅನ್ನು ತರುತ್ತದೆ ಎಂದು ಭಾವಿಸೋಣ. 

// 1. async ಬಳಸಿ ಪೂರ್ಣ ಫಂಕ್ಷನ್ ರಚಿಸುವುದು
async function getStudentInformation() {
    
    // 'try' ಬ್ಲಾಕ್: ಎಲ್ಲವೂ ಸರಿಯಾಗಿ ನಡೆದರೆ ಇದು ಕೆಲಸ ಮಾಡುತ್ತದೆ
    try {
        console.log("ಡೇಟಾ ತರಲು ಶುರು ಮಾಡಲಾಗಿದೆ...");

        // ಹಂತ 1: ಸಪ್ಲೈಯರ್ 'student.json' ಫೈಲ್ ತರುವವರೆಗೆ 'ಕಾಯಿರಿ' (await)
        let response = await fetch('student.json');

        // ಹಂತ 2: ಪಾರ್ಸೆಲ್ ಬಂದ ನಂತರ ಮೇಲಿನ ಎಲ್ಲಾ Properties ಗಳನ್ನು ಪರೀಕ್ಷಿಸುವುದು
        console.log("--- ಪಾರ್ಸೆಲ್ ಮೇಲಿನ ವಿವರಗಳು (Properties) ---");
        console.log("1. ವಿಳಾಸ (url): " + response.url);
        console.log("2. ಇಂಟರ್ನೆಟ್ ಕೋಡ್ (status): " + response.status);
        console.log("3. ಕೋಡ್‌ನ ಅರ್ಥ (statusText): " + response.statusText);
        console.log("4. ಬೇರೆ ಕಡೆಯಿಂದ ಬಂದಿದೆಯೇ? (redirected): " + response.redirected);
        console.log("5. ಡೇಟಾ ಪ್ರಕಾರ (type): " + response.type);
        console.log("6. ಪಾರ್ಸೆಲ್ ಓದಿದ್ದೇವಾ? (ಮೊದಲು): " + response.bodyUsed); // false ಇರುತ್ತದೆ

        // ಹಂತ 3: ಆರ್ಡರ್ ಸಿಕ್ಕಿಲ್ಲ ಎಂದರೆ (ಉದಾಹರಣೆಗೆ 404 Error ಬಂದರೆ)
        if (response.ok === false) {
            // throw ಎಂದರೆ ಇಲ್ಲಿಗೆ ಕೆಲಸ ನಿಲ್ಲಿಸಿ 'catch' ವಿಭಾಗಕ್ಕೆ ಎರರ್ ಕಳುಹಿಸು
            throw new Error("ಕ್ಷಮಿಸಿ, ನೀವು ಕೇಳಿದ ಫೈಲ್ ಸಿಗಲಿಲ್ಲ!");
        }

        // ಹಂತ 4: ಎಲ್ಲವೂ ಸರಿಯಿದ್ದರೆ, ಪಾರ್ಸೆಲ್ ಅನ್ನು JSON ಆಗಿ ಬಿಡಿಸಲು ಕಾಯಿರಿ (await)
        let data = await response.json(); 
        // ಸೂಚನೆ: ಒಂದು ವೇಳೆ ಟೆಕ್ಸ್ಟ್ ಫೈಲ್ ಆಗಿದ್ದರೆ response.text() ಬಳಸುತ್ತಿದ್ದೆವು.

        // ಈಗ ಪಾರ್ಸೆಲ್ ಓದಿದ್ದೇವಾ? ಎಂದು ಮತ್ತೊಮ್ಮೆ ಪರಿಶೀಲನೆ
        console.log("7. ಪಾರ್ಸೆಲ್ ಓದಿದ್ದೇವಾ? (ನಂತರ): " + response.bodyUsed); // ಈಗ true ಆಗಿರುತ್ತದೆ

        // ಹಂತ 5: ಬಿಡಿಸಿದ ಡೇಟಾವನ್ನು ಪರದೆಯ ಮೇಲೆ ತೋರಿಸುವುದು
        console.log("--- ಪಾರ್ಸೆಲ್ ಒಳಗಿದ್ದ ಮುಖ್ಯ ಮಾಹಿತಿ ---");
        console.log("ವಿದ್ಯಾರ್ಥಿಯ ಹೆಸರು: " + data.name);
        console.log("ವಿದ್ಯಾರ್ಥಿಯ ಊರು: " + data.city);

    } 
    // 'catch' ಬ್ಲಾಕ್: ದಾರಿಯಲ್ಲಿ ಇಂಟರ್ನೆಟ್ ಹೋದರೆ ಅಥವಾ ಫೈಲ್ ಇಲ್ಲದಿದ್ದರೆ ಇದು ಕೆಲಸ ಮಾಡುತ್ತದೆ
    catch (functionError) {
        console.log("ಏನೋ ತಪ್ಪಾಗಿದೆ, ದಯವಿಟ್ಟು ಗಮನಿಸಿ: " + functionError.message);
    }
}

// 2. ಬರೆದ ಫಂಕ್ಷನ್ ಅನ್ನು ಕೆಲಸ ಮಾಡಲು ಆದೇಶಿಸುವುದು (Call ಮಾಡುವುದು)
getStudentInformation();

ಈ ಕೋಡ್ ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ? (ಹಂತ-ಹಂತವಾಗಿ)

  1. async function: ಈ ಫಂಕ್ಷನ್ ಒಳಗೆ ನಾವು ಇಂಟರ್ನೆಟ್ ಕೆಲಸಗಳಿಗೆ ಕಾಯಬೇಕಾಗುತ್ತದೆ (Time takes) ಎಂದು ಬ್ರೌಸರ್‌ಗೆ ಮೊದಲೇ ತಿಳಿಸುತ್ತದೆ.
  2. await fetch: ಡೇಟಾ ಬರುವವರೆಗೆ ಮುಂದಿನ ಲೈನ್ ಕೋಡ್ ಓಡಬಾರದು ಎಂದು ತಡೆದು ನಿಲ್ಲಿಸುತ್ತದೆ.
  3. Properties ಚೆಕ್ಕಿಂಗ್: ಡೇಟಾ ಬಂದ ತಕ್ಷಣವೇ ಅದರ status, url, ಮತ್ತು ok ಅನ್ನು ಪ್ರಿಂಟ್ ಮಾಡಿ, ಫೈಲ್ ಸುರಕ್ಷಿತವಾಗಿ ಬಂದಿದೆಯೇ ಎಂದು ಖಚಿತಪಡಿಸಿಕೊಳ್ಳುತ್ತೇವೆ.
  4. await response.json(): ಬಾಕ್ಸ್ ಅನ್ನು ಒಡೆದು, JSON ಡೇಟಾವನ್ನು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಓದುವ ಭಾಷೆಗೆ ಬದಲಾಯಿಸಲು ಕಾಯುತ್ತದೆ. (ಈ ಹಂತದಲ್ಲಿ bodyUsed property false ನಿಂದ true ಗೆ ಬದಲಾಗುತ್ತದೆ).
  5. Try / Catch: ಇಂಟರ್ನೆಟ್ ಕಟ್ ಆದರೆ ಅಥವಾ ನಾವು ತಪ್ಪು ಫೈಲ್ ಹೆಸರು ಕೊಟ್ಟರೆ, ಕೋಡ್ ಕ್ರಾಶ್ ಆಗುವುದಿಲ್ಲ. ಬದಲಾಗಿ catch ಬ್ಲಾಕ್ ಒಳಗೆ ಹೋಗಿ ನಮಗೆ ಸುಂದರವಾದ Error ಮೆಸೇಜ್ ನೀಡುತ್ತದೆ.
