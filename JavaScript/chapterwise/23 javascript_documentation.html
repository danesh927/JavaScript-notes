JavaScript ಮಾಸ್ಟರ್‌ಕ್ಲಾಸ್

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ (JavaScript) ಸುಲಭ ಪಾಠಗಳು

ನೀವು ಐಟಿ ಹಿನ್ನೆಲೆಯಿಂದ ಬಂದಿಲ್ಲದಿದ್ದರೂ ಇದನ್ನು ಸುಲಭವಾಗಿ ಅರ್ಥಮಾಡಿಕೊಳ್ಳಬಹುದು.

1. forEach (ಲೂಪ್ಸ್)

forEach ಎನ್ನುವುದು ಒಂದು Array (ಪಟ್ಟಿ) ಯಲ್ಲಿರುವ ಪ್ರತಿಯೊಂದು ಅಂಶಕ್ಕೂ (item) ಒಂದು ನಿರ್ದಿಷ್ಟ ಕೆಲಸವನ್ನು ಮಾಡಲು ಬಳಸುವ ವಿಧಾನ (method).

ಸರಳ ಉದಾಹರಣೆ: ನಿಮ್ಮ ಬಳಿ 5 ಸೇಬುಗಳಿವೆ. ಪ್ರತಿಯೊಂದು ಸೇಬಿಗೂ ಒಂದು ಸ್ಟಿಕ್ಕರ್ ಅಂಟಿಸಬೇಕು. ನೀವು ಒಂದೊಂದೇ ಸೇಬನ್ನು ತೆಗೆದುಕೊಂಡು ಸ್ಟಿಕ್ಕರ್ ಅಂಟಿಸುತ್ತೀರಿ. ಇದನ್ನೇ ಕೋಡಿಂಗ್‌ನಲ್ಲಿ forEach ಮಾಡುತ್ತದೆ!

ಹಂತ 1: ಬಿಗಿನರ್ (Beginner) - ಮೂಲ ಬಳಕೆ

ಒಂದು ಹೆಸರಿನ ಪಟ್ಟಿಯಲ್ಲಿದ್ದ ಪ್ರತಿಯೊಬ್ಬರಿಗೂ "ಹಲೋ" ಎಂದು ಹೇಳಬೇಕು ಎಂದುಕೊಳ್ಳಿ.

let names = ["ರಾಮು", "ಸೋಮು", "ಗೀತಾ"];

// ಪ್ರತಿಯೊಂದು ಹೆಸರಿಗೂ ಹಲೋ ಹೇಳುವುದು
names.forEach(function(name) {
  console.log("ಹಲೋ " + name);
});

ಇದು ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ? ಪಟ್ಟಿಯಲ್ಲಿ 3 ಹೆಸರುಗಳಿವೆ. forEach 3 ಬಾರಿ ಕೆಲಸ ಮಾಡುತ್ತದೆ. ಮೊದಲಿಗೆ name ಅಂದರೆ "ರಾಮು", ಆಮೇಲೆ "ಸೋಮು", ಕೊನೆಗೆ "ಗೀತಾ".

ಹಂತ 2: ಇಂಟರ್ಮೀಡಿಯೇಟ್ (Intermediate) - ಬಾಣದ ಫಂಕ್ಷನ್ (Arrow Function)

ಇದೇ ಕೋಡ್ ಅನ್ನು ಶಾರ್ಟ್ ಕಟ್ (ಆರೋ ಫಂಕ್ಷನ್ - Arrow Function) ಬಳಸಿ ಬರೆಯಬಹುದು. ಮತ್ತು forEach ನಮಗೆ ಕೇವಲ ಐಟಂ ಮಾತ್ರವಲ್ಲದೆ ಅದರ ಜಾಗದ ಸಂಖ್ಯೆ (index) ಕೂಡ ನೀಡುತ್ತದೆ.

let fruits = ["ಸೇಬು", "ಬಾಳೆಹಣ್ಣು", "ಕಿತ್ತಳೆ"];

fruits.forEach((fruit, index) => {
  console.log(index + ": " + fruit);
});

/* ಔಟ್‌ಪುಟ್:
0: ಸೇಬು
1: ಬಾಳೆಹಣ್ಣು
2: ಕಿತ್ತಳೆ
*/

ಗಮನಿಸಿ: ಕಂಪ್ಯೂಟರ್ ಯಾವಾಗಲೂ ಎಣಿಕೆಯನ್ನು 0 ಯಿಂದ ಶುರು ಮಾಡುತ್ತದೆ.

ಹಂತ 3: ಪ್ರೊ (Pro Level) - ಆಳವಾದ ವಿಷಯಗಳು

ನೀವು ಪ್ರೊಫೆಷನಲ್ ಆದಾಗ, forEach ಬಗ್ಗೆ ಈ 3 ಪ್ರಮುಖ ವಿಷಯಗಳು ತಿಳಿದಿರಬೇಕು:

  • 1. ಹೊಸ Array ಅನ್ನು ಹಿಂದಿರುಗಿಸುವುದಿಲ್ಲ (Returns undefined):
    forEach ಕೇವಲ ಕೆಲಸ ಮಾಡುತ್ತದೆ, ಆದರೆ ಹೊಸ ಪಟ್ಟಿಯನ್ನು ಕೊಡುವುದಿಲ್ಲ. ನಿಮಗೆ ಹೊಸ ಪಟ್ಟಿ ಬೇಕಿದ್ದರೆ map() ಬಳಸಬೇಕು.
  • 2. break ಅಥವಾ continue ಕೆಲಸ ಮಾಡುವುದಿಲ್ಲ:
    ನೀವು forEach ಶುರು ಮಾಡಿದರೆ, ಅದು ಪಟ್ಟಿಯ ಕೊನೆಯವರೆಗೂ ಹೋಗಲೇಬೇಕು. ಮಧ್ಯದಲ್ಲಿ ನಿಲ್ಲಿಸಲು (break) ಆಗುವುದಿಲ್ಲ. ನಿಲ್ಲಿಸಬೇಕಿದ್ದರೆ for...of ಲೂಪ್ ಬಳಸಬೇಕು.
  • 3. Objects (ವಸ್ತುಗಳ) ಜೊತೆ ಬಳಕೆ:
    ಬ್ಯಾಂಕ್ ಅಕೌಂಟ್, ವಿದ್ಯಾರ್ಥಿಗಳ ಮಾರ್ಕ್ಸ್ ಮುಂತಾದ ಡೇಟಾಬೇಸ್ ಮಾಹಿತಿಯನ್ನು ನಿರ್ವಹಿಸಲು ಇದು ತುಂಬಾ ಉಪಯುಕ್ತ. 
    let students = [
  { name: "ರಾಮು", marks: 45 },
  { name: "ಗೀತಾ", marks: 80 }
];

// ಪ್ರತಿಯೊಬ್ಬ ವಿದ್ಯಾರ್ಥಿಗೆ 5 ಗ್ರೇಸ್ ಮಾರ್ಕ್ಸ್ ಸೇರಿಸುವುದು
students.forEach((student) => {
  student.marks = student.marks + 5;
});

console.log(students);
// ರಾಮು ಮಾರ್ಕ್ಸ್ 50, ಗೀತಾ ಮಾರ್ಕ್ಸ್ 85 ಆಗಿರುತ್ತದೆ.

2. Timers (ಟೈಮರ್‌ಗಳು)

setTimeout() (ಟೈಮರ್ ಅಥವಾ ಅಲಾರ್ಮ್ ಇಡುವುದು)

ಇದು ಏನು ಮಾಡುತ್ತದೆ? "ಇಷ್ಟು ಸಮಯದ ನಂತರ ಈ ಕೆಲಸ ಮಾಡು" ಎಂದು ವೆಬ್‌ಸೈಟ್‌ಗೆ ಹೇಳಲು ಇದನ್ನು ಬಳಸುತ್ತಾರೆ.

ದೈನಂದಿನ ಉದಾಹರಣೆ: ನೀವು ಒಲೆಯ ಮೇಲೆ ಹಾಲು ಇಟ್ಟು, "5 ನಿಮಿಷದ ನಂತರ ಬಂದು ಆಫ್ ಮಾಡಬೇಕು" ಎಂದು ಅಲಾರ್ಮ್ ಇಟ್ಟುಕೊಳ್ಳುತ್ತೀರಲ್ಲ, ಅದೇ ರೀತಿ.
ವೆಬ್‌ಸೈಟ್ ಉದಾಹರಣೆ: ನೀವು ಕೆಲವು ಸುದ್ದಿ ವೆಬ್‌ಸೈಟ್‌ಗಳನ್ನು ಓಪನ್ ಮಾಡಿ ಸುಮ್ಮನೆ ಓದುತ್ತಾ ಕೂತರೆ, ಸರಿಯಾಗಿ 10 ಸೆಕೆಂಡ್ ಆದ ಮೇಲೆ ತಾನಾಗಿಯೇ ಒಂದು ಜಾಹೀರಾತು (Ad) ಬರುತ್ತದಲ್ಲ, ಅದುವೇ setTimeout(). 
// ಕೋಡ್ ಹೇಗಿರುತ್ತದೆ?
// 3000 ಎಂದರೆ 3 ಸೆಕೆಂಡುಗಳು (3000 ಮಿಲಿಸೆಕೆಂಡು)
setTimeout(function() {
  alert("3 ಸೆಕೆಂಡ್ ಆಯ್ತು!"); 
}, 3000);

setInterval() (ಮತ್ತೆ ಮತ್ತೆ ಕೆಲಸ ಮಾಡುವ ಅಲಾರ್ಮ್)

ಇದು ಏನು ಮಾಡುತ್ತದೆ? setTimeout ಒಮ್ಮೆ ಮಾತ್ರ ಅಲಾರ್ಮ್ ಬಾರಿಸಿ ಸುಮ್ಮನಾಗುತ್ತದೆ. ಆದರೆ setInterval ನೀವು ಹೇಳಿದ ಸಮಯಕ್ಕೆ ಮತ್ತೆ ಮತ್ತೆ ಆ ಕೆಲಸವನ್ನು ಮಾಡುತ್ತಲೇ ಇರುತ್ತದೆ (ನೀವು ನಿಲ್ಲಿಸುವವರೆಗೂ).

ದೈನಂದಿನ ಉದಾಹರಣೆ: ಶಾಲೆಯಲ್ಲಿ ಪ್ರತಿ 45 ನಿಮಿಷಕ್ಕೊಮ್ಮೆ ಬೆಲ್ ಹೊಡೆಯುತ್ತಾರಲ್ಲ, ಹಾಗೆ!
ವೆಬ್‌ಸೈಟ್ ಉದಾಹರಣೆ: ವೆಬ್‌ಸೈಟ್‌ಗಳಲ್ಲಿ ಒಂದು ಗಡಿಯಾರ (Digital Clock) ಇರುತ್ತದಲ್ಲ, ಅದರಲ್ಲಿ ಪ್ರತೀ ಸೆಕೆಂಡಿಗೆ ಸೆಕೆಂಡ್ ಮುಳ್ಳು ಬದಲಾಗುತ್ತಲೇ ಇರುತ್ತದೆ. ಅಲ್ಲಿ ಇದನ್ನು ಬಳಸುತ್ತಾರೆ. 
// ಕೋಡ್ ಹೇಗಿರುತ್ತದೆ?
// ಪ್ರತಿ 10 ನಿಮಿಷಕ್ಕೊಮ್ಮೆ (600,000 ಮಿಲಿಸೆಕೆಂಡು) ಈ ಮೆಸೇಜ್ ಬರುತ್ತಲೇ ಇರುತ್ತದೆ
setInterval(function() {
  alert("10 ನಿಮಿಷ ಆಯ್ತು, ಸ್ವಲ್ಪ ನೀರು ಕುಡಿಯಿರಿ!"); 
}, 600000);

3. Toggle ಮತ್ತು classList

Toggle (ಟಾಗಲ್)

Toggle ಎಂದರೆ ಒಂದೇ ಗುಂಡಿಯನ್ನು ಬಳಸಿ ಎರಡು ವಿರುದ್ಧವಾದ ಕೆಲಸಗಳನ್ನು (On ಮತ್ತು Off) ಮಾಡುವುದು.

ದೈನಂದಿನ ಉದಾಹರಣೆ: ನಿಮ್ಮ ಮನೆಯ ಟಿವಿ ರಿಮೋಟ್‌ನ 'ಪವರ್ (Power)' ಬಟನ್ ಅಥವಾ ಪೆನ್ನಿನ ಮೇಲಿರುವ ಟಿಕ್-ಟಿಕ್ ಗುಂಡಿ.
  • ನೀವು ಪವರ್ ಬಟನ್ ಅನ್ನು ಒಮ್ಮೆ ಒತ್ತಿದರೆ ಟಿವಿ ಆನ್ (ON) ಆಗುತ್ತದೆ.
  • ಅದೇ ಬಟನ್ ಅನ್ನು ಮತ್ತೆ ಒತ್ತಿದರೆ ಟಿವಿ ಆಫ್ (OFF) ಆಗುತ್ತದೆ.

ಒಂದೇ ಗುಂಡಿ, ಆದರೆ ಸಂದರ್ಭಕ್ಕೆ ತಕ್ಕಂತೆ ಎರಡು ಕೆಲಸ ಮಾಡುತ್ತದೆ. ಇದನ್ನೇ ಕಂಪ್ಯೂಟರ್ ಭಾಷೆಯಲ್ಲಿ Toggle ಎನ್ನುತ್ತಾರೆ.

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ Toggle ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ?

ನಾವು classList.toggle("dark-theme") ಎಂದು ಬಳಸಿದಾಗ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ತನ್ನಷ್ಟಕ್ಕೆ ತಾನೇ ಹೀಗೆ ಆಲೋಚಿಸುತ್ತದೆ:

  • 1. ಮೊದಲನೇ ಬಾರಿ ಗುಂಡಿ ಒತ್ತಿದಾಗ: "ಇಲ್ಲಿ dark-theme ಇದೆಯಾ? ಇಲ್ಲ, ಸ್ಕ್ರೀನ್ ಬಿಳಿಯಾಗಿದೆ. ಸರಿ, ಹಾಗಾದರೆ ಕಪ್ಪು ಬಣ್ಣವನ್ನು ಹಾಕುತ್ತೇನೆ (Add)." (ಈಗ ಡಾರ್ಕ್ ಮೋಡ್ ಆನ್ ಆಯ್ತು)
  • 2. ಎರಡನೇ ಬಾರಿ ಅದೇ ಗುಂಡಿ ಒತ್ತಿದಾಗ: "ಈಗ dark-theme ಇದೆಯಾ? ಹೌದು, ಈಗ ಸ್ಕ್ರೀನ್ ಕಪ್ಪಗಿದೆ. ಸರಿ, ಹಾಗಾದರೆ ಆ ಕಪ್ಪು ಬಣ್ಣವನ್ನು ತೆಗೆಯುತ್ತೇನೆ (Remove)." (ಈಗ ಡಾರ್ಕ್ ಮೋಡ್ ಆಫ್ ಆಯ್ತು)

Toggle ಇಲ್ಲದಿದ್ದರೆ ಏನಾಗುತ್ತಿತ್ತು? (Pro Level)

ಒಂದು ವೇಳೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ಈ toggle ಎನ್ನುವ ಮ್ಯಾಜಿಕ್ ಪದ ಇಲ್ಲದೆ ಹೋಗಿದ್ದರೆ, ನಾವು ಡಾರ್ಕ್ ಮೋಡ್ ಮಾಡಲು ಇಷ್ಟೊಂದು ಉದ್ದದ ಕೋಡ್ (If-Else) ಬರೆಯಬೇಕಾಗಿತ್ತು:

// Toggle ಇಲ್ಲದಿದ್ದರೆ ನಾವು ಹೀಗೆ ಕಷ್ಟಪಡಬೇಕಿತ್ತು:
function changeMode() {
  if (document.body.classList.contains("dark-theme")) {
    // ಕಪ್ಪು ಬಣ್ಣ ಮೊದಲೇ ಇದ್ದರೆ, ಅದನ್ನು ತೆಗೆ
    document.body.classList.remove("dark-theme"); 
  } else {
    // ಕಪ್ಪು ಬಣ್ಣ ಇಲ್ಲದಿದ್ದರೆ, ಅದನ್ನು ಹಾಕು
    document.body.classList.add("dark-theme");
  }
}

ಆದರೆ toggle ಇರುವುದರಿಂದ, ಈ 5-6 ಸಾಲಿನ ಕೋಡ್ ಅನ್ನು ನಾವು ಕೇವಲ ಒಂದೇ ಸಾಲಿನಲ್ಲಿ ಮುಗಿಸಬಹುದು!

// Toggle ಬಳಸಿ ಸ್ಮಾರ್ಟ್ ಆಗಿ ಮಾಡುವುದು:
document.body.classList.toggle("dark-theme");

classList ಮತ್ತು ಅದರ ಸಾಧನಗಳು

classList ಮತ್ತು classList.contains() ಇವೆರಡೂ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ ವೆಬ್‌ಸೈಟ್‌ನ ಡಿಸೈನ್ (CSS) ಜೊತೆ ಆಟವಾಡಲು ಬಳಸುವ ಅತ್ಯಂತ ಪ್ರಮುಖ ಸಾಧನಗಳು. ಇದನ್ನು "ಒಬ್ಬ ವ್ಯಕ್ತಿ ಹಾಕಿರುವ ಬಟ್ಟೆಗಳು ಮತ್ತು ಆತನ ಐಡಿ ಕಾರ್ಡ್" ಗೆ ಹೋಲಿಸಿ ಬಹಳ ಸುಲಭವಾಗಿ ಅರ್ಥಮಾಡಿಕೊಳ್ಳಬಹುದು.

1. classList ಎಂದರೇನು?

ಒಂದು HTML ವಸ್ತು (ಉದಾಹರಣೆಗೆ ಒಂದು ಬಟನ್ ಅಥವಾ ಪ್ಯಾರಾಗ್ರಾಫ್) ಮೇಲೆ ಯಾವೆಲ್ಲಾ ಸ್ಟೈಲ್‌ಗಳು (CSS Classes) ಇವೆ ಎಂದು ತೋರಿಸುವ ಪಟ್ಟಿಯೇ classList.

ಉದಾಹರಣೆ: ರಾಮು ಎಂಬ ಹುಡುಗನನ್ನು ಕಲ್ಪಿಸಿಕೊಳ್ಳಿ. ಅವನು "ಶರ್ಟ್", "ಪ್ಯಾಂಟ್" ಮತ್ತು "ಕನ್ನಡಕ" ಹಾಕಿದ್ದಾನೆ. ಇವು ಅವನ ಕ್ಲಾಸ್‌ಗಳು (Classes). ಯಾರಾದರೂ ಬಂದು "ರಾಮು ಏನೇನು ಹಾಕಿದ್ದಾನೆ?" ಎಂದು ಕೇಳಿದರೆ, ನಾವು ಕೊಡುವ ಲಿಸ್ಟ್ (ಪಟ್ಟಿ) ಇದೆಯಲ್ಲ, ಅದೇ classList.

2. classList.contains() ಎಂದರೇನು?

contains ಎಂದರೆ "ಒಳಗೊಂಡಿದೆಯೇ?" ಅಥವಾ "ಇದೆಯಾ?" ಎಂದು ಕೇಳುವ ಒಂದು ಪ್ರಶ್ನೆ. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಕೇವಲ true (ಹೌದು) ಅಥವಾ false (ಇಲ್ಲ) ಎಂದು ಉತ್ತರಿಸುತ್ತದೆ.

ಉದಾಹರಣೆ: ಸಿನಿಮಾ ಥಿಯೇಟರ್ ಬಾಗಿಲಲ್ಲಿ ನಿಂತಿರುವ ಸೆಕ್ಯೂರಿಟಿ ಗಾರ್ಡ್ "ನಿಮ್ಮ ಬಳಿ VIP ಪಾಸ್ ಇದೆಯಾ?" ಎಂದು ಚೆಕ್ ಮಾಡುತ್ತಾನಲ್ಲ, ಅದೇ contains().

// 'myButton' ಎಂಬ ಗುಂಡಿಯನ್ನು ಹುಡುಕಿ ತರುವುದು
let button = document.getElementById("myButton");

// ಗುಂಡಿಯ ಮೇಲೆ 'active' ಎಂಬ ಸ್ಟೈಲ್/ಪಾಸ್ ಇದೆಯಾ ಎಂದು ಚೆಕ್ ಮಾಡುವುದು
if (button.classList.contains("active")) {
  alert("ಹೌದು, ಈ ಗುಂಡಿ ಆಕ್ಟಿವ್ ಆಗಿದೆ!"); // (true)
} else {
  alert("ಇಲ್ಲ, ಇದು ಆಕ್ಟಿವ್ ಆಗಿಲ್ಲ."); // (false)
}

ಪ್ರೊಫೆಷನಲ್ ಡೆವಲಪರ್‌ಗಳ 4 ಮ್ಯಾಜಿಕ್‌ಗಳು:

  • classList.add("ಸ್ಟೈಲ್") (ಹೊಸ ಬಟ್ಟೆ ಹಾಕುವುದು): ಯಾವುದಾದರೂ ಹೊಸ ಸ್ಟೈಲ್ ಸೇರಿಸಲು.
  • classList.remove("ಸ್ಟೈಲ್") (ಬಟ್ಟೆ ತೆಗೆಯುವುದು): ಸ್ಟೈಲ್ ಅನ್ನು ತೆಗೆದುಹಾಕಲು.
  • classList.contains("ಸ್ಟೈಲ್") (ಚೆಕ್ ಮಾಡುವುದು): ಸ್ಟೈಲ್ ಇದೆಯಾ ಎಂದು ಕನ್ಫರ್ಮ್ ಮಾಡಿಕೊಳ್ಳಲು.
  • classList.toggle("ಸ್ಟೈಲ್") (ಸ್ವಿಚ್ ಆನ್/ಆಫ್): ಇದ್ದರೆ ತೆಗೆಯುವುದು, ಇಲ್ಲದಿದ್ದರೆ ಹಾಕುವುದು.

ರಿಯಲ್-ವರ್ಲ್ಡ್ ಉದಾಹರಣೆ: ಯೂಟ್ಯೂಬ್‌ನಲ್ಲಿ 'ಲೈಕ್ (Like)' ಗುಂಡಿ ಒತ್ತಿದಾಗ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಹೀಗೆ ಯೋಚಿಸುತ್ತದೆ: "ಈ ಗುಂಡಿಯ classList ನಲ್ಲಿ ಈಗಾಗಲೇ liked ಎಂಬ ಕ್ಲಾಸ್ ಇದೆಯಾ (contains)? ಹೌದು ಇದ್ದರೆ, ಅದನ್ನು ತೆಗೆದುಹಾಕು (remove). ಇಲ್ಲದಿದ್ದರೆ, ಹೊಸದಾಗಿ ಸೇರಿಸು (add)."

4. Browser Controls (ಬ್ರೌಸರ್ ನಿಯಂತ್ರಣ)

window.scrollTo()
(ಪೇಜ್ ಅನ್ನು ಜಾರಿಸುವುದು)

ಬಳಕೆದಾರರು (ನೀವು) ಸ್ಕ್ರೀನ್ ಅನ್ನು ಬೆರಳಿನಿಂದ ಮೇಲಕ್ಕೆ ಅಥವಾ ಕೆಳಕ್ಕೆ ಎಳೆಯುವ ಬದಲು, ಕಂಪ್ಯೂಟರ್ ತಾನಾಗಿಯೇ ಪೇಜ್ ಅನ್ನು ಮೇಲೆ-ಕೆಳಗೆ ಜಾರುವಂತೆ (Scroll) ಮಾಡುತ್ತದೆ.

ದೈನಂದಿನ ಉದಾಹರಣೆ: ಒಂದು ಉದ್ದವಾದ ಕ್ಯಾಲೆಂಡರ್ ಅನ್ನು ಸುರುಳಿ ಸುತ್ತುತ್ತಾ ಕೆಳಗೆ ಅಥವಾ ಮೇಲಕ್ಕೆ ನೋಡುವುದು.

ವೆಬ್‌ಸೈಟ್ ಉದಾಹರಣೆ: ಅಮೆಜಾನ್ ಓದುತ್ತಾ ತೀರಾ ಕೆಳಗೆ ಹೋದಾಗ, ಅಲ್ಲಿ "Go to Top" ಗುಂಡಿ ಇರುತ್ತದೆ. ಅದನ್ನು ಒತ್ತಿದಾಗ ತಾನಾಗಿಯೇ ವೇಗವಾಗಿ ಮೇಲ್ಭಾಗಕ್ಕೆ ಹೋಗುತ್ತೀರಲ್ಲ, ಅಲ್ಲಿ scrollTo() ಬಳಸಿರುತ್ತಾರೆ.

// ಪೇಜ್ ಅನ್ನು ಅತ್ಯಂತ ಮೇಲ್ಭಾಗಕ್ಕೆ 
// (0, 0 ಜಾಗಕ್ಕೆ) ತಳ್ಳುತ್ತದೆ
window.scrollTo(0, 0);

focus()
(ಬೆಳಕು ಚೆಲ್ಲುವುದು / ಗಮನ ಸೆಳೆಯುವುದು)

ವೆಬ್‌ಸೈಟ್‌ನಲ್ಲಿರುವ ಯಾವುದಾದರೂ ಒಂದು ಟೆಕ್ಸ್ಟ್ ಬಾಕ್ಸ್‌ಗೆ (ಬರೆಯುವ ಜಾಗ) ನೇರವಾಗಿ ಕರ್ಸರ್ (ಟೈಪ್ ಮಾಡುವ ಗೆರೆ) ಹೋಗಿ ನಿಲ್ಲುವಂತೆ ಮಾಡುತ್ತದೆ.

ದೈನಂದಿನ ಉದಾಹರಣೆ: ಕತ್ತಲೆಯಲ್ಲಿ ಸ್ಟೇಜ್ ಮೇಲೆ ನಿಂತಿರುವ ಒಬ್ಬನೇ ವ್ಯಕ್ತಿಯ ಮೇಲೆ ಲೈಟ್ ಬಿಟ್ಟು (Spotlight) ಎಲ್ಲರ ಗಮನ ಸೆಳೆಯುವ ಹಾಗೆ.

ವೆಬ್‌ಸೈಟ್ ಉದಾಹರಣೆ: ಗೂಗಲ್ (Google) ಆಪ್ ಓಪನ್ ಮಾಡಿದ ತಕ್ಷಣ, ಸರ್ಚ್ ಬಾಕ್ಸ್ ಮೇಲೆ ಕ್ಲಿಕ್ ಮಾಡದಿದ್ದರೂ ಕೀಬೋರ್ಡ್ ತಾನಾಗಿಯೇ ಓಪನ್ ಆಗಿ ಟೈಪ್ ಮಾಡಲು ರೆಡಿ ಇರುತ್ತದಲ್ಲ? ಅಲ್ಲಿ focus() ಮಾಡಿರುತ್ತಾರೆ.

// 'searchBox' ಜಾಗಕ್ಕೆ ಹೋಗಿ ಟೈಪ್ ಮಾಡಲು ರೆಡಿಯಾಗು
document.getElementById("searchBox").focus();

5. JSON (ಜೆಸನ್)

JSON (JavaScript Object Notation) ಎನ್ನುವುದು ಕಂಪ್ಯೂಟರ್ ಜಗತ್ತಿನ ಪಾರ್ಸೆಲ್ ಅಥವಾ ಕುರಿಯರ್ ಸರ್ವೀಸ್ ಇದ್ದ ಹಾಗೆ! ಮಾಹಿತಿಯನ್ನು (Data) ಬರೆದಿಡುವ ಮತ್ತು ಕಳುಹಿಸುವ ಒಂದು ಸರಳವಾದ ಶೈಲಿ.

ದೈನಂದಿನ ಉದಾಹರಣೆ: ನೀವು ಮನೆ ಶಿಫ್ಟ್ ಮಾಡುವಾಗ ನಿಮ್ಮ ಎಲ್ಲಾ ವಸ್ತುಗಳನ್ನು ಬೇರೆ ಬೇರೆ ರಟ್ಟಿನ ಬಾಕ್ಸ್‌ಗಳಲ್ಲಿ ಹಾಕಿ, ಅದರ ಮೇಲೆ ಲೇಬಲ್ ಅಂಟಿಸುತ್ತೀರಲ್ಲ? ಅದೇ ರೀತಿ, ಇಂಟರ್ನೆಟ್‌ನಲ್ಲಿ ಒಂದು ಕಂಪ್ಯೂಟರ್‌ನಿಂದ ಇನ್ನೊಂದು ಕಂಪ್ಯೂಟರ್‌ಗೆ ಮಾಹಿತಿಯನ್ನು ಕಳುಹಿಸಲು ಮಾಡುವ 'ಪ್ಯಾಕಿಂಗ್' ಅನ್ನು JSON ಎನ್ನುತ್ತಾರೆ.

1. JSON.stringify()

(ಬಾಕ್ಸ್‌ಗೆ ಹಾಕಿ ಪ್ಯಾಕ್ ಮಾಡುವುದು)

ನಿಮ್ಮ ಬಳಿ ಇರುವ ಮಾಹಿತಿಯನ್ನು (Object) ಒಂದು ಸಾಮಾನ್ಯ ಟೆಕ್ಸ್ಟ್ ಅಥವಾ ವಾಕ್ಯವಾಗಿ (String) ಬದಲಾಯಿಸುತ್ತದೆ. ಇಂಟರ್ನೆಟ್ ಎಂಬ ರಸ್ತೆಯಲ್ಲಿ ಡೇಟಾ ಟೆಕ್ಸ್ಟ್ ರೂಪದಲ್ಲಿ ಮಾತ್ರ ಪ್ರಯಾಣ ಮಾಡಬೇಕು.

ಉದಾಹರಣೆ: ಊರಿಗೆ ಹೋಗುವಾಗ ಬೀರುದಲ್ಲಿರುವ ಬಟ್ಟೆಗಳನ್ನೆಲ್ಲಾ ತೆಗೆದು ಸೂಟ್‌ಕೇಸ್‌ಗೆ ಹಾಕಿ ಜಿಪ್ (Zip) ಹಾಕಿದ ಹಾಗೆ.

let student = { name: "ರಾಮು", marks: 95 };

// ಇಂಟರ್ನೆಟ್‌ನಲ್ಲಿ ಕಳುಹಿಸಲು ಪ್ಯಾಕ್ ಮಾಡೋಣ
let packedData = JSON.stringify(student); 

// ಈಗ ಟೆಕ್ಸ್ಟ್ ಬಾಕ್ಸ್: '{"name":"ರಾಮು","marks":95}'
console.log(packedData);

2. JSON.parse()

(ಪಾರ್ಸೆಲ್ ಓಪನ್ ಮಾಡಿ ಹೊರತೆಗೆಯುವುದು)

ಇಂಟರ್ನೆಟ್‌ನಿಂದ ಬಂದ ಟೆಕ್ಸ್ಟ್ ಬಾಕ್ಸ್ (String) ಅನ್ನು ಒಡೆದು/ಬಿಡಿಸಿ, ಮತ್ತೆ ಬಳಸಬಹುದಾದ ಅಸಲಿ ಮಾಹಿತಿಯನ್ನಾಗಿ (Object) ಮಾಡುತ್ತದೆ.

ಉದಾಹರಣೆ: ಊರಿಗೆ ತಲುಪಿದ ಮೇಲೆ, ಬಸ್‌ನಿಂದ ತಂದ ಸೂಟ್‌ಕೇಸ್ ಓಪನ್ ಮಾಡಿ ಬಟ್ಟೆಗಳನ್ನು ಹೊರತೆಗೆದು ಮತ್ತೆ ಧರಿಸಲು ರೆಡಿ ಮಾಡಿದ ಹಾಗೆ.

let receivedBox = '{"name":"ಗೀತಾ","marks":98}';

// ಇಂಟರ್ನೆಟ್‌ನಿಂದ ಬಂದ ಪಾರ್ಸೆಲ್ ಓಪನ್ ಮಾಡೋಣ
let unpackedData = JSON.parse(receivedBox);

// ಈಗ ಹೆಸರನ್ನು ಸುಲಭವಾಗಿ ತೆಗೆದುಕೊಳ್ಳಬಹುದು
console.log(unpackedData.name); // ಗೀತಾ

ಹಂತ 3: ಪ್ರೊ (Pro Level) - ಪ್ರಾಜೆಕ್ಟ್‌ಗಳಲ್ಲಿ ಇದರ ಬಳಕೆ ಏನು?

ದೊಡ್ಡ ದೊಡ್ಡ ವೆಬ್‌ಸೈಟ್‌ಗಳಲ್ಲಿ (Amazon, Swiggy, WhatsApp) ಈ JSON ಅನ್ನು ಮುಖ್ಯವಾಗಿ 2 ಕಡೆ ಬಳಸುತ್ತಾರೆ:

  • 1. localStorage (ಮೆಮೊರಿ) ಜೊತೆಗೆ:
    ಮೆಮೊರಿಯಲ್ಲಿ ಕೇವಲ 'ಅಕ್ಷರಗಳನ್ನು (Text)' ಮಾತ್ರ ಸೇವ್ ಮಾಡಬಹುದು. ಹಾಗಾಗಿ 'ಶಾಪಿಂಗ್ ಕಾರ್ಟ್' ಸೇವ್ ಮಾಡಬೇಕಾದರೆ, ಮೊದಲು JSON.stringify() ಬಳಸಿ ಪ್ಯಾಕ್ ಮಾಡಿ ಸೇವ್ ಮಾಡಬೇಕು. ಮರುದಿನ ನೋಡುವಾಗ JSON.parse() ಬಳಸಿ ಅನ್‌ಪ್ಯಾಕ್ ಮಾಡಿ ಸ್ಕ್ರೀನ್ ಮೇಲೆ ತೋರಿಸಬೇಕು.
  • 2. fetch() (ಡೆಲಿವರಿ ಬಾಯ್) ಜೊತೆಗೆ:
    ಅಮೆಜಾನ್ ಸರ್ವರ್‌ನಿಂದ ನಿಮ್ಮ ಫೋನ್‌ಗೆ 100 ಮೊಬೈಲ್‌ಗಳ ಲಿಸ್ಟ್ ಬರುವಾಗ, ಅದು JSON ಪ್ಯಾಕೆಟ್ ರೂಪದಲ್ಲಿಯೇ ಬರುತ್ತದೆ. ನಿಮ್ಮ ಫೋನ್ ಆ ಪ್ಯಾಕೆಟ್ ಅನ್ನು ರಿಸೀವ್ ಮಾಡಿಕೊಂಡು JSON.parse() ಮಾಡಿ, ಅದರಲ್ಲಿರುವ ಮೊಬೈಲ್ ಹೆಸರು, ಬೆಲೆಗಳನ್ನು ಬಿಡಿಸಿ ಸುಂದರವಾಗಿ ತೋರಿಸುತ್ತದೆ.

6. ಬ್ರಹ್ಮಾಸ್ತ್ರಗಳು (Pro Developer Tools)

ನೀವು ಈ ಎಲ್ಲಾ ಕಾನ್ಸೆಪ್ಟ್‌ಗಳನ್ನು ಒಟ್ಟುಗೂಡಿಸಿ ಪ್ರಾಜೆಕ್ಟ್ (Swiggy, Amazon ನಂತಹ ವೆಬ್‌ಸೈಟ್) ಮಾಡಲು ಹೊರಟಾಗ, ಪ್ರತಿಯೊಬ್ಬ ಪ್ರೊಫೆಷನಲ್ ಡೆವಲಪರ್ ಬಳಸುವ ಇನ್ನೂ 4 ಪ್ರಮುಖ ಬ್ರಹ್ಮಾಸ್ತ್ರಗಳು ಇವೆ.

1. document.querySelector()

(ಸ್ಮಾರ್ಟ್ ಹುಡುಕಾಟ / GPS ಟ್ರ್ಯಾಕಿಂಗ್)

ವೆಬ್‌ಸೈಟ್‌ನಲ್ಲಿರುವ ಸಾವಿರಾರು ಲೈನ್‌ಗಳ ಕೋಡ್‌ನಲ್ಲಿ, ನಿಮಗೆ ಬೇಕಾದ ಒಂದು ನಿರ್ದಿಷ್ಟ ಗುಂಡಿ, ಫೋಟೋ ಅಥವಾ ಟೆಕ್ಸ್ಟ್ ಅನ್ನು ಹುಡುಕಿ ತರಲು ಇದನ್ನು ಬಳಸುತ್ತಾರೆ. ಇದು getElementById ಗಿಂತ ಬಹಳ ಅಡ್ವಾನ್ಸ್ಡ್.

  • ದೈನಂದಿನ ಉದಾಹರಣೆ: ಸ್ವಿಗ್ಗಿಯಲ್ಲಿ ಊಟ ಆರ್ಡರ್ ಮಾಡಿದಾಗ, ಡೆಲಿವರಿ ಬಾಯ್ ಗೂಗಲ್ ಮ್ಯಾಪ್ (GPS) ಬಳಸಿ ನಿಖರವಾಗಿ ನಿಮ್ಮ ಮನೆ ಬಾಗಿಲಿಗೆ ಬರುತ್ತಾನಲ್ಲ, ಹಾಗೆ.
  • ಪ್ರಾಜೆಕ್ಟ್ ಉದಾಹರಣೆ: ಇನ್‌ಸ್ಟಾಗ್ರಾಮ್‌ನಲ್ಲಿ ಯಾರಿಗಾದರೂ 'ಲೈಕ್' ಮಾಡುವಾಗ, ನೀವು ಒತ್ತಿದ ಆ ನಿರ್ದಿಷ್ಟ ಫೋಟೋದ ಲೈಕ್ ಗುಂಡಿಯನ್ನು ಮಾತ್ರ ಹುಡುಕಲು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಇದನ್ನು ಬಳಸುತ್ತದೆ.
// "submit-btn" ಎಂಬ ಹೆಸರಿರುವ ಗುಂಡಿಯನ್ನು ಹುಡುಕಿ ತಾ
let myButton = document.querySelector(".submit-btn");

2. ಟೆಂಪ್ಲೇಟ್ ಲಿಟರಲ್ಸ್ ${...}

(ಫಿಲ್-ಇನ್-ದಿ-ಬ್ಲಾಂಕ್ಸ್)

ವಾಕ್ಯಗಳ (Text) ಮಧ್ಯೆ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನ ನಂಬರ್ ಅಥವಾ ಮಾಹಿತಿಯನ್ನು ಸುಲಭವಾಗಿ ಸೇರಿಸಲು ಇದನ್ನು ಬಳಸುತ್ತಾರೆ. ಇಲ್ಲಿ ಸಾಮಾನ್ಯ ಕೋಟ್ಸ್ ಬದಲು ಬ್ಯಾಕ್-ಟಿಕ್ (`) ಬಳಸುತ್ತಾರೆ (ಇದು ಕೀಬೋರ್ಡ್‌ನಲ್ಲಿ 1 ನಂಬರ್‌ನ ಪಕ್ಕ ಇರುತ್ತದೆ).

  • ದೈನಂದಿನ ಉದಾಹರಣೆ: ಶಾಲೆಯಲ್ಲಿ ಕೊಡುವ "ಪ್ರೋಗ್ರೆಸ್ ರಿಪೋರ್ಟ್". ಅದರಲ್ಲಿ ಎಲ್ಲಾ ಪ್ರಿಂಟ್ ಆಗಿರುತ್ತದೆ, ಕೇವಲ ನಿಮ್ಮ [ಹೆಸರು] ಮತ್ತು [ಮಾರ್ಕ್ಸ್] ಜಾಗ ಮಾತ್ರ ಖಾಲಿ ಬಿಟ್ಟಿರುತ್ತಾರೆ (Fill in the blanks).
  • ಪ್ರಾಜೆಕ್ಟ್ ಉದಾಹರಣೆ: ಅಮೆಜಾನ್‌ನಲ್ಲಿ ವಸ್ತುಗಳನ್ನು ಕಾರ್ಟ್‌ಗೆ ಸೇರಿಸಿದಾಗ "ನಿಮ್ಮ ಕಾರ್ಟ್‌ನಲ್ಲಿ 5 ವಸ್ತುಗಳಿವೆ, ಒಟ್ಟು ಮೊತ್ತ ₹500" ಎಂದು ತೋರಿಸಲು ಇದನ್ನು ಬಳಸುತ್ತಾರೆ.
let items = 5;
let price = 500;

// ವಾಕ್ಯದ ಮಧ್ಯೆ ವೇರಿಯೇಬಲ್ ಸೇರಿಸಲು ${} ಬಳಸುತ್ತೇವೆ
let message = `ನಿಮ್ಮ ಕಾರ್ಟ್‌ನಲ್ಲಿ ${items} ವಸ್ತುಗಳಿವೆ, ಒಟ್ಟು ಮೊತ್ತ ₹${price}.`;
console.log(message);

3. map()

(ಫ್ಯಾಕ್ಟರಿ ಮಶೀನ್ / ಆಟೋಮ್ಯಾಟಿಕ್ ಕಾಪಿ)

forEach ಕೇವಲ ಕೆಲಸ ಮಾಡುತ್ತದೆ, ಆದರೆ map() ಒಂದು ಪಟ್ಟಿಯಲ್ಲಿರುವ (Array) ಪ್ರತಿಯೊಂದು ವಸ್ತುವನ್ನು ತೆಗೆದುಕೊಂಡು, ಅದನ್ನು ಬದಲಾಯಿಸಿ, ಒಂದು ಹೊಸ ಪಟ್ಟಿಯನ್ನು (New Array) ಕೊಡುತ್ತದೆ.

  • ದೈನಂದಿನ ಉದಾಹರಣೆ: ಫ್ಯಾಕ್ಟರಿ ಮಶೀನ್. ನೀವು 10 ಹಸಿ ಆಲೂಗಡ್ಡೆಗಳನ್ನು ಹಾಕಿದರೆ, ಅದು ಆಲೂಗಡ್ಡೆಯನ್ನು ಕತ್ತರಿಸಿ, ಹುರಿದು, ಇನ್ನೊಂದು ಕಡೆಯಿಂದ 10 'ಪೊಟ್ಯಾಟೊ ಚಿಪ್ಸ್' ಪ್ಯಾಕೆಟ್‌ಗಳನ್ನು ಹೊರಹಾಕುತ್ತದೆ.
  • ಪ್ರಾಜೆಕ್ಟ್ ಉದಾಹರಣೆ: Zomato ಸರ್ವರ್‌ನಿಂದ 10 ಹೋಟೆಲ್‌ಗಳ ಹೆಸರು ಬಂದಾಗ, ಆ 10 ಹೆಸರುಗಳನ್ನು ಸುಂದರವಾದ 10 "HTML ಡಿಸೈನ್ ಕಾರ್ಡ್" ಗಳನ್ನಾಗಿ ಬದಲಾಯಿಸಲು map() ಬಳಸುತ್ತಾರೆ.
let numbers = [1, 2, 3];

// ಪ್ರತಿಯೊಂದು ನಂಬರ್ ಅನ್ನು 2 ರಿಂದ ಗುಣಿಸಿ ಹೊಸ ಪಟ್ಟಿ ಮಾಡುವುದು
let doubledNumbers = numbers.map(num => num * 2);

console.log(doubledNumbers); // ಉತ್ತರ: [2, 4, 6] ಬರುತ್ತದೆ

4. Async / Await

(ಟೋಕನ್ ವ್ಯವಸ್ಥೆ / ಕಾಯುವಿಕೆ)

ಒಂದು ಕೆಲಸ ಆಗಲು ಸಮಯ ಹಿಡಿಯುತ್ತದೆ ಎಂದಾಗ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಆ ಕೆಲಸ ಮುಗಿಯುವವರೆಗೆ ಕಾಯುವಂತೆ ಮಾಡುತ್ತದೆ.

  • ದೈನಂದಿನ ಉದಾಹರಣೆ: ನೀವು ಕಾಫಿ ಶಾಪ್‌ಗೆ ಹೋಗಿ ಕಾಫಿ ಆರ್ಡರ್ ಮಾಡುತ್ತೀರಿ. ಅವರು ತಕ್ಷಣ ಕಾಫಿ ಕೊಡಲ್ಲ, ಟೋಕನ್ ಕೊಡುತ್ತಾರೆ. ನೀವು ಕಾಯುತ್ತೀರಿ (Await). ಕಾಫಿ ರೆಡಿ ಆದಾಗ ತೆಗೆದುಕೊಂಡು ಕುಡಿಯುತ್ತೀರಿ.
  • ಪ್ರಾಜೆಕ್ಟ್ ಉದಾಹರಣೆ: fetch() ಬಳಸಿ ಇಂಟರ್ನೆಟ್‌ನಿಂದ ಡೇಟಾ ತರುವಾಗ, ಅದು ಬರಲು 2 ಸೆಕೆಂಡ್ ಬೇಕು. ಆ ಡೇಟಾ ಸಂಪೂರ್ಣವಾಗಿ ಬರುವವರೆಗೂ ಕಾಯಲು ಇದನ್ನು ಬಳಸುತ್ತಾರೆ. ಇಲ್ಲದಿದ್ದರೆ ಡೇಟಾ ಬರುವ ಮೊದಲೇ ಸ್ಕ್ರೀನ್ ಓಪನ್ ಆಗಿ, "ಮಾಹಿತಿ ಇಲ್ಲ (undefined)" ಎಂದು ಎರರ್ ಬರುತ್ತದೆ.
// async ಎಂದರೆ "ಇದರಲ್ಲಿ ಸಮಯ ಹಿಡಿಯುವ ಕೆಲಸ ಇದೆ" ಎಂದು ಅರ್ಥ
async function getRestaurantData() {
  
  // ಹೋಟೆಲ್ ಲಿಸ್ಟ್ ಬರುವವರೆಗೂ ಇಲ್ಲಿಯೇ ಕಾಯುತ್ತಿರು (await)
  let response = await fetch("https://api.swiggy.com/hotels");
  
  // ಡೇಟಾ ಬಂದ ಮೇಲೆ ಮುಂದಿನ ಕೆಲಸ ಮಾಡು
  console.log("ಹೋಟೆಲ್ ಲಿಸ್ಟ್ ರೆಡಿ ಇದೆ!");
}
