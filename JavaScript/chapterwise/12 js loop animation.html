ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಲೂಪ್‌ಗಳ ಲೈವ್ ಅನಿಮೇಷನ್

ಲೂಪ್‌ಗಳು ಪ್ರೊಸೆಸರ್‌ನಲ್ಲಿ ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತವೆ ಎಂಬುದನ್ನು ಹಂತ-ಹಂತವಾಗಿ ವೀಕ್ಷಿಸಿ.

1. For ಲೂಪ್ ಅನಿಮೇಷನ್

for (
  let i = 1; // 1. ಪ್ರಾರಂಭ
  i <= 3; // 2. ಷರತ್ತು ಪರಿಶೀಲನೆ
  i++ // 4. ಮೌಲ್ಯ ಹೆಚ್ಚಳ
) {
  console.log(i); // 3. ಕೋಡ್ ರನ್
}
ಅನಿಮೇಷನ್ ಪ್ರಾರಂಭಿಸಲು 'ಶುರುಮಾಡಿ' ಒತ್ತಿ.
ಮೆಮೊರಿ (RAM)
i = undefined
ಔಟ್‌ಪುಟ್ (Console)

2. While ಲೂಪ್ ಅನಿಮೇಷನ್

let i = 1;

while (i <= 3) {
  console.log(i);
  i++;
}
ಮೊದಲು ಷರತ್ತು ಪರಿಶೀಲಿಸಿ ನಂತರ ಕೋಡ್ ರನ್ ಆಗುತ್ತದೆ.
ಮೆಮೊರಿ (RAM)
i = undefined
ಔಟ್‌ಪುಟ್ (Console)

3. Do...While ಲೂಪ್ ಅನಿಮೇಷನ್

let i = 1;

do {
  console.log(i);
  i++;
} while (i <= 3);
ಮೊದಲು ಒಮ್ಮೆ ಕೋಡ್ ರನ್ ಆಗಿ, ಆಮೇಲೆ ಷರತ್ತು ಪರಿಶೀಲನೆ!
ಮೆಮೊರಿ (RAM)
i = undefined
ಔಟ್‌ಪುಟ್ (Console)
