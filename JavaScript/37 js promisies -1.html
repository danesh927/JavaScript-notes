JavaScript Promises Masterclass

ಪರಿವಿಡಿ (Contents)

ಪ್ರಸ್ತಾವನೆ 1. ಚಿಕ್ಕ ಮಕ್ಕಳಿಗಾಗಿ ಸರಳ ಉದಾಹರಣೆ 2. Promise ರಚಿಸುವ ಸಿಂಟ್ಯಾಕ್ಸ್ 3. ರಚಿಸಿದ Promise ಬಳಸುವುದು 4. Promise Chaining (ಮಧ್ಯಮ ಮಟ್ಟ) 5. Async / Await (ವೃತ್ತಿಪರ ಮಟ್ಟ) 6. ಬಹುಮುಖ್ಯ Promise ವಿಧಾನಗಳು ಸಾರಾಂಶ (Summary)

ಖಂಡಿತ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ (JavaScript) ನಲ್ಲಿ Promises ಎನ್ನುವುದನ್ನು ನಾನ್-ಐಟಿ (Non-IT) ವಿದ್ಯಾರ್ಥಿಗಳಿಗೆ ಮತ್ತು ಚಿಕ್ಕ ಮಕ್ಕಳಿಗೂ ಅರ್ಥವಾಗುವಂತೆ ಸಂಪೂರ್ಣವಾಗಿ, ಹಂತ-ಹಂತವಾಗಿ, ಒಂದೇ ಸರಣಿಯಲ್ಲಿ ಜೋಡಿಸಿ ವಿವರಿಸಲಾಗಿದೆ.

1. ಚಿಕ್ಕ ಮಕ್ಕಳಿಗಾಗಿ ಸರಳ ಉದಾಹರಣೆ (The Analogy)

ನಿಮ್ಮ ತಂದೆ ನಿಮಗೆ "ನಾಳೆ ಸಂಜೆ ನಿನಗೊಂದು ಹೊಸ ಸೈಕಲ್ ಕೊಡಿಸುತ್ತೇನೆ" ಎಂದು ಭರವಸೆ (Promise) ನೀಡುತ್ತಾರೆ ಎಂದು ಕಲ್ಪಿಸಿಕೊಳ್ಳಿ.

ಈಗ ಆ ಭರವಸೆಯು 3 ಹಂತಗಳಲ್ಲಿ ಯಾವುದಾದರೂ ಒಂದರಲ್ಲಿ ಇರಬಹುದು:

  • 1. Pending (ಬಾಕಿ ಇದೆ / ಕಾಯುವುದು): ನಾಳೆ ಸಂಜೆಯವರೆಗೆ ನೀವು ಕಾಯಬೇಕು. ಸೈಕಲ್ ಸಿಗುತ್ತದೆಯೋ ಇಲ್ಲವೋ ಇನ್ನೂ ತಿಳಿದಿಲ್ಲ.
  • ​2. Resolve / Fulfilled (ಯಶಸ್ಸು): ನಾಳೆ ಸಂಜೆ ಅವರು ಹೇಳಿದಂತೆ ಸೈಕಲ್ ತಂದುಕೊಟ್ಟರು. ನಿಮ್ಮ ತಂದೆ ಭರವಸೆಯನ್ನು ಈಡೇರಿಸಿದರು!
  • 3. Reject (ವಿಫಲ): ಅಂಗಡಿ ಬಂದ್ ಆಗಿದ್ದರಿಂದ ಅಥವಾ ಅವರ ಬಳಿ ಹಣವಿಲ್ಲದ ಕಾರಣ ಸೈಕಲ್ ತರಲು ಸಾಧ್ಯವಾಗಲಿಲ್ಲ. ಭರವಸೆ ಮುರಿಯಿತು.

ಕಂಪ್ಯೂಟರ್ ಜಗತ್ತಿನಲ್ಲಿ, ನಾವು ಆನ್‌ಲೈನ್‌ನಲ್ಲಿ ಸ್ವಿಗ್ಗಿ/ಝೊಮ್ಯಾಟೊ (Swiggy/Zomato) ಮೂಲಕ ಬಿರಿಯಾನಿ ಆರ್ಡರ್ ಮಾಡಿದಾಗಲೂ ಇದೇ ಆಗುತ್ತದೆ. ಆರ್ಡರ್ ಮಾಡಿದ ತಕ್ಷಣ ಆ್ಯಪ್ ನಿಮಗೆ ಊಟ ತಲುಪಿಸುವ ಭರವಸೆ (Promise) ನೀಡುತ್ತದೆ. ಊಟ ಬರುವವರೆಗೆ ನೀವು ಕಾಯುತ್ತೀರಿ (Pending). ಊಟ ಬಂದರೆ ಯಶಸ್ಸು (Resolve), ಹೋಟೆಲ್‌ನಲ್ಲಿ ಊಟ ಖಾಲಿಯಾದರೆ ವಿಫಲ (Reject).

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿಯೂ Promise ಎಂದರೆ ಇದೇ! ಒಂದು ಕೆಲಸ (ಉದಾಹರಣೆಗೆ: ಇಂಟರ್ನೆಟ್‌ನಿಂದ ಡೇಟಾ ತರುವುದು) ಮುಗಿಯುವವರೆಗೆ ಅದು ಕಾಯುತ್ತದೆ ಮತ್ತು ಆ ಕೆಲಸ ಯಶಸ್ವಿಯಾಯಿತೋ ಅಥವಾ ವಿಫಲವಾಯಿತೋ ಎಂದು ನಮಗೆ ತಿಳಿಸುತ್ತದೆ.

2. Promise ರಚಿಸುವ ಸಿಂಟ್ಯಾಕ್ಸ್ (ಕೋಡ್ ಬರೆಯುವ ನಿಯಮ) - Beginner Level

ಮೊದಲು ನಾವು ಕಂಪ್ಯೂಟರ್‌ಗೆ ಭರವಸೆಯನ್ನು ಸೃಷ್ಟಿಸಲು ಹೇಳಬೇಕು. ಊಟ ಆರ್ಡರ್ ಮಾಡುವ ಉದಾಹರಣೆಯನ್ನು ಕೋಡ್ ಮೂಲಕ ನೋಡೋಣ. ಪ್ರತಿಯೊಂದು ಸಾಲನ್ನೂ ಬಿಡಿಸಿ ಹೇಳಲಾಗಿದೆ.

// 1. ಭರವಸೆಯನ್ನು (Promise) ಹುಟ್ಟುಹಾಕುವುದು
let orderFood = new Promise(function(resolve, reject) {
    
    let foodAvailable = true; // ಊಟ ಹೋಟೆಲ್‌ನಲ್ಲಿ ಇದೆಯೋ ಇಲ್ಲವೋ?

    if (foodAvailable === true) {
        resolve("ನಿಮ್ಮ ಬಿರಿಯಾನಿ ಡೆಲಿವರಿ ಆಗಿದೆ! 🍲"); // ಯಶಸ್ಸು
    } else {
        reject("ಕ್ಷಮಿಸಿ, ಬಿರಿಯಾನಿ ಖಾಲಿ ಆಗಿದೆ. ❌"); // ವಿಫಲ
    }
    
});

ಕೋಡ್‌ನ ಪ್ರತಿಯೊಂದು ಸಾಲಿನ ಸಂಪೂರ್ಣ ವಿವರಣೆ (Non-IT ವಿದ್ಯಾರ್ಥಿಗಳಿಗೆ):

  • let orderFood = : orderFood (ಊಟದ ಆರ್ಡರ್) ಎಂಬ ಹೆಸರಿನ ಒಂದು ಡಬ್ಬವನ್ನು (Variable) ನಾವು ಮಾಡುತ್ತಿದ್ದೇವೆ. ಈ ಡಬ್ಬದೊಳಗೆ ನಮ್ಮ ಭರವಸೆಯನ್ನು (Promise) ಇಡುತ್ತೇವೆ.
  • new Promise(...) : ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ಗೆ "ನಾನೀಗ ಒಂದು ಹೊಸ ಭರವಸೆಯನ್ನು (Promise) ಹುಟ್ಟುಹಾಕುತ್ತಿದ್ದೇನೆ" ಎಂದು ಹೇಳುವ ಅಧಿಕೃತ ಪದವಿದು.
  • function(resolve, reject) { ... } : ಇದು ಪ್ರಾಮಿಸ್‌ನ ನಿಯಮ. ಭರವಸೆ ಯಶಸ್ವಿಯಾದರೆ ಬಳಸಲು resolve (ಸಕ್ಸಸ್) ಎಂಬ ಬಟನ್, ಮತ್ತು ಭರವಸೆ ವಿಫಲವಾದರೆ ಬಳಸಲು reject (ಫೇಲ್) ಎಂಬ ಬಟನ್ ಅನ್ನು ನಾವಿಲ್ಲಿ ಸಿದ್ಧಪಡಿಸುತ್ತಿದ್ದೇವೆ.
  • let foodAvailable = true; : ಹೋಟೆಲ್‌ನಲ್ಲಿ ಊಟ ಇದೆಯಾ ಎಂದು ನಾವು ಪರಿಶೀಲಿಸುತ್ತಿದ್ದೇವೆ. true ಎಂದರೆ "ಹೌದು, ಊಟ ಇದೆ" ಎಂದರ್ಥ. (false ಎಂದು ಬರೆದರೆ "ಊಟ ಇಲ್ಲ" ಎಂದರ್ಥ).
  • if (foodAvailable === true) { ... } : ಊಟ ನಿಜವಾಗಿಯೂ ಇದ್ದರೆ (if) ಏನು ಮಾಡಬೇಕು ಎಂದು ಕಂಪ್ಯೂಟರ್‌ಗೆ ಹೇಳುತ್ತಿದ್ದೇವೆ.
  • resolve("ನಿಮ್ಮ ಬಿರಿಯಾನಿ ಡೆಲಿವರಿ ಆಗಿದೆ! 🍲"); : ಊಟ ಇರುವುದರಿಂದ ಆರ್ಡರ್ ಯಶಸ್ವಿಯಾಗಿದೆ. ಹಾಗಾಗಿ ನಾವು resolve ಅನ್ನು ಬಳಸಿ, ಒಂದು ಖುಷಿಯ ಸಂದೇಶವನ್ನು ಕಳುಹಿಸುತ್ತಿದ್ದೇವೆ.
  • else { ... } : ಒಂದು ವೇಳೆ ಊಟ ಇಲ್ಲದಿದ್ದರೆ (else) ಏನು ಮಾಡಬೇಕು ಎಂದು ಹೇಳುತ್ತಿದ್ದೇವೆ.
  • reject("ಕ್ಷಮಿಸಿ, ಬಿರಿಯಾನಿ ಖಾಲಿ ಆಗಿದೆ. ❌"); : ಊಟ ಖಾಲಿಯಾಗಿದ್ದರೆ ಭರವಸೆ ಮುರಿಯುತ್ತದೆ. ಹಾಗಾಗಿ reject ಅನ್ನು ಬಳಸಿ, ಕ್ಷಮೆಯ ಸಂದೇಶವನ್ನು ಕಳುಹಿಸುತ್ತಿದ್ದೇವೆ.

3. ರಚಿಸಿದ Promise ಅನ್ನು ಬಳಸುವುದು ಹೇಗೆ? (Consuming the Promise)

ಮೇಲೆ ನಾವು ಕೇವಲ ಭರವಸೆಯನ್ನು ಸೃಷ್ಟಿಸಿದೆವು. ಈಗ ಆ ಭರವಸೆಯ ಫಲಿತಾಂಶ ಏನಾಯಿತು? ಎಂದು ನೋಡಲು ನಾವು .then (ಯಶಸ್ಸಾದರೆ) ಮತ್ತು .catch (ವಿಫಲವಾದರೆ) ಎಂಬ ಎರಡು ಪದಗಳನ್ನು ಬಳಸುತ್ತೇವೆ.

// 2. Promise ಅನ್ನು ಬಳಸುವುದು
orderFood
    .then(function(message) {
        console.log("ಖುಷಿಯ ವಿಚಾರ: " + message);
    })
    .catch(function(error) {
        console.log("ಬೇಸರದ ವಿಚಾರ: " + error);
    })
    .finally(function() {
        console.log("ಆರ್ಡರ್ ಪ್ರಕ್ರಿಯೆ ಮುಗಿದಿದೆ (ಯಶಸ್ವಿಯಾಗಲಿ ಅಥವಾ ಬಿಡಲಿ)!");
    });

ಇದರ ಪ್ರತಿಯೊಂದು ಸಾಲಿನ ವಿವರಣೆ:

  • orderFood : ನಾವು ಮೇಲೆ ಸೃಷ್ಟಿಸಿದ ಭರವಸೆಯ ಹೆಸರು.
  • .then(function(message) { ... }) : ಭರವಸೆ ಯಶಸ್ವಿಯಾದರೆ (ಅಂದರೆ ಮೇಲೆ resolve ಆಗಿದ್ದರೆ) ಮಾತ್ರ ಈ then ಎನ್ನುವ ಭಾಗ ಕೆಲಸ ಮಾಡುತ್ತದೆ.
  • message : ಮೇಲೆ ನಾವು resolve ಒಳಗೆ ಕಳುಹಿಸಿದ "ನಿಮ್ಮ ಬಿರಿಯಾನಿ ಡೆಲಿವರಿ ಆಗಿದೆ!" ಎಂಬ ವಾಕ್ಯವು ಈ message ಎಂಬ ಪದದೊಳಗೆ ಬಂದು ಕೂರುತ್ತದೆ.
  • console.log(...) : ಇದು ಕಂಪ್ಯೂಟರ್ ಸ್ಕ್ರೀನ್ (ಪರದೆ) ಮೇಲೆ ನಮ್ಮ ಸಂದೇಶವನ್ನು ತೋರಿಸಲು ಬಳಸುವ ಒಂದು ಸಿಂಪಲ್ ಕಮಾಂಡ್.
  • .catch(function(error) { ... }) : ಒಂದು ವೇಳೆ ಭರವಸೆ ಮುರಿದು ಬಿದ್ದರೆ (ಅಂದರೆ ಮೇಲೆ reject ಆಗಿದ್ದರೆ), ಆಗ ಆ ತಪ್ಪನ್ನು ಹಿಡಿಯಲು (catch ಮಾಡಲು) ಇದು ಕೆಲಸ ಮಾಡುತ್ತದೆ.
  • error : reject ಒಳಗೆ ಕಳುಹಿಸಿದ "ಕ್ಷಮಿಸಿ, ಬಿರಿಯಾನಿ ಖಾಲಿ ಆಗಿದೆ" ಎಂಬ ವಾಕ್ಯವು ಈ error ಎಂಬ ಪದದೊಳಗೆ ಬರುತ್ತದೆ.
  • .finally(function() { ... }) : ಫಲಿತಾಂಶ ಏನೇ ಆಗಿರಲಿ (Resolve ಆಗಿರಲಿ ಅಥವಾ Reject ಆಗಿರಲಿ), ಕೊನೆಯಲ್ಲಿ ಇದು ರನ್ ಆಗೇ ಆಗುತ್ತದೆ.

4. ಮಧ್ಯಮ ಮಟ್ಟ (Intermediate Level) - Promise Chaining

ಕೆಲವೊಮ್ಮೆ ನಾವು ಒಂದು ಕೆಲಸ ಮುಗಿದ ತಕ್ಷಣ, ಅದರ ಆಧಾರದ ಮೇಲೆ ಮತ್ತೊಂದು ಕೆಲಸವನ್ನು ಮಾಡಬೇಕಾಗುತ್ತದೆ. ಇದನ್ನು Promise Chaining ಎನ್ನುತ್ತಾರೆ.

ಉದಾಹರಣೆಗೆ: ಸೈಕಲ್ ಸಿಕ್ಕರೆ ಅದನ್ನು ಓಡಿಸಲು ಕಲಿಯಬೇಕು ಓಡಿಸಲು ಕಲಿತರೆ ಗೆಳೆಯರ ಜೊತೆ ರೇಸ್ ಮಾಡಬೇಕು.

// ಸೈಕಲ್ ಸಿಗುವ ಪ್ರಾಮಿಸ್
const getCycle = new Promise(function(resolve, reject) {
    resolve("ಸೈಕಲ್ ಸಿಕ್ಕಿತು");
});

getCycle
    .then(function(result) {
        console.log(result); // ಔಟ್‌ಪುಟ್: ಸೈಕಲ್ ಸಿಕ್ಕಿತು
        return "ಸೈಕಲ್ ಓಡಿಸಲು ಕಲಿತೆ"; // ಮುಂದಿನ ಹಂತಕ್ಕೆ ರವಾನಿಸುತ್ತೇವೆ
    })
    .then(function(nextResult) {
        console.log(nextResult); // ಔಟ್‌ಪುಟ್: ಸೈಕಲ್ ಓಡಿಸಲು ಕಲಿತೆ
        return "ಗೆಳೆಯರ ಜೊತೆ ರೇಸ್ ಮಾಡಿದೆ"; 
    })
    .then(function(finalResult) {
        console.log(finalResult); // ಔಟ್‌ಪುಟ್: ಗೆಳೆಯರ ಜೊತೆ ರೇಸ್ ಮಾಡಿದೆ
    })
    .catch(function(error) {
        // ನಡುವೆ ಎಲ್ಲಿಯಾದರೂ ದೋಷ ಬಂದರೆ ಈ catch ಅದನ್ನು ಹಿಡಿಯುತ್ತದೆ
        console.log("ಏನೋ ತಪ್ಪಾಗಿದೆ: ", error);
    });

ಇಲ್ಲಿ ನಾವು ಒಂದರ ನಂತರ ಒಂದರಂತೆ .then() ಅನ್ನು ಸರಪಳಿಯಂತೆ (Chain) ಜೋಡಿಸುತ್ತೇವೆ.

5. ವೃತ್ತಿಪರ ಮಟ್ಟ (Professional Level) - Async / Await

ಹಿಂದೆ .then() ಮತ್ತು .catch() ಬಳಸುತ್ತಿದ್ದುದು ಹಳೆಯ ವಿಧಾನ. ಈಗಿನ ವೃತ್ತಿಪರ ಡೆವಲಪರ್‌ಗಳು ಕೋಡ್ ಅನ್ನು ಇನ್ನಷ್ಟು ಸುಲಭವಾಗಿ ಮತ್ತು ಕ್ಲೀನ್ ಆಗಿ ಬರೆಯಲು Async / Await ಅನ್ನು ಬಳಸುತ್ತಾರೆ. ಇದು ಪ್ರಾಮಿಸ್ ಅನ್ನು ನಿರ್ವಹಿಸುವ ಆಧುನಿಕ ವಿಧಾನವಾಗಿದೆ.

// ಅದೇ ಸೈಕಲ್ ಉದಾಹರಣೆಯನ್ನು ಈಗ ವೃತ್ತಿಪರ ಶೈಲಿಯಲ್ಲಿ ಬರೆಯೋಣ
function buyCyclePro() {
    return new Promise(function(resolve, reject) {
        let shopOpen = true;
        
        // ಸ್ವಲ್ಪ ಸಮಯದ ನಂತರ (1.5 ಸೆಕೆಂಡ್) ತೀರ್ಮಾನ ಹೇಳಲು setTimeout ಬಳಕೆ
        setTimeout(function() {
            if (shopOpen) {
                resolve("ಯೇ! ಹೊಸ ಸೈಕಲ್ ಸಿಕ್ಕಿತು! 🚲");
            } else {
                reject("ಅಂಗಡಿ ಮುಚ್ಚಿದೆ. 😞");
            }
        }, 1500);
    });
}

// Async ಫಂಕ್ಷನ್ ರಚನೆ
async function getMyCycleTask() {
    try {
        console.log("ಅಪ್ಪ ಅಂಗಡಿಗೆ ಹೋಗಿದ್ದಾರೆ... ದಯವಿಟ್ಟು ಕಾಯಿರಿ...");
        
        // ಇಲ್ಲಿ 'await' ಎಂದರೆ Promise ನ ಫಲಿತಾಂಶ ಬರುವವರೆಗೂ ಕಾಯುವುದು
        const result = await buyCyclePro(); 
        
        console.log("ಫಲಿತಾಂಶ: ", result); // resolve ಆದರೆ ಇಲ್ಲಿಗೆ ಬರುತ್ತದೆ
    } catch (error) {
        console.log("ದೋಷ: ", error); // reject ಆದರೆ ಇಲ್ಲಿಗೆ ಬರುತ್ತದೆ
    }
}

// ಫಂಕ್ಷನ್ ಕರೆ ಮಾಡುವುದು
getMyCycleTask();

ವೃತ್ತಿಪರ ಸಲಹೆ: async ಫಂಕ್ಷನ್ ಒಳಗೆ ಮಾತ್ರ await ಅನ್ನು ಬಳಸಬಹುದು. ದೋಷಗಳನ್ನು (errors) ಹಿಡಿಯಲು try...catch ಬ್ಲಾಕ್ ಅನ್ನು ಬಳಸಲಾಗುತ್ತದೆ. ಇದು .then ಮತ್ತು .catch ಗಿಂತ ಓದಲು ಬಹಳ ಸುಲಭವಾಗಿರುತ್ತದೆ.

6. ಬಹುಮುಖ್ಯ Promise ವಿಧಾನಗಳು (Advanced Level)

ವೃತ್ತಿಪರ ಮಟ್ಟದಲ್ಲಿ ಒಂದೇ ಬಾರಿಗೆ ಹಲವು Promise ಗಳನ್ನು ನಿಭಾಯಿಸಬೇಕಾಗುತ್ತದೆ. ಅದಕ್ಕಾಗಿ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ 4 ಮುಖ್ಯ ವಿಧಾನಗಳನ್ನು ನೀಡಿದೆ:

1. Promise.all([p1, p2])

ಎಲ್ಲಾ ಪ್ರಾಮಿಸ್‌ಗಳು ಯಶಸ್ವಿಯಾದರೆ (resolve) ಮಾತ್ರ ಇದು ಯಶಸ್ವಿಯಾಗುತ್ತದೆ. ಯಾವುದಾದರೂ ಒಂದು ವಿಫಲವಾದರೂ (reject), ಇಡೀ ಪ್ರಕ್ರಿಯೆ ವಿಫಲವಾಗುತ್ತದೆ. (ಎಲ್ಲರೂ ಪಾಸ್ ಆಗಬೇಕು)

2. Promise.race([p1, p2])

ಇದರಲ್ಲಿ ಯಾವ ಪ್ರಾಮಿಸ್ ಅತ್ಯಂತ ವೇಗವಾಗಿ ಮುಗಿಯುತ್ತದೆಯೋ (ಪಾಸ್ ಆಗಲಿ ಅಥವಾ ಫೇಲ್ ಆಗಲಿ), ಅದರ ಫಲಿತಾಂಶವನ್ನು ಮಾತ್ರ ನೀಡುತ್ತದೆ. (ಓಟದ ಸ್ಪರ್ಧೆಯಂತೆ!)

3. Promise.allSettled([p1, p2])

ಎಲ್ಲಾ ಪ್ರಾಮಿಸ್‌ಗಳು ಮುಗಿಯುವವರೆಗೆ ಕಾಯುತ್ತದೆ. ಅವು ಪಾಸ್ ಆಗಿರಲಿ ಅಥವಾ ಫೇಲ್ ಆಗಿರಲಿ, ಎಲ್ಲದರ ವಿವರವಾದ ರಿಪೋರ್ಟ್ ನೀಡುತ್ತದೆ. (ಎಲ್ಲಾ ರಿಸಲ್ಟ್ ಬೇಕು)

4. Promise.any([p1, p2])

ಹಲವು ಪ್ರಾಮಿಸ್‌ಗಳಲ್ಲಿ ಯಾವುದಾದರೂ ಒಂದು ಯಶಸ್ವಿಯಾದರೂ ಸಾಕು, ಇದು ಯಶಸ್ವಿಯಾಗುತ್ತದೆ. ಎಲ್ಲವೂ ವಿಫಲವಾದರೆ ಮಾತ್ರ ಇದು ವಿಫಲವಾಗುತ್ತದೆ. (ಯಾರಾದರೂ ಒಬ್ಬರು ಪಾಸ್ ಆದರೆ ಸಾಕು)

ಉದಾಹರಣೆ (Promise.all):

const task1 = Promise.resolve("ಹೋಮ್‌ವರ್ಕ್ ಮುಗಿಯಿತು");
const task2 = Promise.resolve("ರೂಮ್ ಕ್ಲೀನ್ ಮಾಡಿದೆ");

Promise.all([task1, task2])
    .then(function(results) {
        console.log("ಎಲ್ಲಾ ಕೆಲಸ ಮುಗಿಯಿತು: ", results);
        // ಔಟ್‌ಪುಟ್: ಎಲ್ಲಾ ಕೆಲಸ ಮುಗಿಯಿತು: ["ಹೋಮ್‌ವರ್ಕ್ ಮುಗಿಯಿತು", "ರೂಮ್ ಕ್ಲೀನ್ ಮಾಡಿದೆ"]
    });

ಹೆಚ್ಚುವರಿ ಉದಾಹರಣೆ (Promise.race ಮತ್ತು Promise.any):

// ಯಾರು ಮೊದಲು ಬರುತ್ತಾರೋ ಅವರು ಗೆಲ್ಲುತ್ತಾರೆ (ಓಟದ ಸ್ಪರ್ಧೆ)
const fastTask = new Promise(function(resolve) { setTimeout(resolve, 100, "ಮೊದಲನೇಯವನು ಬಂದ! 🏃‍♂️"); });
const slowTask = new Promise(function(resolve) { setTimeout(resolve, 500, "ನಿಧಾನವಾಗಿ ಬಂದೆ... 🚶‍♂️"); });

Promise.race([fastTask, slowTask]).then(function(winner) {
    console.log("Race Result: ", winner); // ಔಟ್‌ಪುಟ್: ಮೊದಲನೇಯವನು ಬಂದ! 🏃‍♂️
});

ಸಾರಾಂಶ (Summary)

ನೀವು ನೆನಪಿಡಬೇಕಾದ್ದು ಕೇವಲ ಮೂರೇ ಅಂಶಗಳು:

  • 1 ಭರವಸೆ ನೀಡುವುದು: new Promise
  • 2 ಯಶಸ್ಸು ಮತ್ತು ವಿಫಲತೆ: resolve ಮತ್ತು reject
  • 3 ಫಲಿತಾಂಶ ಪಡೆಯುವುದು: .then (ಯಶಸ್ಸಿಗೆ) ಮತ್ತು .catch (ವಿಫಲತೆಗೆ) ಅಥವಾ ವೃತ್ತಿಪರ ಶೈಲಿಯಲ್ಲಿ async/await.
