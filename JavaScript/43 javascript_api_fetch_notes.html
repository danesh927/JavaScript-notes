API & Fetch Concepts

ಮೂಲ ಪರಿಕಲ್ಪನೆ (API & Fetch) ಮೂಲ ವಿಧಾನ (Syntax) ಸಾಂಪ್ರದಾಯಿಕ ಫಂಕ್ಷನ್ ಹಂತ 1: ತಳಮಟ್ಟ ಹಂತ 2: ಮಧ್ಯಮ ಮಟ್ಟ ಹಂತ 3: ವೃತ್ತಿಪರ ಮಟ್ಟ ಹೆಚ್ಚಿನ ಉದಾಹರಣೆಗಳು ಉದಾಹರಣೆ 1: GET Request ಉದಾಹರಣೆ 2: POST Request ಉದಾಹರಣೆ 3: Dynamic Fetch

API ಮತ್ತು Fetch ಎಂದರೇನು? (ಮೂಲ ಪರಿಕಲ್ಪನೆ)

API (Application Programming Interface):

ಸರಳವಾಗಿ ಹೇಳಬೇಕೆಂದರೆ, API ಎಂಬುದು ಇಬ್ಬರು ವ್ಯಕ್ತಿಗಳ ಅಥವಾ ಎರಡು ಸಾಫ್ಟ್‌ವೇರ್‌ಗಳ ನಡುವೆ ಕೆಲಸ ಮಾಡುವ ಒಬ್ಬ "ಮಧ್ಯವರ್ತಿ" (Mediator) ಅಥವಾ "ಸಂದೇಶವಾಹಕ" (Messenger).

ಹೋಟೆಲ್ ಉದಾಹರಣೆ:

ನೀವು ಹೋಟೆಲ್‌ನಲ್ಲಿ ಟೇಬಲ್ ಬಳಿ ಕುಳಿತಿದ್ದೀರಿ (ಇದು ನಿಮ್ಮ ಕಂಪ್ಯೂಟರ್ ಅಥವಾ ವೆಬ್‌ಸೈಟ್ - Frontend). ಹೋಟೆಲ್‌ನ ಒಳಗೆ ದೊಡ್ಡ ಅಡುಗೆ ಮನೆ ಇದೆ, ಅಲ್ಲಿ ಎಲ್ಲ ಬಗೆಯ ಊಟ ಇದೆ (ಇದು ಡೇಟಾಬೇಸ್ ಅಥವಾ ಸರ್ವರ್ - Database/Server). ಆದರೆ ನೀವು ನೇರವಾಗಿ ಅಡುಗೆ ಮನೆಗೆ ಹೋಗಿ ಊಟ ತರುವಂತಿಲ್ಲ. ಅದಕ್ಕಾಗಿ ನೀವು ಒಬ್ಬ ಸಪ್ಲೈಯರ್ ಅಥವಾ ವೈಟರ್ (Waiter) ಅನ್ನು ಕರೆಯುತ್ತೀರಿ. ನಿಮ್ಮ ಆರ್ಡರ್ ಅನ್ನು ಅಡುಗೆ ಮನೆಗೆ ತೆಗೆದುಕೊಂಡು ಹೋಗಿ, ಅಲ್ಲಿಂದ ಊಟವನ್ನು ನಿಮ್ಮ ಟೇಬಲ್‌ಗೆ ತಂದು ಕೊಡುವ ಆ ವೈಟರ್ ಹೆಸರೇ API.

ಗ್ರಾಹಕ / Frontend (ನಿಮ್ಮ ಕಂಪ್ಯೂಟರ್)
ಆರ್ಡರ್ / ಡೇಟಾ
Waiter (API)
ಅಡುಗೆ ಮನೆ / Server (ಡೇಟಾಬೇಸ್)

Fetch (ಫೆಚ್):

ಇಂಗ್ಲಿಷ್‌ನಲ್ಲಿ 'Fetch' ಎಂದರೆ "ಹೋಗಿ ತೆಗೆದುಕೊಂಡು ಬಾ" ಎಂದರ್ಥ. ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ fetch() ಎನ್ನುವುದು ನೀವು ವೈಟರ್‌ಗೆ ಕೊಡುವ ಆದೇಶ (Order). "ಇಂಟರ್ನೆಟ್‌ನಲ್ಲಿರುವ ಆ ನಿರ್ದಿಷ್ಟ ಜಾಗಕ್ಕೆ ಹೋಗಿ, ನನಗೆ ಬೇಕಾದ ಮಾಹಿತಿಯನ್ನು ತೆಗೆದುಕೊಂಡು ಬಾ" ಎಂದು ಪ್ರೋಗ್ರಾಮ್‌ಗೆ ಹೇಳಲು ನಾವು fetch() ಕಮಾಂಡ್ ಅನ್ನು ಬಳಸುತ್ತೇವೆ.

ಇದನ್ನು ಬರೆಯುವ ಮೂಲ ವಿಧಾನ (Syntax):

ಯಾವಾಗಲೂ ನಾವು fetch('ಮಾಹಿತಿ ಇರುವ ವಿಳಾಸ ಅಥವಾ URL') ಎಂದು ಬರೆಯುತ್ತೇವೆ. ಆನಂತರ ಅದು ತರುವ ಮಾಹಿತಿಯನ್ನು ಸ್ವೀಕರಿಸಲು .then(function() {}) ಅಥವಾ async/await ಬಳಸುತ್ತೇವೆ.

(ಇಲ್ಲಿಂದ ನೀವು ನೀಡಿದ್ದ ಮಾಹಿತಿ ಯಥಾವತ್ತಾಗಿ ಪ್ರಾರಂಭವಾಗುತ್ತದೆ)

ಖಂಡಿತ, ಹೊಸಬರಿಗೆ (Non-IT) Arrow Function (=>)ಅರ್ಥಮಾಡಿಕೊಳ್ಳಲು ಸ್ವಲ್ಪ ಗೊಂದಲವಾಗಬಹುದು. ಹಾಗಾಗಿ, ನೀವು ಕೇಳಿದಂತೆಯೇ ಸಂಪೂರ್ಣವಾಗಿ ಸಾಂಪ್ರದಾಯಿಕ ಫಂಕ್ಷನ್ (function() {}) ವಿಧಾನವನ್ನೇ ಬಳಸಿ ಹಂತ-ಹಂತವಾಗಿ ವಿವರಿಸುತ್ತೇನೆ.

ಇಲ್ಲಿ ನಾವು ಅದೇ ಹೋಟೆಲ್ ಉದಾಹರಣೆಯನ್ನು ನೆನಪಿನಲ್ಲಿಟ್ಟುಕೊಳ್ಳೋಣ (ನೀವು ಗ್ರಾಹಕ, Fetch ಸಪ್ಲೈಯರ್, ಡೇಟಾ ಎಂಬುದು ಊಟ). ಪೂರ್ಣ ಫಂಕ್ಷನ್ ಬಳಸುವಾಗ ನಾವು return ಎಂಬ ಪದವನ್ನು ಬಳಸಬೇಕಾಗುತ್ತದೆ. return ಎಂದರೆ ಸಪ್ಲೈಯರ್ ತಂದ ಊಟವನ್ನು ಟೇಬಲ್ ಮೇಲೆ 'ಇಡು' ಅಥವಾ ಮುಂದಿನ ಹಂತಕ್ಕೆ 'ಕೊಡು' ಎಂದರ್ಥ.

ಹಂತ 1: ತಳಮಟ್ಟ - ಸಾಮಾನ್ಯ ಪಠ್ಯವನ್ನು (Text Data) ತರುವುದು

ನಿಮ್ಮ ಬಳಿ message.txt ಎಂಬ ಫೈಲ್ ಇದೆ ಎಂದುಕೊಳ್ಳಿ. ಅದರಲ್ಲಿ "ನಮಸ್ಕಾರ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಜಗತ್ತಿಗೆ ಸ್ವಾಗತ!" ಎಂದು ಬರೆಯಲಾಗಿದೆ. 

// 1. ಸಪ್ಲೈಯರ್‌ಗೆ ಆರ್ಡರ್ ನೀಡುವುದು (Fetch)
fetch('message.txt')
 
  // 2. ಸಪ್ಲೈಯರ್ ತಂದ ಪಾರ್ಸೆಲ್ ಅನ್ನು ಬಿಡಿಸುವುದು
  .then(function(response) {
      // ತಂದಿರುವ ಪಾರ್ಸೆಲ್ ಅನ್ನು ಕೇವಲ ಅಕ್ಷರಗಳ (text) ರೂಪದಲ್ಲಿ ಬದಲಾಯಿಸಿ ಮುಂದಿನ ಹಂತಕ್ಕೆ ಕೊಡು (return)
      return response.text(); 
  })
 
  // 3. ಬಿಡಿಸಿದ ಪಾರ್ಸೆಲ್ (ಡೇಟಾ) ಅನ್ನು ಓದುವುದು
  .then(function(data) {
      // ಈಗ ಡೇಟಾ ಓದಲು ಸಿದ್ಧವಾಗಿದೆ, ಅದನ್ನು ಪರದೆಯ ಮೇಲೆ ತೋರಿಸಿ
      console.log("ಫೈಲ್‌ನಲ್ಲಿದ್ದ ಮಾಹಿತಿ: " + data);
  });

ಹಂತ 2: ಮಧ್ಯಮ ಮಟ್ಟ - JSON ಡೇಟಾ ತರುವುದು (ವ್ಯವಸ್ಥಿತ ಮಾಹಿತಿ)

ಈಗ ಮಾಹಿತಿಯನ್ನು JSON (ಒಂದು ಫಾರ್ಮ್‌ನಂತೆ) ರೂಪದಲ್ಲಿ ತರೋಣ. ನಿಮ್ಮ ಬಳಿ student.json ಫೈಲ್ ಇದೆ (ಅದರಲ್ಲಿ ರಾಮ, 22 ವರ್ಷ, ಬೆಳಗಾವಿ ಎಂಬ ಮಾಹಿತಿ ಇದೆ). 

fetch('student.json')
 
  // ಮೊದಲನೇ ಫಂಕ್ಷನ್: ಮಾಹಿತಿಯನ್ನು ಸ್ವೀಕರಿಸುವುದು
  .then(function(response) {
      // ತಂದಿರುವ ಪಾರ್ಸೆಲ್ ಅನ್ನು JSON ಆಗಿ ಬದಲಾಯಿಸಿ ಮುಂದಿನ ಹಂತಕ್ಕೆ ಕೊಡು (return)
      return response.json(); 
  })
 
  // ಎರಡನೇ ಫಂಕ್ಷನ್: ಮಾಹಿತಿಯನ್ನು ಬಳಸುವುದು
  .then(function(studentData) {
      // ಈಗ ನಮಗೆ ಸರಿಯಾದ ಫಾರ್ಮ್ಯಾಟ್‌ನಲ್ಲಿ ಮಾಹಿತಿ ಸಿಕ್ಕಿದೆ
      console.log("ವಿದ್ಯಾರ್ಥಿಯ ಹೆಸರು: " + studentData.name);
      console.log("ವಯಸ್ಸು: " + studentData.age);
  });

ಹಂತ ಹಂತದ ವಿವರಣೆ:

  • fetch ಸಪ್ಲೈಯರ್‌ನಂತೆ ಹೋಗಿ student.json ಫೈಲ್ ತರುತ್ತದೆ.
  • ಮೊದಲ .then ಒಳಗೆ ಇರುವ function(response) ಆ ಫೈಲ್ ಅನ್ನು JSON ಆಗಿ ಪರಿವರ್ತಿಸಿ, return ಮೂಲಕ ಎರಡನೇ ಹಂತಕ್ಕೆ ಕಳುಹಿಸುತ್ತದೆ.
  • ಎರಡನೇ .then ಒಳಗೆ ಇರುವ function(studentData) ಆ ಮಾಹಿತಿಯನ್ನು ತೆಗೆದುಕೊಂಡು ನಮಗೆ ಬೇಕಾದ ಹೆಸರನ್ನು (studentData.name) ಮಾತ್ರ ಪ್ರಿಂಟ್ ಮಾಡುತ್ತದೆ.

ಹಂತ 3: ವೃತ್ತಿಪರ ಮಟ್ಟ - ಪೂರ್ಣ ಫಂಕ್ಷನ್ ಮತ್ತು Async/Await

ಐಟಿ ಕಂಪನಿಗಳಲ್ಲಿ .then ಅನ್ನು ಪದೇ ಪದೇ ಬಳಸುವುದರಿಂದ ಕೋಡ್ ನೋಡಲು ಉದ್ದವಾಗುತ್ತದೆ ಎಂದು ವೃತ್ತಿಪರರು async ಮತ್ತು await ಬಳಸುತ್ತಾರೆ. ಇದು ಸಂಪೂರ್ಣವಾಗಿ ನೀವು ಕೇಳಿದ "Full Function" ಶೈಲಿಯಲ್ಲೇ ಇರುತ್ತದೆ ಮತ್ತು ಓದಲು ಕಥೆಯಂತೆ ಸರಳವಾಗಿರುತ್ತದೆ. 

// ಒಂದು ಪೂರ್ಣ ಫಂಕ್ಷನ್ ರಚಿಸುವುದು. 
// async ಎಂದರೆ "ಈ ಕೆಲಸ ಸ್ವಲ್ಪ ಸಮಯ ತೆಗೆದುಕೊಳ್ಳುತ್ತದೆ, ಕಾಯಿರಿ" ಎಂದು ಹೇಳುವುದು.
async function getStudentInfo()
{
    try {
        // 1. ಸಪ್ಲೈಯರ್ ಫೈಲ್ ತರುವವರೆಗೆ 'ಕಾಯಿರಿ' (await)
        let response = await fetch('student.json');
 
        // 2. ಒಂದು ವೇಳೆ ಸಪ್ಲೈಯರ್ ಖಾಲಿ ಕೈಯಲ್ಲಿ ಬಂದರೆ (ಫೈಲ್ ಸಿಗಲಿಲ್ಲ ಎಂದರೆ)
        if (response.ok === false) {
            // ಇದು ಕೋಡ್ ಅನ್ನು ಇಲ್ಲೇ ನಿಲ್ಲಿಸಿ, ಕೆಳಗಿರುವ 'catch' ವಿಭಾಗಕ್ಕೆ ಕಳುಹಿಸುತ್ತದೆ
            throw new Error("ಕ್ಷಮಿಸಿ, ನೀವು ಕೇಳಿದ ಫೈಲ್ ಸಿಗಲಿಲ್ಲ!");
        }
 
        // 3. ಫೈಲ್ ಸಿಕ್ಕರೆ, ಅದನ್ನು JSON ಆಗಿ ಪರಿವರ್ತಿಸಲು ಕಾಯಿರಿ (await)
        let data = await response.json();
 
        // 4. ಮಾಹಿತಿಯನ್ನು ಪ್ರಿಂಟ್ ಮಾಡಿ
        console.log("ವೃತ್ತಿಪರ ವಿಧಾನದಲ್ಲಿ ಹೆಸರು: " + data.name);
        console.log("ವೃತ್ತಿಪರ ವಿಧಾನದಲ್ಲಿ ಊರು: " + data.city);
 
    } catch (functionError) {
        // ಏನಾದರೂ ತೊಂದರೆಯಾದರೆ (ಉದಾಹರಣೆಗೆ ಫೈಲ್ ಇಲ್ಲದಿದ್ದರೆ), ಈ ಬ್ಲಾಕ್ ಅದನ್ನು ತೋರಿಸುತ್ತದೆ
        console.log("ಏನೋ ತಪ್ಪಾಗಿದೆ: " + functionError.message);
    }
}
 
// ನಾವೇ ಬರೆದ ಮೇಲಿನ ಫಂಕ್ಷನ್ ಅನ್ನು ಕೆಲಸ ಮಾಡಲು ಆದೇಶಿಸುವುದು
getStudentInfo();

ವೃತ್ತಿಪರ ವಿಧಾನದ ಲಾಭಗಳು:

  1. ಇಲ್ಲಿ ಎಲ್ಲಿಯೂ .then ಅಥವಾ ಗೊಂದಲಮಯ ಕೋಡ್ ಇಲ್ಲ. ಎಲ್ಲವೂ ಸಾಂಪ್ರದಾಯಿಕ function ನಂತೆಯೇ ಇದೆ.
  2. await ಪದವು ಸಪ್ಲೈಯರ್ ಹೋಗಿ ಬರುವವರೆಗೆ ಕೋಡ್ ಅನ್ನು ಮುಂದಕ್ಕೆ ಹೋಗದಂತೆ ತಡೆಯುತ್ತದೆ.
  3. try ಮತ್ತು catch ಬಳಸಿದ್ದರಿಂದ, ಏನೇ ತಪ್ಪು ನಡೆದರೂ (Error) ಪ್ರೋಗ್ರಾಮ್ ಕ್ರ್ಯಾಶ್ ಆಗುವುದಿಲ್ಲ, ಬದಲಾಗಿ ನಮಗೆ ಅರ್ಥವಾಗುವ ಭಾಷೆಯಲ್ಲಿ ತಪ್ಪನ್ನು ತೋರಿಸುತ್ತದೆ.

(ನಿಮ್ಮ ಮಾಹಿತಿ ಇಲ್ಲಿಗೆ ಮುಕ್ತಾಯವಾಯಿತು)

ಇನ್ನಷ್ಟು ಹೆಚ್ಚಿನ ಉದಾಹರಣೆಗಳು (ಸಂಪೂರ್ಣ ವಿವರಣೆಯೊಂದಿಗೆ)

ಈಗ ನೀವು API ಮತ್ತು Fetch ಪರಿಕಲ್ಪನೆಯನ್ನು ಅರ್ಥಮಾಡಿಕೊಂಡಿದ್ದೀರಿ. ಇದನ್ನು ಬಳಸಿ ನಿಜವಾದ ಇಂಟರ್ನೆಟ್‌ನಿಂದ ಮಾಹಿತಿ ಹೇಗೆ ತರುವುದು ಮತ್ತು ಕಳುಹಿಸುವುದು ಎಂಬುದಕ್ಕೆ 3 ಉದಾಹರಣೆಗಳು ಇಲ್ಲಿವೆ:

ಉದಾಹರಣೆ 1: ಇಂಟರ್ನೆಟ್‌ನಲ್ಲಿರುವ ಬೇರೊಂದು ಸರ್ವರ್‌ನಿಂದ (API) ಮಾಹಿತಿ ತರುವುದು GET Request

ನಾವೀಗ ಉಚಿತವಾಗಿ ಸಿಗುವ ಒಂದು ನಿಜವಾದ API ನಿಂದ ಒಬ್ಬ ಬಳಕೆದಾರನ (User) ಮಾಹಿತಿಯನ್ನು ತರೋಣ. 

fetch('https://jsonplaceholder.typicode.com/users/1')
  .then(function(response) {
      // 1. ಮೊದಲು ಸಪ್ಲೈಯರ್ ತಂದ ಮಾಹಿತಿಯನ್ನು (ಪಾರ್ಸೆಲ್) JSON ಫಾರ್ಮ್ಯಾಟ್‌ಗೆ ಬದಲಾಯಿಸುತ್ತೇವೆ.
      return response.json();
  })
  .then(function(userData) {
      // 2. ಈಗ ಆ ಪಾರ್ಸೆಲ್ ಬಿಡಿಸಿದಾಗ ಒಳಗೆ ಆ ಬಳಕೆದಾರನ ಸಂಪೂರ್ಣ ಮಾಹಿತಿ ಇದೆ.
      console.log("ಬಳಕೆದಾರನ ಹೆಸರು: " + userData.name);
      console.log("ಬಳಕೆದಾರನ ಇಮೇಲ್: " + userData.email);
  })
  .catch(function(error) {
      // 3. ಒಂದು ವೇಳೆ ಇಂಟರ್ನೆಟ್ ಇಲ್ಲದಿದ್ದರೆ ಅಥವಾ ಸರ್ವರ್ ಡೌನ್ ಇದ್ದರೆ ಇಲ್ಲಿಗೆ ಬರುತ್ತದೆ.
      console.log("ಮಾಹಿತಿ ತರುವಾಗ ತೊಂದರೆಯಾಗಿದೆ: " + error);
  });

ವಿವರಣೆ: ಇಲ್ಲಿ ನಾವು ನಮ್ಮದೇ ಕಂಪ್ಯೂಟರ್‌ನಲ್ಲಿರುವ ಫೈಲ್ ಬದಲಿಗೆ https://... ಎಂಬ ವೆಬ್‌ಸೈಟ್ ವಿಳಾಸ ಕೊಟ್ಟಿದ್ದೇವೆ. ನಮ್ಮ ವೈಟರ್ (fetch) ಅಲ್ಲಿಗೆ ಹೋಗಿ, 1ನೇ ನಂಬರಿನ ಬಳಕೆದಾರನ ಡೇಟಾ ತಂದು ನಮಗೆ ತೋರಿಸುತ್ತಿದ್ದಾನೆ. ಇದು ಬೇರೆ ವೆಬ್‌ಸೈಟ್‌ಗಳಿಂದ ಮಾಹಿತಿ ಕದಿಯುವುದಲ್ಲ, ಬದಲಾಗಿ ಅವರು ನಮಗಾಗಿ ಮಾಡಿರುವ ವ್ಯವಸ್ಥೆ (API) ಮೂಲಕ ಗೌರವಯುತವಾಗಿ ಕೇಳಿ ಪಡೆಯುವುದು.

ಉದಾಹರಣೆ 2: ನಮ್ಮ ಮಾಹಿತಿಯನ್ನು ಸರ್ವರ್‌ಗೆ ಕಳುಹಿಸುವುದು POST Request

ಇದುವರೆಗೂ ನಾವು ಊಟ ತರಿಸಿದ್ದೆವು. ಈಗ ನಾವು ಅಡುಗೆ ಮನೆಗೆ "ಈ ರೀತಿ ಹೊಸ ಅಡುಗೆ ಮಾಡಿ" ಎಂದು ರೆಸಿಪಿ (ಮಾಹಿತಿ) ಕಳುಹಿಸುವ ಉದಾಹರಣೆ ನೋಡೋಣ. ಇದನ್ನು ವೃತ್ತಿಪರ ಶೈಲಿಯಲ್ಲಿ (async/await) ಬರೆಯೋಣ. ಉದಾಹರಣೆಗೆ: ವೆಬ್‌ಸೈಟ್‌ನಲ್ಲಿ ಹೊಸ ಖಾತೆ (Account) ತೆರೆಯುವುದು. 

async function createNewUser() {
    try {
        // 1. ನಾವು ಸರ್ವರ್‌ಗೆ ಕಳುಹಿಸಬೇಕಾದ ಹೊಸ ಮಾಹಿತಿ
        let myNewData = {
            name: "ವಿಕಾಸ್",
            job: "ಸಾಫ್ಟ್‌ವೇರ್ ಡೆವಲಪರ್"
        };

        // 2. ಸಪ್ಲೈಯರ್‌ಗೆ (fetch) ಎಲ್ಲಿಗೆ ಹೋಗಬೇಕು ಮತ್ತು ಏನು ತೆಗೆದುಕೊಂಡು ಹೋಗಬೇಕು ಎಂದು ಹೇಳುವುದು
        let response = await fetch('https://reqres.in/api/users', {
            method: 'POST', // POST ಎಂದರೆ ಹೊಸದನ್ನು ಸೃಷ್ಟಿಸಲು ಮಾಹಿತಿ ಕಳುಹಿಸುವುದು
            headers: {
                'Content-Type': 'application/json' // ನಾವು ಕಳುಹಿಸುತ್ತಿರುವುದು JSON ಮಾಹಿತಿ ಎಂದು ಹೇಳುವುದು
            },
            body: JSON.stringify(myNewData) // ನಮ್ಮ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಮಾಹಿತಿಯನ್ನು JSON ಆಗಿ ಪ್ಯಾಕ್ ಮಾಡುವುದು
        });

        // 3. ಸರ್ವರ್ ನಮ್ಮ ಮಾಹಿತಿಯನ್ನು ಪಡೆದ ಮೇಲೆ ಅದು ಕೊಡುವ ರಸೀದಿ (Response)
        let serverReply = await response.json();
        
        console.log("ಹೊಸ ಖಾತೆ ಯಶಸ್ವಿಯಾಗಿ ಸೃಷ್ಟಿಯಾಗಿದೆ!");
        console.log("ನಿಮ್ಮ ಹೊಸ ಐಡಿ ನಂಬರ್: " + serverReply.id);

    } catch (functionError) {
        console.log("ಖಾತೆ ಸೃಷ್ಟಿಸುವಲ್ಲಿ ದೋಷ: " + functionError.message);
    }
}

// ಫಂಕ್ಷನ್ ಪ್ರಾರಂಭಿಸುವುದು
createNewUser();

ವಿವರಣೆ: ಇಲ್ಲಿ method: 'POST' ಬಹಳ ಮುಖ್ಯ. ಸಾಮಾನ್ಯ fetch ಕೇವಲ ಮಾಹಿತಿಯನ್ನು ತರುತ್ತದೆ (GET). ಆದರೆ POST ಬಳಸಿದಾಗ ನಮ್ಮ ಡೇಟಾವನ್ನು ಸರ್ವರ್‌ಗೆ ಸೇರಿಸುತ್ತದೆ. body ಒಳಗೆ ನಾವು ನಮ್ಮ ಮಾಹಿತಿಯನ್ನು ಪ್ಯಾಕ್ ಮಾಡಿ ಕಳುಹಿಸುತ್ತೇವೆ. ಸರ್ವರ್ ಆ ಮಾಹಿತಿಯನ್ನು ಸ್ವೀಕರಿಸಿ, ಖಾತೆ ಸೃಷ್ಟಿಸಿ, ನಮಗೊಂದು ID ನಂಬರ್ ವಾಪಸ್ ಕಳುಹಿಸುತ್ತದೆ.

ಉದಾಹರಣೆ 3: ಬಟನ್ ಒತ್ತಿದಾಗ ನಾಯಿಯ ಚಿತ್ರ ತರುವುದು Dynamic API Fetch

ವೆಬ್‌ಸೈಟ್‌ನಲ್ಲಿ "ನಾಯಿಮರಿಯ ಚಿತ್ರ ತೋರಿಸು" ಎಂಬ ಬಟನ್ ಒತ್ತಿದಾಗಲೆಲ್ಲಾ ಬೇರೆ ಬೇರೆ ನಾಯಿಯ ಚಿತ್ರ ಬರುವಂತೆ ಮಾಡುವುದು ಹೇಗೆಂದು ನೋಡೋಣ. 

// ಒಂದು ಬಟನ್ (button) ಕ್ಲಿಕ್ ಮಾಡಿದಾಗ ಕೆಲಸ ಮಾಡುವ ಫಂಕ್ಷನ್
async function getDogPicture() {
    try {
        console.log("ಚಿತ್ರವನ್ನು ಹುಡುಕಲಾಗುತ್ತಿದೆ, ದಯವಿಟ್ಟು ಕಾಯಿರಿ...");

        // 1. ನಾಯಿಯ ಚಿತ್ರಗಳಿಗಾಗಿಯೇ ಇರುವ ಉಚಿತ API ನಿಂದ ಮಾಹಿತಿ ಕೇಳುವುದು
        let response = await fetch('https://dog.ceo/api/breeds/image/random');
        
        // 2. ಡೇಟಾವನ್ನು JSON ರೂಪಕ್ಕೆ ಪರಿವರ್ತಿಸಲು ಕಾಯುವುದು
        let data = await response.json();
        
        // 3. ಸರ್ವರ್ ಯಶಸ್ವಿಯಾಗಿ ಚಿತ್ರದ ಲಿಂಕ್ ಕಳುಹಿಸಿದ್ದರೆ ಅದನ್ನು ಪ್ರಿಂಟ್ ಮಾಡುವುದು
        if (data.status === "success") {
            console.log("ಚಿತ್ರದ ವಿಳಾಸ ಇಲ್ಲಿದೆ: " + data.message);
            // ವಾಸ್ತವದಲ್ಲಿ ಈ 'data.message' ಎಂಬ ಲಿಂಕ್ ಅನ್ನು ನಾವು HTML ನ  ಟ್ಯಾಗ್ ಒಳಗೆ ಹಾಕುತ್ತೇವೆ.
        }

    } catch (error) {
        console.log("ಚಿತ್ರ ತರುವಲ್ಲಿ ವಿಫಲವಾಗಿದೆ!");
    }
}

// ಬಟನ್ ಒತ್ತಿದಾಗ ಈ ಕೆಳಗಿನಂತೆ ಫಂಕ್ಷನ್ ಕರೆಯಬಹುದು
getDogPicture();

ವಿವರಣೆ: ನೀವು ಅಮೆಜಾನ್ ಅಥವಾ ಫ್ಲಿಪ್‌ಕಾರ್ಟ್‌ನಲ್ಲಿ ವಸ್ತುಗಳನ್ನು ಹುಡುಕಿದಾಗ ಪ್ರತಿ ಬಾರಿಯೂ ಬೇರೆ ಬೇರೆ ಪ್ರಾಡಕ್ಟ್‌ಗಳು ಪರದೆಯ ಮೇಲೆ ಬರುವುದು ಇದೇ ರೀತಿ! ನಾವು fetch ಮೂಲಕ ಆಯಾ ವೆಬ್‌ಸೈಟ್‌ನ API ಗೆ ರಿಕ್ವೆಸ್ಟ್ ಕಳುಹಿಸುತ್ತೇವೆ, ಅವರು ಆ ಕ್ಷಣದ ಲೈವ್ ಮಾಹಿತಿಯನ್ನು (ಚಿತ್ರ, ಬೆಲೆ, ಇತ್ಯಾದಿ) ಕಳುಹಿಸುತ್ತಾರೆ, ಅದನ್ನು ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ ಸಹಾಯದಿಂದ ಪರದೆಯ ಮೇಲೆ ತೋರಿಸುತ್ತೇವೆ.
