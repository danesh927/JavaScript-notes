ಪರಿವಿಡಿ (TOC)

ಪ್ರಸ್ತಾವನೆ 1. ತಳಮಟ್ಟದ ಪರಿಕಲ್ಪನೆ ದೃಶ್ಯ ವಿವರಣೆ (Diagram) 2. ಉದಾಹರಣೆಗಳು ಉದಾ: ಟೈಮರ್ ಉದಾ: ಪಾಸ್ / ಫೇಲ್ ಉದಾ: ಸರಪಳಿ ವಿಧಾನ ಉದಾ: Async / Await ಹೆಚ್ಚುವರಿ ಮಾಹಿತಿ ಸಾರಾಂಶ

ಖಂಡಿತ, ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ (JavaScript) ಕಾಲ್‌ಬ್ಯಾಕ್ (Callback) ಗಳನ್ನು ಪ್ರಾಮಿಸ್ (Promise) ಗಳಾಗಿ ಹೇಗೆ ಬದಲಾಯಿಸುವುದು ಎಂಬುದನ್ನು ಚಿಕ್ಕ ಮಕ್ಕಳಿಗೂ ಅರ್ಥವಾಗುವಂತಹ ಸುಲಭವಾದ ಕಥೆಯ ಮೂಲಕ, ತಳಮಟ್ಟದಿಂದ ವೃತ್ತಿಪರ ಮಟ್ಟದವರೆಗೆ ಹಂತ ಹಂತವಾಗಿ ಕಲಿಯೋಣ.

1. ತಳಮಟ್ಟದ ಪರಿಕಲ್ಪನೆ: ಕಥೆಯ ಮೂಲಕ ಅರ್ಥಮಾಡಿಕೊಳ್ಳೋಣ!

ಕಾಲ್‌ಬ್ಯಾಕ್ (Callback) ಎಂದರೆ ಏನು?

ನೀವು ಒಂದು ಹೋಟೆಲ್‌ಗೆ ಹೋಗಿ ಮಸಾಲೆ ದೋಸೆ ಆರ್ಡರ್ ಮಾಡುತ್ತೀರಿ ಎಂದುಕೊಳ್ಳಿ. ವೇಟರ್ ನಿಮ್ಮ ಬಳಿ ಬಂದು, "ಸರ್, ದೋಸೆ ರೆಡಿಯಾದ ತಕ್ಷಣ ನಾನೇ ಬಂದು ನಿಮಗೆ ಕರೆಯುತ್ತೇನೆ, ಅಲ್ಲಿಯವರೆಗೆ ಕಾಯಿರಿ" ಎನ್ನುತ್ತಾರೆ.

ಇಲ್ಲಿ ವೇಟರ್ ನಿಮಗೆ ದೋಸೆ ತಂದುಕೊಡುವುದು ಒಂದು Callback. ಆದರೆ ಇಲ್ಲಿ ಒಂದು ಸಮಸ್ಯೆಯಿದೆ: ನೀವು ಆ ವೇಟರ್ ಮೇಲೆಯೇ ಸಂಪೂರ್ಣವಾಗಿ ಅವಲಂಬಿತರಾಗಿರುತ್ತೀರಿ.

ಪ್ರಾಮಿಸ್ (Promise) ಎಂದರೆ ಏನು?

ಈಗ ನೀವು ಅದೇ ಮಸಾಲೆ ದೋಸೆಯನ್ನು ಕೆಎಫ್‌ಸಿ (KFC) ಅಥವಾ ಮೆಕ್‌ಡೊನಾಲ್ಡ್ಸ್‌ನಂತಹ ಫಾಸ್ಟ್ ಫುಡ್ ಅಂಗಡಿಯಲ್ಲಿ ಆರ್ಡರ್ ಮಾಡುತ್ತೀರಿ. ಅವರು ನಿಮಗೆ ಒಂದು 'ಬಿಲ್ ರಶೀದಿ ಅಥವಾ ಟೋಕನ್ ನಂಬರ್' ಕೊಡುತ್ತಾರೆ.

ಈ ಟೋಕನ್ ನಿಮಗೆ ನೀಡಿದ ಭರವಸೆ (Promise).

ದೃಶ್ಯ ವಿವರಣೆ: Callback vs Promise

Callback
ವೇಟರ್ ಮೇಲೆ ಅವಲಂಬಿತ (Blocked)
Promise
TOKEN #42
ಸ್ವತಂತ್ರ - ಕೆಲಸ ಮುಗಿದಾಗ ಸೂಚನೆ (Async)

ಈ ಪ್ರಾಮಿಸ್‌ಗೆ 3 ಹಂತಗಳಿವೆ:

  • Pending (ಕಾಯುವಿಕೆ): ದೋಸೆ ಇನ್ನೂ ತಯಾರಾಗುತ್ತಿದೆ.
  • Resolved / Fulfilled (ಯಶಸ್ಸು): ನಿಮ್ಮ ಟೋಕನ್ ನಂಬರ್ ಬಂತು, ದೋಸೆ ಸಿದ್ಧವಾಗಿದೆ!
  • Rejected (ವೈಫಲ್ಯ): ದೋಸೆ ಹಿಟ್ಟು ಖಾಲಿಯಾಗಿದೆ, ನಿಮಗೆ ದೋಸೆ ಸಿಗುವುದಿಲ್ಲ ಎಂದು ಅವರು ಹೇಳುವುದು.

ನಾವು ಕಾಲ್‌ಬ್ಯಾಕ್ ಅನ್ನು ಪ್ರಾಮಿಸ್ ಆಗಿ ಏಕೆ ಬದಲಾಯಿಸಬೇಕು?

ಬಹಳಷ್ಟು ಕಾಲ್‌ಬ್ಯಾಕ್‌ಗಳನ್ನು ಒಟ್ಟಿಗೆ ಬಳಸಿದಾಗ, ಕೋಡ್ ನೋಡಲು ಬೆಟ್ಟದಂತೆ (Pyramid of Doom / Callback Hell) ಗೊಂದಲಮಯವಾಗುತ್ತದೆ. ಪ್ರಾಮಿಸ್ ಬಳಸಿದರೆ ಕೋಡ್ ತುಂಬಾ ಸ್ವಚ್ಛವಾಗಿ ಮತ್ತು ಸುಲಭವಾಗಿ ಅರ್ಥವಾಗುತ್ತದೆ.

ಕಾಲ್‌ಬ್ಯಾಕ್ ಅನ್ನು ಪ್ರಾಮಿಸ್ ಆಗಿ ಬದಲಾಯಿಸುವ 4 ಪ್ರಮುಖ ಹಂತಗಳು:

  1. ಫಂಕ್ಷನ್ ಒಳಗಡೆ return new Promise((resolve, reject) => { ... }) ಎಂದು ಬರೆಯಿರಿ.
  2. ನಿಮ್ಮ ಹಳೆಯ ಕೆಲಸವನ್ನು (ಅಸಿಂಕ್ರೊನಸ್ ಕೆಲಸ) ಇದರ ಒಳಗೆ ಹಾಕಿ.
  3. ಕೆಲಸ ಯಶಸ್ವಿಯಾದರೆ ಕಾಲ್‌ಬ್ಯಾಕ್ ಬದಲಿಗೆ resolve(data) ಬಳಸಿ.
  4. ಕೆಲಸ ವಿಫಲವಾದರೆ reject(error) ಬಳಸಿ.

2. ಉದಾಹರಣೆಗಳು ಮತ್ತು ಹಂತ-ಹಂತದ ವಿವರಣೆ

ಈಗ 4 ವಿಭಿನ್ನ ಉದಾಹರಣೆಗಳೊಂದಿಗೆ ಇದನ್ನು ಕಲಿಯೋಣ.

ಉದಾಹರಣೆ 1: ಅತಿ ಸರಳವಾದ ಟೈಮರ್ (Basic Timer)

ಒಂದು ಕೆಲಸ 2 ಸೆಕೆಂಡುಗಳ ನಂತರ ಆಗಬೇಕು ಎಂದುಕೊಳ್ಳೋಣ.

❌ ಹಳೆಯ ಕಾಲ್‌ಬ್ಯಾಕ್ ವಿಧಾನ:

function waitAndSpeakCallback(callback) {
    setTimeout(() => {
        callback("2 ಸೆಕೆಂಡ್ ಮುಗಿಯಿತು!");
    }, 2000);
}

waitAndSpeakCallback((message) => {
    console.log(message);
});

✅ ಹೊಸ ಪ್ರಾಮಿಸ್ ವಿಧಾನ:

function waitAndSpeakPromise() {
    // ಹಂತ 1: ಪ್ರಾಮಿಸ್ ಅನ್ನು ರಿಟರ್ನ್ ಮಾಡುವುದು
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            // ಹಂತ 2: ಕಾಲ್‌ಬ್ಯಾಕ್ ಬದಲು resolve ಬಳಸುವುದು
            resolve("2 ಸೆಕೆಂಡ್ ಮುಗಿಯಿತು!"); 
        }, 2000);
    });
}

// ಹಂತ 3: ಭರವಸೆ ಈಡೇರಿದಾಗ .then() ಬಳಸುವುದು
waitAndSpeakPromise().then((message) => {
    console.log(message);
});

ವಿವರಣೆ: ಇಲ್ಲಿ ನಾವು callback ಕಳುಹಿಸುವ ಬದಲು, ಒಂದು Promise ಅನ್ನು ಹಿಂದಿರುಗಿಸುತ್ತೇವೆ. ಅದು ಯಶಸ್ವಿಯಾದಾಗ resolve ಅನ್ನು ಕರೆಯುತ್ತದೆ. ನಂತರ ನಾವು .then() ಬಳಸಿ ಉತ್ತರ ಪಡೆಯುತ್ತೇವೆ.

ಉದಾಹರಣೆ 2: ಯಶಸ್ಸು ಮತ್ತು ವೈಫಲ್ಯದ ಪರೀಕ್ಷೆ (Success and Failure)

ವಿದ್ಯಾರ್ಥಿ 35ಕ್ಕಿಂತ ಹೆಚ್ಚು ಅಂಕ ಪಡೆದರೆ 'ಪಾಸ್' (Success), ಇಲ್ಲದಿದ್ದರೆ 'ಫೇಲ್' (Failure).

❌ ಹಳೆಯ ಕಾಲ್‌ಬ್ಯಾಕ್ ವಿಧಾನ:

function checkMarksCallback(marks, successCallback, errorCallback) {
    if (marks >= 35) {
        successCallback("ನೀವು ಪಾಸ್ ಆಗಿದ್ದೀರಿ! 🎉");
    } else {
        errorCallback("ನೀವು ಫೇಲ್ ಆಗಿದ್ದೀರಿ. 😢");
    }
}

checkMarksCallback(40, 
    (msg) => console.log(msg), 
    (err) => console.log(err)
);

✅ ಹೊಸ ಪ್ರಾಮಿಸ್ ವಿಧಾನ:

function checkMarksPromise(marks) {
    return new Promise((resolve, reject) => {
        if (marks >= 35) {
            resolve("ನೀವು ಪಾಸ್ ಆಗಿದ್ದೀರಿ! 🎉"); // ಯಶಸ್ಸು
        } else {
            reject("ನೀವು ಫೇಲ್ ಆಗಿದ್ದೀರಿ. 😢"); // ವೈಫಲ್ಯ
        }
    });
}

checkMarksPromise(30)
    .then((msg) => console.log(msg))    // resolve ಆದಾಗ ಇದು ಕೆಲಸ ಮಾಡುತ್ತದೆ
    .catch((err) => console.log(err));  // reject ಆದಾಗ ಇದು ಕೆಲಸ ಮಾಡುತ್ತದೆ

ವಿವರಣೆ: ಪ್ರಾಮಿಸ್‌ನಲ್ಲಿ ಯಶಸ್ಸಾದರೆ .then() ಮತ್ತು ವಿಫಲವಾದರೆ ಹಿಡಿಯಲು .catch() ಬಳಸಲಾಗುತ್ತದೆ. ಇದು ತುಂಬಾ ಶಿಸ್ತುಬದ್ಧವಾದ ವಿಧಾನ.

ಉದಾಹರಣೆ 3: ಕಾಲ್‌ಬ್ಯಾಕ್ ನರಕದಿಂದ (Callback Hell) ಮುಕ್ತಿ - ಸರಪಳಿ ವಿಧಾನ

ಆರ್ಡರ್ ಮಾಡುವುದು -> ಅಡುಗೆ ಮಾಡುವುದು -> ಬಡಿಸುವುದು. ಇವು ಒಂದಾದ ಮೇಲೊಂದು ಆಗಬೇಕು.

❌ ಹಳೆಯ ಕಾಲ್‌ಬ್ಯಾಕ್ ವಿಧಾನ (ನೋಡಲು ಗೊಂದಲ):

orderFood(function(order) {
    cookFood(order, function(cookedFood) {
        serveFood(cookedFood, function(served) {
            console.log("ಊಟ ಮುಗಿಯಿತು!");
        });
    });
});

✅ ಹೊಸ ಪ್ರಾಮಿಸ್ ವಿಧಾನ (ಚೈನಿಂಗ್ - Chaining):

ನೋಡಲು ಒಂದೇ ನೇರ ಸಾಲಿನಲ್ಲಿ ಬರುತ್ತದೆ.

orderFoodPromise()
    .then((order) => {
        return cookFoodPromise(order);
    })
    .then((cookedFood) => {
        return serveFoodPromise(cookedFood);
    })
    .then((served) => {
        console.log("ಊಟ ಮುಗಿಯಿತು!");
    })
    .catch((error) => {
        console.log("ಸಮಸ್ಯೆ ಉಂಟಾಗಿದೆ:", error);
    });

ವಿವರಣೆ: ಪ್ರತಿಯೊಂದು ಕೆಲಸ ಮುಗಿದ ನಂತರ ಮುಂದಿನ ಕೆಲಸಕ್ಕೆ ಕಳುಹಿಸಲು ನಾವು .then() ಅನ್ನು ಸರಪಳಿಯಂತೆ (Chain) ಜೋಡಿಸುತ್ತೇವೆ. ಎಲ್ಲಾದರೂ ತೊಂದರೆಯಾದರೆ ಕೊನೆಯಲ್ಲಿರುವ ಒಂದೇ ಒಂದು .catch() ಅದನ್ನು ಹಿಡಿಯುತ್ತದೆ.

ಉದಾಹರಣೆ 4: ವೃತ್ತಿಪರ ಮಟ್ಟ - Async / Await (Modern JavaScript)

ಪ್ರಾಮಿಸ್‌ಗಳನ್ನು ಇನ್ನೂ ಸುಲಭವಾಗಿ ಮತ್ತು ನೋಡಲು ಸಾಧಾರಣ (Synchronous) ಕೋಡ್‌ನಂತೆ ಬರೆಯಲು ವೃತ್ತಿಪರ ಡೆವಲಪರ್‌ಗಳು async ಮತ್ತು await ಅನ್ನು ಬಳಸುತ್ತಾರೆ.

ಮೇಲಿನ 3ನೇ ಉದಾಹರಣೆಯನ್ನು ವೃತ್ತಿಪರರು ಹೇಗೆ ಬರೆಯುತ್ತಾರೆ ನೋಡಿ:

✅ ವೃತ್ತಿಪರ Async/Await ವಿಧಾನ:

async function haveDinner() {
    try {
        // await ಎಂದರೆ ಆ ಕೆಲಸ ಮುಗಿಯುವವರೆಗೂ 'ಕಾಯಿರಿ' ಎಂದರ್ಥ
        const order = await orderFoodPromise();
        const cookedFood = await cookFoodPromise(order);
        const served = await serveFoodPromise(cookedFood);
        
        console.log("ಊಟ ಮುಗಿಯಿತು!");
    } catch (error) {
        // ಏನಾದರೂ ತಪ್ಪಾದರೆ ಇಲ್ಲಿಗೆ ಬರುತ್ತದೆ
        console.log("ಸಮಸ್ಯೆ ಉಂಟಾಗಿದೆ:", error);
    }
}

haveDinner(); // ಫಂಕ್ಷನ್ ಕರೆಯುವುದು

ವಿವರಣೆ:

  • async ಎಂಬ ಪದವನ್ನು ಫಂಕ್ಷನ್ ಹಿಂದೆ ಬರೆಯುವುದರಿಂದ ಅದು ಪ್ರಾಮಿಸ್ ಅನ್ನು ಬೆಂಬಲಿಸುತ್ತದೆ.
  • await ಎಂಬ ಪದ ಪ್ರತಿಯೊಂದು ಕೆಲಸ ಮುಗಿಯುವವರೆಗೂ ಕಾಯುವಂತೆ ಮಾಡುತ್ತದೆ.
  • ಇಲ್ಲಿ .then() ಅಥವಾ .catch() ಬರೆಯುವ ಕಿರಿಕಿರಿ ಇರುವುದಿಲ್ಲ. try { } catch { } ಬಳಸಿ ಸುಲಭವಾಗಿ ತಪ್ಪುಗಳನ್ನು ಹಿಡಿಯಬಹುದು.

ವೃತ್ತಿಪರ ವಿಸ್ತರಣೆ (Expert Expansion): Fetch & Promise.all()

ನೈಜ ಪ್ರಪಂಚದ ಪ್ರಾಜೆಕ್ಟ್‌ಗಳಲ್ಲಿ, ನಾವು ಬಹುತೇಕ Promises ಗಳನ್ನು API ಗಳಿಂದ ಡೇಟಾ ತರಲು ಬಳಸುತ್ತೇವೆ. ಇದಕ್ಕಾಗಿ fetch() ಉಪಯೋಗಿಸಲಾಗುತ್ತದೆ.

// ಒಂದು API ನಿಂದ ಡೇಟಾ ತರುವ ವೃತ್ತಿಪರ ವಿಧಾನ
async function getUserData() {
    try {
        // fetch() ತಾನಾಗಿಯೇ ಒಂದು Promise!
        const response = await fetch('https://jsonplaceholder.typicode.com/users/1');
        const user = await response.json(); 
        console.log(`ನಮಸ್ಕಾರ, ${user.name}!`);
    } catch (error) {
        console.log("ಡೇಟಾ ತರಲು ಸಾಧ್ಯವಾಗಲಿಲ್ಲ!", error);
    }
}

ಸಾರಾಂಶ:

ಕಾಲ್‌ಬ್ಯಾಕ್ ಎಂಬುದು ಹಳೆಯ ಕಾಲದ ಟೆಲಿಗ್ರಾಮ್ ಇದ್ದಂತೆ. ಪ್ರಾಮಿಸ್ ಮತ್ತು Async/Await ಎಂಬುದು ಈಗಿನ ಸ್ಮಾರ್ಟ್‌ಫೋನ್ ಇದ್ದಂತೆ! ಕೋಡ್ ಬರೆಯಲು ಮತ್ತು ಓದಲು ತುಂಬಾ ಸುಲಭವಾಗಿರುತ್ತದೆ.
