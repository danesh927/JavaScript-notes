ನಮಸ್ಕಾರ! ನೀವು ನಾನ್-ಐಟಿ (Non-IT) ಹಿನ್ನೆಲೆಯಿಂದ ಬಂದಿರುವುದರಿಂದ, ಯಾವುದೇ ಕಷ್ಟಕರವಾದ ಟೆಕ್ನಿಕಲ್ ಪದಗಳನ್ನು ಬಳಸದೆ, ನಮ್ಮ ದೈನಂದಿನ ಜೀವನದ ಉದಾಹರಣೆಗಳ ಮೂಲಕ Synchronous, Asynchronous ಮತ್ತು Callback ಎಂದರೆ ಏನು ಎಂದು ಹಂತ ಹಂತವಾಗಿ ವಿವರಿಸುತ್ತೇನೆ.

1. Synchronous ಮತ್ತು Asynchronous ಎಂದರೇನು?

Synchronous (ಸಿಂಕ್ರೋನಸ್) ಎಂದರೆ:

ಒಂದು ಕೆಲಸ ಸಂಪೂರ್ಣವಾಗಿ ಮುಗಿದ ಮೇಲೆಯೇ ಇನ್ನೊಂದು ಕೆಲಸ ಶುರು ಮಾಡುವುದು.

  • ಉದಾಹರಣೆ: ನೀವು ಎಟಿಎಂ (ATM) ಕ್ಯೂನಲ್ಲಿ ನಿಂತಿದ್ದೀರಿ. ನಿಮ್ಮ ಮುಂದಿರುವ ವ್ಯಕ್ತಿ ಹಣ ಡ್ರಾ ಮಾಡುವವರೆಗೂ ನೀವು ಕಾಯಲೇಬೇಕು. ಅವರು ಹೊರಬಂದ ಮೇಲೆಯೇ ನೀವು ಒಳಗೆ ಹೋಗಬೇಕು. ಇದು Synchronous. ಇಲ್ಲಿ ಕಾಯುವಿಕೆಯಲ್ಲಿ (Waiting) ಸಮಯ ವ್ಯರ್ಥವಾಗುತ್ತದೆ.
ವ್ಯಕ್ತಿ 1 (ATM ನಲ್ಲಿ)
ನೀವು (ಕಾಯುತ್ತಿದ್ದೀರಿ)

Asynchronous (ಅಸಿಂಕ್ರೋನಸ್) ಎಂದರೆ:

ಒಂದು ಕೆಲಸಕ್ಕೆ ಹೆಚ್ಚು ಸಮಯ ಹಿಡಿಯುತ್ತದೆ ಎಂದರೆ, ಅದಕ್ಕಾಗಿ ಕಾಯದೆ ಮುಂದಿನ ಕೆಲಸವನ್ನು ಮುಂದುವರಿಸುವುದು. ಮೊದಲನೇ ಕೆಲಸ ಮುಗಿದ ತಕ್ಷಣ ಅದರ ಫಲಿತಾಂಶವನ್ನು ಪಡೆಯುವುದು.

  • ಉದಾಹರಣೆ: ನೀವು ರೆಸ್ಟೋರೆಂಟ್‌ಗೆ ಹೋಗಿ ದೋಸೆಗೆ ಆರ್ಡರ್ ಮಾಡುತ್ತೀರಿ. ದೋಸೆ ಬರಲು 10 ನಿಮಿಷ ಬೇಕು. ಆ 10 ನಿಮಿಷ ನೀವು ಸುಮ್ಮನೆ ಕೂರುವುದಿಲ್ಲ, ಬದಲಾಗಿ ನಿಮ್ಮ ಸ್ನೇಹಿತರ ಜೊತೆ ಮಾತನಾಡುತ್ತೀರಿ ಅಥವಾ ಮೊಬೈಲ್ ನೋಡುತ್ತೀರಿ. ದೋಸೆ ರೆಡಿಯಾದ ತಕ್ಷಣ ವೇಟರ್ ತಂದು ಕೊಡುತ್ತಾರೆ. ಇಲ್ಲಿ ನೀವು ಕಾಯುತ್ತಾ ಸಮಯ ವ್ಯರ್ಥ ಮಾಡಲಿಲ್ಲ. ಇದು Asynchronous.
ದೋಸೆ ಬೇಯುತ್ತಿದೆ (ಹಿನ್ನೆಲೆಯಲ್ಲಿ)
ನೀವು ಮೊಬೈಲ್ ನೋಡುತ್ತಿದ್ದೀರಿ (ಮುಂದಿನ ಕೆಲಸ)

2. Callback (ಕಾಲ್‌ಬ್ಯಾಕ್) ಎಂದರೇನು?

ಹೆಸರಿನಲ್ಲೇ ಇರುವಂತೆ "Call-back" (ಮತ್ತೆ ಕರೆ ಮಾಡು). ನೀವು ಒಂದು ಕೆಲಸವನ್ನು ಯಾರಿಗಾದರೂ ವಹಿಸಿ, "ಆ ಕೆಲಸ ಮುಗಿದ ಮೇಲೆ ನನಗೆ ತಿಳಿಸು (ಕಾಲ್ ಮಾಡು), ನಾನು ಮುಂದಿನ ಕೆಲಸ ಮಾಡುತ್ತೇನೆ" ಎಂದು ಹೇಳುವುದೇ ಕಾಲ್‌ಬ್ಯಾಕ್.

  • ಉದಾಹರಣೆ: ನೀವು ಮೆಕ್ಯಾನಿಕ್‌ಗೆ ಬೈಕ್ ರಿಪೇರಿಗೆ ಕೊಡುತ್ತೀರಿ. "ಬೈಕ್ ರಿಪೇರಿ ಮುಗಿದ ಮೇಲೆ ನನಗೆ ಕಾಲ್ ಮಾಡಿ, ನಾನು ಬಂದು ತೆಗೆದುಕೊಂಡು ಹೋಗುತ್ತೇನೆ" ಎಂದು ಹೇಳುತ್ತೀರಿ. ಬೈಕ್ ರೆಡಿಯಾದ ತಕ್ಷಣ ಮೆಕ್ಯಾನಿಕ್ ನಿಮಗೆ ಕಾಲ್ ಮಾಡುವುದು 'Callback'.

ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿ, ಒಂದು ಫಂಕ್ಷನ್ (ಕೆಲಸ) ಮುಗಿದ ನಂತರ, ಮತ್ತೊಂದು ಫಂಕ್ಷನ್ ಅನ್ನು ರನ್ ಮಾಡಲು ಈ Callback ಅನ್ನು ಬಳಸಲಾಗುತ್ತದೆ.

3. Callback ಬರೆಯುವ ವಿಧಾನ (Syntax)

ಬಹಳ ಸರಳವಾಗಿ ಹೇಳಬೇಕೆಂದರೆ, ಒಂದು ಫಂಕ್ಷನ್ ಒಳಗೆ ಮತ್ತೊಂದು ಫಂಕ್ಷನ್ ಅನ್ನು ಪಾಸ್ ಮಾಡುವುದು:

// ಮುಖ್ಯ ಕೆಲಸದ ಫಂಕ್ಷನ್
function doSomething(callbackFunction) {
    // 1. ಇಲ್ಲಿ ಮುಖ್ಯ ಕೆಲಸ ನಡೆಯುತ್ತದೆ...
    
    // 2. ಆ ಕೆಲಸ ಮುಗಿದ ಮೇಲೆ ಕಾಲ್‌ಬ್ಯಾಕ್ ಫಂಕ್ಷನ್ ಅನ್ನು ಕರೆಯಲಾಗುತ್ತದೆ
    callbackFunction(); 
}

4. ಉದಾಹರಣೆಗಳು ಮತ್ತು ಹಂತ-ಹಂತದ ವಿವರಣೆ

ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿ ಇದು ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ ಎಂಬುದನ್ನು ನೋಡೋಣ. (ಇಲ್ಲಿ ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್ - JavaScript ಕೋಡ್ ಬಳಸಲಾಗಿದೆ, ಆದರೆ ಅರ್ಥಮಾಡಿಕೊಳ್ಳಲು ತುಂಬಾ ಸುಲಭವಾಗಿದೆ).

ಉದಾಹರಣೆ 1: ಸರಳ ಕಾಲ್‌ಬ್ಯಾಕ್ (Synchronous Callback)

ಒಬ್ಬರ ಹೆಸರು ಹೇಳಿ, ನಂತರ ಅವರಿಗೆ ವಿದಾಯ ಹೇಳುವ ಪ್ರೋಗ್ರಾಮ್.

// 1. ಮುಖ್ಯ ಫಂಕ್ಷನ್
function greet(name, callback) {
    console.log("ನಮಸ್ಕಾರ, " + name); // ಮೊದಲ ಕೆಲಸ
    callback(); // ಎರಡನೇ ಕೆಲಸ (ಕಾಲ್‌ಬ್ಯಾಕ್)
}

// 2. ಕಾಲ್‌ಬ್ಯಾಕ್ ಆಗಿ ಕೆಲಸ ಮಾಡುವ ಫಂಕ್ಷನ್
function sayGoodbye() {
    console.log("ಮತ್ತೆ ಸಿಗೋಣ! ಬಾಯ್.");
}

// 3. ಫಂಕ್ಷನ್ ಅನ್ನು ಕರೆಯುವುದು
greet("ರಾಮು", sayGoodbye);
ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ?
  1. ನಾವು greet ಫಂಕ್ಷನ್‌ಗೆ "ರಾಮು" ಎಂಬ ಹೆಸರು ಮತ್ತು sayGoodbye ಎಂಬ ಫಂಕ್ಷನ್ ಅನ್ನು ಕಳುಹಿಸುತ್ತೇವೆ.
  2. ಮೊದಲು ಕಂಪ್ಯೂಟರ್ "ನಮಸ್ಕಾರ, ರಾಮು" ಎಂದು ಪ್ರಿಂಟ್ ಮಾಡುತ್ತದೆ.
  3. ತಕ್ಷಣವೇ ಕಾಲ್‌ಬ್ಯಾಕ್ ಆದ sayGoodbye ಅನ್ನು ರನ್ ಮಾಡಿ "ಮತ್ತೆ ಸಿಗೋಣ! ಬಾಯ್." ಎಂದು ಪ್ರಿಂಟ್ ಮಾಡುತ್ತದೆ.

ಉದಾಹರಣೆ 2: ಅಸಿಂಕ್ರೋನಸ್ ಕಾಲ್‌ಬ್ಯಾಕ್ (ಪಿಜ್ಜಾ ಆರ್ಡರ್ ಮಾಡುವ ಉದಾಹರಣೆ - Asynchronous Callback)

ಇದು ನಿಜವಾದ ಕಾಲ್‌ಬ್ಯಾಕ್‌ನ ಶಕ್ತಿ. ಪಿಜ್ಜಾ ತಯಾರಾಗಲು ಸಮಯ ಬೇಕು, ಅಲ್ಲಿಯವರೆಗೆ ಕಾಯದೆ ನಾವು ಬೇರೆ ಕೆಲಸ ಮಾಡಬಹುದು.

(ಸೂಚನೆ: setTimeout ಎಂಬುದು ಪ್ರೋಗ್ರಾಮಿಂಗ್‌ನಲ್ಲಿ ಸ್ವಲ್ಪ ಸಮಯ ಕಾಯಲು ಬಳಸುವ ಟೂಲ್ ಆಗಿದೆ)

function orderPizza(callback) {
    console.log("1. ಪಿಜ್ಜಾ ಆರ್ಡರ್ ಮಾಡಲಾಗಿದೆ. ತಯಾರಾಗುತ್ತಿದೆ...");

    // ಪಿಜ್ಜಾ ತಯಾರಾಗಲು 3 ಸೆಕೆಂಡ್ (3000 milliseconds) ಬೇಕು
    setTimeout(function() {
        console.log("2. ಬಿಸಿ ಬಿಸಿ ಪಿಜ್ಜಾ ರೆಡಿಯಾಗಿದೆ!");
        callback(); // ಪಿಜ್ಜಾ ರೆಡಿಯಾದ ಮೇಲೆ ಈ ಕಾಲ್‌ಬ್ಯಾಕ್ ರನ್ ಆಗುತ್ತದೆ
    }, 3000);
}

function eatPizza() {
    console.log("3. ನಾನು ಪಿಜ್ಜಾ ತಿನ್ನುತ್ತಿದ್ದೇನೆ. ಸೂಪರ್ ಆಗಿದೆ!");
}

// ಪ್ರೋಗ್ರಾಮ್ ಶುರು
orderPizza(eatPizza);
console.log("ನಾನು ಪಿಜ್ಜಾ ಬರುವವರೆಗೆ ಟಿವಿ ನೋಡುತ್ತಿದ್ದೇನೆ...");
ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ? (ಹಂತ ಹಂತವಾಗಿ):
  1. ಮೊದಲು orderPizza ಶುರುವಾಗುತ್ತದೆ. "1. ಪಿಜ್ಜಾ ಆರ್ಡರ್ ಮಾಡಲಾಗಿದೆ..." ಎಂದು ಪ್ರಿಂಟ್ ಆಗುತ್ತದೆ.
  2. ಪಿಜ್ಜಾ ತಯಾರಾಗಲು 3 ಸೆಕೆಂಡ್ ಬೇಕು (ಇದು ಅಸಿಂಕ್ರೋನಸ್ ಕೆಲಸ). ಕಂಪ್ಯೂಟರ್ 3 ಸೆಕೆಂಡ್ ಸುಮ್ಮನೆ ಕಾಯುವುದಿಲ್ಲ!
  3. ಅದು ನೇರವಾಗಿ ಕೊನೆಯ ಸಾಲಿಗೆ ಬಂದು "ನಾನು ಪಿಜ್ಜಾ ಬರುವವರೆಗೆ ಟಿವಿ ನೋಡುತ್ತಿದ್ದೇನೆ..." ಎಂದು ಪ್ರಿಂಟ್ ಮಾಡುತ್ತದೆ.
  4. ಸರಿಯಾಗಿ 3 ಸೆಕೆಂಡ್ ಆದ ಮೇಲೆ, ಪಿಜ್ಜಾ ರೆಡಿಯಾಗುತ್ತದೆ. ಆಗ "2. ಬಿಸಿ ಬಿಸಿ ಪಿಜ್ಜಾ ರೆಡಿಯಾಗಿದೆ!" ಎಂದು ಪ್ರಿಂಟ್ ಆಗುತ್ತದೆ.
  5. ಕೊನೆಯದಾಗಿ ಕಾಲ್‌ಬ್ಯಾಕ್ (eatPizza) ರನ್ ಆಗಿ "3. ನಾನು ಪಿಜ್ಜಾ ತಿನ್ನುತ್ತಿದ್ದೇನೆ..." ಎಂದು ಬರುತ್ತದೆ.

ಉದಾಹರಣೆ 3: ಫೋಟೋ ಡೌನ್‌ಲೋಡ್ ಮಾಡುವುದು (ಇಂಟರ್ನೆಟ್ ಉದಾಹರಣೆ)

ನೀವು ವಾಟ್ಸಾಪ್‌ನಲ್ಲಿ ಒಂದು ದೊಡ್ಡ ವಿಡಿಯೋ ಅಥವಾ ಫೋಟೋ ಡೌನ್‌ಲೋಡ್ ಮಾಡುವಾಗ, ಅದು ಡೌನ್‌ಲೋಡ್ ಆಗುವವರೆಗೂ ನೀವು ಬೇರೆ ಮೆಸೇಜ್ ಓದುತ್ತಿರುತ್ತೀರಿ ಅಲ್ಲವೇ? ಅದೇ ರೀತಿ ಇದು ಕೆಲಸ ಮಾಡುತ್ತದೆ.

function downloadPhoto(photoName, callback) {
    console.log("-> " + photoName + " ಡೌನ್‌ಲೋಡ್ ಶುರುವಾಗಿದೆ...");

    // ಡೌನ್‌ಲೋಡ್ ಆಗಲು 4 ಸೆಕೆಂಡ್ ಬೇಕು ಎಂದು ಕಲ್ಪಿಸಿಕೊಳ್ಳಿ
    setTimeout(function() {
        console.log("-> " + photoName + " ಸಂಪೂರ್ಣವಾಗಿ ಡೌನ್‌ಲೋಡ್ ಆಯ್ತು.");
        callback(photoName); // ಡೌನ್‌ಲೋಡ್ ಮುಗಿದ ಮೇಲೆ ಓಪನ್ ಮಾಡಲು ಹೇಳುವುದು
    }, 4000); 
}

function openPhoto(photo) {
    console.log("-> ಈಗ ನಾನು " + photo + " ಅನ್ನು ಗ್ಯಾಲರಿಯಲ್ಲಿ ನೋಡುತ್ತಿದ್ದೇನೆ.");
}

// ಕೆಲಸ ಶುರು
downloadPhoto("ಪ್ರವಾಸದ_ಫೋಟೋ.jpg", openPhoto);
console.log("-> ಫೋಟೋ ಡೌನ್‌ಲೋಡ್ ಆಗುವವರೆಗೆ ನಾನು ಬೇರೆ ಮೆಸೇಜ್ ಓದುತ್ತಿದ್ದೇನೆ...");
ಹೇಗೆ ಕೆಲಸ ಮಾಡುತ್ತದೆ?
  1. ಮೊದಲು ಫೋಟೋ ಡೌನ್‌ಲೋಡ್ ಶುರುವಾಗುತ್ತದೆ.
  2. ಡೌನ್‌ಲೋಡ್ ಆಗಲು 4 ಸೆಕೆಂಡ್ ತೆಗೆದುಕೊಳ್ಳುತ್ತದೆ, ಆ ಸಮಯದಲ್ಲಿ ಸಿಸ್ಟಮ್ ಬ್ಲಾಕ್ ಆಗುವುದಿಲ್ಲ. ತಕ್ಷಣವೇ "ನಾನು ಬೇರೆ ಮೆಸೇಜ್ ಓದುತ್ತಿದ್ದೇನೆ" ಎಂಬ ಸಾಲು ರನ್ ಆಗುತ್ತದೆ.
  3. 4 ಸೆಕೆಂಡ್ ನಂತರ, ಡೌನ್‌ಲೋಡ್ ಮುಗಿಯುತ್ತದೆ ಮತ್ತು ಕಾಲ್‌ಬ್ಯಾಕ್ ಮೂಲಕ ಆ ಫೋಟೋ ಗ್ಯಾಲರಿಯಲ್ಲಿ ಓಪನ್ ಆಗುತ್ತದೆ.

ಒಟ್ಟಾರೆಯಾಗಿ ಸಾರಾಂಶ (Summary):

  • Synchronous: ಒಬ್ಬರಾದ ಮೇಲೆ ಒಬ್ಬರು (ಕ್ಯೂನಲ್ಲಿ ನಿಂತಂತೆ).
  • Asynchronous: ಕಾಯದೆ ಬೇರೆ ಕೆಲಸ ಮಾಡುವುದು ಮತ್ತು ಮೊದಲ ಕೆಲಸ ಮುಗಿದ ಮೇಲೆ ಅದರ ಫಲಿತಾಂಶ ಪಡೆಯುವುದು (ರೆಸ್ಟೋರೆಂಟ್‌ನಲ್ಲಿ ಆರ್ಡರ್ ಮಾಡಿದಂತೆ).
  • Callback: "ನನ್ನ ದೊಡ್ಡ ಕೆಲಸ ಮುಗಿದ ಮೇಲೆ, ಈ ಸಣ್ಣ ಕೆಲಸವನ್ನು ಮಾಡು" ಎಂದು ಪ್ರೋಗ್ರಾಮ್‌ಗೆ ಹೇಳುವ ಒಂದು ಅದ್ಭುತ ವಿಧಾನ.
ಹೆಚ್ಚುವರಿ ಪ್ರಮುಖ ಮಾಹಿತಿಗಳು (Bonus Core Concepts)

5. DOM & Element Selection (ಡಾಮ್ ಮತ್ತು ಎಲಿಮೆಂಟ್ ಆಯ್ಕೆ)

ಜಾವಾಸ್ಕ್ರಿಪ್ಟ್‌ನಲ್ಲಿ HTML ಪೇಜ್ ಅನ್ನು ಒಂದು "ಮನೆ" ಅಥವಾ ಬ್ಲೂಪ್ರಿಂಟ್ (Blueprint) ರೀತಿ ನೋಡಲಾಗುತ್ತದೆ. ಇದನ್ನೇ DOM (Document Object Model) ಎನ್ನುತ್ತೇವೆ.

  • Element Selection: ಮನೆಯಲ್ಲಿ ನಿಮ್ಮ ನಿರ್ದಿಷ್ಟ ಕೋಣೆಯನ್ನು ಹುಡುಕುವಂತೆ, ವೆಬ್‌ಸೈಟ್‌ನಲ್ಲಿರುವ ಒಂದು ಬಟನ್ ಅಥವಾ ಟೆಕ್ಸ್ಟ್ ಅನ್ನು ಹುಡುಕಿ ಬದಲಾಯಿಸುವುದು.
  • ಉದಾಹರಣೆ: ನಿಮ್ಮ ಸ್ನೇಹಿತ "ಕೆಂಪು ಶರ್ಟ್" ಹಾಕಿದ್ದಾನೆ ಎಂದು ಹುಡುಕುವುದು (Class Name ಮೂಲಕ ಹುಡುಕಿದಂತೆ).

Event Handling (ಘಟನೆಗಳನ್ನು ನಿರ್ವಹಿಸುವುದು)

ಯಾರಾದರೂ ನಿಮ್ಮ ಮನೆಯ ಕಾಲಿಂಗ್ ಬೆಲ್ ಒತ್ತಿದಾಗ, ನೀವು ಬಾಗಿಲು ತೆಗೆಯುತ್ತೀರಿ. ಇಲ್ಲಿ "ಬೆಲ್ ಒತ್ತುವುದು" ಒಂದು Event (ಘಟನೆ), ಮತ್ತು "ಬಾಗಿಲು ತೆಗೆಯುವುದು" ನೀವು ನೀಡುವ ಪ್ರತಿಕ್ರಿಯೆ.

// 1. HTML ನಲ್ಲಿರುವ 'bellButton' ಅನ್ನು ಹುಡುಕುವುದು (Element Selection)
const button = document.getElementById("bellButton");

// 2. ಬೆಲ್ ಒತ್ತಿದಾಗ (click event) ಏನು ಆಗಬೇಕು ಎಂದು ಬರೆಯುವುದು (Event Handling)
button.addEventListener("click", function() {
    console.log("ಬಾಗಿಲು ತೆಗೆಯಲಾಗಿದೆ! ಸ್ವಾಗತ.");
    // ಬಣ್ಣ ಬದಲಾಯಿಸುವುದು (DOM Manipulation)
    document.body.style.backgroundColor = "lightgreen";
});

6. Client-side Storage (ಬ್ರೌಸರ್‌ನಲ್ಲಿ ಡೇಟಾ ಸೇವ್ ಮಾಡುವುದು)

ಇಂಟರ್ನೆಟ್ ಕಟ್ ಆದರೂ, ನೀವು ಯಾವ ಥೀಮ್ (Dark/Light) ಬಳಸುತ್ತಿದ್ದೀರಿ ಅಥವಾ ಲಾಗಿನ್ ಆಗಿದ್ದೀರಾ ಎಂದು ಬ್ರೌಸರ್ ನೆನಪಿಟ್ಟುಕೊಳ್ಳಲು ಈ Storage ಬಳಸುತ್ತೇವೆ.

LocalStorage (ಶಾಶ್ವತ)

ಇದು ನಿಮ್ಮ ಪರ್ಸನಲ್ ಡೈರಿ ಇದ್ದಂತೆ. ಬ್ರೌಸರ್ ಮುಚ್ಚಿದರೂ ಇದರಲ್ಲಿರುವ ಮಾಹಿತಿ ಅಳಿಸಿಹೋಗುವುದಿಲ್ಲ.

SessionStorage (ತಾತ್ಕಾಲಿಕ)

ಇದು ಸಿನಿಮಾ ಟಿಕೆಟ್ ಇದ್ದಂತೆ. ಬ್ರೌಸರ್ (ಥಿಯೇಟರ್) ಮುಚ್ಚಿದ ತಕ್ಷಣ ಡೇಟಾ ಡಿಲೀಟ್ ಆಗುತ್ತದೆ.

// ಡೈರಿಯಲ್ಲಿ (LocalStorage) ಹೆಸರು ಸೇವ್ ಮಾಡುವುದು
localStorage.setItem("username", "ರಾಮು");

// ಸೇವ್ ಮಾಡಿದ ಡೇಟಾ ವಾಪಸ್ ಪಡೆಯುವುದು
const name = localStorage.getItem("username");
console.log("ಮತ್ತೆ ಸ್ವಾಗತ, " + name); 

// ಬೇಡವಾದಾಗ ಡೇಟಾ ಡಿಲೀಟ್ ಮಾಡುವುದು
localStorage.removeItem("username");
